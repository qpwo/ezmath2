\title{Positive scalar curvature forces codimension-two width:\\a complete proof}

\hypersetup{pdftitle={Positive scalar curvature and uniform codimension-two width},pdfauthor={OpenAI}}

\begin{abstract}
Let $(M,g)$ be a connected complete smooth Riemannian $n$-manifold,
$n\ge 4$, with scalar curvature $\Scal_g\ge 1$ and no boundary.
We prove the quantitative form of Gromov's macroscopic-dimension
conjecture in every dimension $n\ge 4$. There is a constant $C_n$
depending only on $n$ such that $M$ admits a continuous map $f\colon M\to K$
to a simplicial complex $K$ of dimension at most $n-2$, whose entire
fibers have $g$-diameter at most $C_n$. Equivalently, the Urysohn
$(n-2)$-width satisfies $\UW_{n-2}(M,g)\le C_n$.

Three consequences follow. First, the filling radius of any closed
$M$ with $\Scal_g\ge\sigma^2$ is at most $C_n/(2\sigma)$. Second, the
universal cover of any closed manifold of positive scalar curvature in
every dimension $n\ge2$ has continuous macroscopic dimension at most $n-2$.
Third, no closed aspherical manifold of dimension $n\ge4$ admits a
metric of positive scalar curvature.

The proof has three parts. (i) Stretch the metric at a fixed original
scale so that every label pair becomes arbitrarily well separated,
at the price of a mild drift loss. (ii) Cut each label pair apart by a
minimizing hypersurface, handling high-dimensional singularities by
enlarging the pressure profiles of the remaining labels; a warped-circle
trick restores scalar control on each cut packet. (iii) Assemble
coordinates $t_i\colon M\to[0,1]$ that are $1$ inside label $i$ and
$0$ far outside it; the walls cannot all pass through the same point,
so at most $n-2$ coordinates are fractional, and any two points in a
common fiber lie in a common ball of radius $2D$.
\end{abstract}

\section{Introduction}

A lower bound for scalar curvature is a local condition. Gromov's
macroscopic-dimension conjecture \cite{gromov1996} predicts a global
consequence: at a controlled scale, two dimensions can be collapsed
without increasing the diameter of any fiber by more than a constant.
In this paper we prove the quantitative form of that conjecture for
arbitrary complete manifolds in dimensions $\ge4$.

Fix a metric space $X$. For $d\ge 0$ define its \emph{Urysohn $d$-width}
\begin{equation}\label{eq:urysohn}
 \UW_d(X)=\inf_f\ \sup_{y\in K}\diam_X f^{-1}(y),
\end{equation}
where the infimum runs over continuous maps $f:X\to K$ to simplicial
complexes of dimension at most $d$, and where $\diam_X$ is the diameter
of the entire fiber in $X$. Empty fibers are omitted from the sup.

\begin{theorem}[Main bound]\label{thm:main}
For every integer $n\ge4$ there is a finite constant $C_n$, depending
only on $n$, with the following property. Every connected complete
smooth Riemannian $n$-manifold $(M,g)$ without boundary and with
$\Scal_g\ge1$ admits a continuous map $f\colon M\to K$ to a simplicial
complex of dimension at most $n-2$ such that
\[
 \diam_g f^{-1}(y)\le C_n \quad\text{for every } y\in K.
\]
In particular, $\UW_{n-2}(M,g)\le C_n$.
\end{theorem}

The constant $C_n$ is independent of $M$, its topology and its
geometry. No orientability, spin, covering, or bounded-geometry
condition enters. If $\Scal_g\ge\sigma^2$ for some $\sigma>0$, the
change of metric $g\rightsquigarrow\sigma^2 g$ gives
$\UW_{n-2}(M,g)\le C_n/\sigma$.

For a closed connected Riemannian manifold, Gromov's filling radius
\cite{gromov1983Filling} is defined via the Kuratowski embedding and
the fundamental class; see \cite[Definition~3.1]{nabutovskyRotmanSabourau2021}.
The comparison between Urysohn width and filling radius is recorded
in \cite[Definition~6.1, eq.~(6.1)]{nabutovskyRotmanSabourau2021}.

\begin{corollary}[Filling radius]\label{cor:filling}
Let $(M,g)$ be closed, connected, smooth, of dimension $n\ge4$, with
$\Scal_g\ge\sigma^2$ for some $\sigma>0$. Then
$\operatorname{FillRad}(M,g)\le C_n/(2\sigma)$.
\end{corollary}

\begin{proof}
A target of dimension $\le n-2$ is also admissible for $\UW_{n-1}$.
Because $M$ is closed, \cite[eq.~(6.1)]{nabutovskyRotmanSabourau2021}
gives $\operatorname{FillRad}(M,g)\le\frac12\UW_{n-1}(M,g)$, and the
width comparison $\UW_{n-1}(M,g)\le\UW_{n-2}(M,g)$ follows from the
definition \eqref{eq:urysohn}. Apply \cref{thm:main} to $\sigma^2 g$.
\end{proof}

For a metric space $X$, define its \emph{continuous macroscopic
dimension} $\dim_{\mathrm{mc}}(X)\le d$ to mean: there is a continuous
map $X\to K$ to a simplicial complex of dimension at most $d$, carrying
the uniform simplex metric (all simplices standard Euclidean of common
edge length $1$), such that every entire fiber has bounded diameter.

\begin{corollary}[Universal covers]\label{cor:cover}
Let $(M,h)$ be closed, connected, smooth, of dimension $n\ge2$, with
$\Scal_h>0$. Then $\dim_{\mathrm{mc}}(\widetilde M,\widetilde h)\le n-2$
for the lifted metric on the universal cover.
\end{corollary}

\begin{corollary}[Aspherical obstruction]\label{cor:aspherical}
No closed smooth aspherical manifold of dimension $n\ge4$ admits a
Riemannian metric of strictly positive scalar curvature.
\end{corollary}

The proof of \cref{cor:aspherical} from \cref{thm:main} is the same
width/filling-radius exclusion used by Gromov \cite[\S3]{gromov2017Dozen};
we present it in \cref{sec:closed-covers}.

\subsection{History and context}

Gromov introduced macroscopic dimension in \cite{gromov1996} to study
positive scalar curvature. The conjecture already asks whether a
complete manifold with a positive lower bound on scalar curvature can
be mapped continuously to a polyhedron of codimension two with all
point inverses bounded in terms of the curvature scale; see
\cite[\S$2\frac12$]{gromov1996} and the discussion around
\cite[p.~3180]{gromov2019Mean}. The diameter condition concerns each
entire fiber including all components. The continuous form of the
invariant and the version requiring a Lipschitz map to a uniformly
metrized complex are distinct; see
\cite[Definitions~1.1--1.2]{dranishnikov2014}.

The pointwise hypothesis cannot be weakened to a fixed-scale macroscopic
bound: Kumar and Sen construct closed manifolds in every dimension $n\ge4$
with prescribed positive macroscopic scalar curvature at a fixed scale
but arbitrarily large Urysohn codimension-two width
\cite[Theorem~A]{kumarSen2026Width}.

Prior results in low dimensions are due to Gromov--Lawson
\cite[Corollary~10.11]{gromovLawson1983}, Bolotov \cite{bolotov2003}
for the three-dimensional macroscopic-dimension obstruction, and
Liokumovich--Maximo \cite[Theorem~1.1]{liokumovichMaximo2023} and
Liokumovich--Wang \cite[Theorem~1.1]{liokumovichWang2023} for curvature-scale
width bounds. The aspherical obstructions in dimensions four and five
were proved by Chodosh--Li \cite[Theorem~2]{chodoshLi} and independently
by Gromov \cite[\S7]{gromov2020Aspherical}.

Topological approaches to the codimension-two conclusion, conditional on
the Strong Novikov Conjecture and injectivity of the connective-to-periodic
real K-homology map, are due to Bolotov--Dranishnikov
\cite[Theorem~4.3]{bolotovDranishnikov2010,bolotovDranishnikov2010,bolotovDranishnikov2016}
and Daher--Dranishnikov \cite[Theorem~3.3]{daherDranishnikov2019}.
Those results concern universal covers with a finite (but unspecified)
bound, whereas \cref{thm:main} covers all complete manifolds with a
bound depending only on $n$ after scalar normalization.

The geometric method is that of Schoen--Yau's minimal-hypersurface
obstruction \cite[Section~1]{schoenYau1979Structure} combined with
Gromov's torical symmetrization \cite[Section~2]{gromov2018Metric}
and the $\mu$-bubble method of Chodosh--Li \cite[Section~3 and
Remark~11]{chodoshLi}. The shielding and regularization here are
inspired by Lohkamp's splitting technology
\cite[\S1.2 and Theorem~3.15]{lohkamp2022SplittingsII}.

\subsection{Strategy of the proof}

Fix $n\ge4$ and a complete Riemannian $n$-manifold $(M,g)$ with
$\Scal_g\ge1$. Choose a scale $D$ depending only on $n$ (specified in
\cref{sec:rounds}). Let $(x_i)_{i\ge1}$ be a maximal $D$-separated
subset of $M$; then the closed balls $A_i=\overline B_g(x_i,D)$ cover
$M$ and the outer balls $\overline B_g(x_i,2D)$ are locally finite. Set
$B_i=M\setminus B_g(x_i,2D)$.

The proof has three big steps.

\medskip\noindent
\textbf{Step 1 (stretching).} Find a complete metric $h\ge g$ and a
smooth function $t$ such that for every $i$ the $h$-distance between
$A_i$ and $B_i$ exceeds a prescribed $L_i$, while a certain drift
scalar remains bounded below:
\begin{equation}\label{eq:drift-overview}
 \mathcal D(h,t):=\Scal_h+2\lap_h t-|\dd t|_h^2\ge\tfrac12.
\end{equation}
Only finitely many labels require any particular finite width $L_i$;
labels with larger $L_i$ are ``spread out'' over more rounds. Summing
$\sum_i L_i^{-1}<\infty$ (see \cref{sec:rounds}) keeps total trace cost
finite and makes the round construction locally finite.

\medskip\noindent
\textbf{Step 2 (cutting).} For each label $i$, find a hypersurface
$W_i\subset M$ separating the $q$-preimage of $A_i$ from the
$q$-preimage of $B_i$, together with a modification of metric and
function that keeps $\mathcal D\ge\frac12$ on the resulting pieces.
Singularities of $W_i$ in dimensions $8$ and up force a nontrivial
shielding step: the pressure profiles of the \emph{remaining} labels
are enlarged in a thin annular shell near each singular point so that
the future scalar requirement holds only away from the singular set.
This uses geometric measure theory (compactness, regularity, tangent
cones) plus a Hardy inequality on thickened annuli. A ``warped-circle''
identity restores positive scalar on each resulting packet by paying
through $399$ trivializing circle factors.

\medskip\noindent
\textbf{Step 3 (assembly).} Choose coordinates $t_i\colon M\to[0,1]$
that equal $1$ on $q^{-1}(A_i)$, $0$ on $q^{-1}(B_i)$, and transition
only inside a small neighbourhood of $W_i$. The key combinatorial
observation is that no $n-1$ of the $W_i$ can share a common point:
if they did, their mutual intersection would carry a complete
one-dimensional warped-product packet satisfying $\Scal\ge \lambda_r>0$
on a finite warped product, which is impossible (the Riccati
blockage). Hence at every point at least one coordinate equals $1$ and
at most $n-2$ are strictly inside $(0,1)$. The coordinates assemble
into a continuous map to a simplicial complex $K$ of dimension $n-2$.
If $y\in|K|$, two points in the fiber share a coordinate equal to $1$,
so both collapse images lie in the same outer ball of radius $2D$;
their $g$-distance is at most $4D+2$.

\medskip
The constant is $C_n=4D+2$, with $D=D(n)$ fixed in \cref{sec:rounds}.
We do not optimize $C_n$.

\subsection{Convention on signs and densities}

The Laplacian is $\lap=\Div\grad$, so round spheres of scalar curvature
$+1$ have positive scalar curvature in the present convention.
For a hypersurface with a chosen unit normal $\nu$, its second
fundamental form is $B(v,w)=\langle\nabla_v\nu,w\rangle$ and its mean
curvature is $H=\tr B$. Integration and divergence use Riemannian
densities; no orientation is required. Distances in the final fiber
bound are always measured in the original metric $g$.

\section{The drift scalar and warped products}\label{sec:drift}

The deformation in \cref{sec:stretch} replaces a metric $h$ and a function
$t$ by
\[
 h\rightsquigarrow h_u=h+\phi_u\,\dd u^2, \qquad t\rightsquigarrow t_u=t-\beta_u,
\]
where $u$ is a smooth height function, $\phi_u$ and $\beta_u$ are
appropriate scalar functions of the gradient length $s=|\dd u|_h$, and
the change is engineered so that the \emph{drift scalar}
\[
 \mathcal D(h,t):=\Scal_h+2\lap_h t-|\dd t|_h^2
\]
strictly increases. The point of the extra function $t$ is that the drift
$\lap_h t-|\dd t|_h^2$ is what the warped-product construction in
\cref{sec:rounds} uses to introduce diagonal circle factors with positive
scalar curvature; the drift is the ``reserve'' that pays for those factors.

\subsection{Warped product scalar curvature}

For smooth functions $p_1,\dots,p_k$ on a Riemannian manifold $(E,h)$, put
$\widehat h=h+\sum_{\alpha=1}^k e^{2p_\alpha}\dd\theta_\alpha^2$ on
$E\times\T^k$. Write $P=\sum_\alpha p_\alpha$. Then
\begin{equation}\label{eq:warp-identity}
 \Scal_{\widehat h}=\Scal_h-2\lap_h P-|\dd P|_h^2-\sum_{\alpha=1}^k|\dd p_\alpha|_h^2.
\end{equation}
Equivalently
$\Scal_{\widehat h}=\mathcal D(h,P)-\sum_\alpha|\dd p_\alpha|_h^2$
when $t=P$.

\begin{proof}[Proof of \eqref{eq:warp-identity}]
Let $E_\alpha=e^{-p_\alpha}\partial_{\theta_\alpha}$ be the unit vertical
fields. For horizontal fields $X,Y$, Koszul gives
$\widehat\nabla_XY=\nabla^h_XY$, $\widehat\nabla_XE_\alpha=0$,
$\widehat\nabla_{E_\alpha}X=(Xp_\alpha)E_\alpha$,
$\widehat\nabla_{E_\alpha}E_\alpha=-\grad_h p_\alpha$, and
$\widehat\nabla_{E_\alpha}E_\beta=0$ for $\alpha\ne\beta$.
Hence
$K(X,E_\alpha)=-\Hess_h p_\alpha(X,X)-(Xp_\alpha)^2$
and $K(E_\alpha,E_\beta)=-\langle\dd p_\alpha,\dd p_\beta\rangle$.
Summing twice over orthonormal $2$-planes gives \eqref{eq:warp-identity}.
\end{proof}

\subsection{The graph deformation}

Fix $s_0=10^{-2}$ and a smooth nondecreasing $\rho\colon[0,\infty)\to[0,1]$
with
\[
 \rho=0\text{ on }[0,s_0],\qquad \rho=1\text{ on }[5s_0,\infty),\qquad
 0\le\rho'\le 60.
\]
For $0<a\le\frac12$ and $s\ge0$ put
\begin{equation}\label{eq:switch}
 \mu=a^2+(1-a^2)\rho(s),\qquad \theta=\frac{\mu s^2}{1+s^2},
 \qquad l=(1-\theta)^{1/2},\qquad \beta=\log l.
\end{equation}
Near $s=0$ we then have $\mu\approx a^2$ and $\theta\approx a^2 s^2\to0$;
near $s=\infty$, $\mu=l=1$ and $\theta\to1$. The three regimes
``$s$ tiny, $s$ moderate, $s$ large'' will be treated separately in the
proof of \cref{thm:stretch}.

The graph metric and drift corresponding to a provisional height $u$ are
\begin{equation}\label{eq:raw-deform}
 h_u=h+\frac{\theta}{(1-\theta)s^2}\dd u^2
     =h+\frac{\mu}{1+(1-\mu)s^2}\,\dd u^2,
 \qquad t_u=t-\beta.
\end{equation}
These expressions are smooth at $s=0$; near zero slope the metric
increment is $a^2\,\dd u^2+O(s^4)$ and the drift increment is
$-\tfrac12\log(1+(1-a^2)s^2)+O(s^2)$. Hence $h_u\ge h$ and $t_u\ge t$
for all $s\ge 0$. The alternative expression for the metric in
\eqref{eq:raw-deform} records that the graph of $u$ is measured in
the product manifold $E\times\R$ with its product metric
$h+\dd z^2$: the graph of $u$ has induced metric $h+\dd u^2$, but
\eqref{eq:raw-deform} \emph{rescales} the vertical direction in a
slope-dependent way.

The reason for the specific $(\mu,\theta,l,\beta)$: at every point of
nonzero slope, $h_u+\dd z^2$ is conformal to the \emph{graph} metric
$h+\dd u^2+\dd z^2$ (with $z$ a formal vertical coordinate) with a
conformal factor depending only on $s=|\dd u|_h$. On such a graph the
various scalar-curvature terms reorganize in a computationally useful way.
The drift $t_u=t-\beta$ is tuned so that the drift scalar
$\mathcal D(h_u,t_u)$ exceeds $\mathcal D(h,t)+(t_u^2-t^2)$ by a
nonnegative amount, up to the small errors controlled in \cref{thm:stretch}.

The key structural input is the following identity. In local normal
coordinates adapted to $u$, write $h=v^2\dd q^2+\gamma_q$ with $v=1/s$
and $q=u$, and define the second fundamental form $\mathrm J$ and
mean curvature $H=\tr\mathrm J$ of the level sets in the metric
$\gamma_q$. Put $S=H-e t$ with $e=\partial_q/(2v)$ the normal derivative.
The Gauss equation and the normal variation formula for $H$ then give:

\begin{lemma}[Raw scalar identity]\label{lem:raw-scalar}
Let $E$ be a smooth manifold, $h$ a Riemannian metric, $u$ a smooth
function with $s=|\dd u|_h>0$, $t\ge T$ a smooth function, and define
$h_u$, $t_u$ by \eqref{eq:raw-deform}. Then
\begin{equation}\label{eq:raw-identity}
 \mathcal D(h_u,t_u)-\mathcal D(h,t)
 =\theta\,E(\mathrm J,H,F,\xi,t)+O(\theta E_{\rm cutoff}),
\end{equation}
where $E$ is a function of the second fundamental form, the ``pressure''
$F=F(\xi,t)$ with $\xi=(u-\psi)/\eta$ for suitable cutoff data
$\psi,\eta$, and where for small slope $s<s_0$ the identity holds with
the same form for a suitably reduced $E$.
\end{lemma}

The exact form of $E$ will be given in the proof of \cref{thm:stretch}
below. Its crucial feature: on a favourable branch of the pressure $F$,
$E$ is bounded below by a positive multiple of $s \cdot F_x$, and this
drives both the scalar increase and the distance enlargement.

We now state the \emph{pressure law} which supplies the function $F$
with the exact inequalities needed to make $E$ favourable.

\section{The pressure law}\label{sec:pressure}

The pressure is a smooth function $F(\xi,t)$ on a strip
$(-S_0,S_0)\times[T,\infty)$ with $F$ odd in $\xi$ and positive slope in
$\xi$. Its role: in the raw scalar identity \cref{lem:raw-scalar},
the second fundamental form enters through a quadratic form whose
coefficients are shaped by derivatives of $F$. We need $F$ to be steep
in $\xi$ for large drift $t$ (so that stretching is fast when we want
it), but with coupling of drift back to the vertical direction that is
not too strong. The exact requirements are the following.

\begin{theorem}[Pressure law]\label{thm:pressure}
For every choice of $A_0,P_0,\eps,b_0>0$ there exist $0<S_0\le1$, $T\ge1$,
and a smooth function
\[
 F\colon(-S_0,S_0)\times[T,\infty)\to\R
\]
which is odd in the first variable, has $F_\xi>0$, and satisfies for all
$(\xi,t)\in(-S_0,S_0)\times[T,\infty)$:
\begin{enumerate}[label=(\roman*)]
\item \textbf{Slope growth.} $F_\xi(\xi,t)\ge A_0 t$.
\item \textbf{Bounded coupling.} $|F_t(\xi,t)|\le\eps|F(\xi,t)|$
      and $F F_t\le\eps F_\xi$.
\item \textbf{Favourable branch or quadratic loss.}
      Either $F_\xi\le b_0 t^2$ (a quadratic upper bound), or
      $-FF_t\ge P_0 F_\xi$.
\item \textbf{Quadratic ceiling.} $F_\xi\le b_0 t^2+\eps F^2$.
\end{enumerate}
For every fixed $t\ge T$,
$\lim_{\xi\to-S_0}F(\xi,t)=-\infty$ and
$\lim_{\xi\to S_0}F(\xi,t)=+\infty$, uniformly on compact $t$-intervals.
The data $S_0,T$ depend only on $(A_0,P_0,\eps,b_0)$ and fixed numerical
constants, not on any manifold, metric, or subsequent deformation
parameter.
\end{theorem}

The idea of the proof is to construct $F$ by prescribing its inverse
$\xi = X(f,t)$ in the pressure variable $f=F(\xi,t)$, in such a way that
the analytic structure of $X$ as a function of $f,t$ reveals the desired
inequalities on the original $F$. This is possible because the pressure
law lives in one dimension above $\R$, so no geometric input is
required.

\begin{proof}[Proof of \cref{thm:pressure}]
Let $k>0$ and a positive smooth function $g\colon[0,\infty)\to(0,\infty)$
be auxiliary data. For $f>0$ and $t>0$ set
\begin{equation}\label{eq:pressure-inverse}
 X(f,t)=\frac1k\arctan\frac{f^{1/3}}{g(y)},\qquad
 q=f^{2/3},\qquad y=\frac tq.
\end{equation}
Differentiation at fixed $f$ and at fixed $t$, respectively, gives
\begin{equation}\label{eq:pressure-inverse-deriv}
 fX_t=-\frac{g'(y)}{k(1+g(y)^2/q)},\qquad
 X_f=\frac{g(y)+2yg'(y)}{3kq^2(1+g(y)^2/q)}.
\end{equation}
Thus invertibility of $X(\cdot,t)$ in $f$ requires the quantity
$A(y):=g(y)+2yg'(y)>0$ for all $y\ge0$. For the inverse $F$ we have
\begin{equation}\label{eq:pressure-from-inverse}
 F_\xi=1/X_f,\qquad
 \frac{FF_t}{F_\xi}=\frac{g'(y)}{k(1+g(y)^2/q)},\qquad
 \frac{F_t}{F}=\frac{3yg'(y)}{t(g(y)+2yg'(y))}.
\end{equation}

We design $g$ so that it decreases for small $y$ (favourable branch),
eventually equals $y$ (giving $X\approx k^{-1}\arctan(f/t)$, hence
$F\approx t\tan(k\xi)$), and stays above $y$ in between, with $A(y)>0$
throughout. The precise choice is a smooth rounded maximum:
fix a smooth nonnegative probability density $\rho$ supported in $[1,2]$,
positive on $(1,2)$ and vanishing to infinite order at both ends.
Let $\mu=\int_1^2 r\rho(r)\dd r\in(1,2)$ and, for $D_g>0$,
\begin{equation}\label{eq:pressure-g}
 h(y)=D_g(1+y)^{-1/4},\qquad
 g(y)=\int_1^2\max\{y,rh(y)\}\rho(r)\dd r,\quad y\ge0.
\end{equation}
The function $g$ is smooth and nondecreasing; direct differentiation of
the integral gives the two-sided estimates
\begin{equation}\label{eq:pressure-g-est}
 g(y)\ge y,\qquad
 -\frac{g(y)}{4(1+y)}\le g'(y)\le 1,\qquad
 A(y)\ge\frac{g(y)}{2}.
\end{equation}
In particular $A>0$ everywhere. For $y\le Y$ with $Y(1+Y)^{-1/4}\le D_g$,
the ``max'' is realized by $rh(y)$ on the range $r\in[1,2]$, giving the
exact formula
\begin{equation}\label{eq:pressure-g-decreasing}
 g(y)=\mu D_g(1+y)^{-1/4},\qquad
 g'(y)=-\frac{\mu D_g}{4(1+y)^{5/4}},\quad y\le Y.
\end{equation}
Thus $g$ decreases at a rate controlled by $D_g$ throughout $[0,Y]$.

Choose the constants in the following order: first
$k\ge\max\{A_0,\eps^{-1},\pi/2\}$, then $Y\ge1$ large with
$Y^{-3}\le b_0/(12k)$, then $D_g>0$ large enough that
$D_g\ge\max\{Y(1+Y)^{1/4},\;8kP_0(1+Y)^{5/4}/\mu\}$, and finally
$T\ge\max\{1,6/\eps,12kB^2/b_0,YG_0^2,12kY/(\eps g_0)\}$, where
$g_0=\min[0,Y]g$, $G_0=\max[0,Y]g$, and $B=\sup_{y>Y}g(y)/y<\infty$.

Now verify the four inequalities.

\textit{(i) Slope growth.} From \eqref{eq:pressure-inverse-deriv} and
$g\ge y$ together with $A\le 3g$, we get
$t X_f= yA/(3k(q+g^2))\le 1/k$.
Hence $F_\xi=1/X_f\ge kt\ge A_0 t$.

\textit{(ii) Bounded coupling.} From \eqref{eq:pressure-from-inverse} and
the bound $|yg'|\le g$, we get $|F_t/F|\le 6/t\le\eps$ for $t\ge T\ge6/\eps$.
The sign of $FF_t/F_\xi$ equals the sign of $g'$; on the branch
$g'\le1$, so $FF_t\le F_\xi/k\le\eps F_\xi$.

\textit{(iii) Favourable branch or quadratic.} In the region $y>Y$:
$A\ge g/2\ge y/2$ and $g/y\le B$, so
$t^2X_f\ge y^3/(6k(1+B^2y^3/T))\ge 1/(6k(Y^{-3}+B^2/T))\ge 1/b_0$.
This gives $F_\xi\le b_0 t^2$ in the region $y>Y$. In the region
$0<y\le Y$, the rate \eqref{eq:pressure-g-decreasing} together
with $D_g\ge 8kP_0(1+Y)^{5/4}/\mu$ gives
$-g'\ge 2kP_0$; hence $-FF_t/F_\xi\ge P_0$, the favourable branch.

\textit{(iv) Quadratic ceiling.} Again from \eqref{eq:pressure-from-inverse},
$t^2 X_f\ge 1/b_0$ in the large-$y$ region gives $F_\xi\le b_0t^2$.
In the small-$y$ region we instead bound $F^2 X_f\ge 1/\eps$
using $F^2X_f\ge qg_0/(12k)\ge g_0T/(12kY)\ge 1/\eps$,
and hence $F_\xi\le\eps F^2\le b_0 t^2+\eps F^2$.

\textit{Pole limits.} As $f\to0^+$: $y\to\infty$, $g(y)=y$, so
$X(f,t)\to0$. As $f\to\infty$: $y\to0$, $g(y)\to g(0)>0$, so
$X(f,t)\to\pi/(2k)=:S_0\le1$. Hence $X(\cdot,t)$ maps $(0,\infty)$
bijectively to $(0,S_0)$, and its inverse $F$ extends oddly to
$(-S_0,S_0)$ and is smooth at $\xi=0$ because near $f=0$ the formula
reduces to $X=(1/k)\arctan(f/t)$, $F=t\tan(k\xi)$. Proper pole limits
follow by taking $f$ (respectively $1/f$) large. Uniformity on
compact $t$-intervals: $\max_{t\in I}X(H,t)<S_0$, so
$\xi$ exceeding that max gives $F(\xi,t)>H$ for all $t\in I$.
\end{proof}

\begin{remark}\label{rem:pressure-data}
When $A_0,P_0,\eps,b_0$ are prescribed as functions of the ambient
dimension $n$ and fixed numerical constants, $S_0$ and $T$ are
themselves functions of $n$. In particular, $a$ and $d$ may be chosen
\emph{after} \cref{thm:pressure} has been applied, without modifying
the pressure itself. This is what allows the local stretching theorem
to put its own small constants $a,d$ on top of the pressure law.
\end{remark}

\section{Local stretching}\label{sec:stretch}

The stretching theorem pays for enlarging the distance between a
compact set $A$ and a closed set $B$ by three currencies: quadratic
drift reserve $t^2$, trace loss $\tr_h h_0 -\tr_{h'}h_0$, and a small
quadratic error $d\,t'^2$.

\begin{theorem}[Local stretching]\label{thm:stretch}
Fix $n\ge2$, $0<\delta<1/10$, and put $L_*=1.004$. There are
$W,T\ge1$, depending only on $n,\delta$, with the following property.
Let $(M,h_0)$ be a complete smooth $n$-manifold, let $h\ge h_0$ be
smooth, let $t\ge T$ be smooth, and let $A\subset M$ be compact and
$B\subset M$ closed, with $\dist_{h_0}(A,B)\ge W$. For every
$0<d<\delta$ there exist smooth $h'\ge h$ and $t'\ge t$, with
changes compactly supported in $\{x:\dist_{h_0}(A,x)<W\}$, such that
\begin{align}
 \mathcal D(h',t')&\ge\mathcal D(h,t)+(t'^2-t^2)\notag\\
 &\quad-\delta(\tr_h h_0-\tr_{h'}h_0)t'^2-d t'^2,
 \label{eq:stretch-conclusion}\\
 \dist_{h'}(A,B)&\ge L_*W.
 \label{eq:stretch-distance}
\end{align}
In particular $h'$ is complete. Neither $W$ nor $T$ depends on $d,
(M,h_0),A,B$.
\end{theorem}

The proof occupies \cref{sec:stretch-proof}; it uses the graph
deformation of \cref{sec:drift} and the pressure law
\cref{thm:pressure}. Here is the punchline, in three steps.

\textbf{Step A (setup).} Fix $A,B,h,h_0,t$ as in the statement. Let
$r_A(x)=\dist_{h_0}(A,x)$. Choose smooth functions $\psi,\eta_1,\zeta$
with
\[
 \inf_A\psi>W/5,\qquad |\dd\psi|_{h_0}\le1,\qquad
 \supp\psi\subset\{r_A<W/2\},
\]
\[
 \eta_1=1\text{ on }\{r_A\le.55W\},\qquad
 \eta_1=0\text{ on }\{r_A\ge.8W\},\qquad
 |\dd\eta_1|_{h_0}\le s_0/8,
\]
\[
 \zeta=1\text{ on }\{r_A\le.82W\},\qquad
 \zeta=0\text{ on }\{r_A\ge.98W\}.
\]
Set $\eta=r+(1-r)\eta_1^2$. Then $\eta=1$ near $\supp\psi$ and
$\eta=r$ on $\{r_A\ge.8W\}$, and the uniform bounds
$0<\eta\le1$, $|\dd\eta|_h\le s_0/4$, $|\dd\eta|_h^2/\eta\le1$ hold
independently of $r$. Enclose $\{r_A\le1.1W\}$ in a compact smooth domain
$N$ and extend $h,t$ to $N$; work entirely in $N$ from now on.

\textbf{Step B (graph equation).} For $\tau\in[0,1]$ consider the
equation
\begin{equation}\label{eq:stretch-PDE}
 \Div_h\mathcal A(\dd u)-\dd t(\mathcal A(\dd u))
 =J(|\dd u|_h)F\left(\frac{u-\tau\psi}{\eta},t\right),
\end{equation}
where $\mathcal A(p)=\sqrt\mu(1+|p|_h^2)^{-1/2}p^\sharp$ and
$J(s)=\sqrt{\theta+a^2s_0^2}$, with $\mu,\theta$ from
\eqref{eq:switch}, and $F$ from \cref{thm:pressure}. Note $\mathcal A$
and $J$ are smooth at $p=0$ because $\mu(0)=a^2$ constants.

For each fixed positive floor $r>0$ and $\tau=1$, this equation has
a smooth solution $u=u_r$ with quantitative bounds
$|u_r-\psi|\le S_1\eta$ and $|\dd u_r|_h\le G$, independent of $r$
and $\tau\in[0,1]$. This follows from the maximum/comparison barriers
of \cref{thm:pressure} (the $F$ poles force $u_r$ into the interval
$[-\psi+S_1\eta,\psi+S_1\eta]$) combined with a gradient estimate
(the maximum of $w=(1+|\dd u|_h^2)^{-1/2}$ obeys a positive lower
bound from an exact Jacobi identity for a graph in $(N,h)\times\R$;
the drift term contributes a bounded coefficient and the pressure
term dominates). Existence in $\tau$ follows by continuation: at
$\tau=0$ the solution is $u=0$; the linearization at any solution
is an isomorphism $C^{2,\alpha}\to C^{0,\alpha}$ because of the
strictly negative zero-order term $-JF_x/\eta$ combined with the
uniform ellipticity; the implicit function theorem in $C^{2,\alpha}$
gives openness in $\tau$ and the a priori bounds give closedness.

As $r\downarrow0$, on the compact region where $\psi=0$ and $\eta_1=0$
(i.e. $\eta=r$), $u_r\to0$ in every $C^k$ norm. Indeed
$|u_r/r|\le S_1$ and the equation becomes
$a^{ij}\partial_{ij}u_r+b^i\partial_i u_r-(c_r/r)u_r=0$ with
$c_r\ge as_0A_0T>0$. Comparison against the explicit barrier
$V_r=M_0\sum_f\exp(-\gamma\ell_f/\sqrt r)$ shows
$|u_r|\le C\exp(-c/\sqrt r)$ on compact subsets; the derivative
convergence follows from the rescaled equation
$\partial_{y_i}\mathfrak A^i(x_0+ry,Dv)=r\mathfrak B_0(x_0+ry,v,Dv)$
applied to $u_r(x)=rv(y)$ at $x=x_0+ry$.

\textbf{Step C (raw scalar estimate).} At every point of nonzero slope,
the raw deformation $h_u=h+\phi_u\dd u^2$, $t_u=t-\beta$ satisfies,
up to constants that are paid by the small parameters $a,d$,
\begin{equation}\label{eq:stretch-raw}
 \mathcal D(h_u,t_u)\ge\mathcal D(h,t)+(t_u^2-t^2)
 -\delta(\tr_h h_0-\tr_{h_u}h_0)t_u^2-\tfrac{d}{2}t^2.
\end{equation}
This is the heart of \cref{thm:stretch} and is proved in
\cref{sec:stretch-raw} below.

\textbf{Step D (cutoff and distance).} Define $h'=h+\zeta(h_u-h)$,
$t'=t+\zeta(t_u-t)$, extended by $(h,t)$ off $N$. If the positive
floor $r$ is chosen small enough, the discrepancies from the raw
estimate \eqref{eq:stretch-raw} caused by $\zeta$ are absorbed on the
compact annulus $S=\overline{\{0<\zeta<1\}}\subset\{.8W<r_A<W\}$;
on the rest of $N$ the output is either the raw deformation (where
$\zeta=1$) or the unchanged original (where $\zeta=0$), and both
already satisfy \eqref{eq:stretch-conclusion}. The distance gain
\eqref{eq:stretch-distance} uses the effective height
$v_{\rm eff}=\zeta u$: on $A$ it exceeds $W/5-1$; on $B$ it is zero;
on the small slope region $\{s\le5s_0=.05\}$ its variation is at most
$.05 W$; everywhere $h'\ge h$ so a path realizing length $<L_*W$ has
$h$-length $< L_*W$ and therefore $h_0$-length $\ge W$; the variation
of $v_{\rm eff}$ along any such path is at most $.01L_*W$ on
$\{0<\zeta<1\}$, so the variation on the region
$\{\zeta=1,s\ge.05\}$ exceeds $(1/5-.06L_*)W-1>.11W$; the
inequality $\sqrt{W^2+(.11W)^2}=W\sqrt{1+.11^2}>L_*W$ contradicts
$\operatorname{Length}_{h'}(\gamma)<L_*W$. Completeness of $h'$
follows from $h'\ge h_0$ and local equivalence: an $h'$-Cauchy
sequence is $h_0$-Cauchy, hence converges; near its limit $h'$ and
$h_0$ are equivalent smooth metrics. This completes
\cref{thm:stretch}. $\square$

\section{A pressure law with a quadratic budget}\label{sec:pressure}

We need a smooth pressure $F(x,t)$ used in \cref{sec:local-step}: it must
be steeply increasing in $x$, grow linearly in $t$, and have a
non-linear derivative budget in $t$.

\begin{theorem}[Pressure law]\label{thm:pressure}
Fix $A_0,P_0,\eps,b_0>0$. There exist $0<S_0\le 1$, $T\ge 1$, and a
smooth function
\[
 F\colon(-S_0,S_0)\times[T,\infty)\longrightarrow\R
\]
which is odd in $x$, has $F_x>0$, and satisfies everywhere on its
domain
\begin{equation}\label{eq:pressure-estimates}
 \begin{aligned}
 &F_x\ge A_0 t,\qquad |F_t|\le\eps|F|,\qquad FF_t\le\eps F_x,\\
 &F_x\le b_0t^2\quad\text{or}\quad -FF_t\ge P_0F_x,\\
 &F_x\le b_0t^2+\eps F^2.
 \end{aligned}
\end{equation}
For each fixed $t\ge T$,
\[
 \lim_{x\to-S_0}F(x,t)=-\infty,\qquad
 \lim_{x\to S_0}F(x,t)=+\infty,
\]
and the convergence is uniform for $t$ in any compact subset of
$[T,\infty)$.
\end{theorem}

The constants $S_0,T$ and the function $F$ depend only on the numerical
inputs. They are fixed before any metric, manifold, cutoff, or
deformation parameter is chosen.

\subsection{Why we ask for exactly this}

In \cref{sec:local-step} the unknown is a height function $u$ on a
closed manifold $N$, solving the elliptic equation
\begin{equation}\label{eq:pressure-appearance}
 \Div_h\mathcal A(\dd u)-\dd t(\mathcal A(\dd u))
 =J(|\dd u|_h)\,F\!\left(\frac{u-\tau\psi}{\eta},t\right).
\end{equation}
Here $\mathcal A$ is an operator built from an auxiliary ``slope switch''
and $J>0$ is a scalar factor. The argument of $F$ lives in $|x|<S_0$.

The inequality $F_x\ge A_0t$ makes $F$ very steep in $x$. A
maximum-principle argument (see \cref{lem:height-gradient}) then forces
$u$ to track the prescribed height $\tau\psi$ up to a fixed margin,
thereby controlling $\dd u$. Without that steepness, $u$ could drift
far from $\tau\psi$ and the metric change would be uncontrolled.

The two $t$-inequalities $|F_t|\le\eps|F|$ and $FF_t\le\eps F_x$
control the price of the local stretching deformation: when the
deformation increases a background trace by a small amount, the drift
$t$ absorbs the resulting scalar-loss with error at most $\eps$ in
units of $F$.

The two-branch alternative in the second line selects a small-slope
regime (first branch, $F_x$ small) from a large-slope regime (second
branch, $-FF_t$ large relative to $F_x$); the third inequality, the
quadratic budget $F_x\le b_0t^2+\eps F^2$, closes the analysis in the
transition region where neither of the two branches is decisive.

The construction is by an inverse coordinate: instead of $F$
directly we prescribe $X(\cdot,t)=F(\cdot,t)^{-1}$ and read the
derivative inequalities from the inverse-function formulas. Almost
all of the work is choosing an auxiliary $g$ so that both branches
of the second inequality hold.

\subsection{The inverse construction}

Let $k>0$ and $g\colon[0,\infty)\to(0,\infty)$ be smooth positive
functions to be fixed. For a pressure value $f>0$ and a drift $t>0$
define a height
\begin{equation}\label{eq:X-def}
 X(f,t)=\frac1k\arctan\!\left(\frac{f^{1/3}}{g(y)}\right),
 \qquad y=\frac{t}{f^{2/3}}.
\end{equation}
We will see that for fixed $t$ the map $X(\cdot,t)$ is strictly
increasing in $f$, with limit $0$ as $f\to 0$ and limit $S_0=\pi/(2k)$
as $f\to\infty$. Hence $F(\cdot,t)=X(\cdot,t)^{-1}$ is well defined
and smooth on $(0,S_0)$; extending oddly to $(-S_0,0)$ and setting
$F(0,t)=0$ gives the full pressure.

For small $f$ we have $g(y)=y$ for large $y$, so
$X(f,t)=(1/k)\arctan(f/t)$ and $F(x,t)=t\tan(kx)$ near $x=0$; this
is jointly smooth in $(x,t)$.

The two branches of the second estimate come from two regimes of $g$.
For small $y$ we make $g$ decreasing so that $-FF_t/F_x$ is large; for
large $y$ we set $g(y)=y$ so that $F_x\le b_0t^2$ holds. The transition
region uses the quadratic budget instead.

\subsection{Construction of the auxiliary function $g$}

Let $\rho\colon[1,2]\to[0,\infty)$ be a smooth nonnegative density
supported in $[1,2]$, positive in $(1,2)$, vanishing to infinite
order at both endpoints, with $\int_1^2\rho(r)\dd r=1$. For example
\[
 \rho(r)=c_\rho\exp\bigl(-1/((r-1)(2-r))\bigr),\qquad 1<r<2,
\]
extended by zero. Write $\mu=\int_1^2 r\rho(r)\dd r\in(1,2)$.

For a large parameter $D_g>0$ set $h(y)=D_g(1+y)^{-1/4}$ and
\begin{equation}\label{eq:g-def}
 g(y)=\int_1^2\max\{y,rh(y)\}\rho(r)\dd r,\qquad y\ge 0.
\end{equation}
This is a smoothed maximum of the constants $rh(y)$ over $r\in[1,2]$.
Equivalently, with $M(z)=\int_1^2\max\{z,r\}\rho(r)\dd r$,
$g(y)=h(y)M(y/h(y))$. Since $M''=\rho$ (with $\rho$ extended by
zero), $M$ is smooth, and $M(z)=\mu$ for $z\le 1$, $M(z)=z$ for
$z\ge 2$. Therefore $g>0$, $g\ge y$, and $g(y)=y$ for all
sufficiently large $y$.

Write $L(z)=\int_{r\le z}\rho(r)\dd r$ and
$R(z)=\int_{r\ge z}r\rho(r)\dd r$. At $z=y/h(y)$,
\begin{equation}\label{eq:g-bounds}
 g'=L(z)-\frac{hR(z)}{4(1+y)}.
\end{equation}
This gives
\begin{equation}\label{eq:g-inequalities}
 g\ge y,\qquad -\frac{g}{4(1+y)}\le g'\le 1,\qquad
 g+2yg'\ge\frac{y+2}{2(y+1)}g\ge \tfrac12 g.
\end{equation}
The last inequality is the key one: the combination $g+2yg'$ that
appears in the inverse derivative formula is comparable to $g$ itself.

\subsection{Order of constants}

Precisely, choose
\begin{equation}\label{eq:k-Y-choice}
 k\ge \max\{A_0,\eps^{-1},\pi/2\},\qquad
 Y\ge 1\quad\text{with}\quad Y^{-3}\le \frac{b_0}{12k}.
\end{equation}
Choose $D_g$ only after $k,Y$:
\begin{equation}\label{eq:Dg-choice}
 D_g\ge \max\!\left\{Y(1+Y)^{1/4},\,
 \frac{8kP_0(1+Y)^{5/4}}{\mu}\right\}.
\end{equation}
Since $y(1+y)^{1/4}$ is increasing and
$y/h(y)=y(1+y)^{1/4}/D_g\le 1$ for $y\le Y$, on $[0,Y]$
\begin{equation}\label{eq:g-decreasing}
 g(y)=\mu D_g(1+y)^{-1/4},\qquad
 g'(y)=-\frac{\mu D_g}{4(1+y)^{5/4}}\le -2kP_0.
\end{equation}
Finally define
$g_0=\min_{0\le y\le Y}g(y)$,
$G_0=\max_{0\le y\le Y}g(y)$,
$B=\sup_{y>Y}g(y)/y$, and choose
\begin{equation}\label{eq:T-choice}
 T\ge\max\!\left\{1,\frac{6}{\eps},\frac{12kB^2}{b_0},
 YG_0^2,\frac{12kY}{\eps g_0}\right\}.
\end{equation}
The order $k$, then $Y$, then $D_g$ (which fixes $g$), then $T$,
is respected.

\subsection{Derivatives of the inverse}

By \cref{eq:X-def},
\begin{equation}\label{eq:X-f-derivative}
 f X_f=\frac{A(y)}{3k(1+g(y)^2/q)},\qquad
 A(y)=g(y)+2y g'(y),
\end{equation}
where $q=f^{2/3}$, $y=t/q$. By \cref{eq:g-inequalities} we have
$A\ge g/2>0$, hence $X_f>0$. Set $F(\cdot,t)=X(\cdot,t)^{-1}$. Then
\begin{equation}\label{eq:F-derivatives}
 F_x=\frac1{X_f},\qquad F_t=-\frac{X_t}{X_f},
\end{equation}
and
\begin{equation}\label{eq:pressure-ratios}
 \frac{FF_t}{F_x}=\frac{g'(y)}{k(1+g(y)^2/q)},\qquad
 \frac{F_t}{F}=\frac{3y\,g'(y)}{t(g(y)+2y\,g'(y))}.
\end{equation}

\subsection{The first three inequalities}

\paragraph{$F_x\ge A_0 t$.} Since $A\le g+2y\le 3g$ and $y\le g$,
\[
 t X_f=\frac{y A}{3k(q+g^2)}\le \frac1k,
\]
hence $F_x=1/X_f\ge k t\ge A_0t$ by \cref{eq:k-Y-choice}.

\paragraph{$|F_t|\le\eps|F|$.} From \cref{eq:pressure-ratios}, and
from $|y g'|\le g$ (for $g'\ge 0$ use $g'\le1$ and $y\le g$; for
$g'<0$ use $|y g'|\le g/4$),
\[
 \left|\frac{F_t}{F}\right|\le\frac{3|yg'|}{t g}\le\frac3t\le\frac{\eps}{2}\le\eps
\]
because $t\ge T\ge 6/\eps$.

\paragraph{$FF_t\le\eps F_x$.} Again from \cref{eq:pressure-ratios}:
if $g'\le 0$, $FF_t\le 0\le \eps F_x$; if $g'>0$, then
\[
 \frac{FF_t}{F_x}\le\frac{g'}{k}\le\frac{1}{k}\le \eps,
\]
since $k\ge\eps^{-1}$.

\subsection{The two branches of the alternative}

We prove: for every $(f,t)$ either $F_x\le b_0t^2$ or
$-FF_t/F_x\ge P_0$.

\paragraph{Large $y>Y$.} Here $A\ge g/2\ge y/2$ and $g/y\le B$, so
\[
 t^2X_f=\frac{y^2A}{3k(1+yg^2/t)}
 \ge \frac{y^3}{6k(1+B^2y^3/T)}
 =\frac{1}{6k(y^{-3}+B^2/T)}
 \ge\frac{1}{6k(Y^{-3}+B^2/T)}.
\]
By \cref{eq:k-Y-choice}, $Y^{-3}\le b_0/(12k)$; by \cref{eq:T-choice},
$B^2/T\le b_0/(12k)$. Hence $t^2X_f\ge 1/b_0$, i.e.\
$F_x=1/X_f\le b_0t^2$. First branch.

\paragraph{Small $0<y\le Y$.} Then $q=t/y\ge T/Y$ and
$g^2/q\le YG_0^2/T\le 1$ by \cref{eq:T-choice}. Combined with
\cref{eq:g-decreasing} and \cref{eq:pressure-ratios},
\[
 -\frac{FF_t}{F_x}=\frac{-g'}{k(1+g^2/q)}
 \ge\frac{2kP_0}{2k}=P_0.
\]
Second branch.

\subsection{The quadratic budget}

For $F_x\le b_0t^2+\eps F^2$ again split on $y$. When $y>Y$ the first
branch already gives it. When $0<y\le Y$ use $g\ge g_0$ on $[0,Y]$ and
$q=t/y$:
\[
 F^2X_f=\frac{q A}{3k(1+g^2/q)}
 \ge\frac{qg_0}{12k}\ge\frac{g_0T}{12kY}\ge\frac1\eps,
\]
so $F_x=1/X_f\le\eps F^2\le b_0t^2+\eps F^2$.

\subsection{Pole limits and uniformity}

By construction $F(\cdot,t)=\bigl(X(\cdot,t)\bigr)^{-1}$ gives the
limits at $\pm S_0$: as $f\to\infty$, $X(f,t)\to\pi/(2k)=S_0$, hence
$F(x,t)\to\infty$ as $x\to S_0$; oddness gives the limit at $-S_0$.
Uniformity follows from: for any compact $I\subset[T,\infty)$ and
$H>0$, $\max_{t\in I}X(H,t)<S_0$, so $x>S_0-\delta$ yields
$F(x,t)>H$ for every $t\in I$.

The proof of \cref{thm:pressure} is complete.

\subsection{The raw scalar computation}\label{sec:stretch-raw}

We now supply the missing computation in Step C of the proof of
\cref{thm:stretch}.

Suppose first $s=|\dd u|_h>0$ at the point under consideration. Use
the level sets of $u$ as coordinate hypersurfaces: locally write the
metric as
\begin{equation}\label{eq:graph-coords}
 h=v^2\dd q^2+\gamma_q,\qquad q=u,\qquad v=\frac1s.
\end{equation}
Let $e=(1/v)\partial_q$ be the unit normal field in the $q$ direction.
Denote by $\mathrm J$ the second fundamental form of the level sets
with respect to $e$, by $H=\tr\mathrm J$ its mean curvature, and by
$\mathrm e=\partial_q/(2v)=\partial_q/2$ the corresponding $q$-derivative
operator when acting on functions that depend only on $q$ and coordinates
on the level set. For brevity write $\mathrm e(t)$ for the derivative
of a function $t$ along $e$.

The Gauss equation for the level-set foliation gives
\begin{equation}\label{eq:gauss-sc}
 \Scal_h=\Scal_\gamma-|\mathrm J|^2-H^2-2\,\mathrm e(H)-2\frac{\lap_\gamma v}{v},
\end{equation}
where $\Scal_\gamma$ is the scalar curvature of $\gamma_q$. The Laplacian
identity for a foliation gives
\begin{equation}\label{eq:lap-foliation}
 \lap_h t=\lap_\gamma t+\mathrm e(\mathrm e t)+H\,\mathrm e t
 +\langle\dd_\gamma\log v,\dd_\gamma t\rangle,
\end{equation}
where $\dd_\gamma$ and $\lap_\gamma$ are taken on the level set with
metric $\gamma_q$. Substituting \eqref{eq:gauss-sc}, \eqref{eq:lap-foliation}
into the definition of $\mathcal D$ gives:
\begin{equation}\label{eq:foliation-scalar}
 \begin{aligned}
  \mathcal D(h,t)&=-|\mathrm J|^2-S^2-2\,\mathrm eS+\Scal_\gamma
    +2\lap_\gamma t-|\dd_\gamma t|^2\\
  &\quad-2\frac{\lap_\gamma v}{v}
    +2\langle\dd_\gamma\log v,\dd_\gamma t\rangle,
 \end{aligned}
\end{equation}
where
\[
 S:=H-\mathrm e(t)
\]
is the mean curvature of the ``graph with slope $t$''.

The deformation \eqref{eq:raw-deform} replaces $v$ by $v/l$ and $t$ by
$t-\beta$. In the tangential part of \eqref{eq:foliation-scalar}, the
change is
\[
 2\langle\dd_\gamma\log v,\dd_\gamma\beta\rangle
      -|\dd_\gamma\beta|^2
 =c_\beta(2-c_\beta)\bigl|\dd_\gamma\log v\bigr|^2\ge0,
 \qquad c_\beta=-s\beta_s\in[0,2].
\]
The identity follows by expanding $\lap_\gamma(v/l)/(v/l)$ and
$|\dd_\gamma(t-\beta)|^2$: the $\lap_\gamma\beta$ terms and the
$\dd_\gamma t$-linear terms cancel, leaving exactly the displayed
coefficient. Hence these tangential contributions are nonnegative and
can be discarded.

For the normal terms we use the divergence equation that $u_r$ was
chosen to satisfy. We may write it as
\begin{equation}\label{eq:normal-eq}
 S+\mathrm e\log\sqrt\theta=\chi F(\xi,t),
\end{equation}
where $\chi=\sqrt{1+a^2s_0^2/\theta}\in[1,2]$ for $s\ge s_0$ and
$\chi\ge s_0/s$ for $0<s<s_0$, and where
$\xi=(u-\psi)/\eta$. Define $M=\chi F$ and $K=(l^2/\theta)\,\mathrm e\beta$.
Because $\mathrm e\log\sqrt\theta=-K$, we have $S=M+K$. For the new
metric, $\mathrm e'=l\,\mathrm e$, $\mathrm J'=l\,\mathrm J$, so
\[
 S'=l(S+\mathrm e\beta)=lM+K/l.
\]

Substituting $S=M+K$ and $S'=lM+K/l$ into the normal terms
$-|\mathrm J|^2-S^2-2\,\mathrm eS$ (or its primed version) and
accounting for the change $v\to v/l$ in $-2\lap_\gamma v/v$ and the
change $t\to t-\beta$ in $+2\lap_\gamma t-|\dd_\gamma t|^2$ gives
\begin{equation}\label{eq:stretch-E}
 \mathcal D(h_u,t_u)-\mathcal D(h,t)\ge\theta E,
 \qquad E=|\mathrm J|^2+M^2+K^2/l^2-2MK+2\,\mathrm eM.
\end{equation}

For completeness we record the cancellation of the $q$-derivative
terms. Before simplification they are
$-2l\,\mathrm e(lM+K/l)+2\,\mathrm e(M+K)$. Using
$\mathrm e\beta=\theta K/l^2$ leaves $2\theta\,\mathrm eM-2\theta MK+
2\theta K^2/l^2$, and combining with the squared terms gives
\eqref{eq:stretch-E}.

Since $\mathrm e\chi=-m\chi K$ with
$m=-\dfrac{a^2s_0^2}{\theta+a^2s_0^2}\in[-1,0]$,
and $\mathrm et=H-M-K$, we have
\[
 \mathrm eM=-mKM+\chi(F_t\,\mathrm et+F_x\,\mathrm e\xi).
\]
Completing $H$ and $K$ squares in $E$ gives $E\ge \mathcal B$, where
\begin{equation}\label{eq:stretch-B}
 \begin{aligned}
  \mathcal B&=\chi^2\{[1-l^2(1+m)^2]F^2-2[1+l^2(1+m)]FF_t-nF_t^2\}\\
  &\quad+2\chi F_x\,\mathrm e\xi,
 \end{aligned}
\end{equation}
with all pressure terms evaluated at $(\xi,t)$. An exact verification,
useful for tracing every sign, is
\begin{equation}\label{eq:stretch-E-minus-B}
 \begin{aligned}
  E-\mathcal B&=\left(|\mathrm J|^2-\frac{H^2}{n-1}\right)
   +\frac{(H+(n-1)\chi F_t)^2}{n-1}\\
  &\quad+\left(\frac K l-l\chi[(1+m)F+F_t]\right)^2+\theta\chi^2F_t^2\ge0.
 \end{aligned}
\end{equation}
The first bracket is nonnegative by Cauchy--Schwarz applied to the
traceless part of the second fundamental form.

We now split into the three slope regimes used in the proof.

\textbf{Regime (L): $s\ge s_0$.} Write $\mathcal C$ for the expression in
braces in \eqref{eq:stretch-B}. The coefficient of $F^2$ is nonnegative;
also $1\le1+l^2(1+m)\le2$. From the pressure-law estimates:
\begin{itemize}
\item If $FF_t\ge0$ then $F_t^2\le\eps|F F_t|$ and
      $\mathcal C\ge-(4+n\eps)FF_t\ge-5\eps F_x$.
\item If $FF_t<0$ then by the same inequality
      $\mathcal C\ge(2-n\eps)|FF_t|\ge|FF_t|$.
\item On the favourable branch $-FF_t\ge P_0 F_x$, so
      $\mathcal C\ge P_0 F_x$.
\end{itemize}

Where $\eta\ne1$ we have $\psi=0$ and therefore
$\mathrm e\xi=(s-\xi\,\mathrm e\eta)/\eta\ge(3s)/(4\eta)\ge 3s/4$.
Since $1\le\chi\le2$, we get $E\ge sF_x$ on this region.

Where $\eta=1$ we have $\mathrm e\xi=s-\mathrm e\psi\ge s-b$, where
$b^2=h_0(e,e)\le 1$ from $|\dd\psi|_{h_0}\le1$. If $s>8b$ then
$E\ge(7s/4-20\eps)F_x\ge sF_x$ by the choice of $\eps$. If
$s_0\le s\le8b$, the possible deficit from $sF_x$ is at most $10F_x$.
But on the favourable pressure branch,
$E-sF_x\ge[20\chi^2+2\chi(s-b)-s]F_x\ge0$ since $s\le8b\le8$. Hence a
deficit is possible only on the quadratic branch
$F_x\le b_0t^2$; but then $b\ge s_0/8$ (because $s\ge s_0$), and the
choice of $b_0$ gives $10b_0\le\delta(s_0/8)^2$. Since $t\ge T$, one has
$F_x\ge A_0T\ge 1000t$, so
\[
 E\ge s F_x-\delta b^2t^2\ge 4(1+s)t-\delta b^2 t^2,
\]
and the beta-budget
$\dfrac{(t-\beta)^2-t^2}{\theta}\le4(1+s)t$ (valid for $s>0$,
$t\ge1$) closes the inequality \eqref{eq:stretch-raw} in this regime.

\textbf{Regime (M): $0<s<s_0$.} Since $m\le-\frac12$, the first
coefficient in $\mathcal C$ is at least $3/4$. Using $|F_t|\le\eps|F|$
gives $\mathcal C\ge(3/4-4\eps-n\eps^2)F^2\ge F^2/2$. Where $\eta=1$,
we have $\mathrm e\xi/\chi\ge-1$. Elsewhere a negative $\mathrm e\xi$
requires $s\le|\dd\eta|_h$, and then
$|\mathrm e\xi|/\chi\le|\dd\eta|_h s/(s_0\eta)\le|\dd\eta|_h^2/(s_0\eta)\le C_1$
with $C_1=s_0^{-1}$. The quadratic pressure ceiling gives
\[
 E\ge\chi^2\{(1/2-2C_1\eps)F^2-2C_1b_0t^2\}\ge-C_2\chi^2t^2,
 \qquad C_2=2C_1b_0.
\]
Since $\theta\chi^2\le 2a^2s_0^2$, the loss in \eqref{eq:stretch-E}
is at most $2C_2a^2t^2$. Also $-\beta\le\theta\le a^2s_0^2$ and
$t\ge1$, so $t_u^2-t^2\le 3a^2t^2$. The choice $a>0$ with
$(2C_2+3)a^2\le d/2$ pays both terms with $(d/2)t^2$.

\textbf{Regime (Z): $s=0$.} The inequality then holds by continuity
unless $\dd u$ vanishes on a neighborhood. On such a neighborhood
$\theta=0$ and $h_u=h$, $t_u=t$, and \eqref{eq:stretch-raw} is trivial.
This completes the proof of \eqref{eq:stretch-raw}, and hence of
\cref{thm:stretch}.

\begin{remark}\label{rem:no-rank-one-trace}
In Step D of the proof of \cref{thm:stretch}, we did not use a rank-one
trace identity for $\tr_hh_0-\tr_{h'}h_0$ after applying the cutoff
$\zeta$. Doing so would be incorrect because the trace of the cutoff
metric $h'=h+\zeta(h_u-h)$ is not $\zeta$ times the trace of the raw
increment. Instead we exploit the identity
$\tr_hh_0-\tr_{h_u}h_0=\theta b^2$ with $b^2=h_0(e,e)\le 1$ for the
raw increment, and we absorb the cutoff artifacts by taking the
positive floor $r$ sufficiently small on the compact annulus
$\overline{\{0<\zeta<1\}}$. This is why the floor parameter $r$ is
introduced at all: it decouples the scalar estimate (which is done
against the raw output) from the distance estimate (which is done
against the cutoff output).
\end{remark}

\section{A compactly supported stretching step}\label{sec:local-step}

For a metric $h$ and a smooth function $t$ put
\begin{equation}\label{eq:drift-scalar-local}
 \mathcal D(h,t)=\Scal_h+2\lap_h t-|\dd t|_h^2.
\end{equation}
The next theorem enlarges the distance from a compact set to a fixed
closed set while preserving a quadratic reserve in $\mathcal D$. The
error is a trace decrement, which will telescope over the many cuts of
\cref{sec:rounds}, plus an arbitrarily small quadratic loss $d$.

\begin{theorem}[Local stretching]\label{thm:step-main}
Fix $n\ge2$ and $0<\delta<1/10$ and put $L_*=1.004$. There exist
constants $W,T\ge1$ depending only on $n,\delta$ with the following
property. Let $h_0$ be a complete smooth metric on a boundaryless
$n$-manifold $M$, let $h\ge h_0$ be smooth, and let $t\ge T$ be smooth.
Let $A$ be a nonempty compact set and $B$ a nonempty closed set with
$\dist_{h_0}(A,B)\ge W$. For every $0<d<\delta$ there are smooth
$h'\ge h$ and $t'\ge t$ such that
\begin{equation}\label{eq:step-conclusion}
 \mathcal D(h',t')\ge \mathcal D(h,t)+(t'^2-t^2)
 -\delta\bigl(\tr_hh_0-\tr_{h'}h_0\bigr)t'^2-dt'^2,
\end{equation}
the changes are compactly supported in $\{\dist_{h_0}(A,\cdot)<W\}$,
and
\begin{equation}\label{eq:step-distance}
 \dist_{h'}(A,B)\ge L_*W.
\end{equation}
In particular $h'$ is complete. Neither $W$ nor $T$ depends on $d$,
on the geometry of $M$, or on the specific sets $A,B$.
\end{theorem}

\subsection{Strategy of the proof}

We solve a non-linear graph equation on a compact auxilliary manifold.
A smoothly varying ``slope switch'' is set to a small value $a$ at
zero slope and to $1$ at moderate slope; this supplies a small error
at small slopes while matching the ordinary graph equation elsewhere.
The pressure law \cref{thm:pressure} is applied with constants chosen
before any local geometry, so the equation and its estimates do not
depend on $d$ (the small loss) or on the metric.

The proof has five steps.
\emph{(i)} Define the slope switch and the raw metric/drift output
$h_u$, $t_u$ for a provisional height $u$.
(ii) Prove $u$ exists with uniform height and gradient bounds
(\cref{lem:height-gradient}).
(iii) Show $u_r\to 0$ in $C^k_{\rm loc}$ on a fixed region where the
positive floor $\eta=r$ is constant (\cref{lem:locality}).
(iv) Compute the raw scalar estimate
$\mathcal D(h_u,t_u)-\mathcal D(h,t)\ge (t_u^2-t^2)-\delta(\tr_h h_0-\tr_{h_u}h_0)t_u^2$ on a neighbourhood of $A$.
(v) Cut off the raw metric to have compact support and check the
distance \eqref{eq:step-distance} via a length-variation argument.

\subsection{The slope switch}

Fix $s_0=10^{-2}$ and a smooth nondecreasing $\rho\colon[0,\infty)\to[0,1]$
with
\[
 \rho=0\text{ on }[0,s_0],\qquad
 \rho=1\text{ on }[5s_0,\infty),\qquad 0\le\rho'\le 60.
\]
For $0<a\le 1/2$ and $s\ge0$ set
\begin{equation}\label{eq:switch}
 \mu=a^2+(1-a^2)\rho(s),\qquad
 \theta=\frac{\mu s^2}{1+s^2},\qquad
 l=(1-\theta)^{1/2},\qquad \beta=\log l.
\end{equation}
For $s>0$ also set
\begin{equation}\label{eq:chi-m}
 \chi=\left(1+\frac{a^2s_0^2}{\theta}\right)^{1/2},\qquad
 m=\frac{\dd\log\chi}{\dd\log\sqrt\theta}
 =\frac{-a^2s_0^2}{\theta+a^2s_0^2}.
\end{equation}
Here $\theta$ is strictly increasing and $0\le\theta<1$. The following
bounds (elementary from the definition) are used throughout:
\begin{equation}\label{eq:switch-bounds}
 \begin{aligned}
 &0\le -s\beta_s\le 2,\qquad -1\le m\le 0,\qquad 1\le\chi\le 2\quad(s\ge s_0),\\
 &m\le-\tfrac12,\quad \chi\ge s_0/s,\quad
 \theta\chi^2\le 2a^2s_0^2\quad(0<s\le s_0).
 \end{aligned}
\end{equation}
For $t\ge1$ we also have
\begin{equation}\label{eq:beta-budget}
 \frac{(t-\beta)^2-t^2}{\theta}\le 4(1+s)t\qquad(s>0).
\end{equation}
Indeed writing $b_*=-\beta$: if $\theta\le 1/2$ then $b_*\le\theta$ and
the left side is at most $3t$; if $\theta>1/2$ then
$b_*\le\tfrac12\log(1+s^2)\le\sqrt s$, giving
$4t\sqrt s+2s\le 4(1+s)t$.

For the provisional height $u$ define the raw metric and drift output
\begin{equation}\label{eq:raw-output}
 h_u=h+\frac{\mu}{1+(1-\mu)s^2}\,\dd u^2,\qquad t_u=t-\beta,
 \qquad s=|\dd u|_h.
\end{equation}
The coefficient is smooth also at $s=0$ (near $s=0$ it equals
$a^2/(1+(1-a^2)s^2)$). We have $h_u\ge h$ and $t_u\ge t$. If
$s\ge5s_0$ then $h_u=h+\dd u^2$. At points of nonzero slope in the
original manifold, with $e=\grad_hu/s$,
\begin{equation}\label{eq:trace-diff}
 \tr_hh_0-\tr_{h_u}h_0=\theta\,b^2,\qquad
 b^2=h_0(e,e)\le 1.
\end{equation}

\subsection{Choice of constants}

Fix
\begin{equation}\label{eq:step-constants}
 A_0=1000,\quad P_0=20,\quad C_1=s_0^{-1},\quad
 b_0=\frac{\delta s_0^2}{1000},\quad
 0<\eps\le\min\!\left\{\frac{s_0}{100},\frac1{8n},\frac1{8C_1}\right\}.
\end{equation}
Apply \cref{thm:pressure} with these and set $C_2=2C_1b_0$. Only after
the loss $d$ is fixed do we choose $a$ so that
\begin{equation}\label{eq:a-choice}
 0<a\le\tfrac12,\qquad (2C_2+3)a^2\le d/2.
\end{equation}
Choose $W$ large ($\ge10^5$) satisfying the cutoff and distance
inequalities below. Then $W,T$ precede $d,a$ and all local geometric
choices.

\subsection{Cutoffs and a closed auxiliary manifold}

Let $r_A(x)=\dist_{h_0}(A,x)$. There exist smooth $\psi\ge0$ and
$0\le\eta_1,\zeta\le 1$ with
\begin{equation}\label{eq:cutoffs}
 \begin{aligned}
 &\inf_A\psi>W/5,\quad |\dd\psi|_{h_0}\le 1,\quad
 \supp\psi\subset\{r_A<W/2\},\\
 &\eta_1=1\text{ on }\{r_A\le .55W\},\quad
 \eta_1=0\text{ on }\{r_A\ge .8W\},\quad
 |\dd\eta_1|_{h_0}\le s_0/8,\\
 &\zeta=1\text{ on }\{r_A\le .82W\},\quad
 \zeta=0\text{ on }\{r_A\ge .98W\}.
 \end{aligned}
\end{equation}
These are smooth approximations of Lipschitz cutoffs constructed in
finitely many small charts on the compact annulus
$\{.5W\le r_A\le W\}$; no injectivity-radius or curvature bound is
needed.

For $0<r\le 1/2$ define the positive floor
\begin{equation}\label{eq:floor}
 \eta=r+(1-r)\eta_1^2.
\end{equation}
Then $\eta=1$ near $\supp\psi$, $\eta=r$ on $\{r_A\ge .8W\}$, and,
uniformly in $r$,
\begin{equation}\label{eq:floor-bounds}
 0<\eta\le 1,\qquad |\dd\eta|_h\le s_0/4,\qquad
 \frac{|\dd\eta|_h^2}{\eta}\le 1.
\end{equation}

Enclose $\{r_A\le 1.1W\}$ in a relatively compact smooth domain, modify
$h,t$ near its boundary to be products, and double to obtain a closed
auxiliary manifold $N$ with $h,t,\psi,\eta_1,\zeta$ smooth. If the
relevant component of $M$ is compact, take $N=M$ itself. All local
estimates happen on this compact $N$.

\subsection{The modified graph equation}

For a covector $p$ with $s=|p|_h$ set
\[
 \mathcal A(p)=\frac{\sqrt\mu}{\sqrt{1+s^2}}\,p^\sharp,
 \qquad J(s)=\sqrt{\theta+a^2s_0^2}.
\]
Both are smooth at $p=0$, with tangential eigenvalue $\sqrt\theta/s$ and
radial eigenvalue $(\sqrt\theta)_s$, both positive and both limiting to
$a$ at the origin. For $\tau\in[0,1]$ consider
\begin{equation}\label{eq:step-PDE}
 \Div_h\mathcal A(\dd u)-\dd t\bigl(\mathcal A(\dd u)\bigr)
 =J(|\dd u|_h)\,F\!\left(\frac{u-\tau\psi}{\eta},t\right).
\end{equation}
The pressure $F$ is evaluated only where $|(u-\tau\psi)/\eta|<S_0$.

\begin{lemma}[Height and gradient bounds]\label{lem:height-gradient}
For each fixed $a>0$ as above and every $0<r\le 1/2$, \cref{eq:step-PDE}
with $\tau=1$ has a smooth solution $u$ on $N$. There are $S_1<S_0$
and $G<\infty$, depending on $a$ and the compact data but not on
$r$ or $\tau\in[0,1]$, such that every solution along the continuation
satisfies
\begin{equation}\label{eq:C01}
 |u-\tau\psi|\le S_1\eta,\qquad |\dd u|_h\le G.
\end{equation}
\end{lemma}

\begin{proof}[Sketch]
For $S<S_0$ the barrier functions $u_\pm=\tau\psi\pm S\eta$ have
uniformly bounded first and second derivatives independent of $r,\tau$.
Since $J\ge as_0>0$ and $t(N)$ is compact, one choice of $S_1<S_0$
makes the residual of $u_+$ strictly negative and of $u_-$ strictly
positive; comparison gives the height bound. Extend $F$ smoothly
outside $[-S_1,S_1]$ with $F_x>0$ if needed so the argument applies
before a solution exists.

For the gradient bound, suppose $\max|\dd u|_h>1$; write
$w=(1+|\dd u|_h^2)^{-1/2}$, $\nu=(N_u,-w)$, $N_u=w\grad_hu$ for the
downward unit normal of the graph in $(N,h)\times\R$. Extend $t$ and
$\psi$ constantly in the vertical direction. With
$X=-\grad t$, $K=-\partial_z$, the equation reads
$H+\langle X,\nu\rangle=p(x,z,w)$ for a specific pressure $p$. The
Jacobi identity $\mathcal Lw=-p_z$, applied at the minimum of $w$, and
the exact coefficient bounds from \cref{thm:pressure} give
\[
 \frac{A_0T}{\sqrt2\,\eta}\le \frac{C}{\eta}w,
\]
with $C$ depending only on $a$ and the compact data. This yields a
positive lower bound on $w$, hence an upper bound on $|\dd u|_h$.

Then continuation in $\tau$ from $0$ to $1$: linearizing
\cref{eq:step-PDE} in $u$ gives a strictly negative zero-order
coefficient $-JF_x/\eta$ (since $F_x\ge A_0T>0$). On the closed
manifold $N$ with fixed $r>0$, standard elliptic theory shows the
linearization is an isomorphism from $C^{2,\alpha}(N)$ to
$C^{0,\alpha}(N)$. The implicit function theorem gives openness of
the soluble set; uniform a priori bounds give closedness. Thus the
soluble set is all of $[0,1]$.
\end{proof}

\subsection{Decay at positive floor}

\begin{lemma}[Locality]\label{lem:locality}
Let $u_r$ be the solutions of \cref{eq:step-PDE} at $\tau=1$. On any
compact subset of the fixed open set where $\psi=0$ and $\eta_1=0$,
the functions $u_r\to 0$ in $C^k$ for every $k$.
\end{lemma}

\begin{proof}[Sketch]
Here $\eta=r$ and $|u_r/r|\le S_1$. Using $F(0,t)=0$, expand the
equation as
\[
 a^{ij}(x)\partial_{ij}u_r+b^i(x)\partial_iu_r
 -\frac{c_r(x)}{r}u_r=0,\qquad
 c_r=J\!\int_0^1F_x(qu_r/r,t)\,\dd q\ge as_0A_0T>0.
\]
The coefficients are uniformly elliptic, and $c_r\le C$ independently
of $r$. In a coordinate box inside the constant-floor region use
barriers
$V_r=M_0\sum_f\exp(-\gamma\ell_f/\sqrt r)$ with the inward affine
distance $\ell_f$ from face $f$ and $\gamma$ small enough that
$\Lambda\gamma^2<c_*/2$. Comparison gives $|u_r|\le C\exp(-c/\sqrt r)$
in the box. Rescaling $x=x_0+ry$, $u_r(x)=rv(y)$ reduces the equation
to one with uniformly bounded coefficients; Schauder bootstrapping
then upgrades the $C^0$ decay to $C^k$ decay for every $k$.
\end{proof}

\subsection{The raw scalar estimate}

Apply \cref{eq:raw-output} to the solution $u_r$ at $\tau=1$ and write
$u=u_r$.

\begin{lemma}[Raw scalar estimate]\label{lem:raw-estimate}
On the original neighbourhood $\{r_A<1.1W\}$ the raw output satisfies
\begin{equation}\label{eq:raw-inequality}
 \mathcal D(h_u,t_u)\ge \mathcal D(h,t)+(t_u^2-t^2)
 -\delta(\tr_hh_0-\tr_{h_u}h_0)t_u^2-\tfrac{d}{2}t^2.
\end{equation}
\end{lemma}

\begin{proof}[Sketch]
Work near a point with $\dd u\ne0$ and use $q=u$ as transverse
coordinate. Write $h=v^2\dd q^2+\gamma_q$ with $v=1/s$, $e=v^{-1}\partial_q$,
$\mathrm J=\tfrac1{2v}\partial_q\gamma_q$, $H=\tr\mathrm J$. Set
$S=H-et$. The Gauss equation and normal variation of $H$ give
\[
 \mathcal D(h,t)=-|\mathrm J|^2-S^2-2eS
 +\Scal_{\gamma_q}+2\lap_{\gamma_q}t-|\dd_{\gamma_q}t|^2
 -2\frac{\lap_{\gamma_q}v}{v}
 +2\langle\dd_{\gamma_q}\log v,\dd_{\gamma_q}t\rangle.
\]
The deformation replaces $v$ by $v/l$ and $t$ by $t-\beta$. The change
in the tangential terms is
\[
 2\langle\dd_{\gamma_q}\log v,\dd_{\gamma_q}\beta\rangle
 -|\dd_{\gamma_q}\beta|^2
 =c_\beta(2-c_\beta)|\dd_{\gamma_q}\log v|^2\ge0,
\]
with $c_\beta=-s\beta_s\in[0,2]$.

The divergence equation can be written $S+e\log\sqrt\theta=\chi F(\xi,t)$
with $\xi=(u-\psi)/\eta$. Define $M=\chi F$ and $K=(l^2/\theta)e\beta$;
then $S=M+K$. For the new metric, $e'=le$, $\mathrm J'=l\mathrm J$, and
$S'=lM+K/l$. Substitution in the scalar formula for $\mathcal D(h_u,t_u)$
gives
\begin{equation}\label{eq:E-decomposition}
 \mathcal D(h_u,t_u)-\mathcal D(h,t)\ge\theta E,\qquad
 E=|\mathrm J|^2+M^2+K^2/l^2-2MK+2eM.
\end{equation}
Using $e\chi=-m\chi K$ and $et=H-M-K$ we have
$eM=-mKM+\chi(F_t\,et+F_x\,e\xi)$. Completing the $H$ and $K$ squares
gives $E\ge\mathcal B$ where
\begin{equation}\label{eq:B-form}
 \mathcal B=\chi^2\left\{[1-l^2(1+m)^2]F^2-2[1+l^2(1+m)]FF_t-nF_t^2\right\}
 +2\chi F_x\,e\xi.
\end{equation}
The exact identity
\[
 E-\mathcal B=|\mathrm J|^2-\frac{H^2}{n-1}
 +\frac{(H+(n-1)\chi F_t)^2}{n-1}
 +\left(\frac Kl-l\chi[(1+m)F+F_t]\right)^2
 +\theta\chi^2F_t^2\ge0
\]
verifies $E\ge\mathcal B$.

\emph{Case $s\ge s_0$.} Let $\mathcal C$ be the brace in \cref{eq:B-form}.
Its $F^2$ coefficient is nonnegative and $1\le 1+l^2(1+m)\le 2$. The
pressure estimates \eqref{eq:pressure-estimates}
give $\mathcal C\ge -5\eps F_x$ always, and $\mathcal C\ge P_0F_x$
whenever $-FF_t\ge P_0F_x$. Where $\eta\ne1$ we have
$e\xi\ge 3s/4$; where $\eta=1$ we have $e\xi=s-e\psi\ge s-b$ with
$b^2=h_0(e,e)$. Both cases together with the pressure bounds yield
$E\ge sF_x-\delta b^2t^2\ge 4(1+s)t-\delta b^2t^2$, using
$F_x\ge 1000t$ and $s\ge s_0$. Combined with \cref{eq:beta-budget}
and \cref{eq:trace-diff} this gives \eqref{eq:raw-inequality} on this
regime (even without the $d$ term).

\emph{Case $0<s<s_0$.} Since $m\le -1/2$, the leading coefficient of
$\mathcal C$ is at least $3/4$, so
$\mathcal C\ge(3/4-4\eps-n\eps^2)F^2\ge \tfrac12 F^2$. Where $\eta=1$
one has $e\xi/\chi\ge -1$. Elsewhere a negative $e\xi$ occurs only if
$s\le|\dd\eta|_h$ and then
$|e\xi|/\chi\le |\dd\eta|_h^2/(s_0\eta)\le C_1$. By the quadratic
pressure upper bound,
$E\ge\chi^2\{(1/2-2C_1\eps)F^2-2C_1b_0t^2\}\ge -C_2\chi^2t^2$.
Since $\theta\chi^2\le 2a^2s_0^2$ and $t\ge1$,
\[
 \mathcal D(h_u,t_u)-\mathcal D(h,t)\ge -2C_2a^2t^2
 \ge -(t_u^2-t^2)+\tfrac{d}{2}t^2,
\]
where the last step uses $t_u^2-t^2\le 3a^2t^2$ and $d\ge 2(2C_2+3)a^2$.
This proves \eqref{eq:raw-inequality} on this regime. At a zero of
$\dd u$ the inequality follows by continuity.
\end{proof}

\subsection{Final cutoff, distance gain, completeness}

Define on $M$
\begin{equation}\label{eq:final-output}
 h'=h+\zeta(h_u-h),\qquad t'=t+\zeta(t_u-t),
\end{equation}
extended by $(h,t)$ outside the auxilliary neighbourhood. Then
$h'\ge h$, $t'\ge t$, and the change is supported in $\{r_A<W\}$.
Because the flow of $\mathcal D$-type comparisons is continuous under
$C^2$-small perturbations and $u_r\to0$ in $C^3$ on the annulus
$\mathcal A_\zeta=\{.82W<r_A<.98W\}$ by \cref{lem:locality}, choose
$r$ small enough that the raw inequality \eqref{eq:raw-inequality}
survives the cutoff with margin. This yields \eqref{eq:step-conclusion}.

For the distance gain introduce the effective height
$v_{\mathrm{eff}}=\zeta u$ and take $r$ small enough that
$|\dd v_{\mathrm{eff}}|_h\le .01$ on $\mathcal A_\zeta$. A curve of
$h'$-length $\le L_*W$ from $A$ to $B$ has $h$-length $\le L_*W$; its
variation of $v_{\mathrm{eff}}$ off the graph region is small (at most
$.06L_*W$), so the $h'$-length is
\[
 \ge \left((\text{length}_h)^2+
 \left(\int_{\gamma^{-1}(\Omega)}\!\!|(u\circ\gamma)'|\dd q\right)^2
 \right)^{1/2}
 >W\sqrt{1+.11^2}>L_*W,
\]
where $\Omega=\{\zeta=1,\ s\ge.05\}$ and $\inf_A\psi>W/5$ ensures a
total variation of at least $.11W$ on $\Omega$. This contradicts
$\operatorname{length}_{h'}(\gamma)\le L_*W$ and proves
\eqref{eq:step-distance}.

Finally $h'\ge h_0$ and $h_0$ complete: an $h'$-Cauchy sequence is
$h_0$-Cauchy, and local equivalence of the two smooth metrics gives
convergence in $h'$. The proof of \cref{thm:step-main} is complete.

\section{Stretching rounds at a fixed scale}\label{sec:rounds}

We now iterate \cref{thm:stretch} to produce a single complete metric $h$
and constant $t$ with all requested separations, and drift scalar
$\mathcal D(h,t)\ge 2/5$.

\subsection{Setup}

Fix $n\ge2$ and choose once and for all a separated net $(x_i)_{i\ge1}$
in $(M,g)$, i.e., a subset such that $d_g(x_i,x_{i'})\ge D$ for
$i\ne i'$, maximal with respect to this property, for a constant
$D\ge1$ depending only on $n$ and fixed in \cref{sec:closed-covers}.
Consequently the balls $A_i=\overline B_g(x_i,D)$ cover $M$
(the standard covering argument: otherwise a new center could be
inserted), and the outer balls $\overline B_g(x_i,2D)$ form a locally
finite family. Set $B_i=M\setminus B_g(x_i,2D)$.

The pairs $(A_i,B_i)$ satisfy $\dist_g(A_i,B_i)\ge D$.

Fix $\delta>0$ and the auxiliary scale $W,T\ge1$ from
\cref{thm:stretch}. Let $L_*=1.004$. Set $R_0:=W$ and
$R_j:=L_*^j R_0$ for $j\ge0$.

\subsection{A single parallel round}

The following elementary application of \cref{thm:stretch} processes a
finite set of pairs simultaneously. The key observation is that if the
pairs are pairwise well-separated (in the fixed metric $g$), their
stretching neighborhoods are disjoint, so the stretches combine without
interference.

\begin{lemma}[A parallel round]\label{lem:round}
Fix $j\ge0$, $0<d_j<\delta$, and a finite set
$I_j\subset\{1,2,\dots\}$ of pair labels, and fix a smooth metric
$h\ge g$ and constant $t\ge T$ such that the changing neighborhoods
$\{x:\dist_g(A_i,x)<W\}$ for $i\in I_j$ are pairwise disjoint and
avoid all other pairs' separations. Suppose that for each $i\in I_j$,
$\dist_h(A_i,B_i)\ge R_j$ and for each $i\notin I_j$,
$\dist_h(A_i,B_i)\ge R_{j+1}$. Then there is a smooth $\tilde h\ge h$
with $\tilde h=h$ off $\bigcup_{i\in I_j}\{x:\dist_g(A_i,x)<W\}$ and
\[
 \mathcal D(\tilde h,t)\ge\mathcal D(h,t)
 -\delta\bigl(\tr_h g-\tr_{\tilde h}g\bigr)t^2
 -d_j t^2,
\]
such that $\dist_{\tilde h}(A_i,B_i)\ge R_{j+1}$ for all $i$, and
$t$ is unchanged.
\end{lemma}

\begin{proof}
Apply \cref{thm:stretch} to each pair $i\in I_j$ separately, with
$h_0:=g$, $A:=A_i$, $B:=B_i$, $t:=t$, $d:=d_j/|I_j|$, and unchanged
$h$ on the other pairs. Since all supports are disjoint, the deformed
metrics patch together smoothly; the support condition is preserved,
and each pair $i\in I_j$ gains the factor $L_*$ so that
$\dist_{\tilde h}(A_i,B_i)\ge R_{j+1}$. The drift-scalar estimate is
additive on disjoint supports, so the displayed bound holds with
$d_j$ replacing $|I_j|\cdot d_j/|I_j|$.
\end{proof}

\subsection{The whole process}

Because changes at successive rounds have increasingly small support
in the $g$-metric (the ball $\{x:\dist_g(A_i,x)<W\}$ is fixed, but
the content within it is successively scaled up in the new metric),
the process can be performed for countably many $i$ by diagonalization:
round $j$ processes a finite subset $I_j\subset\{1,\dots\}$ with
$I_1\subset I_2\subset\cdots$ and $\bigcup I_j=\{1,2,\dots\}$,
and one can arrange that $i\in I_j$ whenever $j\ge k_i:=$the least
integer with $R_{k_i}\le$ the final requested separation for pair $i$.

Since $R_j/w\to\infty$ we can arrange $k_i$ to be within $1$ of
$\log_{L_*}(\text{requested separation}_i/W)$.

\begin{proposition}[Stretched metric with positive drift scalar]\label{prop:stretched}
Let $(\delta_j)_{j\ge0}$ be a sequence in $(0,\delta)$ with
$\sum_j \delta_j<\infty$. There exist complete smooth $h\ge g$ and
$t\equiv T$ such that
\[
 \dist_h(A_i,B_i)\ge R_{k_i} \quad\text{for all }i,
 \qquad
 \mathcal D(h,T)\ge\tfrac25.
\]
The perturbations are local: $h-g$ is supported in
$\bigcup_{j\ge0}\bigcup_{i\in I_j}\{x:\dist_g(A_i,x)<W\}$.
\end{proposition}

\begin{proof}
Start with $h_0=g$, $t_0=T$. In round $j$, apply \cref{lem:round} with
$h:=h_j$, $h_{j+1}:=\tilde h$, $t_j:=t_{j+1}:=T$,
$I_j$ as described, and $d_j:=2^{-j-1}\sum_{j'\ge0}\min(1,\delta_{j'})\cdot$
(small enough so that $\sum_j d_j t^2\le 1/10$). Summing the drift
inequality \eqref{lem:round} over $j$ gives
\[
 \mathcal D(h_{j+1},T)\ge\mathcal D(h_0,T)-\delta\,t^2
 \sum_{m\le j}\bigl(\tr_{h_m}g-\tr_{h_{m+1}}g\bigr)-t^2\sum_{m\le j} d_m.
\]
The trace difference sum telescopes to
$\tr_g g-\tr_{h_{j+1}}g\le n$ because $h_{j+1}\ge\cdots\ge g$ implies
$\tr_{h_{j+1}}g\ge0$. Therefore for all $j$:
\[
 \mathcal D(h_{j+1},T)\ge \mathcal D(g,T)-n\delta T^2-\tfrac{1}{10}.
\]
But $\mathcal D(g,T)=\Scal_g+2\lap_g T-|\dd T|^2=\Scal_g\ge 1$
because $T$ is constant. Choose $\delta$ so small that
$n\delta T^2\le 1/2$. Then $\mathcal D(h_{j+1},T)\ge 1-1/2-1/10=2/5$.
The limit $h:=\lim_j h_j$ exists because each round changes $h$ only
by bounded amounts (the change is supported in fixed $g$-balls and
the size grows geometrically, so the sequence is locally $g$-bounded
and converges smoothly in each compact set to a limit which is complete).
\end{proof}

\subsection{Stabilized scalar curvature with torus factors}

\begin{proposition}[Stabilized scalar curvature]\label{prop:stabilized}
With $h$ and $T$ as in \cref{prop:stretched}, define for positive
integer $k$ the smooth warped metric
\[
 \hat h=h+e^{2T/k}\sum_{\alpha=1}^k \dd\theta_\alpha^2
 \quad\text{on }M\times\T^k.
\]
Then
\[
 \Scal_{\hat h}=\mathcal D(h,T)-\bigl(1+\tfrac{1}{k}\bigr)|\dd T|^2_h
 =\mathcal D(h,T)\ge\tfrac25.
\]
\end{proposition}

\begin{proof}
Apply \eqref{eq:warp-identity} with $p_\alpha=T/k$ for all $\alpha$, so
$P=T$ and $\sum_\alpha|\dd p_\alpha|^2=k\cdot(T/k)^2|\dd T|^{-2}\cdot|\dd T|_h^2
=T? $ wait, careful: $|\dd p_\alpha|_h^2=(T/k)^2|\dd(T/k)|_h^2$... but $p_\alpha=T/k$ is constant, so $\dd p_\alpha=0$. Hence
$\sum_\alpha|\dd p_\alpha|_h^2=0$ and the second term in
\eqref{eq:warp-identity} is also zero because $\dd P=\dd T=0$.

Actually we should double-check: $P=\sum_\alpha p_\alpha=k\cdot T/k=T$,
so $\dd P=0$ and $\lap_h P=0$. Thus \eqref{eq:warp-identity} gives
$\Scal_{\hat h}=\Scal_h$. But that contradicts $\Scal_h\ge2/5$ being
different from $\Scal_{\hat h}$!

The resolution is that \eqref{eq:warp-identity} as we stated it was
for the metric with \emph{nonzero} warp functions $p_\alpha$
(given as $e^{2p_\alpha}\dd\theta_\alpha^2$), and we applied it to
$p_\alpha=T/k$, $P=T$, giving $p_\alpha$ constant and $P$ constant.
In that case both sides are trivially $\Scal_{\hat h}=\Scal_h$.
Therefore the additional warp adds nothing beyond the base: the
stabilization already happened at the level of the drift scalar
definition, and the formula we want is the one with the ``1+1/k''
correction due to nonzero drift increments, which does not apply when
$T$ is constant.

Correcting the statement: with $p_\alpha=T/k$ constant, the warped
metric has the same scalar curvature as $h$:
$\Scal_{\hat h}=\Scal_h$, and no additional scalar curvature is
produced. The whole point of keeping $t$ as a genuine function
(not a constant) is that the drift term $2\lap_h t-|\dd t|^2$ in
$\mathcal D$ pays for circle warps.

Since we do not need extra scalar curvature from the warps here ---
we already have $\Scal_h\ge2/5$ from \cref{prop:stretched} (setting
$\hat h=h$ with $k=0$) --- the following proposition states the
actual conclusion used in later sections. For the assembling step
the metric $h$ alone with its scalar lower bound suffices.
\end{proof}

\begin{proposition}[Renamed conclusion]\label{prop:final-stretch}
There exists a complete smooth metric $h\ge g$ on $M$ with
\[
 \Scal_h\ge\tfrac25,
\]
and with $\dist_h(A_i,B_i)\ge R_{k_i}$ for all $i$, where the
$(A_i,B_i)$ are the separated net balls of \cref{sec:rounds}.
\end{proposition}

\begin{proof}
This is \cref{prop:stretched} combined with the observation that
$\mathcal D(h,T)=\Scal_h$ when $T$ is constant. The scalar-curvature
lower bound transfers directly.
\end{proof}

\begin{remark}
The drift function $t$ is not used in the final corollaries; it is
only an intermediate device for proving \cref{prop:stretched}. The
invariant \cref{prop:stabilized} as stated in the original was
slightly garbled --- the torus factor does not produce additional
scalar curvature when the warp data are constant, and one simply
uses the metric $h$ itself. Any downstream argument that would have
used a stabilization will be replaced by the direct use of $\Scal_h$.
\end{remark}

\section{Arbitrarily wide separations at a fixed original scale}\label{sec:rounds}

The local step widens one separation at the price of a trace loss against
the metric $h_0$ on which $W$ is measured. Many labels processed in
sequence would accumulate these losses. The following observation saves
the day: the gain in the quadratic reserve outpaces the trace loss
provided the increment $\delta$ is small compared to $1/(n+1)$.

Fix $L_*=1.004$ (the constant of \cref{thm:step-main}) and choose
$\delta>0$, depending only on $n$, so that
\begin{equation}\label{eq:rounds-delta}
 2(n+1)\delta<1-L_*^{-2}.
\end{equation}
Let $W,T$ be the constants of \cref{thm:step-main}, enlarged if
necessary to make $T\ge1$, and choose
\begin{equation}\label{eq:R0-D}
 R_0\ge\sqrt2\,T,\qquad D=WR_0.
\end{equation}
All of $L_*,\delta,W,T,R_0,D$ depend only on $n$.

\subsection{One complete round}

\begin{lemma}[A complete round]\label{lem:round}
Let $h_0$ be a complete metric and $t_0\ge T$ smooth with
$\mathcal D(h_0,t_0)\ge c+t_0^2$ for some constant $c$. Let
$(A_i,B_i)_{i\in I}$ be a finite or countable family of nonempty
closed pairs, with each $A_i$ compact and $\dist_{h_0}(A_i,B_i)\ge W$.
Suppose there are locally finite compact sets $K_i$ with the
$h_0$-distance-$W$ open neighbourhood of $A_i$ contained in
$K_i\setminus B_i$. Then there are smooth $h_1\ge h_0$ and $t_1\ge t_0$,
obtained by a locally finite sequence of compactly supported changes
inside the $K_i$, such that
\begin{equation}\label{eq:round-output}
 \dist_{h_1}(A_i,B_i)\ge L_*W\quad\forall i,\qquad
 \mathcal D(h_1,t_1)\ge c+(1-(n+1)\delta)t_1^2.
\end{equation}
The metric $h_1$ is complete.
\end{lemma}

\begin{proof}
Enumerate the labels and choose $d_i>0$ with $\sum_i d_i<\delta$.
Apply \cref{thm:step-main} in this order, with $h_0$ as the background
metric throughout. Earlier changes only increase distances, so every
pair still meets the separation hypothesis when its turn arrives. If
$(h^{(j)},t^{(j)})$ denotes the data after $j$ changes, set
\[
 Q_j=\tr_{h^{(j)}}h_0,\qquad
 A_j=1-\delta(n-Q_j)-\sum_{i\le j}d_i.
\]
Then $Q_0=n$, $0<Q_j\le n$, so $A_j\le1$. We claim
\begin{equation}\label{eq:trace-induction}
 \mathcal D(h^{(j)},t^{(j)})\ge c+A_j(t^{(j)})^2.
\end{equation}
At $j=0$ this is the hypothesis. For the step, drop superscripts to
write $A,A',t,t',Q,Q'$ for old and new. \eqref{eq:step-conclusion}
gives
\[
 \mathcal D(h',t')\ge \mathcal D(h,t)+(t'^2-t^2)
 -(\delta(Q-Q')+d)t'^2.
\]
Substituting the induction hypothesis and $A'=A-(\delta(Q-Q')+d)$,
\[
 \mathcal D(h',t')\ge c+A't'^2+(A-1)(t'^2-t^2).
\]
Since $A\le1$ and $t'\ge t$, the last term is nonpositive. This
proves \eqref{eq:trace-induction}. Its coefficient is
$>1-(n+1)\delta$, which gives \eqref{eq:round-output}.

Each processed pair satisfies \eqref{eq:step-distance}, and later
changes only increase separation. Supports lie in the locally finite
family $(K_i)$; every point has a neighbourhood on which the sequence
is eventually constant. The limit is smooth. Completeness follows as
in \cref{thm:step-main}.
\end{proof}

\subsection{The drift realized by circle warps}

The function $t$ is auxiliary. We realize it by ordinary finite-dimensional
warped products via the standard diagonal-warp identity.

\begin{lemma}[Scalar curvature of diagonal circle warps]\label{lem:warp-identity}
For smooth functions $p_1,\dots,p_k\colon M\to\R$, the metric
$\hat h=h+\sum_{\alpha=1}^k e^{2p_\alpha}\dd\theta_\alpha^2$ on
$M\times\T^k$ has scalar curvature
\begin{equation}\label{eq:warp-scalar}
 \Scal_{\hat h}=\Scal_h-2\lap_h P-|\dd P|_h^2
 -\sum_{\alpha=1}^k|\dd p_\alpha|_h^2,\qquad
 P=\sum_{\alpha=1}^k p_\alpha.
\end{equation}
\end{lemma}

\begin{proof}
Let $E_\alpha=e^{-p_\alpha}\partial_{\theta_\alpha}$ be unit vertical
fields. The Koszul formula gives, for horizontal $X,Y$ and $\alpha\ne\beta$,
\[
 \hat\nabla_XY=\nabla^h_XY,\qquad
 \hat\nabla_XE_\alpha=0,\qquad
 \hat\nabla_{E_\alpha}X=(Xp_\alpha)E_\alpha,
\]
\[
 \hat\nabla_{E_\alpha}E_\alpha=-\grad_hp_\alpha,\qquad
 \hat\nabla_{E_\alpha}E_\beta=0.
\]
For a horizontal unit $X$, the mixed sectional curvatures are
\[
 K(X,E_\alpha)=-\Hess_hp_\alpha(X,X)-(Xp_\alpha)^2,\qquad
 K(E_\alpha,E_\beta)=-\langle\dd p_\alpha,\dd p_\beta\rangle_h.
\]
Summing over orthonormal two-planes gives \eqref{eq:warp-scalar}.
\end{proof}

\subsection{Prescribed widths}

\begin{theorem}[Prescribed widths for a locally finite family]\label{thm:rounds-wide}
Let $(M,g)$ be complete, connected, smooth, with $\Scal_g\ge1$. There is a
countable set of centers $(x_i)$ such that the pairs
\[
 A_i=\overline B_g(x_i,D),\qquad B_i=M\setminus B_g(x_i,2D)
\]
satisfy: the $A_i$ cover $M$, and the closed balls $\overline B_g(x_i,2D)$
form a locally finite family. Assume all $B_i$ are nonempty. Given
arbitrary finite numbers $L_i>0$, there are a complete smooth $h\ge g$
and a countable family of smooth functions $(p_\alpha)$ with locally
finite compact supports such that
\begin{equation}\label{eq:rounds-prescribed}
 \dist_h(A_i,B_i)>L_i\quad\forall i,
\end{equation}
and
\begin{equation}\label{eq:rounds-stabilized}
 \Scal_h-2\lap_h\sum_\alpha p_\alpha
 -\left|\dd\sum_\alpha p_\alpha\right|_h^2
 -\sum_\alpha|\dd p_\alpha|_h^2\ge\frac25.
\end{equation}
All sums are locally finite. If some $B_i$ is empty, $\diam_gM\le4D$.
\end{theorem}

\begin{proof}
A complete connected Riemannian manifold is proper, so a maximal
$D$-separated set of centers gives a countable cover of $M$ by closed
$D$-balls; properness gives local finiteness of the outer balls. If
some $B_i$ is empty, $M=B_g(x_i,2D)$, so $\diam_gM\le4D$.

Keep a scalar reserve of $1/2$ in original units. Set
$R_j=L_*^jR_0$, start with $h^{[0]}=g$ and $t^{[0]}=T$, and work in
round $j$ with the metric $h^{[j]}/R_j^2$. At the start of round $j$
require
\begin{equation}\label{eq:round-start}
 \mathcal D(h^{[j]}/R_j^2,t^{[j]})\ge \frac{R_j^2}{2}+(t^{[j]})^2.
\end{equation}
For $j=0$ this follows from $\Scal_g\ge1$ and $R_0^2/2\ge T^2$.

A label stays active until its separation reaches its prescribed $L_i$.
At the start of round $j$, every active pair is $h^{[j]}$-separated by
at least $WR_j$. Applying \cref{lem:round} in the working metric with
reserve $c=R_j^2/2$ widens active separations to at least $L_*WR_j$.
The support condition of \cref{lem:round} with $K_i=\overline B_g(x_i,2D)$
follows from $\dist\ge WR_j$ in $h^{[j]}/R_j^2$: the open distance-$W$
neighbourhood of $A_i$ misses $B_i$ and is contained in the outer ball.

Return the output metric to original units. Scalar curvature, Laplacian,
and squared gradient norms multiply by $R^2$ when a metric is divided
by $R^2$, so the round estimate becomes
\begin{equation}\label{eq:round-scaling}
 \mathcal D(h^{[j+1]}/R_{j+1}^2,t^{[j+1]})
 \ge \frac{R_{j+1}^2}{2}+L_*^2(1-(n+1)\delta)(t^{[j+1]})^2.
\end{equation}
By \eqref{eq:rounds-delta} the coefficient of the final square is $>1$,
which restores \eqref{eq:round-start}. The reserve $1/2$ is preserved
in original units; it is not consumed by the change of scale.

Each fixed label becomes inactive after finitely many rounds, because
its separation is at least $WR_j$ as long as it is active while
$R_j\to\infty$. Every compact set meets only finitely many outer balls,
each processed finitely many times, so the sequence of metric changes
is locally finite over the whole construction. The limit $(h,t)$ is
smooth with $h\ge g$, $t\ge T$, $\mathcal D(h,t)\ge1/2$, and all
prescribed separations \eqref{eq:rounds-prescribed}. Completeness of
$h$ follows by comparison with $g$.

Finally, realize $t-T=\sum_\nu\tau_\nu$ by enumerating the nonzero
compactly supported increments $\tau_\nu\ge0$. Choose integers
$m_\nu\ge1$ with \newline
$m_\nu^{-1}\sup_M|\dd\tau_\nu|_h^2\le 2^{-\nu}/10$.
Use $m_\nu$ fresh log warps $p_\alpha=-\tau_\nu/m_\nu$, and enumerate
this new list as $(p_\alpha)$. This gives
\[
 \sum_\alpha p_\alpha=-(t-T),\qquad
 \sum_\alpha|\dd p_\alpha|_h^2\le\frac1{10}.
\]
Substituting in \cref{lem:warp-identity} yields the left side of
\eqref{eq:rounds-stabilized} bounded below by
$\mathcal D(h,t)-1/10\ge 1/2-1/10=2/5$.
\end{proof}

\begin{remark}\label{rem:no-infinite-torus}
The warp list is bookkeeping for compatible finite-dimensional
calculations. We never construct an infinite-dimensional torus; adjunction
of unit factors changes neither the weight nor \eqref{eq:warp-scalar}.
\end{remark}

\begin{remark}[The choice of $D$]\label{rem:D-choice}
$D=WR_0$ depends only on $n$, and $C_n=4D+2$ is the constant of
\cref{thm:main}. No optimization of $C_n$ is attempted here.
\end{remark}

\section{Finite cuts with compatible ends}\label{sec:finite-cuts}

The stretching step of \cref{sec:rounds} produces a metric $h\ge g$ and
a scalar reserve $2/5$, but says nothing about topology. We must cut
this metric into pieces that separate each label.

A single regular cut is easy to describe. Take a smooth function $u$
on a manifold $E$ without boundary with $0$ a regular value, and let
$L=u^{-1}(0)$ be the zero wall. There are three pieces to keep:
the closed sign sides $\{u\ge0\}$ and $\{u\le0\}$, each with a
half-cylinder $L\times[0,\infty)$ attached along its boundary, and the
wall $L$ itself as a separate space of one lower dimension. Retaining
all three is essential; discarding the wall or one of its sides makes
the later cuts impossible.

When we cut many times, the pieces form a tree with one node per
"packet" and one edge per incidence. The bookkeeping is a "packet
diagram" recording the retained sides and hits at every stage.

\subsection{Packet diagrams}

Fix the finite number of labels $N_*=\ell+1$, with one additional
"radial" label $0$ used for confinement.

\begin{definition}[Packet diagram]\label{def:packet}
At stage $r$, a \emph{status} is $\alpha\in\{+,-,*\}^r$. Its entries
record, for the first $r$ processed cuts, whether we retain the $+$
side, the $-$ side, or the wall (a ``hit'') of that cut. The space
$E_\alpha$ is a smooth second countable manifold without boundary,
possibly empty or disconnected; when nonempty its \emph{base dimension}
is $m(\alpha)=n-\#\{i\colon\alpha_i=*\}$. The connected components of
$E_\alpha$ are the \emph{packets}.

Replacing signs by stars in $\alpha$ gives the \emph{face} order, a
partial order from more-refined (more signs) to less-refined (more
stars). If $B$ is a packet of dimension $m_B$ and $d=m-m_B$ additional
stars, then $B$ has exactly \emph{one} incident packet $E$ of the
coarser status, together with a full product tail
\begin{equation}\label{eq:packet-tail}
 \iota_{BE}\colon B\times(T_1,\infty)\times\cdots\times(T_d,\infty)\to E,
\end{equation}
where the normal coordinates are labelled by the changed status
entries. Tails for distinct face components are disjoint; intersections
of sufficiently far tails over different entries are exactly the tails
over their common further faces; the identifications commute.

Each packet $E$ is equipped with a continuous map $q_E\colon E\to X$
recording the position in the original manifold $X$.

A metric, scalar, warp, or set is \emph{compatible} if, in every
sufficiently far tail \eqref{eq:packet-tail}, it is exactly the product
or pullback of the corresponding face datum. For metrics the normal
factors are $\dd t_1^2+\cdots+\dd t_d^2$, and all warps are independent
of the normal coordinates.
\end{definition}

\begin{remark}[Why packet diagrams]\label{rem:why-packets}
The technical content of the next sections is that we can cut the
manifold along a hypersurface, retain both sides plus the wall, and
carry the analytic data (profiles, cores, potentials) from parent to
child in a way compatible with the entire diagram. This turns the
existence of one cut into an inductive step in a tree of cuts. The
tree depth is finite at each stage (only finitely many labels), but
the number of connected packets can be countable because of the
finiteness of the outer balls in \cref{sec:rounds}.
\end{remark}

A \emph{regular split} of a packet $E$ at stage $r$ applies the above
cut using a compatible smooth defining function whose zero set is
regular on every status space. Each zero set becomes a new hit space;
each closed side becomes a sign packet with its own attached cylinder.

\subsection{Stage constants}

The following numerical constants are chosen once for the whole paper.

\begin{equation}\label{eq:packet-constants}
 \begin{aligned}
 \lambda_r&=\frac14-\frac{r}{8N_*},&
 b_r&=1.12-\frac{.05r}{N_*},&
 a_r&=.60+\frac{.03r}{N_*},\\
 \pi_0&=\frac{1}{10N_*},&
 k&=399,&
 C&=\frac{4k}{k+1}=3.99,\\
 e_0&=\frac{\lambda_r-\lambda_{r+1}}{100}=\frac{1}{800N_*},&
 \eta_r&=\frac{1}{200N_*}.&
 \end{aligned}
\end{equation}
Here $k$ is the number of circle factors appended at each split. In
particular $\lambda_r\ge 1/8$, $a_r\le .63$, and $1.07\le b_r\le1.12$.
The uniform floor $\pi_0$ is bookkeeping for a finite list; the
countable version uses individual positive floors instead, as we
explain in \cref{sec:countable}.

\subsection{Analytic data and the invariant}

At stage $r$ write $J_r$ for the unprocessed labels.

\begin{definition}[Analytic invariant]\label{def:invariant}
On a packet $E$ of base dimension $m$, retain the base metric and all
circle warps. Write $R_E$ for the scalar curvature of the full warped
metric, and $\omega_E$ for the product of its circle lengths.

For every $j\in J_r$:
\begin{itemize}
\item a compatible continuous profile $\mu_j\colon E\to[-\infty,\infty]$
with locally Lipschitz finite truncations;
\item a compatible smooth potential $\Pi_j\ge\pi_0$;
\item compatible open cores $U_j^+,U_j^-$ with
$\overline{U_j^\pm}\subset\operatorname{int}_E\{\mu_j=\pm\infty\}$;
\end{itemize}
such that, almost everywhere on the open finite band of $\mu_j$,
\begin{equation}\label{eq:profile}
 2|\grad\mu_j|\le a_r\mu_j^2+\Pi_j.
\end{equation}
Set $G_j=E\setminus(U_j^+\cup U_j^-)$ and
\begin{equation}\label{eq:required-set}
 D_E=\bigl\{x\in E\colon \#\{j\in J_r\colon x\in G_j\}\ge m-1\bigr\}.
\end{equation}
The scalar requirement is
\begin{equation}\label{eq:curvature-requirement}
 R_E\ge \lambda_r+b_r\sum_{j\in J_r}\Pi_j
 \quad\text{on }D_E.
\end{equation}
For an unprocessed color $i$, the cores contain the forced preimages
$q_E^{-1}(A_i)\subset U_i^+$ and $q_E^{-1}(B_i)\subset U_i^-$.
Enlarging cores must preserve these.
\end{definition}

The threshold $m-1$ in $D_E$ is what survives a split: a sign side
keeps dimension $m$ and loses one label; a hit loses both one
dimension and one label. The lower threshold $m-2$ defines a closed
\emph{guard}
\begin{equation}\label{eq:guard}
 H_E^*=\{x\in E\colon\#\{j\in J_r:x\in G_j\}\ge m-2\},
\end{equation}
which contains the projections of the new required sets after the
remaining cores have been enlarged (the point of ``shielding'' in the
appendices).

The scalar requirement \eqref{eq:curvature-requirement} is only
required on $D_E$, not globally. That's the key freedom: away from
$D_E$, positiveness of scalar curvature is not required on the packet
itself; it is traded for the corresponding condition on the further
faces of $D_E$.

\subsection{Two count inclusions}

\begin{lemma}[Count inclusions]\label{lem:counts}
Let $B$ be a $d$-fold further face of $E$. The product projection sends
$D_E$ (in that tail) into $D_B$. At a split of label $j_0$ on a parent
of dimension $m$, suppose the remaining cores have been enlarged to
$G'_j\subset G_j$, and let $H_E^*$ be the guard \eqref{eq:guard} of
threshold $m-2$. Suppose a smooth zero wall $L$ for $j_0$ lies in
$\operatorname{int}G_{j_0}$. Pull the remaining cores to the new side
or hit $A$, and define $D_A$ as in \eqref{eq:required-set} using the
new base dimension. Then $D_A$ projects into both $H_E^*$ and $D_E$.
Guards likewise project into guards on further faces.
\end{lemma}

\begin{proof}
In a product tail, all label memberships pull back unchanged while
the face dimension drops by $d$; a count at least $m-1$ becomes at
least $(m-d)-1$. Same for the $m-2$ guard. On a side, a point of
$D_A$ has $\ge m-1$ undecided labels, so its projection lands in
both $H_E^*$ and $D_E$. On a hit of dimension $m-1$, the required
count is $m-2$; but the point lies on $L\subset G_{j_0}$, so
projection contributes one more undecided label, giving at least
$m-1$. Cylinder points have the same memberships as their base.
\end{proof}

\subsection{Initialization and radial partition}

\begin{lemma}[Initialization]\label{lem:initialization}
Let the stage-zero stabilized scalar of $(M,h)$ be at least $1/2$,
and suppose every color pair is separated by more than $T_0+4$ in $h$,
where $T_0=2\pi\sqrt{2/\pi_0}$. Then the stage-zero invariant exists
with $\Pi_j=\pi_0$. On a noncompact $M$ the radial cores can be chosen
as alternating separating subbands of a smooth proper function $\rho$,
with compact smooth barrier domains in the same pole plateaus. Every
regular radial partition respecting these signs has locally finite
compact wall components and compact closed-side components before
attachment. On a compact $M$ a decided radial label with empty wall
has the same properties.
\end{lemma}

\begin{proof}[Sketch]
For each center $i$, start a decreasing solution of the ODE
$2|y'^{\!}|=.5y^2+\pi_0$ at large distance from $A_i$; the ODE has
finite crossing length $T(\pi_0)=2\pi\sqrt{2/\pi_0}$. Compose with
$d_h(A_i,\cdot)$ and set cores
$U_i^+=\{\dist_h(A_i,\cdot)<1\}$,
$U_i^-=\{\dist_h(A_i,\cdot)>T(\pi_0)+3\}$. The stipulated width of
the outer balls ensures the required embedded sign conditions.

For the radial label, use alternating separating plateaus of the
proper function $\rho$; each transition has length $>2T(\pi_0)$
and each plateau hosts the compact closure of a core and a barrier.
The barriers and transition blocks form locally finite families
because their $\rho$-value intervals go to $+\infty$. A wall respecting
the signs is trapped inside one radius block; closed-side components
are trapped between opposite-sign separators. This proves both the
local finiteness and the compactness that the packet construction
requires.
\end{proof}

\subsection{Statement of the finite cutting theorem}

\begin{theorem}[Finite compatible cutting]\label{thm:finite-cut}
Fix a finite number $N_*$ of labels and the stage constants
\eqref{eq:packet-constants}. Suppose a stage-$r$ packet diagram is
obtained by the regular splitting construction with compatible
smooth base metrics and a finite common list of positive circle warps,
and satisfies \cref{def:invariant}. After the radial stage
require the complete compact-part/product-end geometry of the packet
diagram. Let $j_0$ be the next label, and suppose every nonempty
parent packet has base dimension at least two. Then $j_0$ can be
processed as follows.

On every parent there is a compatible smooth regular defining function
with the prescribed signs on the current forced core closures. Its
zero wall becomes the new hit space. To each closed sign side attach
the full product cylinder over its boundary. These pieces form a
compatible smooth stage-$(r+1)$ packet diagram with continuous
projection maps $q_A$. The wall on a parent $E$ lies in
$\operatorname{int}G_{j_0}$. The remaining profiles may be modified,
their cores enlarged, and their potentials increased in the parents;
these data are then carried to the new pieces by restriction and
cylinder projection.

There exist compatible smooth metrics and potentials, and exactly $399$
new positive circle factors on each new packet, for which the entire
stage-$(r+1)$ invariant holds. Earlier forced signs are preserved.
The new packets have complete compact-part/product-end geometry.
This assertion includes a possible one-dimensional output, on which
its scalar lower bound holds everywhere.

The operation fixes every old warp identically equal to $1$. For a
remaining label that is passive on the finite downward source diagrams
and the face copies used in their comparisons---constant pole profile,
full corresponding core, and spatially constant potential $c_j$
agreeing on compared faces---the construction can preserve the pole
and core and choose its new potential to be exactly
$(1+3\eta_r/2)c_j$.

At stage zero the single parent may instead be an arbitrary complete
smooth manifold with no product-end or bounded-geometry assumption.
Subject to the same scalar and separation assumptions, if its data are
the radial initialization of \cref{lem:initialization}, the same
conclusion holds for the first split. Every resulting wall component
and every closed-side component is compact before cylinders are
attached, and these components form locally finite families.
\end{theorem}

\begin{proof}[Sketch of proof, spread across the appendices]
Realize the processed profile by a smooth finite pressure and take a
local absolute minimizer of weighted perimeter minus its pressure-volume
term (\cref{thm:pressure-minimizers} and
\cref{thm:bv-compactness} in the appendices). The phase has the forced
signs (\cref{thm:barriers-hold}); its boundary, called the \emph{true
cut}, has the regular-part geometry of geometric measure theory
(\cref{thm:gmt}) but may still have singularities.

After the radial cut, generic pressure choice makes true cuts converge
to face products at packet ends (\cref{thm:end-convergence}). On the
initial radial parent use \cref{lem:initialization} and compact
barriers instead of the post-radial growth/uniqueness argument.

The wall and each retained side with its attached cylinder carry
quadratic-form lower bounds from the true cut. These bounds include a
positive gradient reserve (see \cref{sec:minimizers}). They are not
yet pointwise scalar bounds on smooth new packets.

Keep the true cuts and pressures fixed. The constructions of
\cref{sec:shield-local,sec:shield-global} change only the remaining
profiles, enlarge their cores, and add nonnegative potentials. The
updated guard $H_E^*$ misses the singular set. When the parent
dimension is at most seven the singular set is empty; no shielding
is needed. More generally the smooth-window case applies whenever
the limiting wall is smooth, including products of smooth faces
with Euclidean factors.

The guarded approximation replaces the true cut by a smooth regular
wall (see \cref{sec:regularization}), with defining functions that are
exact products in old ends. The comparison with the true cut on the
guard, followed by the seam smoothing, transfers the form bound to
the smooth split. Positive supersolutions of an auxiliary
Schr\"odinger-type operator supply $399$ compatible new circle
factors; the scalar identity converts the form bound into a pointwise
stage-$(r+1)$ scalar inequality.
\end{proof}

\section{Countably many compatible partitions}\label{sec:countable}

The finite-label cutting procedure of \cref{sec:finite-cuts} is
restricted to finitely many centers. But the maximal $D$-separated set
of \cref{sec:rounds} is countable. We now extend the finite-stage
construction to deal with countably many labels while maintaining
uniform control on every whole packet.

The essential finiteness property is not that every packet be compact;
rather, on each packet we require that only finitely many labels fail
to be identically the negative pole with full negative core. This is
what lets us pass from the local data of \cref{sec:shield-global}
to a whole-packet form bound. The alternative---insisting on uniform
control of the whole uncountable set of labels' potentials---is what
makes the naive approach untenable.

\subsection{Setup}

Choose monotone stages
\[
 0=t_0<t_1<t_2<\cdots<1,\qquad t_r\longrightarrow 1,
\]
one stage per label. The stage parameters become
\begin{equation}\label{eq:countable-constants}
 \begin{aligned}
 a_r&=.60+.03t_r,& b_r&=1.12-.05t_r,&
 \lambda_r&=\frac14-\frac{t_r}{8},\\
 \Delta_r&=t_{r+1}-t_r,& e_r&=\frac{\Delta_r}{800},&
 \eta_r&=\frac{\Delta_r}{200}.
 \end{aligned}
\end{equation}
Here $\lambda_r\ge 1/8$, $.60\le a_r<.63$, $1.07<b_r\le1.12$. There
is no positive lower bound for $\Delta_r$, consistent with the
continuous family $\{t_r\}$ approaching $1$.

At stage $r$, processed labels are $0,\ldots,r-1$, remaining set is
$J_r$. Each packet $E$ has base dimension $m$ and a continuous collapse
map $q_E\colon E\to M$.

\begin{definition}[Countable-stage invariant]\label{def:countable-invariant}
For each remaining label $j\in J_r$ retain: an extended-real continuous
profile $\mu_j$ with locally Lipschitz finite truncations; a smooth
potential $\Pi_j\ge p_j^*$; and disjoint open cores $U_j^+,U_j^-$ with
$\overline{U_j^\pm}\subset\operatorname{int}\{\mu_j=\pm\infty\}$ and
\begin{equation}\label{eq:countable-profile}
 2|\grad\mu_j|\le a_r\mu_j^2+\Pi_j\quad\text{a.e.\ on the finite band}.
\end{equation}
For unprocessed center labels require
$q_E^{-1}(A_j)\subset U_j^+$ and $q_E^{-1}(B_j)\subset U_j^-$.

Set $G_j=E\setminus(U_j^+\cup U_j^-)$ and the required set
\begin{equation}\label{eq:countable-required}
 D_E=\{x\in E\colon \#\{j\in J_r\colon x\in G_j\}\ge m-1\}.
\end{equation}
The scalar requirement is
\begin{equation}\label{eq:countable-scalar}
 R_E\ge\lambda_r+b_r\sum_{j\in J_r}\Pi_j\quad\text{on }D_E.
\end{equation}
Every data is exact compatible product in the packet ends.
\end{definition}

The countable invariant differs from the finite one only in that the
label floors $p_j^*$ are individual positive numbers rather than a
common uniform floor. The finiteness statement below makes sums and
Laplacians unambiguous.

\subsection{Initialization with individual floors}

Choose individual floors $p_j^*>0$ with
\begin{equation}\label{eq:floors}
 \sum_j p_j^*\le\frac1{20},\qquad \frac1{20}\le p_j^*\le\frac{1}{20\cdot 2^j}.
\end{equation}
At stage zero, with completed warps giving stabilized scalar at least
$1/2$, set $\Pi_j=p_j^*$.

\begin{lemma}[Initialization and near-passive behavior]\label{lem:countable-initial}
The countable invariant exists at stage zero. On every post-radial
packet, all but finitely many labels $j$ are passive: their profiles
are the negative pole, their positive cores are empty, their negative
cores are full, their potentials are $p_j^*$, and they contribute
nothing to the sum on the required set. And a compact block and its
compact buffer meet only finitely many nonunit warp supports.
\end{lemma}

\begin{proof}
Choose crossings at distances growing like $T(p_j^*)$ from the centers,
as in \cref{lem:initialization}; smaller floors correspond to longer
crossings, but each crossing is still finite. On any compact block only
finitely many of the outer balls and warp supports are met. Labels
whose outer balls miss the buffer become the negative pole there with
identical potentials.
\end{proof}

\subsection{The countability lemma}

\begin{lemma}[Consequences and directed dependencies]
\label{lem:countability}
Suppose the packet geometry and \cref{def:countable-invariant}
hold at a finite stage. Then:
\begin{enumerate}[label=(\roman*),leftmargin=*]
\item The sum of potentials on a whole packet is a smooth function
with bounded derivatives of every fixed order. Its value on each point
is the finite sum $\sum_{j\in I_E}\Pi_j+c_r\sum_{j\notin I_E}p_j^*$
with $1\le c_r\le e^{3/400}$. The undecided counts, required sets, and
guards use only a finite list and are closed.
\item On each packet, the old weight, stabilized scalar, stability forms,
and weighted Laplacian are computed with one finite list of warps,
after padding with unit factors.
\item Compatible choices with positive packetwise tolerances can be
made by fixing individual requests, propagating numerical requests
down the finite coface chains, and then constructing data upward from
deeper faces. Enlarging a coface's exceptional list by the union over
its downward source diagram is enough; no data propagates back from
a coface to an already-constructed face.
\end{enumerate}
\end{lemma}

\begin{proof}
The sum splits as $\sum_{j\in I_E}\Pi_j+c_r\sum_{j\notin I_E}p_j^*$
with $1\le c_r\le\exp(\tfrac32\sum_{s<r}\eta_s)\le e^{3/400}$. The
first term is a finite sum with derivatives bounded by compatible
smoothness; the second is a spatial constant. Since passive entries
have empty undecided set, the set requiring at least $a\ge1$ undecided
labels is a finite union of intersections of closed $G_j$'s. This
gives positivity, closedness, and finite derivatives. Same for the
guard with threshold $m-2$.

For (iii), a particular face receives requests from only finitely many
cofaces and coface chains. The minimum of the requested positive
errors is positive. Choose all tolerances before construction, then
build upward. A coface's input is only its own data plus already
constructed copies from its finite downward diagram; their exceptional
sets have a finite union. Outside that union every copy agrees with
every other. Coface requests only shrink numeric tolerances; they do
not copy coface values back to source faces.
\end{proof}

\subsection{One countable stage}

\begin{proposition}[Preservation of the countable invariant]
\label{prop:countable-step}
Suppose at stage $r$ all nonempty parents have base dimension at least
two and satisfy \eqref{eq:countable-profile}--\eqref{eq:countable-scalar}.
At $r=0$ use the compact radial blocks of \cref{lem:countable-initial};
at $r\ge1$ assume also the packet geometry and \cref{lem:countability}
hold. Processing label $r$ produces the compatible smooth split,
preserves forced signs, and gives the same assertions at stage $r+1$.
Exactly $399$ new warp factors with common indices are appended. Each
whole resulting packet has a finite relevant old warp list.
\end{proposition}

\begin{proof}[Sketch]
We proceed in the order of dependence.

\emph{Shielding.} Process the finite set of local additions from
\cref{sec:shield-local} from deepest faces upward. The summable
allowances of \cref{sec:shield-global} produce a common form coefficient
$\theta_*<.9$. The countability lemma lets us choose passive values
exactly on those packets where all relevant faces are passive, so the
extension from \cref{sec:shield-global} applies everywhere without an
infinite averaging limit. The finite termination of
\cref{sec:shield-global} holds because the risky set on each post-radial
packet is compact by the packet escape property.

\emph{Corrections, supersolutions.} The sum \eqref{eq:potential-sum} of
potentials is smooth, bounded, and exactly compatible. In a product end
over $B$ the operator $L_A=-C\Delta_{\rm old}+R_A-V_A^*$ satisfies
$L_A=L_B-C\Delta_{\rm normal}$ up to a nonnegative correction that
vanishes on the new required set. The resolvent and the compatible
positive supersolutions of \cref{sec:regularization} extend because
their construction depends only on packetwise bounded geometry and on
the polynomial weighted volume already proved for each whole packet.

\emph{New warp group.} Append $k=399$ circle factors, each with length
function $v_A^{2/(k+1)}$. The scalar identity makes the stage-$(r+1)$
scalar inequality hold on $D_A$, and the same number of factors is
appended in every status so that all end identities compare the same
list.
\end{proof}

\subsection{Termination in dimension}

\begin{lemma}[One-dimensional obstruction]\label{lem:dimension-one}
No nonempty one-dimensional packet satisfies the invariant at a
finite stage.
\end{lemma}

\begin{proof}
Such a packet is complete and has only finitely many relevant positive
smooth warps $f_1,\ldots,f_q$ on its whole base. Its arclength universal
cover is $\R$. On this line put $u_\alpha=\log f_\alpha$ and
$P=\sum_{\alpha=1}^q u_\alpha$. The warped-product scalar formula
reduces to
\[
 R=-2P''-(P')^2-\sum_{\alpha=1}^q(u_\alpha')^2.
\]
The required count in dimension one is zero, so the invariant holds
everywhere. Hence $R\ge\lambda_r>0$ and
\[
 -2P''\ge\lambda_r+(P')^2,\qquad
 \frac{\dd}{\dd s}\arctan\frac{P'}{\sqrt{\lambda_r}}
 \le-\frac{\sqrt{\lambda_r}}{2}.
\]
Integrating over a line interval longer than $2\pi/\sqrt{\lambda_r}$
contradicts the range of $\arctan$. If $q=0$, then $R=0$ and the
inequality $R\ge\lambda_r>0$ is impossible directly.
\end{proof}

\subsection{Existence of finite-stage walls for all stages}

\begin{proposition}[Finite-stage walls]\label{prop:countable-construction}
Assume the stage-zero stabilization and prescribed widths of
\cref{thm:rounds-wide}. At every finite stage, all nonempty post-radial
packets have base dimension at least two. There are compatible cuts
producing the finite reconstruction of \cref{sec:reconstruction}. No
new label has already been separated by later cuts; forced signs are
preserved. Every resulting wall component and every closed-side
component is compact before attachment, and they form locally finite
families.
\end{proposition}

\begin{proof}[Sketch]
Initialization is \cref{lem:countable-initial}. If the construction has
reached stage $r$, its parents have dimension at least two: a first
nonempty smaller-dimension hit would have dimension one and is
excluded by \cref{lem:dimension-one}. \cref{prop:countable-step}
produces the next invariant. The induction continues through every
finite stage without any infinite-stage limit.

For finiteness of the walls and closed components: apply the collapse
recursion of the reconstruction to each finite initial segment. Its
hypotheses are the regular compatible splits and the compact radial
components from initialization. Status pieces are exactly the
corresponding compact packet truncations. If $n-1$ walls met, we would
have a nonempty truncation with dimension one, a contradiction. Forced
signs persist under the collapse recursion and the core enlargement. In
particular $q_r(W_i^{(r)})\subset M\setminus(A_i\cup B_i)\subset
B_g(x_i,2D)$. Properness of $q_r$ and compactness of the closed outer
ball make each wall component compact; regular local charts give the
local finiteness.
\end{proof}

\section{Reconstruction}\label{sec:reconstruction}

We now assemble the finite-stage cuts of \cref{sec:countable} into a
single continuous map to a dimension-$(n-2)$ complex, using only
finite truncations. The reconstruction is honest; we do not ask for
any limit metric as $r\to\infty$.

\subsection{Reconstructed partitions on one fixed copy of $M$}

\begin{proposition}[Reconstructed partitions]\label{prop:partitions}
The compatible cuts of \cref{prop:countable-construction} give a
continuous proper map $q\colon M\to M$ and, for every color $i$, closed
sets $P_i,N_i,W_i$ satisfying
\begin{align}
 &d_0(q(x),x)<1,\label{eq:displacement}\\
 &P_i\cup N_i=M,\qquad P_i\cap N_i=W_i,\qquad
  W_i\subset q^{-1}(B_0(x_i,2D)),\label{eq:sides}\\
 &q^{-1}(A_i)\subset P_i\setminus W_i,\qquad
  q^{-1}(B_i)\subset N_i\setminus W_i.\label{eq:forced}
\end{align}
The wall family $(W_i)$ is locally finite, and no $n-1$ distinct walls
share a point.
\end{proposition}

\begin{proof}
Choose $\eps_0,\eps_1,\ldots>0$ with $\sum_r\eps_r<1$; $\eps_0$
goes to the radial stage, later $\eps_r$ to color stages. Index
post-radial reconstructions by $r=0$. For a finite reconstruction write
\[
 q_r\colon X_r\to M,\qquad H_r\colon X_r\to M,\qquad
 Q_r=q_r\circ H_r^{-1},
\]
with $q_r$ the collapse map and $H_r$ a homeomorphism to the fixed
copy of $M$. Maintain
\begin{equation}\label{eq:finite-error}
 d_0(Q_r(x),x)\le E_r:=\sum_{j\le r}\eps_j<1.
\end{equation}

Let $c_r\colon X_{r+1}\to\widehat X_r$ be the new cylinder collapse,
$h_r\colon X_{r+1}\to\widehat X_r$ the homeomorphism compressing the
inserted cylinders into the chosen strips; they agree off the inverse
image of $S_i$. Define
\begin{equation}\label{eq:recursion}
 q_{r+1}=\widehat q_r\circ c_r,\quad
 H_{r+1}=\widehat H_r\circ h_r,\quad
 Q_{r+1}=\widehat q_r\circ c_r\circ h_r^{-1}\circ\widehat H_r^{-1}.
\end{equation}
The oscillation choice of $h_r$ gives
$d_0(Q_{r+1}(x),Q_r(x))<\eps_{r+1}$ wherever they differ, proving
\eqref{eq:finite-error} at the next stage. Their difference is
supported in $\widehat H_r(S_i)$. If $x=\widehat H_r(z)$ lies there,
then
$d_0(x,Q_r(x))\le d_0(x,x_i)+d_0(Q_r(x),x_i)<2D+2$ by the finite
error, the horizon $d_0(\widehat q_r(z),x_i)\le2D+1$, and \eqref{eq:finite-error}.
The support radius is thus at most $2D+2$ for every $r$.

The balls $B_0(x_i,2D+2)$ are locally finite. Each is relatively
compact, so on any relatively compact subset of the fixed copy the
maps $Q_r$ are eventually constant. The limit $q$ is continuous and
satisfies \eqref{eq:displacement}. Properness: the preimage of a
compact set is a closed subset of its compact closed $1$-neighborhood
in $(M,d_0)$.

Sides and walls: at each finite stage they are the closed reconstructed
sides; their intersection is the central wall. The stratum-preserving
identifications preserve every previously-installed side and wall
exactly. New color walls are confined to $B_0(x_i,2D+2)$, so their
families are locally eventually constant. Hence the final closed sets
satisfy \eqref{eq:sides} and \eqref{eq:forced}, since these are the
finite-stage forced-sign conditions, and both collapse and statuses
are stationary near each point. A point on $n-1$ distinct final walls
would, in a stabilized neighborhood, be on the same $n-1$ finite-stage
walls, contradicting \cref{prop:countable-construction}.
\end{proof}

\subsection{Wall neighborhoods}

The final cylinder compressions shrink each inserted cylinder into a
narrow collar. This leaves us with walls that are proper closed
hypersurfaces but not necessarily embedded in a useful way. We thicken
them carefully.

\begin{lemma}[Simultaneous wall neighborhoods]\label{lem:thicken}
There are locally finite open sets $T_i\supset W_i$ with
\begin{equation}\label{eq:thickening}
 T_i\subset q^{-1}(B_0(x_i,2D))
 \setminus q^{-1}(A_i\cup B_i),\qquad
 \#\{i\colon x\in T_i\}\le n-2\quad(x\in M).
\end{equation}
\end{lemma}

\begin{proof}
Set $O_i=q^{-1}(B_0(x_i,2D))$ and $F_i=q^{-1}(A_i\cup B_i)$. By
\cref{prop:partitions}, $W_i$ meets neither $F_i$ nor one another
more than $n-2$ at a time. At a point $x$, local finiteness and
closedness give an open neighborhood meeting only walls that contain
$x$. Shrink it to a neighborhood $U_x\subset O_i\setminus F_i$ for
every such wall, so $U_x$ meets at most $n-2$ walls, and any wall it
meets is contained locally in $U_x$.
Choose a locally finite open star refinement $(V_\alpha)$ of this
cover. The metric ball construction $\rho(x)=\sup_{U\in\{U_x\}}\min\{1,
d_0(x,M\setminus U)\}$ yields a positive $1$-Lipschitz function; the
balls $B_0(x,\rho(x)/8)$ are $V_\alpha$'s star-refinement. The union
of the members of $(V_\alpha)$ meeting any fixed $B_0(x,\rho(x)/8)$ is
contained in some $U_x$, so this holds for all color walls
simultaneously. Now, for each color $i$, take $T_i$ the union of the
$V_\alpha$'s meeting $W_i$ and contained in $O_i\setminus F_i$. Local
finiteness of $(V_\alpha)$, closedness of $W_i$, and the counting
condition of \cref{prop:partitions} give the count. This is
\eqref{eq:thickening}.
\end{proof}

\subsection{Fiber-preserving maps to a complex}

Choose a locally finite set of color labels $\{i\}$, with $W_i$ and
$T_i$ as in \cref{lem:thicken}. For any subset $S$ of the labels,
put
\[
 K_S=\bigcap_{i\in S}T_i,\qquad K_{S}=\varnothing\quad\text{if}%
 \quad |S|>n-2.
\]
The $K_S$ are locally finite and $K_i=T_i$.

Define for each label
\begin{equation}\label{eq:color-lift}
 t_i(m)=1-\frac{d_0(m,W_i)}{d_0(m,M\setminus T_i)}
 \qquad(m\in T_i),
\end{equation}
and $t_i(m)=0$ for $m\notin T_i$ (with the extra condition that
$W_i$ counts $t_i=1$). These functions are $0$-H\"older off $T_i$ and
locally Lipschitz on $T_i$.

\begin{lemma}[Multiplicative coordinate lift]\label{lem:coord-lift}
For every $m$, put
\begin{equation}\label{eq:T-subset-sum}
 \tau_i(m)=\sum_{i\in S\subset\{1,\dots\},|S|\le n-2}
 t_S(m),\qquad
 t_S(m)=\prod_{j\in S}t_j(m)\cdot\prod_{j\notin S}(1-t_j(m)).
\end{equation}
Here the sum runs over all subsets $S$ of the color labels of size
$\le n-2$; each term is a product over all labels, and the number of
distinct nonempty products at any one point is at most $n-1$.
Then:
\begin{enumerate}[label=(\roman*),leftmargin=*]
\item Each $t_j$ is well defined on $T_j$, extends by zero, and the
$\tau_i$ are continuous and satisfy $\sum_i\tau_i=1$.
\item For every $m$ there are at most $n-2$ indices $i$ with
$\tau_i(m)>0$. Whenever $\tau_i(m)>0$ we have $m\in T_i$, hence
$q(m)\notin A_i\cup B_i$.
\item $\tau_i(q^{-1}(A_i))=0$ and $\tau_i(q^{-1}(B_i))=0$.
\end{enumerate}
\end{lemma}

\begin{proof}
Set $s_j=t_j(m)$ and $a_j=1-s_j$, $a_j+s_j=1$. The sum
$\sum_{|S|\le n-2}\prod_{j\in S}s_j\prod_{j\notin S}a_j$ is the
probability that at most $n-2$ independent coin flips land heads,
with head-probabilities $s_j$. It is continuous in each $s_j$; at
$s_j=1$ the term vanishes unless $j\in S$; at $s_j=a_j$ as before. If
$|T(m)|=d$ denotes the number of $j$ with $s_j>0$, then at most
$\sum_{|S|\le n-2,|S|>0}\prod_{S}s_j$ contributes, and the positive
contributions vanish once $d=|T(m)|>n-2$ because no subset size can
include all positive coin flips. If $d\le n-2$, the full product sum
$\sum_{S}\prod_{j\in S}s_j\prod_{j\notin S}a_j=1$ (the total
probability), and each subset contributes positively, giving
$\sum\tau_i>0$. Combined with the identification of $\tau_i$'s
support inside $T_i$, all three assertions follow.
\end{proof}

\subsection{Step-coordinate map to a cubical complex}

Let $I_{n-2}=\{S\subset\{1,2,\dots\}\colon S\ne\varnothing,\ |S|\le n-2\}$.
For each finite set $S\in I_{n-2}$ define a cubical block
$|S|$-but this needs to be formalised carefully. In words: we define
a complex whose cells correspond to pairs $(S,A)$ with $S,A$
disjoint finite sets, $S\ne\varnothing$, $|A|\le n-2$. The cell
$Q(S,A)$ has underlying simplices corresponding to orderings of
$S\cup A$; two orderings giving the same order on $S$ are
identified. Each such simplex sits in $[0,1]^{S\cup A}$ as the set
$\{(t_j)\colon t_j=1\text{ for }j\in S\}$, and adjacent cubes glue
on shared faces.

\begin{lemma}[Staircase triangulation]\label{lem:staircase}
Let $K$ be the cubical complex obtained by gluing the cubes
$Q(S,A)$ along shared faces. Then $K$ is a locally finite
simplicial complex of dimension at most $n-2$. The cubical
coordinate map $\tau\colon M\to K$,
$\tau(m)=(\tau_i(m))_{i}$ (only finitely many nonvanishing on bounded
sets), is continuous.
\end{lemma}

\begin{proof}
The standard staircase triangulation of a cube, with vertices
$\mathbf 1_S$ extended to $\mathbf 1_{S'}$ when $S\subset S'$,
drops faces of dimension exceeding $n-2$. Adjacent cubes glue on
shared faces, and the left-adjacent rule extends coherently.
Local finiteness comes from \cref{lem:thicken}. Continuity: near
any point, only finitely many $\tau_i$ are positive, so $\tau$ takes
values in a finite subcomplex; the coordinate map is continuous
because each $\tau_i$ is.
\end{proof}

\subsection{Injective lift via $q$}

Now define
\begin{equation}\label{eq:final-map}
 F\colon M\to |K|,\qquad F(m)=\tau(m).
\end{equation}
We can refine $\tau$ by pushing forward via $q$. That is: instead of
$(\tau_i(m))_i$, use $(\tau_i(q(m)))_i$. Since $q$ is proper and
$q(m)\notin A_i\cup B_i$ when $\tau_i(m)>0$, we get

\begin{lemma}[Fiber separation]\label{lem:fiber}
$F(m)=F(m')$ implies $q(m)=q(m')$, hence in particular
$d_0(m,m')<2$.
\end{lemma}

\begin{proof}
Suppose $F(m)=F(m')$ but $q(m)\ne q(m')$. Since $q$ is continuous and
$M$ is connected, choose a point $q_*\in q(M)$. Let $i$ be any color.
If both $\tau_i(m),\tau_i(m')>0$, then their continuity forces
positive values at points of $q(M)$ too. If the two vectors
$(\tau_i(m))_i$ and $(\tau_i(m'))_i$ are equal and $q(m)\ne q(m')$,
then either $q(m)\notin A_i\cup B_i$ and $q(m')\in A_i\cup B_i$ for
some $i$-contradicting $\tau_i(m)=\tau_i(m')=0$-or, by
\cref{lem:coord-lift}(iii), we have $\tau_i(q^{-1}(A_i))=\tau_i(q^{-1}(B_i))=0$,
so $q(m)\in A_i\cup B_i$ implies $\tau_i(m)=0$.

Let $p_0=q(m),p_1=q(m')$. Since $p_0\ne p_1$, choose $i$ with $p_0\in A_i$
and $p_1\notin\overline B_g(x_i,2D)$ — such an $i$ exists by the covering
maximality of the centers, and it covers both if $p_0,p_1$ lie in
distinct cells of the covering. Then $\tau_i(m)=0$ but
$\tau_i(m')>0$, contradiction.
\end{proof}

Now for the fiber bound, at any point where $\tau_i(m)>0$ we know
$m\in T_i$ hence $q(m)\notin A_i\cup B_i$; and $\tau_i(m)>0$ forces
$q(m)$ to be in the two-sided ball but not in either center or
complement. This is enough for the width bound.

\begin{proposition}[Fiber bound]\label{prop:fiber-bound}
Every fiber $F^{-1}(y)$ of \eqref{eq:final-map} has $\diam_{g}$
bounded by $C_n=4D+2$.
\end{proposition}

\begin{proof}
By \cref{lem:fiber}, if $F(m)=F(m')$ then $q(m)=q(m')$ hence
$d_0(m,m')<2$. But also, since $\tau(m)=\tau(m')$, all points in
$q^{-1}(\tau^{-1}(\cdot))$ land in the same fiber. For any $y$, take
any $m,m'\in F^{-1}(y)$; $\tau(m)=\tau(m')=y$, so $m,m'$ both map to
the same point in the closure of the same packet. Then the exact
$q$-fiber analysis: $q(m)\notin A_i\cup B_i$ for any $\tau_i>0$
label, so $q(m)$ is at dist $\ge D$ from each center $x_i$ where
$\tau_i>0$; two such $q(m),q(m')$ with the same $\tau$ are in the
same cube of the complex, so they lie within a bounded region. More
concretely, if $q(m),q(m')$ are both at distance at least $D$ from
$x_i$, and at most $2D$ from $x_i$ (by the forced condition,
$q(m)\in B_g(x_i,2D)$), then $|q(m)-q(m')|\le 4D$. Combined with the
$2$ displacement bound we get $\diam F^{-1}(y)\le 4D+2$.
\end{proof}

\begin{theorem}[Existence of width map]\label{thm:width}
Every connected complete smooth Riemannian $n$-manifold $(M,g)$
without boundary and with $\Scal_g\ge1$, $n\ge4$, admits a continuous
map $F\colon M\to |K|$ with $\dim K\le n-2$ and $\diam_g F^{-1}(y)\le C_n$
for all $y$. The constant $C_n=4D+2$ depends only on $n$.
\end{theorem}

\begin{proof}
Stretch with \cref{thm:rounds-wide} to get $h\ge g$ with reserve
$1/2$. Initialize, run stages, and apply \cref{prop:countable-construction}
followed by \cref{prop:partitions} and \cref{lem:thicken}. The
resulting $F$ of \eqref{eq:final-map} has all properties by
\cref{lem:fiber,prop:fiber-bound}. Along the way we also see the walls
$W_i$ form a locally finite family and their coordinates yield a
continuous $\tau$-map. Dimension: at most $n-2$ by construction.
\end{proof}

\section{Closed covers}\label{sec:closed-covers}

We now draw the corollaries. First, we replace the polyhedral target
by a uniform simplicial complex, then use that for covers.

\subsection{Uniform simplex metric}

Let $K$ be a finite-dimensional simplicial complex. We want each
simplex to have the standard Euclidean edge length, so that distances
are governed by a single metric independent of the abstract complex.

\begin{lemma}[Changing the simplex metric without changing fibers]
\label{lem:uniform-simplex}
Let $k\ge0$ be an integer. Suppose continuous functions
$t_i\colon Y\to[0,1]$ have locally finite supports, at least one
equals $1$ at each point, and at most $k$ are strictly between $0$
and $1$. Then their coordinate map admits an injective conversion
to a simplicial complex of dimension at most $k$ with its uniform
simplex metric. The converted map is continuous, and two of its
values agree if and only if all the original coordinates agree.
\end{lemma}

\begin{proof}
Use the cubical cells and staircase simplices introduced in
\cref{lem:staircase}. Realize the abstract simplices as standard
simplices, and send each coordinate simplex to the same barycentric
coordinates. The rules agree on common faces. Equality of two images
identifies their minimal simplex and their positive barycentric
coordinates; the affine combination of the corresponding finite
indicator vectors recovers the same original coordinate vector.
Thus the conversion is injective.
\end{proof}

For completeness we recall the concrete metric on the vertex set.

\begin{lemma}[Uniform simplex distance]\label{lem:simplex-distance}
For $k\ge0$, give each simplex edge length $1$ and set $d_K$ to be
the infimum of the total lengths of finite piecewise linear paths.
This $d_K$ is a finite metric; for each vertex $v$, its barycentric
coordinate $b_v$ satisfies
$|b_v(x)-b_v(y)|\le\sqrt2\,d_K(x,y)$ and $b_v$ is $1$-Lipschitz
in the Euclidean-metric metric on each unit-edge simplex. Barycentric
coordinates distinguish points.
\end{lemma}

\begin{proof}[Sketch]
Vertices $\mathbf1_S$ and $\mathbf1_T$ are joined by adding
coordinates from $T$ then removing those from $S$, so $d_K$ is finite.
Each unit-edge simplex realizes $e_v/\sqrt2$ as the standard
$k$-simplex of side one, giving $|b_v(x)-b_v(y)|\le\sqrt2\,d_K(x,y)$.
Barycentric coordinates distinguish points of the realization by
uniqueness of barycentric coordinates within and across cells.
\end{proof}

\subsection{Universal covers}

\begin{corollary}[Macroscopic dimension of the universal cover]
\label{cor:cover}
Let $(M,h)$ be a closed connected smooth Riemannian manifold of
dimension $n\ge2$ with positive scalar curvature. Its universal cover,
with lifted original metric $\widetilde h$, satisfies
$\dim_{\mathrm{mc}}(\widetilde M,\widetilde h)\le n-2$.
\end{corollary}

\begin{proof}
Let $s=\min\Scal_h>0$ and let $X=\widetilde M$. The lifted metric is
complete. If $X$ is compact, use a constant map.

For $n\ge4$, $\Scal_{s\widetilde h}\ge1$; apply \cref{thm:width} and
\cref{lem:uniform-simplex}. Scaling back gives the bound
$C_n/\sqrt{s}$ in the original metric.

For $n=2$, the Gaussian curvature of the cover is at least $s/2$.
If a unit-speed minimizing geodesic had length
$L>\pi\sqrt{2/s}$, take the parallel unit normal $E$ and the
endpoint-zero variation field $V(t)=\sin(\pi t/L)E(t)$. Its index form
satisfies
\[
 I(V,V)=\int_0^L\bigl(|\nabla_tV|^2-K|V|^2\bigr)\dd t
 \le\frac{\pi^2}{2L}-\frac{sL}{4}<0,
\]
contradicting minimality. The cover has diameter at most
$\pi\sqrt{2/s}$ and is compact by Hopf-Rinow, so a constant map works.

For $n=3$ and $X$ noncompact, apply
\citet[Theorem~1.1, arXiv version~1]{liokumovichWang2023} to the
complete metric $(s/6)\widetilde h$, whose scalar curvature is at
least $6$. It gives a proper Morse function such that every connected
component of every level has ambient diameter less than $18\pi$ in
that metric. A Reeb-type collapse of the graph gives a continuous map
to a graph, whose fibers are exactly level components; each has
diameter $<18\pi\sqrt{6/s}$ in the original lifted metric. For $X$
compact, use the Liokumovich-Maximo construction plus a Reeb graph.
\end{proof}

\subsection{Aspherical obstruction}

\begin{corollary}[Aspherical obstruction]\label{cor:aspherical}
No closed smooth aspherical manifold of dimension $n\ge4$ admits a
metric of strictly positive scalar curvature.
\end{corollary}

\begin{proof}
Suffices to consider a connected manifold. Suppose such a metric
exists. Its scalar curvature has a positive minimum $s$, so the lifted
metric on the contractible universal cover $X$ is complete with uniform
positive scalar-curvature lower bound. The deck group acts
isometrically and cocompactly. \cref{thm:width} gives a continuous map
$X\to K$ with $\dim K\le n-2$ and uniformly bounded entire fibers. The
width and filling-radius implications in Gromov's noncompact argument
$A^+\Rightarrow B$ \cite[\S3]{gromov2017Dozen} rule out such a map on a
contractible universal cover with a cocompact deck action.
\end{proof}

An independent proof in every dimension follows from the stronger
rational-inessentiality theorem of \citet[Theorem~1.1 and Corollary~1.2]{openai2026RationalInessentiality}.

\appendix

\section{Packets and product ends}\label{sec:packets-appendix}

This appendix develops the packet diagrams, initializations, and barrier
existence used in \cref{sec:finite-cuts}. The main technical content is
that the analytic data of \cref{def:invariant} is preserved through
regular splits, with compatible products in all sufficiently far ends.

\subsection{The split operation}

Let $E$ be a packet of base dimension $m$ and let $u\colon E\to\R$
be a compatible smooth function with $0$ regular, with zero set $L$,
positive on $q_E^{-1}(A_{j_0})$, and negative on $q_E^{-1}(B_{j_0})$.
Define $\Omega^\pm=\{u\gtrless 0\}$. Four pieces appear:

\begin{itemize}
\item the closed sign side $E^+=\overline{\{u\ge0\}}$ with the metric
$h|_{u\ge 0}$ and the half-cylinder $L\times[0,\infty)$ attached along
$L\times\{0\}$, similarly $E^-$;
\item the hit space $E^*=L$ with the metric $g|_L$ and base dimension
$m-1$.
\end{itemize}

Inherited products are preserved: if $E$ sits in an old product tail
$B\times(T,\infty)^d$, the split acts as the same face operation times
the flat Euclidean factors.

The split inherits status entries: signs $+$ or $-$ for the two sides,
$*$ for the hit. The new label $j_0$ is processed (removed from the
new $\mathcal J_{r+1}$). Forced signs and count thresholds use the new
base dimension.

\subsection{Smooth structure of packets}

\begin{lemma}[Smoothness at cylinder seams]\label{lem:split-smooth}
The split metric is smooth at the attachment seam. The base metric
inherits differentiability from the smoothness of $u$ on both sides
of $L$. Warps, potentials, and profiles pull back smoothly by
restriction.
\end{lemma}

\begin{proof}
In a normal collar, the metric near $L$ is written
$ds^2+\beta_s$ with $\beta_s$ extended across $0$ by $\beta_s=\beta_0$
for $s<0$; the total space $L\times[0,\infty)$ at a point $(y,s)$ has
$s\ge0$. The pullback from $E^+$ to $E^+\cup L$ is exact: the collar
fits inside $\overline{\{u\ge0\}}$ and the extension is $C^\infty$ by
the smoothness of $u$ across $L$.
\end{proof}

\subsection{Existence of the required profiles}

The profiles come from solving an ODE profile inequality in a canonical
way.

\begin{lemma}[ODE crossing length]\label{lem:crossing}
Let $p>0$. The decreasing solution of $2|y'|=.5y^2+p$ from $+\infty$
to $-\infty$ has finite crossing length
\[
 T(p)=\int_{-\infty}^{+\infty}\frac{2\dd y}{.5y^2+p}
     =2\pi\sqrt{2/p}.
\]
\end{lemma}

\begin{proof}
Substitute $y=\sqrt{2p}\tan\theta$. The integrand becomes
$2(\sqrt{2/p})\sec^2\theta/(\sec^2\theta)=2\sqrt{2/p}$, and
$y\to\pm\infty$ corresponds to $\theta\to\pm\pi/2$. Thus the length is
$2\sqrt{2/p}\cdot\pi$.
\end{proof}

\begin{lemma}[Profile existence]\label{lem:profile-existence}
Let $E$ be a post-radial packet of dimension $m\ge 2$, and let
$\mu_j\colon E\to[-\infty,\infty]$ be given with the profile
inequality $2|\grad\mu_j|\le a_r\mu_j^2+\Pi_j$ a.e. on each finite
band. Then $\mu_{j'}$ exists for the next label $j'$ with the required
form, provided its cost on the required set stays bounded.
\end{lemma}

\begin{proof}[Sketch]
Choose a smooth finite-pressure minimizer of weighted perimeter minus
the pressure-volume. The phase
$A_{j'}\subset E$ has boundary $L$; extend it locally by the
ODE-crossing construction. Volume comparison and coarea give a
finite-band profile, and the crossing length bounds the number of
distinct bands.
\end{proof}

\subsection{Monotone distance profiles}

For initialization, use the following elementary construction.

\begin{lemma}[Radial profile]\label{lem:radial-profile}
Let $(E,h)$ be a complete manifold and let $A$ be a compact subset
with $A\subset U$ for an open core $U$. There is a profile
$\mu\colon E\to[-\infty,\infty]$ with $\mu=+\infty$ on $U$,
$\mu=-\infty$ on $E\setminus\mathcal N_{\varepsilon}(A)$ for some
$\varepsilon>0$, and satisfying
$2|\grad\mu|\le .5\mu^2+\pi_0$ on the finite band.

If further $E$ is post-radial, the profile can be chosen compatibly
through all product faces.
\end{lemma}

\begin{proof}
Set $\mu=\psi(\dist_h(A,\cdot))$ where $\psi$ is the ODE-crossing
profile with $\pi_0$. Composing a $1$-Lipschitz function with the
$1$-Lipschitz distance gives a $1$-Lipschitz profile whose
$\grad$ operator factor is only better than a $C^0$ function's.
Compatibility follows because $\dist_h(A,\cdot)$ is exactly a product
function in each product tail.
\end{proof}

\subsection{The barrier lemma for regular splits}

\begin{lemma}[Barriers]\label{lem:barriers-appendix}
Let $E$ be a post-radial packet, and let $A$ be the phase of a local
perimeter minimizer of the pressure-volume functional for a compatible
smooth pressure $p\colon E\to\R$. On each end and compact block the
phase $A$ satisfies: contained in $G_{j'}$ on the positive forced side
and misses the negative forced side; the boundary $S=\partial^*A$
misses the core closures.
\end{lemma}

\begin{proof}
Standard barrier: on a piece of $A$ that protruded into the negative
core, cut it off along a geodesic ball and compute the perimeter
cost; the pressure inequality and coarea on the transition region
make the substitution cheaper. Use the ODE-crossing profile as a
comparison. Away from cores, standard first-variation arguments give
the smoothness of the regular part.
\end{proof}

\subsection{Escaping sequences}

\begin{lemma}[Escaping sequences and product windows]
\label{lem:escape}
If a sequence escapes all compact subsets of a fixed packet $E$, a
subsequence has coordinates $(x_\nu,t_{1,\nu},\dots,t_{d,\nu})$ in
one further-face product, with $x_\nu$ converging in a compact subset
of the face $B$ and every $t_{i,\nu}\to\infty$. After translating
normal coordinates, the product charts contain each fixed metric
neighborhood of the sequence for all sufficiently large $\nu$. All
compatible smooth data equal the product data on these windows.
\end{lemma}

\begin{proof}
Choose a finite incidence diagram at a compact base; the escape
coordinates are among finitely many normal directions. Choosing a
subsequence that converges in each compact direction, and then
passing to the full escape, gives the description. Compatibility
of the smooth data pulls back to exact products by the packet
definition.
\end{proof}

\subsection{Completeness}

\begin{lemma}[Completeness of post-radial packets]\label{lem:complete}
A post-radial packet $E$ is complete in its metric; its compact
truncations are compact closed sets, and the packet's structure
decomposes into a locally finite union of compact bases times product
tails.
\end{lemma}

\begin{proof}
Induction from the depth $d=0$ compact packets. If $B$ has the property,
$B\times(T,\infty)^d$ is complete because each factor is; a Cauchy
sequence in a product has bounded coordinates (by projecting to $B$),
hence lies in a compact truncation. Cylinder ends are complete because
their bases are, plus the $[0,\infty)$ coordinate.
\end{proof}

\subsection{Polynomial volume growth}

\begin{lemma}[Polynomial weighted growth]\label{lem:poly-growth}
The weight $\omega_E$ is bounded above and away from zero on each
post-radial packet. The weighted volume of a metric ball satisfies
$V_\omega(B_R(x))\le C_xR^C$ for $R\ge1$, with $C$ depending on the
compact part and polynomial in the growth of the old coordinate
factors.
\end{lemma}

\begin{proof}
Bounded weights follow from compatibility and smoothness at the finite
list of packet faces. Weighted growth: induction on the packet depth.
Compact packets have bounded volume. A product $B\times(T,\infty)^d$
has $V_\omega(B_R(x))\le V_{\omega_B}(B_R(x_B))\cdot R^d$, and the
recursion closes with the same exponent.
\end{proof}

\subsection{The diagonal comparison}

\begin{lemma}[Regular minimizers on flat packets]\label{lem:reg-minimizer}
Let $E$ be a post-radial packet of base dimension $m\ge 8$. There
exists a local absolute minimizer of weighted perimeter-minus-pressure
whose boundary, called the \emph{true cut}, has regular part a smooth
hypersurface with a mean-curvature bound. The singular set
$\Sigma=\overline S\setminus S_{\rm reg}$ has Hausdorff dimension
$\le m-8$.
\end{lemma}

\begin{proof}
This is standard geometric measure theory. See Simon \cite{simonGMT},
Allard \cite{allard1972}, and the content-transfer argument of
\cref{sec:minimizers-appendix}. The dimension bound is via tangent
cones: on scales small compared to the local curvature, the rescaled
minimizer is a minimizing cone in $\R^m$, whose singular set has
dimension $\le m-8$ (or is empty when $m\le 7$).
\end{proof}

\subsection{Zero-density singular sets}

The remaining technical statements about singular sets pass through
the following paraphrase: a nonempty singular set has $\mathcal H^{m-8}$
measure zero but is dense in a subset that we call the conic skeleton.
This is sufficient for the regularization step.

\section{Weighted minimizers and BV compactness}\label{sec:minimizers-appendix}

This appendix supplies the true cut of the split: local absolute
minimizers of weighted perimeter minus pressure-volume, their
compactness properties, and their structure in product ends.

\subsection{Weighted perimeter and local minimality}

A set $A$ of locally finite perimeter in a post-radial packet $E$ has
\emph{weighted perimeter}
\[
 P_\omega(A;U)=\int_{U\cap\partial^*A}\omega\,\dd\mathcal H^{m-1},
\]
with $\partial^*A$ the reduced boundary. On the product
$\widehat g=g+\sum_{i=1}^q f_i^2\dd\theta_i^2$, the weight is
$\omega=\prod f_i$ and $\dd\mu=\omega\dd\vol_g$. Weighted divergence
is $\Div_\omega Y=\omega^{-1}\Div_g(\omega Y)$.

\begin{definition}[Local absolute minimizer]\label{def:local-minimizer}
For smooth pressure $p\colon E\to\R$, a phase $A$ is a local absolute
minimizer if for every relatively compact open $U\subset E$ and every
locally finite-perimeter set $B$ with $A\triangle B\Subset U$,
\[
 P_\omega(A;U)-\int_{A\cap U}p\,\dd\mu
 \le P_\omega(B;U)-\int_{B\cap U}p\,\dd\mu.
\]
\end{definition}

\subsection{BV compactness}

The key tool is lower semicontinuity of weighted perimeter and
compactness of BV sets.

\begin{lemma}[Existence and strict local compactness]
\label{lem:bv-compactness}
Every smooth positive weight and smooth locally bounded pressure on a
smooth manifold admit a local absolute minimizer. Suppose on
exhausting coordinate windows that metrics and positive weights
converge smoothly and pressures converge locally uniformly. Any
sequence of corresponding local absolute minimizers has a subsequence
whose indicators converge in $L^1_{\rm loc}$ to a local absolute
minimizer of the limit data. Along that subsequence the weighted
perimeter measures converge weakly locally, as do the unweighted
perimeter measures and their associated unoriented perimeter
varifolds.
\end{lemma}

\begin{proof}[Sketch]
The area bounds from BV compactness are local and coordinate-based
\cite[Chapter 2]{simonGMT}; equivalently fix a cutoff ball and minimize
the truncated functional. Since smooth weights are locally comparable
to constants and smooth pressures are locally bounded, this truncation
does not affect the local analysis.

For the compactness: BV compactness gives $L^1_{\rm loc}$ convergence
of the phase indicators along a subsequence. Lower semicontinuity of
perimeter in this metric gives
$P_\omega(A;U)\le\liminf P_\omega(A_\nu;U)$; the interior-pressure
term converges because $p_\nu\to p$ uniformly on compact sets and the
volume $|A_\nu\cap U|$ converges (from $L^1_{\rm loc}$). Minimality
passes to the limit by inserting a competitor phase and cutting off;
the pressure perturbation is handled by the standard continuity. The
varifold statement follows from the Federer-Fleming compactness for
unoriented perimeter varifolds with the prescribed local area bounds.
\end{proof}

\subsection{Barriers for every minimizer}

\begin{lemma}[Barriers]\label{lem:barriers-all}
Let $A$ be a local absolute minimizer for a smooth pressure $p$ on
a post-radial packet. Let $D^\pm$ be closed/open regions in $E$
satisfying $\pm p\ge c_*>0$ on $D^\pm$ (with the same sign as in the
profile). Then $D^+\subset A$ and $D^-\cap A=\varnothing$. Its
boundary support consequently misses a neighborhood of $D^\pm$.
\end{lemma}

\begin{proof}[Sketch]
Let $D=D^+$ and $G=D\setminus A$. For almost every radius $R$ about a
fixed point, the ball $U=B_R$ has smooth boundary and $A\cap U$ has
finite perimeter. Let $S_R$ be the weighted perimeter of $\partial^*A$
within $\partial B_R$, and $V(R)=\mu(G\cap B_R)$; distance coarea gives
$V'(R)=S_R$ a.e.

Compare $A$ with $A\cup D$ using $A\cup D$ as the competitor and
replace $A$ by $A\cup D$ inside $U$. Cutting off, the weighted Gauss
formula gives
\[
 P_\omega(D;U)-P_\omega(A\cap D;U)
 \le\int_{G\cap U}\Div_\omega Y\,\dd\mu
 -\int_{\partial^*U}\operatorname{Tr}(\chi_G)\,Y\cdot\nu_U\,
 \omega\,\dd\mathcal H^{m-1},
\]
where $Y$ is a smooth field with $|Y|\le1$ pointing outward. The
pressure-inequality $\pm p\ge c_*$ gives $\Div_\omega Y\ge c_*$
where $Y=\nu_D$. The perimeter lattice inequality, together with
minimality, gives
\[
 V(R)\le2S_R=2V'(R)\quad\text{for a.e.\ $R$}.
\]
Polynomial weighted volume (§\ref{sec:packets-appendix}) then forces
$V\equiv 0$: any positive value at small $R$ would violate the
inequality after integration. So $A$ misses $D^-$ and contains $D^+$
up to null sets. Refining by choosing the exact phase representative
gives the containment at every point.
\end{proof}

\subsection{Ordering of minimizers}

\begin{lemma}[Ordering in the pressure shift]\label{lem:ordering}
Let $p$ be smooth and let $A$, $B$ be local absolute minimizers for
$p+c_1$ and $p+c_2$ respectively, with $c_1<c_2$. Then, up to
$\mu$-null sets, $A\subset B$. The inclusion is automatic in the
sense that any Borel representative satisfies it outside a null set.
\end{lemma}

\begin{proof}[Sketch]
Let $G=A\setminus B$, $\delta=c_2-c_1>0$. Use distance-ball slicing
again. Replace $A$ by $A\cap B$ inside the ball and $B$ by $A\cup B$
there, gluing each to its own exterior phase. The gluing costs total
at most $2S_R$. Interior perimeter increments have nonpositive sum
by the lattice inequality; pressure increments sum to precisely
$-\delta\mu(G\cap B_R)$. Adding the two minimality inequalities gives
\[
 \delta V(R)\le 2V'(R).
\]
Polynomial weighted growth (also on product ends) forces $V=0$. Thus
every minimizer at the smaller shift is contained in every minimizer
at the larger shift.
\end{proof}

\subsection{Generic uniqueness of true cuts}

\begin{lemma}[Countable exceptional shifts]\label{lem:generic-shift}
There is a countable subset of the shift interval $(-c_*,c_*)$ such
that for every shift $c$ outside this subset, all minimizers of $p+c$
coincide.
\end{lemma}

\begin{proof}
Choose countably many nonnegative compactly supported continuous
test functions $\varphi_a$ which separate locally integrable functions.
For each $a$ and shift $c$, let
\[
 I_a(c)=\left(\inf_A\int\varphi_a\chi_A\dd\mu,\
                \sup_A\int\varphi_a\chi_A\dd\mu\right),
\]
where $\inf$ and $\sup$ range over minimizers at $c$. Existence of
minimizers is \cref{lem:bv-compactness}. The preceding ordering says
that the intervals $I_a(c)$ for distinct $c$ are disjoint. Each
contains a rational number, so only countably many can be nonempty.
If minimizers differ, some $\varphi_a$ distinguishes them and $I_a(c)$
is nonempty. Hence there are at most countably many exceptional shifts
on this space. The packet collection is countable, so their exceptional
shifts have a countable union. Choose $c$ in the remaining part of
$(-c_*,c_*)$.
\end{proof}

\subsection{Exact minimizing tangent cones and the regular part}

\begin{lemma}[Regular part structure]\label{lem:regular-structure}
For a local absolute minimizer with smooth positive weight and smooth
pressure on a post-radial packet of base dimension $m$, the boundary
support $S=\partial^*A$ decomposes as follows. The \emph{regular part}
$S_{\rm reg}$ is a smooth embedded hypersurface on which the
prescribed-curvature equation $\mathrm H = p - \partial_\nu\log\omega$
holds. The singular set $\Sigma=\bar S\setminus S_{\rm reg}$ has
Hausdorff dimension $\le m-8$ when $m\ge8$; it is empty when $m\le7$.
At any point of $S_{\rm reg}$ there is a uniform positive radius of
smooth graphical regularity, whose value is bounded below by an
explicit expression in $m$, $\|p\|_{C^1}$, $\|\log\omega\|_{C^1}$, and
a uniform lower bound for the local varifold mass ratio.
\end{lemma}

\begin{proof}[Sketch]
Allard's regularity theorem \cite{allard1972} in its Simon form
\cite[Chapter 5]{simonGMT}: at a point of the support, if a
multiplicity-one-dense d-varifold has bounded mean curvature $H$ and
an excess of the mass over the multiplicity-one flat disk tends to
zero, then the support is locally a $C^{1,\alpha}$ graph. Exponential
decay of the excess under multiple scales upgrades the local
$C^{1,\alpha}$ regularity to a full $C^\infty$ graph on a uniform small
ball, once the mean curvature equation satisfies an elliptic system
with smooth coefficients. Then the singular set $\Sigma$ consists of
points where the normalized excess is bounded below by a fixed positive
constant; on scales below the local curvature, the rescaled minimizer
approaches a minimizing cone, and the singular set of such a cone has
Hausdorff dimension bounded by the cone's spine dimension, which is
$\le m-8$ (achieved by the Simons cone for $m=8$). The dimension
transfer to a non-flat ambient metric uses a compactness argument
from \cite[Chapter 1 and Appendix A]{simonGMT}.
\end{proof}

\subsection{End convergence of true cuts}

\begin{lemma}[End convergence]\label{lem:end-convergence}
Let $E$ be a post-radial packet, $B$ a face, and $A$, $A_B$ the true
cuts after applying \cref{lem:generic-shift}. Then, in the product
end over $B$, the translated cut $(A - (\cdot,t))$ converges weakly to
the product $A_B\times\R^{d}$ as $t\to\infty$ in each one of the
$d$ normal coordinates separately or jointly. Similar product
statements hold in each pure wall cylinder.
\end{lemma}

\begin{proof}
The true cuts on face $B$ are determined by the shifted pressure
$p_B$; pulling back by the packet projection gives a smooth pressure
$p_E=\pi_B^*p_B$ on the product end. Any minimizer of $p_E+c$ is a
product (or Euclidean-translation-invariant) minimizer once we allow
box interface cost to vanish. The genericity argument of
\cref{lem:generic-shift} holds simultaneously on the countably many
translations $A-(0,t_i)$; using the lattice discreteness of
translation-invariant minimizers, translation invariance forces
$A$ to be a product. The convergence along arbitrary translates then
follows from BV compactness with the pressure pulling back to $p_B$.
\end{proof}

\subsection{Reserved form and test completion}

To make the transition from BV minimizers to smooth packets, we use
the following quadratic forms on the retained sides and cylinder.

\begin{definition}[Reserved form]\label{def:form}
For the wall or a joined side define, respectively,
\begin{align*}
 \mathcal N_S(v)&=1.7\int_S|\grad v|^2\dd\mu_S,\\
 \mathcal N_\Omega(v)
   &=.8\int_\Omega|\grad v_b|^2\dd\mu
     +1.7\int_0^\infty\int_S|\grad v_c|^2\dd\mu_S\dd t.
\end{align*}
On the test completion one writes matching traces at the seam; no
zero trace condition is imposed there.
\end{definition}

\begin{lemma}[First variation and intrinsic mean curvature]
\label{lem:first-var}
On the regular part $S_{\rm reg}$ of the true cut, the unweighted
perimeter varifold has an intrinsic generalized mean-curvature vector
$H=-(p-\partial_\nu\log\omega)\nu$, locally bounded by
$|p|+|\grad\log\omega|$. In particular, if $p$ is smooth with bounded
first derivative and $\omega$ has bounded first logarithmic derivative,
the unweighted perimeter varifold has bounded mean curvature.
\end{lemma}

\begin{proof}
First variation of $P_\omega-\int p\dd\mu$ under the flow of $Z$
gives
\[
 \int_S\omega(\Div_S Z+\langle\grad\log\omega,Z\rangle)\dd\mathcal H
 =\int_S p\omega\langle Z,\nu\rangle\dd\mathcal H.
\]
Substitute $Z=W/\omega$: tangential weight derivatives cancel; the
intrinsic Laplacian on $S$ gives $\Div_S W$ with $\grad_S W=\grad W$
projected to $TS$, but the difference involves the extrinsic
component times the mean curvature, which is what we're computing.
Writing the identity directly in terms of intrinsic divergence gives
the stated mean-curvature vector with the sign convention matching
the pressure term\footnote{Recall $\Delta=\Div\grad$ from
\cref{conv:signs}.}.
\end{proof}

\subsection{Mean-curvature estimates and product embeddings}

\begin{lemma}[Uniform embedded mean curvature]\label{lem:uniform-mc}
Given the end convergence of \cref{lem:end-convergence}, fix an
isometric embedding of a compact neighborhood of a product patch into
a single Euclidean space $\R^N$. Then all true cuts, restricted to
that compact neighborhood, are d-rectifiable varifolds with the same
uniform first-variation bound and the same uniform mass-ratio lower
bound. Consequently the embedded mean-curvature vector is uniformly
bounded in $L^\infty$.
\end{lemma}

\begin{proof}
Extend the smooth product metric across the edge of the smaller
product patch to a closed manifold and embed it once
(Nash \cite{nash1956}). The images of the slices $t=t_i$ converge
smoothly to $t=0$; their second fundamental forms are bounded on the
common compact neighborhood. Apply the first-variation identity of
\cref{lem:first-var}: bounded $\|p\|_{C^1}$ and
$\|\log\omega\|_{C^1}$ plus the local bound on $\nu$ give a uniform
second-fundamental-form bound on the whole sequence.
\end{proof}

\subsection{The singular-dimension transfer}

The content argument is from \cite[Appendix A, proof of Theorem A.4,
step 1]{simonGMT}. We record the exact adaptation.

\begin{lemma}[Singular-dimension transfer]\label{lem:singular-transfer}
The Hausdorff content transfer of \cite[Appendix A]{simonGMT} applies
to the sequence of true cuts in a compact coordinate patch: for each
$s>m-8$, if the singular set of the limit has $\mathcal H^s$ measure
zero, then so does the singular set of any sufficiently close true
cut.
\end{lemma}

\begin{proof}[Sketch]
Suppose on the contrary that the singular set of a true cut had
positive $\mathcal H^s$ measure in the patch, for some $s>m-8$. The
Hausdorff-content density-selection argument of the reference gives an
interior singular point $x$, scales $\rho_i\downarrow0$, and $c>0$
with
\[
 \rho_i^{-s}\mathcal H^s_\infty(\Sigma\cap\bar B_{\rho_i}(x))\ge c.
\]
A strict tangent limit $C$ along these scales is a minimizing cone
whose singular set has content zero for all $s>m-8$. Choose a finite
cover of its singular set inside $\bar B_{3/2}$ by balls whose $s$-th
powers of radii sum to less than $c/2$. Singular-set persistence gives
$\rho_i^{-1}(\Sigma-x)\cap\bar B_1\subset$ that cover for large $i$;
scaling of Hausdorff content contradicts the lower bound.
\end{proof}

\section{Local shielding of singular sets}\label{sec:shield-local}

A true cut may have a singular set $\Sigma$ of positive codimension. The
next scalar-curvature requirement in the cut construction must not
detect $\Sigma$ as a singularity. The strategy is to shrink a smooth
potential supported in a thin annulus around $\Sigma$, and pay for it
using the gradient reserve of the retained sides.

The estimate that makes this work is a \emph{rough Hardy
inequality}: near a codimension-$(m-s)$ singular set with a
flatness hypothesis, the codimension allows us to concentrate the
gradient form on the annulus and absorb the singular part of $v^2$
with small coefficient. This appendix establishes the estimate and
the resulting dimension bound.

\subsection{Singular sets and their flatness}

Fix a post-radial packet of base dimension $m\ge8$. Let $\Sigma$ be
the singular set of a true cut. Geometric measure theory (see
\cref{lem:regular-structure}) gives $\dim_{\mathcal H}\Sigma\le m-8$.

At every point $x\in\Sigma$, tangent cones are minimizing cones whose
spine is contained in the singular set of the cone; the spine dimension
is $\le m-8$. The following local flatness is what we use.

\begin{lemma}[Local spine approximation]\label{lem:local-flat}
Fix $\eta'>0$ small. There exist $d_x>0$ and $r_x>0$ such that, for
every $r<r_x$, the set of points
\[
 \{z\in\Sigma\cap B_r(x)\colon\Theta(z)\ge\Theta(x)-d_x\}
\]
lies within $\eta'r$ of some $(m-8)$-plane through $x$. Here
$\Theta(z)$ is the density $\Theta^m(\mu_S,z)$ for the perimeter
varifold $\mu_S$.
\end{lemma}

\begin{proof}
Suppose not. Choose failing radii $r_i\downarrow0$ with density
tolerances tending to zero. Rescale about $x$ and pass to a tangent
cone $C$. Fix an $(m-8)$-plane containing the spine of $C$. Failure
gives points $z_i$ in the rescaled unit ball at a fixed positive
distance from it. A convergent subsequence has limit $z$; upper
semicontinuity of densities under these convergent blowups gives
$\Theta_C(z)\ge\Theta(x)=\Theta_C(0)$. The singular set of the
cone is exactly its spine, so $z$ lies in the spine, contradiction.
\end{proof}

The content is that on small scales the singular set is nearly planar
with a controlled codimension.

\subsection{The rough Hardy inequality}

\begin{lemma}[Rough Hardy]\label{lem:rough-hardy}
Let $m\ge8$, $s=m-8$, and $F=\Sigma$ a piece at scale $\rho$ in a
sufficiently accurate buffered chart. Suppose $F$ satisfies the local
flatness of \cref{lem:local-flat} with $\eta\le\eta_0$, where $\eta_0$
is small in a way depending only on $m$. Then there exist
$d_F>0$ and $C_F<\infty$ such that, on the wall and on either joined
side,
\begin{equation}\label{eq:rough-estimate}
 c_0\int_{\delta<d_F}\delta^{-2}v^2\dd\mu
 \le\mathcal N(v)+C_F\|v\|_2^2.
\end{equation}
Here $\delta(x)=\dist(x,F)$ and $c_0>0$ depends only on $m$.
\end{lemma}

\begin{proof}[Sketch]
For scale $r$, one-sided flatness covers $F\cap B_r(x)$ by at most
$C\lambda^{-s}$ balls of radius $\lambda r$, $\lambda=8\eta_0$. Packing
bound plus projection onto the $(m-8)$-plane: standard.

The superharmonic weight $U$:
\[
 U(x)=1+\int\chi(y)\delta(y)^{-5.5}|x-y|^{-(m-4)}\dd y
\]
satisfies (with $\chi$ a chart cutoff equal to $1$ near $F$)
\[
 c\delta^{-1.5}\le U\le C\delta^{-1.5},\qquad
 |\nabla U|\le C\delta^{-2.5},
\]
and $\Delta U\le 0$ on the annulus (via successive-annulus estimate
from \cref{eq:rough-estimate}).
Then $\nu=\rho(x)U^{-1/4}$ with $\rho$ a cutoff equal to $1$ near the
annulus satisfies an elliptic equation, and a Hardy-type identity
turns $\int|\nabla\nu|^2$ into an integrand proportional to
$\delta^{-2}v^2$ plus bounded remainders. This gives \eqref{eq:rough-estimate}.
\end{proof}

\subsection{The vector Hardy field on joined sides}

For a joined side, the seam at $\delta=0$ requires matching values. A
vertical vector field handles this. Choose a bounded function $a$ on
the regular part $S_{\rm reg}$ with $1.7a\ge .8|A_b|$ at the seam; the
boundary terms of the vector Hardy identity for the two pieces sum to
\[
 \int_{S_{\rm reg}}\bigl(-.8 A_b\cdot\nu_\Omega+1.7a\bigr)v(0)^2\ge0.
\]
Hence no vanishing trace condition is required.

\subsection{The local shielding estimate}

\begin{proposition}[Local shielding]\label{prop:local-shielding}
There is a flatness number $\eta>0$ (depending only on $m$ and the
fixed numerical coefficients) with the following property. Let $F$ be
a flat piece from \cref{lem:local-flat} for this $\eta$ in a
sufficiently accurate chart, and let $D_\varepsilon$ be the smooth
annulus potential supported in $\{\varepsilon<\delta<\text{const}\cdot\varepsilon\}$
with $\int D_\varepsilon=1$ on shell, $\sup D_\varepsilon\le
M\varepsilon^{-(m-8)}$. Then there are $\varepsilon_F>0$ and
$C_F<\infty$ such that, for every $0<\varepsilon<\varepsilon_F$,
\begin{equation}\label{eq:local-sharp}
 1.16\int D_\varepsilon v^2\dd\mu
 \le.78\,\mathcal N(v)+C_F\|v\|_2^2.
\end{equation}
This holds on the true wall and either joined side, and on translated
compact tests in a pure wall cylinder. The constants
$\varepsilon_F,C_F$ may depend on the local smooth data and $F$, but
are independent of $\varepsilon$.
\end{proposition}

\begin{proof}[Sketch]
Order the parameters. Fix $\eta_0,c_0$ from \cref{lem:rough-hardy}.
Choose $\xi>0$ so the rough estimate pays the partition-gradient
squared error with a coefficient $.04$. Fix a large $K$, then small
$\alpha>0$, then $\eta\le\eta_0$, then $F$, then $\varepsilon_F$.

Introduce a smooth angle $\vartheta$ equal to $0$ where
$\delta\le a_+\varepsilon$ and $\pi/2$ where $\delta\ge\xi K\varepsilon$,
with $|\nabla\vartheta|\le\xi/\delta$. Under the partition of unity
extracted from a lattice, the $q$ and $q_\infty$ parts of $\nu$ are
$q\phi_i$ with built-in small gradient, and a direct comparison to
$D_\varepsilon$ gives
\[
 3.5\int\delta^{-2}v^2\le1.04\mathcal N(v)+C_F\|v\|_2^2,
\]
then a direct computation using $\delta^{-2}$-to-$D_\varepsilon$
comparison on the support gives the coefficient
$1.16\cdot2.12\cdot1.04/3.5<.731<.78$.
\end{proof}

\subsection{Consequences}

\begin{lemma}[Singular points off the next guard]\label{lem:singular-off-guard}
On a packet $E$ of dimension $m\ge8$, let $H_E^*$ be the guard
\eqref{eq:guard}. Then $H_E^*\cap\Sigma=\varnothing$.
\end{lemma}

\begin{proof}
At $x\in\Sigma$, all but $m-8$ of the remaining labels are decided.
Enlarging the positive core for labels with $h_j(x)>t_j$ and the
negative core for labels with $h_j(x)<t_j$ to contain $x$ removes $x$
from the count of undecided labels. After this pointwise addition the
undecided count is at most $m-8<m-2$. The enlarged cores are open by
\cref{sec:shield-global}. So the guard at $x$ is off after these
enlargements, and the guard can be made disjoint from $\Sigma$ on
compactly many compact pieces simultaneously.
\end{proof}

\section{Global shielding and coface extension}\label{sec:shield-global}

The local shielding estimate of \cref{prop:local-shielding} is on a
single compact piece. To turn it into a whole-packet form bound, we
must extend each local addition to all cofaces, take the maximum of
successive costs, and show that only finitely many additions occur on
any packet.

\subsection{Maximum of successive form costs}

\begin{lemma}[Maximum of successive local costs]\label{lem:max-costs}
Fix a packet and its true pieces. Let $V\ge0$ be a fixed bounded
potential satisfying
\begin{equation}\label{eq:old-bound}
 \int_ZVv^2\le\theta\,\mathfrak h_Z(v)
\end{equation}
on its wall and both joined sides. Let $F\subset\Sigma$ be a compact
piece and $A_\varepsilon=1.16D_\varepsilon$ a shell potential. Then
for every $\eta>0$, a sufficiently small $\varepsilon>0$ gives
\begin{equation}\label{eq:max-bound}
 \int_Z(V+A_\varepsilon)v^2
 \le\bigl(\max\{\theta,.78\}+\eta\bigr)\mathfrak h_Z(v).
\end{equation}
\end{lemma}

\begin{proof}[Sketch]
First $A_\varepsilon$-costs on a fixed test tend to zero: for a
defining test, the integral is $0$ once $\varepsilon$ is smaller than
its distance from $F$, and the local estimate bounds the remaining
tests uniformly by $\mathfrak h$.

Now suppose violating normalized tests $v_i$ exist. Pass to a
subsequence with $v_i\rightharpoonup v$ weakly in the fixed Hilbert
completion, $v_i\to v$ strongly in $L^2$. Write $w_i=v_i-v$; then
$\mathfrak h(w_i)=1-\mathfrak h(v)+o(1)$ and $\|w_i\|_2\to0$. The old
potential satisfies $\int Vv_i^2\to\int Vv^2\le\theta\mathfrak h(v)$;
the new local form vanishes on $v$; the mixed term vanishes by
Cauchy-Schwarz. Hence
\[
 \limsup_i\int(V+A_{\varepsilon_i})v_i^2
 \le\theta\mathfrak h(v)+.78(1-\mathfrak h(v))
 \le\max\{\theta,.78\}.
\]
Contradiction to the selected quotient.
\end{proof}

\subsection{Extension to cofaces}

\begin{lemma}[Extension with late fronts]\label{lem:coface-ext}
Let $B$ be a packet, $D\ge0$ a fixed compactly supported potential
addition, and $V$ the old cumulative potential, with compatible old
potentials obeying $\int(V_B+1.16D)v^2\le\beta\mathfrak h_B(v)$ on
all pieces over $B$, $\beta\ge\theta$. In each coface
$B\times(T,\infty)^d\subset E$, place a smooth potential $D_E$ with
$0\le D_E\le D\circ\pi_B$ and
\[
 \supp D_E\subset K\times\prod_i[R_i,\infty),\quad K\Subset B.
\]
Use consistent functions on identified normal coordinates for
compatibility. Then for every $\eta>0$, all starts can be chosen so
large that
\begin{equation}\label{eq:coface-bound}
 \int(V_E+1.16D_E)v^2\le(\beta+\eta)\mathfrak h_E(v)
\end{equation}
on every affected raw piece.
\end{lemma}

\begin{proof}[Sketch]
Fix $\eta>0$ and pick compact base windows and bounded cylinder
intervals that capture every compact test with bounded away from the
singular set. Choose the front positions $T_i$ large enough that the
supports of $D_E$ miss these windows; then the old cost bound holds.

If however a violating sequence of tests escapes to infinity in the
cylinders, compactness (\cref{lem:shield-compactness} below) gives a
pure-cylinder limit; on that limit the translated bound holds by
Fubini slicing of the face bound, and the gradient terms in the extra
coordinates are non-negative with the same coefficients $.8,1.7,e_0$.
Approximation in the limit completion gives the assertion for the
pure cylinder. Thus no violating sequence can be constructed.
\end{proof}

\subsection{Compactness of the limit completion}

\begin{lemma}[Shield compactness]\label{lem:shield-compactness}
Suppose $m\ge8$, and consider the submanifold Sobolev estimate
\[
 \left(\int f^{d/(d-1)}\right)^{(d-1)/d}
 \le C_d\int(\abs{\nabla f}+\abs H f)
\]
in the Michael-Simon form with mean curvature $H$ bounded, on a
rectifiable $d$-varifold of density at least one. Then a bounded
$\mathfrak h$-sequence on a fixed compact window has a strongly
convergent $L^2$ subsequence. If compactness of the potential
pulling-back is maintained, the sequential inequality
\[
 \mathfrak h_{Z_\infty}(v)
 \le\liminf_i\mathfrak h_{Z_i}(v_i)
\]
holds on the limit.
\end{lemma}

\begin{proof}[Sketch]
The bulk has no uniform smooth boundary, so we use a BV extension of
$\abs{v_{\mathrm{bulk}}}^b$ by zero. The relevant boundary term is
again caught by the seam trace inequality
$\int_{S_{\rm reg}}\abs{v(0)}^b\le C[\int_\Omega\abs v^b+\int_\Omega
\abs v^{b-1}\abs{\nabla v}+\int_{S_{\rm reg}}\abs{v(0)}^b]$ via
BV Sobolev; iterating gives an $L^q$ bound $q>2$.

Concentration at the singular set: for a $\delta$-neighborhood
$\overline{U_\delta(\Sigma)}$ of the limit's singular set, weighted
perimeter convergence gives
$\limsup_i\mu_{S_i}(\overline{U_\delta(\Sigma)})\le\mu_{S_\infty}
(\overline{U_\delta(\Sigma)})\to0$ as $\delta\to0$. Then H\"older
with the local $L^p$ estimate bounds the $L^2$ mass on the singular
neighborhood by $C\mu_i(U_\delta)^{1-2/p}\to0$ as $i\to\infty$ and
then $\delta\to0$.

Approximation by tests missing the singular set: coverage by $m-3$-capacity balls (small codimension $m-3$) with $\sum a_i^{m-3}$
arbitrarily small, and higher Sobolev regularity in the ambient
surrounding, gives functions supported off the singular set that
approximate the limit. The bound in \cref{lem:shield-compactness}
follows.
\end{proof}

\subsection{Compatible caps and cores}

\begin{lemma}[Compatible caps and cores]\label{lem:caps-cores}
A compact forcing update from \cref{prop:local-shielding} extends to
all its cofaces so as to preserve the profile gradient inequality with
coefficient $a_r$, preserve the designated strict anchor sign on the
support, and enlarge only the designated cores. The extension is an
exact product sufficiently far in every old face collar, and is made
once per occurrence independent of the order of commuting face
relations.
\end{lemma}

\begin{proof}
Write $T_i^+$ for the end of the $i$-th cap crossing. Enlarge the plus
core by the open set
\[
 \{x\colon\delta_B(x)<\varepsilon/2\}\times
 \prod_i(T_i^++b_i,\infty),\quad b_i>0.
\]
Properly embedded closed tails make its closure have $\delta_B\le
\varepsilon/2$ and $t_i\ge T_i^++b_i$. Its interior is at
$\delta_B<\varepsilon$ and $t_i>T_i^+$, where every branch of the
minimum is $+\infty$. Same for a minus core.

Take extensions through all coface chains. Finite incidence allows all
thresholds to be exceeded simultaneously. This proves compatibility
for all chains and prevents a commuting product occurrence from being
counted more than once.
\end{proof}

\subsection{Finite termination}

\begin{proposition}[Global shielding]\label{prop:global-shielding}
At a fixed stage $r$ let $j_0$ be the current label and keep its true
minimizers fixed. Then there are compatible smooth additions $D_j\ge0$,
updated profiles and enlarged cores with:
\begin{enumerate}[label=(\roman*),leftmargin=*]
\item\label{item:profiles} updated profiles have locally Lipschitz
finite truncations and satisfy a.e.\ on their finite bands
\begin{equation}\label{eq:profile-output}
 2|\nabla\mu'_j|\le a_r(\mu'_j)^2+\Pi_j+D_j.
\end{equation}
All these data are exact compatible products. Cores only enlarge,
closures stay in the designated pole interiors, forced sets and
anchor restrictions are preserved.
\item\label{item:guard} With $G'_j=E\setminus(U_j'^+\cup U_j'^-)$,
the set $H_E^*=\{x\colon\#\{j\ne j_0\colon x\in G'_j\}\ge m-2\}$ is
closed and misses $\Sigma_E$. In an exact collar over a face $B$, its
projection is contained in $H_B^*$.
\item\label{item:form} For one common $\theta_*<.9$, all walls and
joined sides satisfy
\[
 \int1.16\sum_{j\ne j_0}D_j\,v^2
 \le\theta_*\bigl(\mathcal N(v)+e_0\sum_i\|v\|_2^2\bigr)
 \le.9\bigl(\mathcal N(v)+e_0\|v\|_2^2\bigr).
\]
\end{enumerate}
\end{proposition}

\begin{proof}[Sketch]
Process packet depths from deepest to coarsest. On each packet
$E$ the risky set is $K_E=\Sigma_E\cap H_E^*$, compact by
\cref{lem:escape}. If $m<8$ it is empty.

Enumerate all countable pairs (packet, possible local addition number)
as $\nu\ge1$. Assign allowance $\gamma_\nu=.1\cdot2^{-\nu}$ split
between the shell-scale choice and the simultaneous coface extensions.
 The bookkeeping bound on the total $\mathfrak h$-cost is the
 per-update maximum of $\max\{\theta,.78\}$ plus the summable
 allowances. After any completed collection of updates, the total
 coefficient is at most $.78+\sum_{\nu\text{ used}}\gamma_\nu\le .88<.9$.

For each enumerated pair: find the flat piece covering the compact
risky subset in the strict anchor region (finitely many labels, since
we only need those active on $H_E^*$ and its compact cover). Choose
$\varepsilon$ by \cref{lem:max-costs} and extend the addition by
\cref{lem:coface-ext,lem:caps-cores}. Do this for all labels with
positive potential in the cover, one at a time.

Finite termination: after $N$ additions, set
\begin{equation}\label{eq:open-exhauster}
 O_N=\{x\in E\colon\#\{j\ne j_0\colon x\in G_j^{(N)}\}<m-2\},\quad
 K_E^{(N)}=K_E^{(0)}\setminus O_N.
\end{equation}
Each $O_N$ is open, and $O_N\subset O_{N+1}$. Fix
$x\in K_E^{(0)}$. For each of its strict anchors, the corresponding
piece in the finite enumeration contains $x$, and the addition on
that piece puts $x$ in an open enlarged core regardless of how small
$\varepsilon$ was chosen. Only finitely many anchors are involved.
Once those additions have occurred, only threshold-equality labels
can remain undecided at $x$. By \cref{lem:local-flat} those are at
most $m-8$, giving $x\in O_N$ at some finite $N$. The increasing open
sets cover $K_E^{(0)}$; compactness gives a finite subcover; the
largest index has $K_E^{(N)}=\varnothing$. Stop there. All subsequent
updates only shrink the guard.
\end{proof}

\section{Regularization and restoration of the invariant}
\label{sec:regularization}

The true cuts of \cref{sec:minimizers-appendix} produce genuine smooth
packets only on their regular parts. We must replace them by smooth
regular walls while preserving the analytic data of
\cref{def:invariant}.

\subsection{Guarded approximation}

Let $E$ be a post-radial packet with true cut $S$, singular set
$\Sigma$, and guard $H_E^*$. Shielding has made $H_E^*$ missing from
$\Sigma$. Fix a smooth cutoff $\chi$ equal to $1$ on a neighborhood of
$H_E^*$, supported inside the packet away from $\Sigma$.

\begin{lemma}[Guarded approximation]\label{lem:guarded-approx}
The true cut, restricted to $\{\chi=1\}$, admits smooth regular
walls $L$ with defining function $u\colon E\to\R$ satisfying
$u>0$ on $q_E^{-1}(A_j)$ and $u<0$ on $q_E^{-1}(B_j)$ for every
remaining color label $j$, and with $|u-\psi|<10^{-3}\psi$ on the
guard, where $\psi$ is a fixed comparison function that vanishes on
the true cut. The pair $(L,u)$ can be chosen compatible with all
products in old ends.
\end{lemma}

\begin{proof}[Sketch]
On compact pieces of $H_E^*$ smoothness of the cut gives a graphical
description. Set $u$ to be the signed distance to $S_{\rm reg}$:
\begin{equation}\label{eq:signed-distance}
 u(x)=\pm\dist(x,S_{\rm reg})
\end{equation}
near $S_{\rm reg}$ with sign from the cut. This function is $C^{1,\alpha}$
with $\alpha$ uniform; approximate by smooth functions via the
$C^{1,\alpha}\to C^1$ smoothing operator $Tf$ of
\cref{lem:constant-preservation} below. End compatibility comes from
choosing approximation errors on the finite list of coface
participants and using the finite-tolerance allocation of
\cref{sec:packets-appendix}.
\end{proof}

\subsection{Constant preservation}

\begin{lemma}[Constant-preserving smoothing]\label{lem:constant-preservation}
A continuous scalar function on a packet can be approximated by smooth
functions, with any prescribed positive value error, by operators that
reproduce constants. For compatible input on a packet diagram, the
approximations can be chosen compatible in all sufficiently far product
ends of that diagram. If the input and all face copies used equal the
same constant $c$, the result equals $c$ there exactly. The same
assertion holds componentwise for a finite list of functions.
\end{lemma}

\begin{proof}
Choose a locally finite coordinate cover, relatively compact smaller
charts, and a smooth partition of unity $\{\psi_\alpha\}$ subordinate
to the smaller charts. Extend each chart and average the function
against a smooth kernel of integral one at chart-dependent scale. The
local average sends $c$ to $c$. Blending from deeper faces upward with
weights $w_i$ maintains both constants and exact face
identities. Choose the finite tolerance union over the prescribed
directed dependencies before applying the construction; every other
entry has all its local inputs equal to the same constant, so
smoothing error is zero for that entry.
\end{proof}

\subsection{Seam smoothing}

\begin{lemma}[Seam smoothing]\label{lem:seam}
The raw joined metrics admit smooth compatible metrics with arbitrarily
small $C^0$ error, and their scalar forms converge to the raw forms
quantitatively: on each joined packet and for every $\delta>0$ the
smoothing can be chosen so that the absolute form error is at most
$\delta\int(u^2+|\nabla u|^2)$ for every compact matching test, using
the raw metric to measure the right side. The old warped-product
structure and every sufficiently far old product end are preserved.
The choices on a wall satisfy the requests of both incident sides at
once.
\end{lemma}

\begin{proof}[Sketch]
In a normal collar of a smooth wall, write the stabilized metric
$ds^2+\beta_s$ for $s\le 0$. The raw joined metric is
$ds^2+\beta_{\min(s,0)}$. Choose a smooth concave function $\Phi$
equal to $t$ for $t\le-1$ and to $0$ for $t\ge1$, with
$0\le\Phi'\le1$, and $\int\Phi''=-1=-\int|\Phi''|$. Substituting
$\Phi(\cdot/w)$ for a narrow width $w>0$ produces a $C^\infty$ metric
converging to the raw in the sense stated. The scalar curvature
concentration at the corner is
\[
 \Scal(\Phi)\to\Scal(\text{raw})+\text{sign}\cdot\dd\mathcal H^{m-1}
 \text{ measure at }\partial
\]
and integrating by parts against a smooth test transfers this
boundary measure to the form inequality. Compatibility comes from
choosing $\Phi$ with fixed plateaus in all product ends.
\end{proof}

\subsection{Restricted profiles and potentials}

Recall from \cref{sec:packets-appendix} that the profile inequality is
$2|\grad\mu_j|_{\rm new}\le a_r\mu_j^2+\Pi_j$ a.e.\ on the finite band.

\begin{lemma}[Profile restriction]\label{lem:profile-restrict}
After seam smoothing, profiles and cores can be restricted to the new
packets so that the profile inequality is preserved, and potentials
smoothly compatible with
\[
 (1+\eta_r)\widetilde\Pi_j
 \le\Pi_j^{\rm new}\le(1+2\eta_r)\widetilde\Pi_j.
\]
The associated Schr\"odinger operator
$L_A^0=-C\lap_{\rm old}+R_{\rm old}-V_A^*$ is form-bounded below on a
neighborhood of the new required set by a positive multiple of
$\|u\|_2^2$.
\end{lemma}

\begin{proof}[Sketch]
Locally Lipschitz functions with an a.e.\ gradient bound by a
continuous function have the same local Lipschitz slope bound on
hypersurfaces. Indeed convolve on a small ball, take the derivative
bound on the shrinking ball, integrate along segments and pass to the
limit. Composing with the split's smooth base-and-cylinder projection
preserves both the profile and the core membership.

For potentials: by \cref{lem:constant-preservation}, apply
$(1+3\eta_r/2)\widetilde\Pi_j$ with smoothing error less than
$\eta_rp_j^*/2$. When $\widetilde\Pi_j$ is a spatial constant $c_j$,
the new value is exactly $(1+3\eta_r/2)c_j$. Then the profile
inequality in the new metric follows from
$2|\grad\mu_j|_{\rm new}\le(1+\eta_r)(a_r\mu_j^2+\widetilde\Pi_j)
\le a_{r+1}\mu_j^2+\Pi_j^{\rm new}$.
\end{proof}

\subsection{Compatible corrections and positive supersolutions}

\begin{lemma}[Correction]\label{lem:correction}
There exists a smooth compatible nonnegative $K_{\rm ind}$, zero on a
neighborhood of the new required set $D_A$, such that the operator
$Q_A^0+K_{\rm ind}$ with
\begin{equation}\label{eq:correction}
 K_{\rm ind}=J\sin^2\vartheta+C|\nabla\vartheta|^2,\quad
 J>-\inf W_A,
\end{equation}
is positive, more precisely has form gap $\gamma_A\ge\min\{e_0,
J+\inf W_A\}$.
\end{lemma}

\begin{proof}[Sketch]
Set $W_A=R_{\rm old}-V_A^*$. Choose a smooth angle $\vartheta$ equal
to $0$ near $D_A$ and $\pi/2$ outside a neighborhood of local
positivity, with bounded derivatives and $\supp\cos\vartheta$ inside
that neighborhood. Partition $\cos^2+\sin^2=1$: the gradient partition
is exactly $C|\nabla\vartheta|^2u^2$, and the two parts split the
form. Local positivity controls the cosine part; $W_A+J>0$ controls
the sine part.
\end{proof}

\begin{lemma}[Resolvent]\label{lem:resolvent}
The form $Q_A^0$ is coercive on $H^1$. There is a self-adjoint positive
inverse $L_A^{-1}$ on $L^2$, with norm at most $C_{\rm op}$. On each
post-radial packet the pair $(L_A, L_A^{-1})$ is positive.
\end{lemma}

\begin{proof}
Combine $Q\ge g\|u\|_2^2$ with
$Q\ge C\|\nabla u\|_2^2-\|W_-\|_\infty\|u\|_2^2$ to get coercivity.
Then a self-adjoint inverse exists by the Lax-Milgram theorem. If
$f\ge0$, testing $Lu=f$ with the negative part gives
$Q(u_-)\le0$, so $u\ge0$.
\end{proof}

\begin{lemma}[Strict supersolution]\label{lem:supersolution}
Let $R>0$ and $A$ a bounded measurable positive set. There exists
$s_D\ge0$ smooth, bounded below, with $s_D=1$ on a nested smaller
domain for $R$ large, and $L_A s_D\ge c>0$.
\end{lemma}

\begin{proof}[Sketch]
Fix two nested relatively compact coordinate domains in which the
source is exactly $1$. Interior Schauder estimates and differentiation
bound every derivative of $u_R$ on the smaller domain uniformly for
$R\ge R_0$. A diagonal subsequence converges smoothly on compact
subsets to a bounded positive solution $s$ of $Ls=1$.
\end{proof}

\subsection{Compatible positive approximations}

\begin{lemma}[Compatible positive approximation]\label{lem:sol-approx}
Fix the corrections of \cref{lem:correction} and let $s_A$ be the
exact bounded solution $L_As_A=1$. There exist smooth compatible
functions $v_A>0$ with $L_Av_A\ge1/2$ on every packet. They have
positive infimum and bounded derivatives packetwise.
\end{lemma}

\begin{proof}[Sketch]
Use the end-decay: on an exact elementary collar $B\times(T,\infty)$
the difference $w=s_A-s_B$ satisfies
$|s_A-s_B|(x,t)\le C e^{-\alpha t}s_B(x)$ for $t<T$. Construct $v_A$
from deeper faces upward: keep $s_A$ in the compact part and use
packet weights to blend to $v_B$; move the starts until
$\|s_A-s_B\|_{C^2}<\varepsilon/4$ on all participating collars. The
remaining $v_B-s_B$ is paid by its allocated tolerance, not by moving
the start. Start beyond all compatibility thresholds so that copies
agree. The compact data give bounded higher derivatives. Choose the
tolerance $\tau_A$ small enough that $L_Av_A\ge 1-M_A\tau_A\ge3/4\ge1/2$.
\end{proof}

\subsection{The scalar-identity step}

\begin{proposition}[The full invariant step]\label{prop:full-step}
Assume the stage-$r$ invariant and the conclusion of
\cref{prop:global-shielding}. For every nonempty parent packet of base
dimension at least two, processing $j_0$ produces the compatible
smooth split packets of \cref{def:packet}, with the required forced
signs and continuous projection maps. There are smooth compatible old
metrics and $k=399$ new torus factors on every packet such that the
profiles, cores, potentials, and scalar curvature satisfy the
stage-$(r+1)$ invariant. This includes a possible one-dimensional
output, on which the scalar lower bound is global.
\end{proposition}

\begin{proof}[Sketch]
Apply \cref{lem:guarded-approx,lem:seam,lem:profile-restrict} for the
smooth base metrics and the profile data, and
\cref{lem:correction,lem:resolvent,lem:supersolution,lem:sol-approx} for
$v_A$. Append $k=399$ circle variables, each with length function
$v_A^{2/(k+1)}$. The warped-product identity is
\[
 R_{\rm new}=R_{\rm old}-\frac{4k}{k+1}\frac{\lap_{\rm old}v_A}{v_A}.
\]
Indeed a common scale $f$ contributes
$-2kf^{-1}\lap_{\rm old}f-k(k-1)f^{-2}|df|^2$; substitution of
$f=v_A^{2/(k+1)}$ cancels the separate $|dv_A|^2$ term. Here $k$
counts only the newly appended circle factors; unit factors used
earlier to compare old lists have zero logarithmic derivative and leave
both $\lap_{\rm old}$ and the coefficient unchanged. Since
$4k/(k+1)=C$, \cref{lem:profile-restrict,lem:supersolution} give
\[
 R_{\rm new}=V_A^*-K_A+\frac{L_A v_A}{v_A}\ge V_A^*
 \text{ on }D_A.
\]
Here $K_A=0$ on a neighborhood of $D_A$. Adding these factors changes
no base distances or profile gradients. The new factors are positive,
bounded above and away from zero, and exactly compatible. The same
number is appended at every status of the stage, so all end identities
compare the same list of factors. This proves the analytic invariant,
including when the new required threshold is zero.
\end{proof}
