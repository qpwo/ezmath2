\title{A torsion-free hyperbolic group that is not residually finite}
\author{}
\date{}

\begin{document}

\begin{abstract}
We construct a torsion-free word-hyperbolic group that is not residually finite, answering the residual finiteness question for hyperbolic groups negatively. The proof is complete and self-contained; the standard metric and van Kampen tools we need are stated and proved from scratch.
\end{abstract}

\section{Introduction}

A group $G$ is \emph{residually finite} if every nonidentity element has a nonidentity image in some finite quotient; equivalently, the intersection $R_f(G)$ of the kernels of all homomorphisms to finite groups, the \emph{finite residual}, is trivial. A finitely generated group is \emph{word-hyperbolic} if in some (equivalently every) Cayley graph, geodesic triangles are uniformly thin. The residual finiteness question asks whether every word-hyperbolic group is residually finite.

\begin{theorem}\label{thm:main}
There is a torsion-free word-hyperbolic group that is not residually finite.
\end{theorem}

The group is $G=\pi_1(K,O)$ for a finite Euclidean triangle complex $K$. We exhibit a finite list $\mathcal R$ of nonidentity elements and prove that no single finite quotient separates all of them. It follows that some one element $w\in\mathcal R$ lies in every finite quotient, i.e.\ $w\in R_f(G)$; since $w\neq1$, $G$ is not residually finite.

\paragraph{The idea in one paragraph.} We build a finite surface-like object $K$ glued from Euclidean triangles so that every vertex link is a graph with no short cycles. That much positive angle excess already forces $\pi_1(K)$ to be torsion-free and word-hyperbolic. Inside $K$ we plant many four-edge loops (rectangular words) and prove each is a nonidentity group element. One finite quotient can separate only finitely many elements, so if $K$ were residually finite some finite quotient would have to keep all these loops nontrivial at once. A rank count over a fixed finite field rules that out: each loop array contributes a matrix of large rank, yet the local triangle relations force the same matrix to have small rank. So some one loop dies in every finite quotient, and $G$ is not residually finite.

\paragraph{How to read this.} Sections~\ref{sec:criterion}--\ref{sec:rank} build $K$ and run the rank contradiction, taking the geometric criterion (Theorem~\ref{thm:triangle-complex}) for granted. A reader willing to accept that criterion may stop after Section~\ref{sec:rank}. Sections~\ref{sec:metric}--\ref{sec:thin} prove the criterion from scratch.

\paragraph{Constants.} Throughout, $h=20$, $c=100^{-h}$, and $r$ is a fixed integer with $cr\ge100$. The prime $q$ is chosen large and then fixed before any finite quotient is considered.

\begin{corollary}\label{cor:nonlinear}
For the group $G$ of Theorem~\ref{thm:main}, for every commutative field $k$, every $n\ge1$, and every homomorphism $\rho:G\to GL_n(k)$, we have $R_f(G)\subseteq\ker\rho$. Hence some fixed nonidentity element is killed by every such representation, and $G$ is linear over no commutative field.
\end{corollary}
\begin{proof}
The finitely generated linear group $\rho(G)$ is residually finite (Malcev). If $g\in R_f(G)$ had $\rho(g)\neq1$, residual finiteness of $\rho(G)$ would give a finite quotient separating $\rho(g)$, contradicting $g\in R_f(G)$. So $R_f(G)\subseteq\ker\rho$; the nonidentity element of $R_f(G)$ is thus killed by every $\rho$.
\end{proof}

\section{The geometric criterion}\label{sec:criterion}

\begin{definition}
A \emph{finite Euclidean triangle complex} is a finite graph with finitely many Euclidean triangles attached side-to-side along full edges, with positive compatible lengths and isometric gluings. Each closed cell embeds; distinct cells may meet in a union of faces. The \emph{angular link} at a vertex has one vertex per incident edge-germ and one edge per incident triangle corner, of length equal to the corner angle. A link walk is \emph{reduced} if it has no immediate reversal, and \emph{cyclically reduced} if this holds also across its closing seam. Link distance between distinct components is $+\infty$.
\end{definition}

The \emph{intrinsic metric} on such a complex is the infimum of lengths of polygonal paths made of straight segments inside cells. A \emph{turn} of a closed edge path compares the direction backward along the incoming edge with the direction forward along the outgoing edge, as points of the vertex link.

\begin{theorem}[Triangle complex criterion]\label{thm:triangle-complex}
Let $K$ be a connected finite Euclidean triangle complex. Suppose there is $\eta>0$ such that every nonempty cyclically reduced closed edge walk in every vertex link has angular length at least $2\pi+\eta$. Then $\pi_1(K)$ is torsion-free and word-hyperbolic. Moreover, a nonempty closed edge path whose every cyclic turn has link distance at least $\pi$ is not nullhomotopic.
\end{theorem}

The conclusions are proved in Propositions~\ref{prop:closed-paths} (nontriviality), \ref{prop:no-torsion} (torsion-freeness), and \ref{prop:word-hyperbolicity} (hyperbolicity).

\section{Punctured lines with few short cycles}\label{sec:incidence}

Put $P=\F_q^h$, $m=q^h$, and let $S\subset\F_q$ have $|S|=\lfloor q/100\rfloor$. A \emph{mark} is a point of $S^h$. A \emph{labelled line} is an affine line $\ell\subset P$ together with a marked point $a\in S^h\cap\ell$; write $\ell_i$ and $a_i$ for label $i$, and $I(z)$ for the set of labels with $z\in\ell_i\setminus\{a_i\}$.

\begin{lemma}[Incidence]\label{lem:incidence}
For all sufficiently large primes $q$ there are $N=(c+o(1))m$ labelled lines, indexed by $I$, and a set $E\subset P$ such that:
\begin{enumerate}[label=\textup{(\roman*)}]
\item the bipartite incidence graph on $P\sqcup I$, with $z\sim i$ iff $z\in\ell_i\setminus\{a_i\}$, has no cycle of length $\le12$;
\item $|I(z)|\ge4r^2$ for $z\notin E$;
\item $|E|=o(m)$.
\end{enumerate}
For these data we may assume $q>r$ and $\tfrac{Nr}{4}>2m+r|E|$.
\end{lemma}

\begin{proof}
Begin with all points of $S^h$ as marks, and for each mark choose its line direction independently and uniformly among the $J=(q^h-1)/(q-1)$ directions of lines through a point of $P$.

\emph{Degrees.} For fixed $z$, each other mark lies on a line through $z$ with probability $1/J$, so $|I(z)|$ is binomial with mean $\mu_z\sim cq$ uniformly in $z$ and variance $\le\mu_z$. For large $q$, $\mu_z\ge8r^2$, and the second-moment inequality gives $\Pr(|I(z)|<4r^2)\le\mu_z(\mu_z-4r^2)^{-2}=O(q^{-1})$. Hence the expected number of low-degree points is $O(m/q)$, and by Markov's inequality, with probability $\to1$, there are at most $mq^{-1/2}$ of them.

\emph{Short cycles.} For $2\le s\le6$ count simple cycles of length $2s$ (with $s$ points and $s$ labels). There are at most $m^s$ ordered choices of the $s$ points. Consecutive points lie on a common line, which is thereby determined geometrically, and the corresponding mark has at most $q$ possible positions on it. Finally the direction of each of the $s$ labels must equal the forced direction, an event of probability $J^{-s}$, independent across labels. So the expected number of such cycles is at most
\[
 m^s\,q^s\,J^{-s}\le q^{2s},
\]
since $J\ge q^{h-1}$. Summing over $s$ gives expected total at most $6q^{12}\le q^{13}$ cycles of length $\le12$.

Pick an outcome satisfying both bounds and delete the label of every line on one of these cycles: at most $6q^{13}$ labels. Let $E$ contain the surviving low-degree points and every point of a deleted line. Deleting labels creates no cycle and, off $E$, changes no degree; hence
\[
 |E|\le mq^{-1/2}+6q^{14}=o(m),\qquad N=(c+o(1))m.
\]
Since $cr\ge100$, for large $q$ we have $N/m\ge c/2$ and $r|E|/m\le1$, while $cr/8\ge12.5>3$; thus $\tfrac{Nr}{4}>2m+r|E|$, as wanted.
\end{proof}

The data of Lemma~\ref{lem:incidence} are fixed once and for all; no finite quotient encountered later may change $q$ or them.

\section{Building the complex}\label{sec:build}

Fix the data of Lemma~\ref{lem:incidence}.

\paragraph{Blocks.} For $z\notin E$ let $d=|I(z)|\ge4r^2$. Choose $n_2\in\{r,\dots,2r-1\}$ with $n_2\equiv d\pmod r$ and put $n_1=(d-(r+1)n_2)/r$. Then
\[
 d=rn_1+(r+1)n_2,\qquad n_1=\frac{d-(r+1)n_2}{r}\ge\frac{4r^2-(r+1)(2r-1)}{r}=\frac{2r^2-r+1}{r}\ge r .
\]
Partition $I(z)$ into two \emph{blocks} $I_b$ with $|I_b|=s_bt_b$ and $s_b,t_b\ge r$: one block has $s_b=r$, $t_b=n_1$; the other $s_b=r+1$, $t_b=n_2$. Index blocks at different points separately; let $p(b)$ be the point of block $b$, and make no blocks at points of $E$. For each block fix a bijection
\begin{equation}\label{eq:chart}
 i\longmapsto(\alpha_b(i),\beta_b(i))\in\Z/s_b\Z\times\Z/t_b\Z .
\end{equation}

\paragraph{The complex $K$.} Let $[r]=\{0,\dots,r-1\}$ and take vertices $O,V,W$ and edges
\[
 x_{iuv}:V\to W\ (i\in I,\ u,v\in[r]),\quad
 L_{b,u,j}:O\to V\ (u\in[r],\,j\in\Z/s_b\Z),\quad
 R_{b,v,k}:O\to W\ (v\in[r],\,k\in\Z/t_b\Z).
\]
For every block $b$, every $i\in I_b$, and all $u,v\in[r]$, attach one triangle with sides
\begin{equation}\label{eq:sides}
 x_{iuv},\qquad L_{b,u,\alpha_b(i)+v},\qquad R_{b,v,\beta_b(i)+u},
\end{equation}
with angles $3\pi/5$ at $O$ and $\pi/5$ at $V$ and at $W$, all of the same fixed shape. Each triangle has three distinct vertices, so closed cells embed; assign matching lengths to unused edges. Regular points exist, so $K$ is connected, finite, with finitely many edges and triangles.

\section{The link bound and nontrivial rectangles}\label{sec:links}

\begin{lemma}\label{lem:links}
$K$ satisfies Theorem~\ref{thm:triangle-complex} with $\eta=2\pi/5$. At $V$ or $W$, the link distance between distinct $x$-directions carrying the same label $i$ is at least $14\pi/5$ (or infinite).
\end{lemma}

\begin{proof}
\emph{At $O$.} For a fixed block $b$, a triangle~\eqref{eq:sides} joins $L_{b,u,j}$ to $R_{b,v,k}$ exactly when $\alpha_b(i)=j-v$ and $\beta_b(i)=k-u$ for a unique $i\in I_b$ (the chart~\eqref{eq:chart} is a bijection), so each pair of an $L$- and an $R$-direction bounds exactly one corner. Thus the link at $O$ is, on each block, a complete bipartite graph on its $rs_b$ $L$-directions and $rt_b$ $R$-directions, and distinct blocks are disjoint components. Every nonempty reduced closed walk has at least four edges, each of angle $3\pi/5$, with angular length $\ge4\cdot3\pi/5=12\pi/5=2\pi+2\pi/5$.

\emph{At $V$ and $W$.} The link at $V$ is bipartite between $x$-germs and $L$-germs. Map $\pi_V:\mathrm{link}(V)\to\Gamma$ (the incidence graph) by $x_{iuv}\mapsto i$ and $L_{b,u,j}\mapsto p(b)$; this is well defined since $i\in I_b$ means $i\in I(p(b))$. It is injective on edges at each vertex: at $x_{iuv}$ the incident link edges lead to $L_{b,u,\alpha_b(i)+v}$ over the unique block at each point containing $i$, giving distinct $p(b)$; at $L_{b,u,j}$ the incident $x$-edges correspond to labels $i\in I_b$ with $j=\alpha_b(i)+v$ for at most one $v$, since $s_b\ge r$. The same argument with $t_b\ge r$ handles the link at $W$.

A cyclically reduced walk of length $L$ in such a link maps to a walk in $\Gamma$ reduced at internal vertices. A shortest positive subwalk with equal endpoints is then a genuine cycle of $\Gamma$ of length $\le L$; a subwalk of length two would be an internal reversal. Since $\Gamma$ has no cycle of length $\le12$ and is bipartite, that cycle has length $\ge14$, so $L\ge14$ and the angular length is $\ge14\pi/5=2\pi+4\pi/5\ge2\pi+\eta$.

\emph{Same-label distances.} A shortest link path between distinct $x$-directions with one label $i$ is reduced; it projects to a positive closed walk at $i$ reduced at internal vertices, which contains a cycle of $\Gamma$ and so has length $\ge14$. Its angular length is at least $14\pi/5$, giving the asserted distance (infinite only if no path exists).
\end{proof}

Set $G=\pi_1(K,O)$. By Theorem~\ref{thm:triangle-complex} and Lemma~\ref{lem:links}, $G$ is torsion-free and word-hyperbolic; it remains to exclude residual finiteness.

\paragraph{Rectangular words.} Choose a path $O\to v$ for each vertex, giving an element $g_e$ for each oriented edge, and abbreviate $g_{iuv}=g_{x_{iuv}}$. Each triangle~\eqref{eq:sides} reads, in path order,
\begin{equation}\label{eq:trirel}
 g_{iuv}=g_{L_{b,u,\alpha_b(i)+v}}^{-1}\,g_{R_{b,v,\beta_b(i)+u}}\qquad(i\in I_b).
\end{equation}

\begin{lemma}\label{lem:rect}
For $i\in I$ and $u\neq u'$, $v\neq v'$ in $[r]$,
\begin{equation}\label{eq:rect}
 w(i,u,u',v,v'):=g_{iuv}\,g_{iu'v}^{-1}\,g_{iu'v'}\,g_{iuv'}^{-1}\neq1 .
\end{equation}
\end{lemma}
\begin{proof}
After cancelling the connector paths, this word is conjugate to the closed four-edge loop $V\xrightarrow{x_{iuv}}W\xrightarrow{x_{iu'v}^{-1}}V\xrightarrow{x_{iu'v'}}W\xrightarrow{x_{iuv'}^{-1}}V$. At each of its four vertices the turn compares two distinct $x$-directions with the common label $i$ (at $W$ the pair $x_{iuv},x_{iu'v}$; at $V$ the pair $x_{iu'v},x_{iu'v'}$; and so on). By Lemma~\ref{lem:links} every cyclic turn has link distance $\ge14\pi/5\ge\pi$, so the closed-path criterion of Theorem~\ref{thm:triangle-complex} makes the loop nontrivial.
\end{proof}

Let $\mathcal R\subset G$ be the finite set of words~\eqref{eq:rect}. Lemma~\ref{lem:rect} says $1\notin\mathcal R$.

\section{The rank obstruction}\label{sec:rank}

\begin{lemma}[Rank under reduction]\label{lem:rankloss}
Let $M$ be an integer matrix whose rows have Euclidean norm at most $\sqrt r$. For every prime $q>r$,
\[
 \rank_{\F_q}M\ge\tfrac12\rank_{\R}M .
\]
\end{lemma}
\begin{proof}
Let $d=\rank_{\R}M$ (clear if $d=0$). A nonzero $d\times d$ minor $B$ satisfies $1\le|\det B|\le r^{d/2}$ by Hadamard's inequality. Let $d'=\rank_{\F_q}B$; subtracting integer lifts of $d'$ spanning rows from the others is determinant-preserving and makes $d-d'$ rows divisible by $q$, so $q^{d-d'}\mid\det B$ and $q^{d-d'}\le r^{d/2}$. As $q>r$, $d-d'<d/2$, i.e.\ $d'\ge d/2$.
\end{proof}

Suppose, for contradiction, that a finite quotient $Q$ of $G$ keeps every $w\in\mathcal R$ nonidentity. Put $D=|Q|$ and let $U(g)$ be the $D\times D$ regular permutation matrix of left multiplication (so $U$ is a homomorphism and $U(g)^{\mathsf T}=U(g^{-1})$). For $i\in I$ form the block matrix
\[
 Y_i=\bigl(U(g_{iuv})\bigr)_{u,v\in[r]},
\]
of size $rD\times rD$, with exactly $r$ unit entries per row, hence row norm $\sqrt r$.

\begin{lemma}\label{lem:Yirank}
$\rank_{\R}Y_i\ge rD/2$ and $\rank_{\F_q}Y_i\ge rD/4$.
\end{lemma}
\begin{proof}
A nonidentity regular permutation has real trace $0$ and the identity has trace $D$. Hence
\[
 \tr(Y_iY_i^{\mathsf T})=r^2D,\qquad
 \tr\bigl((Y_iY_i^{\mathsf T})^2\bigr)=(2r^3-r^2)D .
\]
Indeed $Y_iY_i^{\mathsf T}$ has block $(u,u')=\sum_v U(g_{iuv}g_{iu'v}^{-1})$, so $\tr((Y_iY_i^{\mathsf T})^2)=\sum_{u,u',v,v'}\tr U(g_{iuv}g_{iu'v}^{-1}g_{iu'v'}g_{iuv'}^{-1})$. The argument of $\tr U$ is trivial when $u=u'$ or $v=v'$ and is a word~\eqref{eq:rect}, hence nonidentity, otherwise; the number of trivial index choices is $2r^3-r^2$. Since $Y_iY_i^{\mathsf T}$ is positive semidefinite, Cauchy--Schwarz on its eigenvalues gives
\[
 \rank_{\R}Y_i\ge\frac{(r^2D)^2}{(2r^3-r^2)D}=\frac{r^2D}{2r-1}\ge\frac{rD}{2}.
\]
Lemma~\ref{lem:rankloss} then gives $\rank_{\F_q}Y_i\ge rD/4$.
\end{proof}

Work over $\F_q$ and put $Z(z)=\sum_{i\in I(z)}Y_i$.

\paragraph{Local factorization.} Fix $z\notin E$ with its two blocks. For a block $b$,
\begin{equation}\label{eq:fact}
 \sum_{i\in I_b}Y_i
 =\Bigl(\sum_{j\in\Z/s_b\Z}U(g_{L_{b,u,j}}^{-1})\Bigr)_{u\in[r]}
  \Bigl(\sum_{k\in\Z/t_b\Z}U(g_{R_{b,v,k}})\Bigr)_{v\in[r]},
\end{equation}
a product of an $rD\times D$ block column and a $D\times rD$ block row. Its block $(u,v)$ is $\sum_{j,k}U(g_{L_{b,u,j}}^{-1}g_{R_{b,v,k}})$, which by the bijection $i\mapsto(\alpha_b(i)+v,\beta_b(i)+u)$ equals $\sum_{i\in I_b}U(g_{L_{b,u,\alpha_b(i)+v}}^{-1}g_{R_{b,v,\beta_b(i)+u}})$, i.e.\ $\sum_{i\in I_b}Y_i$ by~\eqref{eq:trirel}. Each block sum thus has rank $\le D$, and
\begin{equation}\label{eq:Zrank}
 \rank Z(z)\le\begin{cases}2D,&z\notin E,\\ rD,&z\in E,\end{cases}
\end{equation}
the second bound being the ambient size.

\paragraph{Interpolation.} For $a_i=(a_{i1},\dots,a_{ih})$ set
\[
 f_i(X)=\prod_{t=1}^h\ \prod_{a\in S\setminus\{a_{it}\}}\frac{X_t-a}{a_{it}-a},\qquad
 f_i(a_{i'})=\delta_{ii'},
\]
so $\deg f_i\le h(|S|-1)$. Let $f=(f_i)_{i\in I}$ and $H(z)=f(z)f(z)^{\mathsf T}$, an $N\times N$ matrix of rank $\le1$ with $H(a_i)=E_{ii}$. Each entry of $H$ has degree $\le2h(|S|-1)<q-1$ and hence, summed over the $q$ points of any affine line, gives $0$ by the finite-field power-sum vanishing. Therefore
\begin{equation}\label{eq:line}
 \sum_{z:\,i\in I(z)}H(z)=-H(a_i)=-E_{ii},
\end{equation}
and, exchanging sums,
\begin{equation}\label{eq:tensor}
 -\bigoplus_{i\in I}Y_i=-\sum_{i\in I}E_{ii}\otimes Y_i=\sum_{z\in P}H(z)\otimes Z(z).
\end{equation}

\paragraph{Rank comparison.} The left side of~\eqref{eq:tensor} has rank $\ge\sum_i\rank_{\F_q}Y_i\ge NrD/4$ by Lemma~\ref{lem:Yirank}. Each summand $H(z)\otimes Z(z)$ has rank $\le\rank Z(z)$, so, summing over the $m$ points and using~\eqref{eq:Zrank},
\[
 \frac{NrD}{4}\le 2D(m-|E|)+rD|E|=2Dm+D|E|(r-2)\le 2Dm+rD|E| .
\]
After cancelling $D>0$, $\tfrac{Nr}{4}\le2m+r|E|$, contradicting the strict budget of Lemma~\ref{lem:incidence}.

Hence no finite quotient preserves all of $\mathcal R$. Were every $w\in\mathcal R$ nontrivial in some finite quotient, the product of those finitely many quotients, followed by passing to its image, would preserve all of $\mathcal R$ at once. So some fixed $w\in\mathcal R$ maps to the identity in every finite quotient, i.e.\ $w\in R_f(G)$; since $w\neq1$, $G$ is not residually finite. This proves Theorem~\ref{thm:main}, subject to Theorem~\ref{thm:triangle-complex}, proved next.

\section{Metrics and planar fillings}\label{sec:metric}

Throughout, $K$ is the complex of Theorem~\ref{thm:triangle-complex}; "original" refers to its cells before subdivision, and links are metric graphs with edge lengths equal to corner angles. Barycentrically subdividing $K$ produces a genuine simplicial complex $S$ whose vertices are the original cells and whose simplices are strict inclusion chains (distinct parallel edges get distinct barycenters, so simplices do not collapse). Realized by straight segments, this preserves the link metric at original vertices; at a new vertex in a triangle interior the link is a circle of length $2\pi$; at a new vertex in an edge interior the two tangent directions are joined by one path of length $\pi$ per incident triangle (or are isolated). Hence every nonempty cyclically reduced closed link walk at a new vertex still has length $\ge2\pi$, and at an original vertex length $\ge2\pi+\eta$. These facts persist under further compatible Euclidean subdivisions.

\begin{lemma}[Metric preliminaries]\label{lem:finite-geodesics}
The intrinsic distance on a connected finite Euclidean triangle complex is a metric inducing the complex topology. The complex is compact, any two points are joined by a shortest path, and every constant-speed local geodesic on a compact parameter interval (closed, if the domain is a circle) has finitely many straight pieces. If such a path lies in the edges of a Euclidean subdivision, the link distance at each of its turns is at least $\pi$.
\end{lemma}

\begin{proof}
Around a point $x$, truncated incident sectors (and possibly isolated radial intervals) are glued along radial sides; the radial coordinate agrees across gluings and any path leaving the star of radius $\rho$ has length $\ge\rho$, while the radial segment has length exactly $\rho$ and is shortest. This proves positivity, identifies metric and star neighborhoods (so the metric induces the topology), and gives contracting neighborhoods. Compactness yields a shortest path between any two points by a standard compactness--equicontinuity (limits of minimizing sequences of proportional-arclength polygonal paths). A sufficiently short unit-speed minimizing arc inside a star has radial coordinate $r(t)=t$; its direction in the finite graph of directions is locally constant, so the arc is straight in one sector. Applying this on both sides of every parameter value gives finitely many straight pieces. Finally, a turn with angular distance $<\pi$ unfolds two nearby equal-radius points into a sector of total angle $<\pi$ and a straight chord shorter than the two radii, contradicting local minimality; this includes the closing seam of a closed geodesic.
\end{proof}

\begin{lemma}[Reduced fillings]\label{lem:reduced-diagrams}
Let $S$ be a Euclidean simplicial subdivision of $K$. Every nonempty nullhomotopic closed edge path in $S$ is the exterior walk of a planar diagram $\Delta$ with a map $\Delta\to S$ taking each edge and triangle isometrically onto an edge and triangle. Each triangle has three distinct vertices and edges. At every interior vertex of $\Delta$ the cyclic sequence of corners maps to a nonempty cyclically reduced closed walk in the target link. Every edge loop in $\Delta$ contracts in $\Delta$.
\end{lemma}

\begin{proof}
The plan is to replace a topological disk filling by a combinatorial one: approximate it by a simplicial map, read off its dual "overlap picture", cancel the cheapest local pattern until none remains, discard pieces the boundary cannot see, and dualize back into a diagram.
Take a continuous disk filling of the path whose boundary is linear on each edge traversal, and mark all boundary breakpoints. Barycentric coordinates of $S$ (extended by zero off stars) are continuous; triangulate the disk finely, respecting marks, so each coordinate varies by $<1/4$ on each small triangle. Label each disk vertex by a target vertex of largest coordinate there. Since $S$ is at most two-dimensional, that largest coordinate is $\ge1/3$, so $1/3-1/4>0$ and the labels of any small triangle lie in a common simplex; extending linearly gives a simplicial map allowing degenerate images. Along a boundary edge only its two endpoints can occur as labels, monotone, changing exactly once; so the nondegenerate boundary steps reproduce the word. Preimages of ties of the largest coordinate form a finite \emph{picture}: arcs meeting at trivalent nodes inside 2-simplices, one arc inside 1-simplices, none inside 0-simplices; regions are labelled by target vertices, an arc separates distinct labels spanning a target edge, and the three labels at a node span a target triangle. Boundary leaves are the word's letters.

Minimize the number of trivalent nodes. No arc joins two nodes whose labels lie in one target triangle: a thin regular neighborhood of the arc has four slots, two on each side, and re-gluing the $a$- and $b$-sides through the third vertex $d$ reduces the node count while preserving all boundary labels (extra arcs between the nodes only create a lens outside the neighborhood). An arc cannot join a node to itself, as the three arcs there carry distinct edge labels.

Erase every picture component off the boundary. The retained picture with the boundary circle is a connected polygonal graph; each inside region is a disk whose boundary is a (possibly repeated) walk, since it can be built by slitting disks with tree edges and then adding arcs. A thin regular neighborhood's region-side boundary curve crosses no arc, so all boundary-incidence labels of a region agree and extend to a region label; erasing cannot increase nodes.

Now take the planar dual: a vertex per region, radii to arc midpoints joined across arcs as dual edges; around each node the three radii bound a triangle whose dual vertices are distinct (labels span a target triangle). An Euler count with $E$ arcs, $N$ nodes, $L>0$ boundary leaves shows the $N$ triangular regions fill exactly the $E-N+1=N$ bounded faces, giving a planar diagram with no holes mapping to $S$; the exterior walk is the prescribed word, and an arc with both ends on the boundary is a spur-edge traversed twice. Reducedness at interior vertices: an immediate reversal would mean two adjacent triangles map to one target triangle, joining their nodes by the arc dual to the shared edge, contradicting minimality (this includes the cyclic seam). Finally every simple edge circuit bounds its filled disk, and any edge loop splits into such circuits and cancellable backtracks, so edge loops contract.
\end{proof}

\section{Curvature, closed paths, and torsion}\label{sec:curv}

For a diagram $\Delta$ of Lemma~\ref{lem:reduced-diagrams} and a vertex $y$, let $\theta_y$ be its total angle and $n_y$ its number of exterior visits. Then $\theta_y\ge2\pi$ when $n_y=0$, and $\theta_y\ge2\pi+\eta$ when also $y$ maps to an original vertex.

\begin{lemma}[Angular Gauss--Bonnet]\label{lem:gaussbonnet}
With $v,e,t$ the numbers of vertices, edges, and triangles of $\Delta$,
\begin{equation}\label{eq:gauss-bonnet}
 \sum_y\bigl((2-n_y)\pi-\theta_y\bigr)=2\pi .
\end{equation}
\end{lemma}
\begin{proof}
Without holes $v-e+t=1$, and counting exterior edge incidences, including both sides of a spur, gives $\sum_y n_y=2e-3t$. As each triangle has angle sum $\pi$,
$\sum_y((2-n_y)\pi-\theta_y)=2\pi v-\pi(2e-3t)-\pi t=2\pi v-2\pi e+2\pi t=2\pi(v-e+t)=2\pi$.
\end{proof}

\begin{proposition}[Closed paths]\label{prop:closed-paths}
In $K$, or any compatible Euclidean simplicial subdivision, a nonempty closed edge path whose every cyclic turn has link distance at least $\pi$ is not nullhomotopic.
\end{proposition}
\begin{proof}
Work in a simplicial subdivision and suppose such a path is nullhomotopic; apply Lemma~\ref{lem:reduced-diagrams}. At an interior vertex the summand of~\eqref{eq:gauss-bonnet} is $\le0$. At a vertex with $n_y=1$ the two interior corners give a link path of length $\theta_y$ between the boundary directions, so the prescribed turn forces $\theta_y\ge\pi$ and the summand is $\le0$; the only no-corner case is a spur tip with coincident directions, excluded. For $n_y\ge2$ the summand is $\le0$ outright. All summands nonpositive contradicts~\eqref{eq:gauss-bonnet}. A path in the original complex passes to the barycentric subdivision with unchanged turn distances (straight-through directions along an original edge have link distance $\pi$ or, for a spur, are excluded), preserving the hypothesis.
\end{proof}

\begin{proposition}\label{prop:no-torsion}
$\pi_1(K)$ is torsion-free.
\end{proposition}
\begin{proof}
Contracting stars and compactness give $\epsilon>0$ so every ball of radius $5\epsilon$ is contractible in $K$; and any two loops that are uniformly $\epsilon$-close are freely homotopic, by coning each quadrilateral through shortest connectors in a ball of radius $3\epsilon$ and gluing along the connectors into an annulus. Let $g\ne1$ and take the infimum $\ell\ge5\epsilon$ of lengths in its free class. A proportional-arclength minimizing sequence has bounded speed; by Lemma~\ref{lem:finite-geodesics} a subsequence converges to a length-$\ell$ minimizer in the same class, and arclength parametrization shows it is a local geodesic: otherwise a subarc of length $<\epsilon$ is replaced by a shorter shortest path with the same endpoints, the difference contracting in a star, shortening the loop in the same class. This closed local geodesic has finitely many straight pieces; insert them into a finite compatible Euclidean subdivision so it lies in the edges, preserving links and turn distances. If $g$ had finite order, a positive repetition would be nullhomotopic, contradicting Proposition~\ref{prop:closed-paths} (its turns, including repeated seams, still have link distance $\ge\pi$).
\end{proof}

\section{Thinness and hyperbolicity}\label{sec:thin}

Now $S$ is the fixed barycentric subdivision of $K$, so every small triangle has an original vertex among its vertices. A reduced diagram over $S$ has, at every interior vertex, total angle $\ge2\pi$ and, at one mapping to an original vertex, total angle $\ge2\pi+\eta$.

\begin{lemma}[Uniform thinness]\label{lem:diagram-thinness}
Choose $M$ larger than every diameter and edge length among the finitely many model triangles and edges, and put $A=M+\tfrac{2M\pi}{\eta}$, $\delta=10(A+1)$. Every geodesic triangle in every reduced diagram is $\delta$-thin.
\end{lemma}
\begin{proof}
Fix a point $p$ of one side not lying on the other two. Let $(a,b)$ be the component of this side off the union of the other two sides containing $p$. There is a simple path $Q$ on the other two sides from $a$ to $b$ using at most two geodesic subarcs (take a subarc if one side holds both $a,b$, else the first meeting of the two arcs toward the common corner). Then $[a,b]\cup Q$ is a simple closed curve of at most three geodesic sides; since diagrams have no holes it bounds a disk $\Omega$. Insert the finitely many straight pieces of the sides and triangulate $\Omega$.

Let $n$ count vertices of $\Omega^\circ$ mapping to original vertices. Each has its full excess $\ge\eta$; other interior vertices have angle $\ge2\pi$; boundary vertices other than the at most three corners have interior angle $\ge\pi$ (else a local shortcut), and corners have deficit $\le\pi$. Lemma~\ref{lem:gaussbonnet} on the disk gives $2\pi\le3\pi-n\eta$, so $n\le\pi/\eta$.

\emph{Inradius.} Let $f(x)=d_\Delta(x,\partial\Omega)$. For interior $x$, join $x$ inside a fixed small triangle to an original vertex, length $<M$; either this meets $\partial\Omega$ ($f(x)<M$) or ends at one of the $n$ counted vertices. So $\{f>M\}$ lies in the union of $n$ ambient balls of radius $M$; as $f$ is $1$-Lipschitz each contributes an interval of length $2M$, and since $f$ is continuous on the connected compact $\Omega$ it realizes $(M,\max f]$ within total length $\le2Mn$. Hence $d_\Delta(x,\partial\Omega)\le A$ for all $x\in\Omega$.

\emph{Level arc.} Suppose $R:=d_\Delta(p,Q)>10(A+1)$. Interpolate $d_\Delta(p,\cdot)$ linearly on a fine subdivision of $\Omega$ to a function $h$ with $|h-d_\Delta(p,\cdot)|<1$, exact on the geodesic side $[a,b]$, and $\ge R-1$ on $Q$. Pick a level $t\in(R/3,2R/3)$ avoiding vertex values. Since $d_\Delta(p,a),d_\Delta(p,b)\ge R$, the level $h=t$ meets the boundary only on $[a,b]$, once on each side of $p$; its component $\gamma$ is an arc with both ends on $[a,b]$. For $y\in\gamma$: $R/3-1<d_\Delta(p,y)<2R/3+1$ and $d_\Delta(y,Q)>R/3-1>A$, so a nearest boundary point to $y$ lies on $[a,b]$ at distance $\le A$. As $y$ runs over $\gamma$, $d_\Delta(a,y)-d_\Delta(a,p)$ changes sign (the ends lie on opposite sides of $p$), so some $y$ has $d_\Delta(a,y)=d_\Delta(a,p)$; pick $z\in[a,b]$ with $d_\Delta(y,z)\le A$. Geodesicity of $[a,b]$ gives $d_\Delta(z,p)=|d_\Delta(a,z)-d_\Delta(a,y)|\le A$, whence $d_\Delta(p,y)\le2A$, contradicting $d_\Delta(p,y)>R/3-1>2A$. So $d_\Delta(p,Q)\le10(A+1)=\delta$, proving thinness.
\end{proof}

\begin{lemma}[Stability of controlled paths]\label{lem:path-stability}
Let $X$ have $\delta$-thin geodesic triangles, $\lambda\ge1$, $C\ge0$. There is $D=D(\delta,\lambda,C)$ such that any rectifiable $\sigma$ with $\length(\sigma|_{[u,v]})\le\lambda d(\sigma(u),\sigma(v))+C$ for all $u\le v$ lies in the $D$-neighborhood of a geodesic $J$ with the same endpoints, and $J$ lies in the $2D$-neighborhood of $\sigma$. Triangles whose three sides obey this subarc bound are $(3D+\delta)$-thin.
\end{lemma}
\begin{proof}
\emph{Projection.} If $R_0>4\delta$, $d(x,J),d(y,J)\ge R_0$, $d(x,y)\le R_0/2$, nearest points $x',y'$ satisfy $d(x',y')\le8\delta$: every $z\in[x',y']$ is within $2\delta$ of one of the other three sides of the geodesic quadrilateral $x,x',y',y$, cannot be near $[x,y]$ (else $d(x,J)\le R_0/2+2\delta<R_0$), and near $[x,x']$ or $[y,y']$ forces $d(z,x')\le4\delta$; apply to the midpoint.

\emph{Excursions.} Take $R_0=1+32\lambda\delta$, so $R_0>4\delta$ and $16\lambda\delta/R_0<1/2$. A component of $\sigma$ beyond distance $R_0$ from $J$ has both ends at distance $R_0$. Partition it by arclength into $\le1+2H/R_0$ pieces of length $\le R_0/2$, with $H$ its length; the projection inequality bounds the end projections' distance by $8\delta(1+2H/R_0)$, and the subarc bound gives $H\le\lambda(2R_0+8\delta)+\tfrac{16\lambda\delta}{R_0}H+C$, so $H\le2[\lambda(2R_0+8\delta)+C]$. Every excursion point is within $H/2$ of an end, so the path lies within $D=R_0+\lambda(2R_0+8\delta)+C$ of $J$. Conversely, for $q\in J$ choose $x\in\sigma$ with $d(s,x)=d(s,q)$ ($s$ the start); a nearest $x'\in J$ gives $d(q,x')=|d(s,x)-d(s,x')|\le D$, so $d(q,x)\le2D$. Replacing path sides by geodesics gives $(3D+\delta)$-thinness.
\end{proof}

\begin{proposition}[Transfer to the Cayley graph]\label{prop:word-hyperbolicity}
$G=\pi_1(K)$ is word-hyperbolic.
\end{proposition}
\begin{proof}
The strategy is a two-sided comparison between distances in reduced diagrams and word distance: expand each letter into a loop, draw a reduced diagram for the resulting nullhomotopic path, and convert uniform thinness of diagrams, via quasigeodesic stability, into uniform thinness of Cayley triangles.
There is $B_1$ with $d_{\Delta^{(1)}}(u,v)\le B_1d_\Delta(u,v)$ for diagram vertices: replace each straight piece through a triangle interior by a route over its boundary, of length $\le1/\sin(\omega/2)$ times the Euclidean distance across a corner of angle $\omega$ (law of cosines), with one factor $c$ over the finitely many model angles; combine partial edge traversals to a full-edge walk of length $\le c\,d_\Delta(u,v)$ and at most $c\,d_\Delta(u,v)/\ell_*$ edges ($\ell_*$ the minimal model edge length).

Fix $o$ and a spanning tree in the $1$-skeleton; for an oriented edge $e:u\to v$ take the element of $\tau_u e\tau_v^{-1}$ and let $S$ be the finite symmetric set of nonidentity values, generating $G$. Choose one edge loop per $s\in S$ bounding all their lengths by $B_2$. Expand a Cayley geodesic triangle $\mathrm{Cay}(G,S)$ letter by letter into these loops in $K$; their concatenation is a nullhomotopic edge path with a reduced diagram, and the letter transitions partition each side into loops of length $\le B_2$ whose marks are the successive group vertices.

For marks $u,v$ with labels $g_u,g_v$, $d_S(g_u,g_v)\le d_{\Delta^{(1)}}(u,v)\le B_1d_\Delta(u,v)$: any diagram edge path has the same displacement as the outside walk (they differ by a contractible edge loop), and inserting tree paths gives at most one letter of $S$ per edge. Along one geodesic side, the letter count between marks equals their group distance, so the expanded subarc between them has length $\le B_1B_2d_\Delta(u,v)$; extending an arbitrary subarc to enclosing marks adds at most $B_2$ at each end. Thus with $\lambda=\max\{1,B_1B_2\}$, $C=2\lambda B_2$, every subarc obeys the bound of Lemma~\ref{lem:path-stability}. Lemma~\ref{lem:diagram-thinness} and \ref{lem:path-stability} give thinness $T=3D+\delta$ of the expanded path triangle; a mark on a side is within $T+B_2$ of a mark on another side, so group vertices satisfy thinness $H_0=B_1(T+B_2)$, $H=H_0+1$. Geodesic polygons thinness and a reduction of arbitrary triangles to vertex polygons (replacing vertexless sides and tails by segments at distance $\le1$) finally give a uniform thinness constant $4H+2$ for all geodesic triangles in $\mathrm{Cay}(G,S)$, which is word-hyperbolicity.
\end{proof}

Together, Propositions~\ref{prop:closed-paths}, \ref{prop:no-torsion}, and \ref{prop:word-hyperbolicity} prove Theorem~\ref{thm:triangle-complex}, hence Theorem~\ref{thm:main}.
