\section{The big picture}\label{sec:picture}
% ============================================================

% This section assumes only that you know what an integral and a
% limit are. Everything else is built here. Read it twice.
%
% The whole paper is about ONE operator. It takes two functions
% F and G of two variables (x,y) and returns a third function of
% (x,y). The recipe: step from (x,y) in the horizontal direction
% through F, step from (x,y) in the vertical direction through G,
% multiply, and weight the step length t by 1/t. Then integrate
% over t. The question: is this operator stable under EVERY way
% of truncating the integration in t, including a way that changes
% from point (x,y) to point (x,y)? The answer is yes, at a cost
% that grows like the square root of the number of truncations.

\subsection{The Hilbert transform in one dimension}\label{subsec:hilbert-1d}

% Start in one variable. For a smooth function f that vanishes
% far away, the Hilbert transform of f is
\begin{equation}\label{eq:HT-1d}
 Hf(x)=\frac1\pi\,\operatorname{p.v.}\!\int_{-\infty}^{\infty}
        \frac{f(x+t)}{t}\,\dd t .
\end{equation}
% Read it out loud. At the point x, look at f at the point x+t,
% stepping t to the right. Weight the value by 1/t. Positive t
% contributes f(x+t)/t; negative t contributes the SAME sign in
% front, and 1/t already has the opposite sign for negative t.
% So the transform measures how lopsided f is around x: it is a
% machine for detecting antisymmetry. A constant has zero Hilbert
% transform. A symmetric bump has zero Hilbert transform.

% Two warnings about (1.1).
%
% (i)  The integral is singular. Near t=0, f(x+t)/t looks like
%      f(x)/t, which has infinite integral. The letters "p.v."
%      stand for principal value: integrate the truncated version
%      over eps<|t|<R and only THEN let eps go to zero and R go
%      to infinity. For smooth compactly supported f the limit
%      exists at every x.
%
% (ii) The transform is not bounded on L^1. It IS bounded on
%      every L^p with 1<p<infinity: a constant C_p satisfies
%          ||Hf||_p <= C_p ||f||_p.
%      "L^p" means the function has finite p-th power integral,
%          ||f||_p = (integral of |f|^p)^(1/p),
%      which you may think of as a measure of size with a knob p.
%      Large p emphasises peaks. Small p emphasises mass.

% The TRUNCATIONS are the objects we are going to "vary":
\begin{equation}\label{eq:HT-trunc}
 H_{\varepsilon,R}f(x)
   =\frac1\pi\int_{\varepsilon<|t|<R}\frac{f(x+t)}{t}\,\dd t ,
 \qquad 0<\varepsilon<R<\infty .
\end{equation}
% Even though Hf exists as a principal value, the truncated
% operators H_{eps,R} are far from innocent. As eps decreases to
% zero and R grows to infinity, the functions H_{eps,R}f do
% converge pointwise almost everywhere -- but proving this
% UNIFORMLY in x is a genuine theorem (a one-sided version of
% the classical Carleson theorem for the Hilbert transform).
% "Almost everywhere" means: for every x outside a set of
% measure zero.
%
% The present paper proves the analogue of that statement in TWO
% dimensions, for a bilinear operator, in the strongest possible
% form: not just pointwise convergence, but control of the total
% amount of oscillation across all truncations at once. That is
% the meaning of "variation".


\subsection{The bilinear Hilbert transform}\label{subsec:BHT}

% Before the two-dimensional version, there is a one-dimensional
% bilayer. Take two functions f and g of ONE variable. Send one
% input to the right and the other to the LEFT by the same amount
% t. Then weight by 1/t and integrate.
\begin{equation}\label{eq:BHT}
 \mathrm{BH}(f,g)(x)
 =\operatorname{p.v.}\!\int_{-\infty}^{\infty}
   f(x+t)\,g(x-t)\,\frac{\dd t}{t}.
\end{equation}
% This is the bilinear Hilbert transform. "Bilinear" means linear
% in f alone and linear in g alone, but not in f+g together. It
% was open for decades whether it is bounded on any nontrivial
% range of L^p spaces; Lacey and Thiele settled this in 1997.
% The relevant point for us: the two inputs move in OPPOSITE
% directions along the SAME axis. That opposition is what makes
% the analysis tractable.

\subsection{The triangular Hilbert transform}
\label{subsec:triangular}

% Now come to the operator of the paper. It is
% two-dimensional: F and G are functions of (x,y). Instead of
% moving both inputs along the same x-axis, we move them along
% ORTHOGONAL directions.
\begin{equation}\label{eq:triangular-def}
 B_{\varepsilon,R}(F,G)(x,y)
 =\int_{\varepsilon<|t|<R}
   F(x+t,y)\,G(x,y+t)\,\frac{\dd t}{t},
 \qquad 0<\varepsilon<R<\infty .
\end{equation}
% Read it out loud. Fix an output point (x,y). Step from (x,y)
% to (x+t,y): that is a HORIZONTAL move, and we read F there.
% Step from (x,y) to (x,y+t): that is a VERTICAL move, and we
% read G there. Multiply the two values, weight by 1/t, sum
% over t.
%
% Three points participate:
%    (x,y)       the corner,
%    (x+t,y)     the horizontal vertex,
%    (x,y+t)     the vertical vertex.
% They form a right triangle. The two inputs sit on the two legs
% that leave the corner. This is why the operator is called
% TRIANGULAR.
%
% Compare (1.3) to the bilinear Hilbert transform (1.2). In (1.2)
% both inputs lie on the SAME line. In (1.3) they lie on
% different lines. The one-dimensional opposition in (1.2) was a
% gift; the two-dimensional orthogonality in (1.3) does not give
% the same gift, and the operator is harder. This paper is the
% first proof that it behaves well at the symmetric endpoint
% (defined in the next subsection).

% There is a subtlety in the word "coherent". F(x+t,y) and
% G(x,y+t) can be highly aligned with the kernel 1/t for large
% stretches of t; only cancellations across changes of scale keep
% the operator bounded. In the classical Hilbert transform, those
% cancellations are packaged by orthogonality of frequencies.
% Here the two different directions break that orthogonality and
% a new mechanism is required. That mechanism is what the paper
% discovers.


\subsection{The scalar simplex form and the symmetric point}
\label{subsec:simplex}

% Dualise the bilinear operator into a TRILINEAR form: multiply
% the output of B by a third function G_0 and integrate.
% Holder's inequality, the standard pairing tool, says
%      ||B(F,G)||_p    <=  C ||F||_3 ||G||_3,
%      || G_0 B(F,G)||_1 <= ||G_0||_q ||B(F,G)||_p
% provided 1/q + 1/p = 1. If p=3/2 then q=3. So for a L^3
% basis, the natural companion G_0 is also in L^3. The scalar
% form we want is exactly
\begin{equation}\label{eq:scalar-annular}
 \Lambda_{\varepsilon,R}(G_0,G_1,G_2)
 =\iiint G_0(a,b)\,G_1(b,c)\,G_2(c,a)\,
    \mathbf 1_{\varepsilon<a+b+c<R}\,
    \frac{\dd a\,\dd b\,\dd c}{a+b+c}.
\end{equation}
% Where does (1.4) come from, and why does the kernel become
% 1/(a+b+c) instead of 1/t? Use the determinant-one substitution
%      a=-x,   b=-y,   t=a+b+c,
% whose inverse is   x=-a,   y=-b,   t=a+b+c.
% The three variables (a,b,c) satisfy c=t+x+y. Then
%      F(x+t,y)   becomes  G_1(b,c)=F(b+c,-b),
%      G(x,y+t)   becomes  G_2(c,a)=G(-a,c+a),
%      G_0(x,y)   becomes  G_0(a,b).
% Each of these maps (F,G) -> (G_1,G_2) and back has matrix
% determinant one, and preserves the L^3 norm. Computing from
% (1.3) yields (1.4): the compactly supported part of
% the kernel is 1/t=1/(a+b+c).
%
% The three inputs in (1.4) each see a DIFFERENT PAIR of the
% three scalar variables (a,b,c):
%      G_0 sees (a,b),
%      G_1 sees (b,c),
%      G_2 sees (c,a).
% The pairs form a CYCLE b->c->a->b. This is the "simplex":
% the graph with vertices a,b,c and edges the three pairs. No
% one input sees all three variables. That is what makes the
% operator "entangled" from the point of view of real-interpolation.

% Now for the exponents. Holder in three variables says:
% the bound
%      |\Lambda| <= C ||G_0||_{p_0} ||G_1||_{p_1} ||G_2||_{p_2}
% applies exactly when
%      1/p_0 + 1/p_1 + 1/p_2 = 1.
% That is a whole SURFACE of choices, not a single point.
%
% The SYMMETRIC POINT of that surface is the choice
%      p_0 = p_1 = p_2 = 3.
% It is the choice in which all three inputs are treated equally,
% with exponent 3 for each. Symmetric-point variation and
% symmetric-point principal values are the endpoint of the whole
% theory: they are what we would want if we did not already
% know a comfortable exponent tuple.

% Which of these was open in 2025?
%    - The lower exponents, p_0,p_1,p_2 close to 2, are easier
%      because there is more room in the Holder relation.
%    - The symmetric point p=3,3,3 is the HARDEST nontrivial
%      choice on the boundary: each input is exactly at the
%      point where the coarea/decomposition tools start to lose.
%    - Even harder is the VARIATION version of the estimate:
%      not just a single annulus eps<|t|<R, but a weighted sum
%      of many disjoint annuli (equation (1.7) below).
% This paper proves the hardest version.


\subsection{Variation}\label{subsec:variation-intro}

% The naive object is the single truncated operator B_{eps,R}.
% But we care about how far the family B_{eps,R} can wander as
% the truncation moves. For a real number r>2 (the "variation
% exponent"), define
\begin{equation}\label{eq:variation-def}
 V_r(F,G)(x,y)
 =\sup_{\substack{J\ge1\\0<t_0<\cdots<t_J\\t_j\in\mathbb Q}}
    \Big(\sum_{j=1}^{J}
     |B_{t_{j-1},t_j}(F,G)(x,y)|^r\Big)^{1/r}.
\end{equation}
% Read the outer object first. For EACH point (x,y), pick any
% partition 0<t_0<t_1<...<t_J of the positive real line, form
% the truncated operators on the slices (t_{j-1},t_j), take the
% absolute value of the scalar at (x,y), raise it to the r-th
% power, sum over the slices, take the r-th root, and THEN take
% the supremum over all partitions. The answer is a function of
% (x,y). The map F,G |-> V_r(F,G) is the r-th variation of the
% family B_{eps,R}.
%
% Two features matter.
%
% (i)  The partition is allowed to depend on (x,y). Different
%      points (x,y),(x',y') may use different partitions; the
%      supremum is taken pointwise FIRST and only afterwards
%      are the L^{3/2} norms taken.
%
% (ii) There is no limit on the number of intervals J. The
%      supremum is over every finite partition. We do not
%      prefer dyadic endpoints.
%
% These features are what make variation strictly stronger than
% pointwise convergence and stronger than the maximal estimate.

\subsection{The count estimate}\label{subsec:count-intro}

% Theorem A below follows from a quantitative estimate for
% finitely many annuli. For an integer n>=1, define
\begin{equation}\label{eq:count-def}
 \mathcal S_n(F,G)(x,y)
 =\sup\Big\{\sum_{j=1}^{m}|B_{\varepsilon_j,R_j}(F,G)(x,y)|
 : 0\le m\le n,\ 0<\varepsilon_j<R_j\in\mathbb Q,\
   (\varepsilon_j,R_j)\text{ pairwise disjoint}\Big\}.
\end{equation}
% Same story. For each point (x,y): pick at most n annuli
% eps_j<|t|<R_j that are pairwise disjoint (endpoints may touch;
% "disjoint" is about the interiors), form the truncated operator
% on each, take absolute values, sum. Take the supremum over all
% choices. The result is \mathcal S_n(F,G), a function of (x,y).
%
% The main theorem is the size bound on \mathcal S_n. It is a
% count estimate: it costs the number of annuli n but nothing
% else. From it, variation follows by a short combinatorial
% step (Section 1.7 below).

% Here is why the count estimate is the right reduction.
% From n annuli we lose sqrt(n). From the count estimate to
% VARIATION at exponent r>2, we sort the increments by size
% and bound the resulting series:
%
%      V_r(F,G)  <=  sum_j  2^{j(1/r-1/2)}  \mathcal S_{2^j}(F,G),
%
% and the series on the right converges whenever r>2. In
% particular, the count estimate on the right directly implies
% the full-variation estimate. The series converges because
% (1/r - 1/2) < 0.


\subsection{From count to variation}\label{subsec:count-to-variation}

% Why does the count estimate imply the variation estimate?
% Take any partition t_0<t_1<...<t_J at a point (x,y). Let
%      d_j = |B_{t_{j-1},t_j}(F,G)(x,y)|
% be the increment sizes. Sort them in decreasing order:
%      d_{(1)} >= d_{(2)} >= ... >= d_{(J)}.
% (The (j) is rank of the increment, not index of the interval;
% the procedure is a pointwise rearrangement.)
%
% For every rank 2^k at which the list still has an entry,
% the first 2^k increments together are a valid choice of 2^k
% pairwise disjoint intervals in the count supremum. Hence
%      d_{(2^k)} <= 2^{-k} * \mathcal S_{2^k}(F,G).
%
% The increments of ranks 2^k <= l < 2^{k+1} form a block of
% size 2^k, and each is at most d_{(2^k)}, so this block's
% l^r sum is at most 2^k * d_{(2^k)}^r, i.e. its l^r norm is at
% most 2^{k/r} d_{(2^k)}. Summing over k gives
\begin{equation}\label{eq:count-to-variation}
 V_r(F,G)
 \le \sum_{k\ge0} 2^{k(1/r-1/2)}\, \mathcal S_{2^k}(F,G)
 \quad\text{pointwise},
\end{equation}
% The coefficient 2^{k(1/r-1/2)} is positive and decreases
% geometrically when r>2, which is exactly the range we need
% for the series to converge. Applying the L^{3/2} norm,
% then Minkowski, and then the count estimate,
%      ||V_r(F,G)||_{3/2}
%      <= C ||F||_3 ||G||_3 \sum_k (1+k)2^{k(1/r-1/2)},
% and the geometric series is finite because 1/r-1/2<0. That is
% the whole content of Theorem B below, aside from measurability.

\section{Statements of the main theorems}\label{sec:statements}
% ============================================================

% Here is the exact content of the paper. Everything else is
% preparation or consequences. We use \R for the real numbers,
% \C for the complex numbers, L^p(\R^2) as usual, and
% ||f||_p for the L^p norm. All functions are complex-valued.

\subsection{The count estimate}

\begin{theorem}[Disjoint-annulus count estimate]\label{thm:count}
There is an absolute constant $C$ such that, for every integer
$n\ge1$ and every complex $F,G\in L^3(\R^2)$,
\begin{equation}\label{eq:count-bound}
 \|\mathcal S_n(F,G)\|_{3/2}
 \le C\sqrt n\,\log(2+n)\,\|F\|_3\,\|G\|_3 .
\end{equation}
\end{theorem}

% Read the right side. There are three factors.
%
%   ||F||_3 ||G||_3  is the size of the inputs. If F=0 or
%   G=0 then the left side is zero.
%
%   sqrt(n) is the price for the number of annuli. Doubling
%   the number of annuli costs at most sqrt(2) more.
%
%   log(2+n) is a negligible extra cost; it appears because we
%   allow arbitrary disjoint annuli, not just dyadic ones.
%
% The constant C is absolute: it does not depend on n, on F, G,
% on the annuli, on the point (x,y), or on anything else.
%
% The answer to "why sqrt(n)" is a rescaling trick. We
% normalize the third input G_0 to unit size, and we can split
% the whole energy of the operator across n annuli. Each annulus
% contributes a bounded amount per switch, but we can choose the
% switch thresholds adaptively. A duality rescaling by a factor
% n^{1/2} absorbs the linear count of switches into the trace
% of the energy (see the smooth-annuli section later).

\subsection{The variation estimate}

\begin{theorem}[Full annular variation]\label{thm:variation}
For every real $r>2$ there is a finite constant $C_r$ such that
\[
 \|V_r(F,G)\|_{3/2}
 \le C_r\,\|F\|_3\,\|G\|_3
\]
for all complex $F,G\in L^3(\R^2)$.
\end{theorem}

% Theorem~\ref{thm:variation} is the headline. It says the
% family B_{eps,R} has bounded r-variation for every r>2,
% uniformly. In particular (take r->infinity formally) it
% implies a maximal estimate, and it implies joint convergence.
% Both consequences are corollaries below.


\subsection{The maximal and principal-value corollaries}
\label{subsec:corollaries}

% Take a single increment in the partition: two endpoints. Then
% variation dominates the maximal operator over all endpoints:
\begin{corollary}[Maximal operator]\label{cor:maximal}
For complex $F,G\in L^3(\R^2)$, define
\[
 B_*(F,G)(x,y)
 =\sup_{0<\varepsilon<R<\infty}|B_{\varepsilon,R}(F,G)(x,y)|
\]
on a common full-measure set (on which every truncated operator
is defined), and set it to zero elsewhere. Then $B_*$ is
measurable and
\begin{equation}\label{eq:hard-maximal-bound}
 \|B_*(F,G)\|_{3/2}\le C_M\|F\|_3\|G\|_3
\end{equation}
with $C_M=C_2$ from Theorem~\ref{thm:variation}.
\end{corollary}

% The supremum is over BOTH hard truncation endpoints, not just
% the upper one. The constant C_M is absolute. From the maximal
% estimate the principal-value theorem follows in one more step.
\begin{corollary}[Joint principal value]\label{cor:bilinear-pv}
For every complex $F,G\in L^3(\R^2)$, the joint limit
\[
 B(F,G)=\lim_{\substack{\varepsilon\downarrow0\\R\uparrow\infty}}
                B_{\varepsilon,R}(F,G)
\]
exists almost everywhere and in $L^{3/2}$, and
$\|B(F,G)\|_{3/2}\le C_M\|F\|_3\|G\|_3$.
Moreover, the entire tail converges in the maximal sense:
\begin{equation}\label{eq:maximal-tail-convergence}
 \Big\|\sup_{\substack{0<\varepsilon<1/n\\R>n}}
 |B_{\varepsilon,R}(F,G)-B(F,G)|\Big\|_{3/2}
 \longrightarrow 0 .
\end{equation}
\end{corollary}

% The maximal operator controls the two-parameter (eps,R)
% truncations simultaneously: the SUPREMUM inside the norm on
% the left of (2.4) shrinks to zero as n grows. So the
% principle-value limit is not merely pointwise; it is
% dominated in L^{3/2} by an integrable envelope.
%
% Section 6 proves all of the above from the count estimate.
% Here we only stated them for context.

\section{Strategy: how a single-annulus bound becomes a count}
\label{sec:strategy}
% ============================================================

% The proof of Theorem~\ref{thm:count} has three layers, each
% built to solve a problem the one beneath it could not.
%
% 1. HEAT FLOW. Replace the hard annulus indicator by a smooth,
%    Gaussian-derived window. The hard truncation is recovered as
%    the endpoint of a Gaussian scale integral. Smooth windows
%    have the miraculous property that their second derivatives
%    are TRACES OF MATRICES with positive spectrum.
%
% 2. MATRIX ENERGY. Sample the trilinear form on a fine grid.
%    It becomes a trace of a product of three matrices times
%    diagonal insertions. The second derivative of the trace
%    trace(R^*R)^{3/2} in the insertion directions is a sum of
%    squares of entries (a Hessian form). A "mixed trace
%    inequality" bounds the cyclic product by those costs.
%
% 3. ROUGH KERNELS. Coming back to the actual annulus
%    indicator, the endpoint errors are bounded using the
%    frequency-block estimate, which exploits disjointness of
%    the annuli. Counting disjointness is exactly what the
%    sqrt(n) loss comes from.
%
% Here is the logic in one sentence: SMOOTHING COSTS A TRACE,
% DISJOINTNESS BOUNDS THE TRACE, AND RANK SORTING
% CONVERTS A COUNT INTO A VARIATION.


\subsection{The two difficulties, and where sqrt(n) comes from}
\label{subsec:difficulties}

% The proof has to handle TWO kinds of error, and each looks at
% first like it costs a factor of n.
%
% difficulty (S) -- SMOOTH SWITCHES. The smooth annuli
% \int_{eps^2}^{R^2} (k_s * k_s * k_s)(t) ds/s have a
% beginning and an end at each truncation endpoint. Summing
% over every interval, the number of switches is 2n at each
% output point. Naively the bound is n times something, not
% sqrt(n). Section~\ref{sec:smooth} shows how to rescale the
% third input to absorb this linear cost into a square root.
%
% difficulty (H) -- HARD ENDPOINT ERRORS. The gap between the
% smooth annulus and the hard Hilbert annulus is a sum of
% "endpoint error" kernels E_eps, one per endpoint. Naively
% there are 2n of them, which again looks linear. Section
% \ref{sec:rough} groups endpoints at comparable scales. Within
% a dyadic group, disjointness gives a uniform pointwise bound
% on the total, so each group behaves like ONE kernel. The
% frequency-block estimate (Section~\ref{sec:frequency}) divides
% the groups into two classes: those with many jumps (there are
% at most 2 sqrt(n) of them) and those with few (which are
% handled by the frequency estimate). The sqrt(n) is the price
% of the changeover.

% The point is that both difficulties are paid for ONCE by
% sqrt(n), and not by n. This is the content of the paper.

% ============================================================
\section{Preliminaries}\label{sec:preliminaries}
% ============================================================

% This section fixes three things: the representatives of
% truncated operators, the change of variables to simplex
% coordinates, and the step-function structure that makes the
% analysis measurable. Then it records the one-dimensional
% maximal estimates that will pay for the switch costs.

\subsection{Finite truncations}\label{subsec:truncations}

% The truncated operator B_{eps,R}(F,G) is a two-dimensional
% integral in t of a product, and it is not obvious whether it
% even makes sense at a given (x,y). The following lemma
% handles this. We call a "null set" a set of measure zero on
% R^2.
\begin{lemma}[Finite truncations]\label{lem:finite-truncations}
For $F,G\in L^3(\R^2)$, all finite annular integrals defining
$B_{\varepsilon,R}(F,G)$ are absolutely convergent outside one
common null set. On its complement they depend continuously on
$(\varepsilon,R)$ in $0<\varepsilon<R<\infty$. Changing the
representatives of $F$ or $G$ changes this family only on a
common null set. Moreover,
\begin{equation}\label{eq:finite-annulus-bound}
 \Big\|\int_{\varepsilon<|t|<R}
 |F(x+t,y)G(x,y+t)|\,\frac{\dd t}{|t|}\Big\|_{3/2}
 \le 2\log(R/\varepsilon)\|F\|_3\|G\|_3 .
\end{equation}
\end{lemma}
\begin{proof}
Fix $t$. Holder's inequality in $(x,y)$ and translation
invariance give
$\|F(\cdot+t,\cdot)G(\cdot,\cdot+t)\|_{3/2}\le\|F\|_3\|G\|_3$.
Minkowski's integral inequality yields \eqref{eq:finite-annulus-bound}.
Apply it to the countably many annuli $1/m<|t|<m$, $m\ge2$;
their union of exceptional null sets works for every finite
annulus. Absolute continuity of the $t$-integral gives endpoint
continuity. A zero-$L^3$-difference estimate on each of those
annuli proves the assertion about representatives.
\end{proof}

% Read (3.1) as a bookkeeping bound: on any finite annulus the
% truncated operator is dominated by $\log(R/\eps)$ times the
% product of input norms. The $\log$ is essential: without it the
% principal value would not be an L^{3/2}-bounded operator on the
% hard annulus. The lemma lets us use any representative we like;
% the truncation acts on L^3 as a well-defined continuous family
% of operators.
%
% In (1.7) the count supremum uses rational endpoints. By
% continuity in $(\varepsilon,R)$, rational endpoints exhaust all
% endpoints; we may (and will) replace rational by arbitrary
% throughout. Similarly, measurability of $B_*$ and $V_r$ comes
% from this exhaustion.


\subsection{Simplex coordinates}\label{subsec:coordinates}

% Most of the analysis is done in simplex coordinates. The
% change of variables below has absolute determinant one, so it
% preserves every L^p norm and commutes with the operator up to
% name changes.
\begin{equation}\label{eq:coords-change}
 x=-a,\quad y=-b,\quad t=a+b+c,
 \quad\text{with inverse}\quad
 a=-x,\quad b=-y,\quad c=t+x+y .
\end{equation}
% Pulled back along this map, the two inputs F and G become
\begin{equation}\label{eq:pair-maps}
 G_1(b,c)=F(b+c,-b),\qquad G_2(c,a)=G(-a,c+a).
\end{equation}
% Each of these two maps is a norm-preserving bijection of
% L^3(\R^2). Its companion change of the dual input is the
% identity: G_0(a,b) is just G_0(a,b). Combining (3.2) with
% (1.3) yields the simplex form (1.4). Thus the bilinear
% theorem on B_{eps,R} and the trilinear theorem on
% \Lambda_{eps,R} are equivalent, at matching constants.
%
% In particular, (1.4) is not merely a formal change of
% variables. It has the SAME optimal constant as the scalar
% flat form (Section 6). We mark this now because it justifies
% working only with the simplex coordinates from here on.

\subsection{Step choices}\label{subsec:step-choices}

% The count and variation suprema are taken pointwise. But the
% operators appearing under the supremum are not arbitrary
% functions of (a,b); they are built from a FINITE list of
% choices. To keep the analysis measurable and Riemann-summable
% we restrict to choices that are locally constant on a
% partition of \R^2 by finitely many horizontal and vertical
% lines.

\begin{definition}[Step choice]\label{def:step-choice}
A \emph{step choice} for the annulus is a choice of data
\[
 \text{ menu }=\big\{\,(\varepsilon_j^{(m)},R_j^{(m)},
                         \beta_j^{(m)})_{j=1,\ldots,n}
 \big\}_{m=1,\ldots,M}
\]
together with a partition of $\R^2$ into finitely many open
rectangles by horizontal and vertical lines, and a map from
rectangles to $m$. The bounded rectangle is where the operator
is genuinely varying; the values on the measure-zero boundaries
of the rectangles are immaterial.
\end{definition}

% In plain words: we pre-commit to a finite list of annuli
% choices, one per row of a big table. The row is chosen
% independently in each rectangle. In the interior of a rectangle
% the annuli and coefficients do not change as (a,b) varies.
%
% Why step choices? Three reasons.
%
% (1) Measurability. A measurable assignment of annuli to (a,b)
%     can be approximated in measure by a step choice. So proving
%     the bound for step choices with uniform constants gives the
%     bound for all measurable assignments by exhaustion.
%
% (2) Riemann sums. Fixing a step choice means we are reduced to
%     finitely many kernels. Riemann-summing against a smooth
%     companion amounts to a finite calculation.
%
% (3) Endpoints at cell boundaries are measure zero, and they
%     contribute nothing to the trilinear form.

% We will be careful about two points.
% - The cell choice can depend on (a,b) and on the specific
%   rectangular partition (of the OUTPUT point), but the finitely
%   many kernels under the sum are fixed once the partition is
%   fixed.
% - Nothing about the step choice needs to be the same at
%   different points (a,b), nor consistent across dyadic scales.


\subsection{Line maximal estimates}\label{subsec:line-maximal}

% Every jump in a matrix energy is controlled by a maximal
% average along a line. We record the two forms below. The
% constants are absolute.

% Let M_R and M_Z denote the CENTERED Hardy--Littlewood
% maximal operators on the real line and on the integer lattice
% respectively: for functions on R,
%      M_R f(x) = sup_{r>0} (2r)^{-1}\int_{x-r}^{x+r}|f(y)|dy,
% and for sequences on Z,
%      M_Z f(j) = sup_{m>=0} (2m+1)^{-1}\sum_{|l-j|<=m}|f(l)|.

\begin{lemma}[Line maximal estimates]\label{lem:line-maximal}
On either $\R$ or $\mathbb Z$, for $1<p<\infty$,
\begin{equation}\label{eq:line-maximal}
 \mu\{M f>\rho\}\le\frac3\rho\|f\|_1,
 \qquad
 \|Mf\|_p^p\le\frac{6p\,2^{p-1}}{p-1}\|f\|_p^p .
\end{equation}
Let $g_\sigma(t)=(2\pi\sigma)^{-1/2}e^{-t^2/(2\sigma)}$. There is
an absolute constant $C$ such that, uniformly in $\sigma$,
\begin{align}
 \sum_m h g_\sigma((m-j)h)|f(m)|
   &\le C M_{\mathbb Z}f(j), &&\sigma\ge h^2,
   \label{eq:discrete-gaussian-maximal}\\
 \int g_\sigma(t)|f(x-t)|\,\dd t
   &\le C M_{\R}f(x), &&\sigma>0 .
   \label{eq:continuous-gaussian-maximal}
\end{align}
\end{lemma}
\begin{proof}
The weak-type estimate: from any finite family of intervals,
greedily select the widest interval not already overlapping
another, and discard any covered by their triple enlargements;
this yields a disjoint sub-covered set of at most 3x the original
measure (also for integer intervals). Combining with the
truncation argument for strong $L^p$ gives
\eqref{eq:line-maximal}.

The Gaussian bounds follow from splitting the sum or integral
into shells $2^{k-1}a<|m-j|\le 2^ka$ with $a=\sqrt\sigma/h$.
Each shell contributes at most $C2^k a M f$, and the resulting
geometric series converges.
\end{proof}

% ============================================================
\section{Dimension-free matrix inequalities}\label{sec:matrix}
% ============================================================

% This is the linear-algebra engine of the proof. Everything
% below is finite-dimensional. The point of this section is to
% introduce a QUADRATIC COST attached to each possible direction
% of variation of a matrix trace, and to prove that this cost
% controls the trilinear form we want to estimate.

% Setup. Spaces $H_0,H_1,H_2$ are finite-dimensional complex
% Hilbert spaces. Maps $W_v:H_{v+1}\to H_v$ are linear (indices
% mod 3). The ROW STAR at vertex $v$ is
\begin{equation}\label{eq:row-star}
 R_v=[\,W_v\quad W_{v-1}^*\,]:H_{v+1}\oplus H_{v-1}\to H_v,
\end{equation}
% with two Gram matrices
\begin{equation}\label{eq:star-grams}
 T_v=R_v^*R_v,\qquad S_v=R_vR_v^* .
\end{equation}
% The two Grams have equal nonzero spectra, so their energies
% coincide:
\[
 \Phi(T_v)=\Phi(S_v),\qquad \Phi(P):=\tr(P^{3/2}) .
\]

% The energy objective is a smooth function of a positive
% semidefinite matrix $P\ge0$, with a natural Hessian cost.
% We first record the Hessian formula.
\begin{lemma}[Supported Hessian]\label{lem:hessian}
On the cone of matrices positive definite on a fixed support,
$\Phi(P)=\tr(P^{3/2})$ is smooth, and its first and second
derivatives at a supported direction $K$ (and at supported
directions $K,L$) are
\begin{equation}\label{eq:hessian}
 d\Phi_P(K)=\frac32\tr(\sqrt P\,K),\qquad
 d^2\Phi_P(K,L)=\mathcal B_P(K,L),
\end{equation}
where the Hessian form is
\begin{equation}\label{eq:hessian-form}
 \mathcal B_P(K,L)
 =\frac32\operatorname{Re}\big\langle K,
     (\mathsf L_{\sqrt P}+\mathsf R_{\sqrt P})^{-1}L\big\rangle_{\HS} .
\end{equation}
% Here \mathsf L_A(X)=AX and \mathsf R_A(X)=XA$ are the left and
% right multiplication operators. In a diagonal basis for $P$
% with positive eigenvalues $p_i$, we have
\begin{equation}\label{eq:hessian-metric}
 b_P(K)=\mathcal B_P(K,K)
 =\frac32\sum_{i,j}\frac{|K_{ij}|^2}{\sqrt{p_i}+\sqrt{p_j}} .
\end{equation}
% The form $b_P$ is symmetric positive-definite on supported
% Hermitian directions, and it satisfies
% $|\mathcal B_P(K,L)|^2\le b_P(K)b_P(L)$ (after Cauchy--Schwarz).
\end{lemma}
\begin{proof}
Restrict to the support $\supp P$ (where $P$ is positive
definite) and set $A=\sqrt P$. Differentiating the squaring map
$A\mapsto A^2$ at $A$ multiplies the $(i,j)$ entry by
$\sqrt{p_i}+\sqrt{p_j}>0$, so the inverse function theorem
gives a smooth positive square root. Then
$d\Phi_P(K)=\frac32\tr(AK)$ by the chain rule, and one more
differentiation gives (4.4)-(4.5). Diagonalising $A$ yields
(4.6). Cauchy--Schwarz on the Hilbert-Schmidt inner product
along eigenvectors of $\mathsf L_A+\mathsf R_A$ gives the
bound $|\mathcal B_P(K,L)|^2\le b_P(K)b_P(L)$ after noting that
$(\mathsf L_A+\mathsf R_A)^{-1}$ is positive definite on
supported Hermitian matrices.
\end{proof}

% The key use is a COMPARISON: if we enlarge the Gram, the
% cost shrinks. This is a monotonicity of the Hessian form,
% analogous to Loewner--Heinz monotonicity of the square root.


% The following lemma is the monotonicity of the Hessian cost
% under a Loewner-type order on the Gram.
\begin{lemma}[Order comparison]\label{lem:order}
If $P,Q\ge0$ on the same space, $a\ge1$, and $P\le aQ$, then
every Hermitian $K$ supported on $P$ is supported on $Q$, and
\begin{equation}\label{eq:matrix-order}
 b_P(K)\ge a^{-1/2}\,b_Q(K).
\end{equation}
\end{lemma}
\begin{proof}
$\ker Q\subseteq\ker P$ follows directly from $0\le P\le aQ$,
which gives the support containment. The order facts: spectrum
calculus and the integral representation
$\sqrt X=\pi^{-1}\int_0^\infty X(X+tI)^{-1}t^{-1/2}\dd t$
combined with inversion order
($0<X\le Y\Rightarrow Y^{-1}\le X^{-1}$) and monotonicity of
the multiplication operator $\mathsf L_A+\mathsf R_A$ in $A$
give $\sqrt{P+\eta I}\le \sqrt a\sqrt{Q+\eta I}$ for all $\eta>0$.
Then
$(\mathsf L_{\sqrt{P+\eta I}}+\mathsf R_{\sqrt{P+\eta I}})^{-1}$,
being the inverse of a positive operator on Hilbert--Schmidt
space, decreases as $\eta\downarrow0$ by monotonicity of
inversion. Taking $\eta\downarrow0$ proves (4.7).
\end{proof}

% The following lemma bounds a matrix cost of the form
% $R^*LR$ (Gram of $R$, insertion $L$) by the size of $L$
% times $\Phi(R^*R)$. It is the single most-used inequality in
% the paper.
\begin{lemma}[Inserted Gram bound]\label{lem:insertion}
For a linear map $R:E\to F$ and a Hermitian map $L:F\to F$,
the matrix $R^*LR$ is supported on $R^*R$, and
\begin{equation}\label{eq:insertion-bound}
 b_{R^*R}(R^*LR)
 \le \frac34\|L\|_{\op}^2\,\Phi(R^*R) .
\end{equation}
\end{lemma}
\begin{proof}
Let $\rho_i>0$ be the positive singular values of $R$ and
$C$ the compression of $L$ to the left singular subspace.
Then
$b_{R^*R}(R^*LR)
=\frac32\sum_{i,j}\frac{\rho_i^2\rho_j^2}{\rho_i+\rho_j}
 |C_{ij}|^2$.
The pointwise inequality
$x^2y^2/(x+y)\le (x^3+y^3)/4$ for $x,y>0$ and the fact that
each row (and column) of $C$ has squared $\ell^2$ norm at most
$\|L\|_{\op}^2$ yield
$\le\frac34\|L\|_{\op}^2\sum_i\rho_i^3$, which equals
the right side of (4.8).
\end{proof}

% Finally, if we form a PROBABILITY AVERAGE of Grams and their
% insertions, the cost can only decrease. This is what lets
% us average over Gaussian center shifts.
\begin{lemma}[Joint convexity]\label{lem:joint-convexity}
Let $E$ be a fixed subspace and $T_\omega\ge0$, $K_\omega$
measurable, $\supp T_\omega=E$, $K_\omega$ supported on
$T_\omega$, and $\int(\|T_\omega\|_{\op}+\|K_\omega\|_{\op})\dd\nu<\infty$.
Then $\overline T=\int T_\omega\dd\nu$ has support $E$,
$\overline K=\int K_\omega\dd\nu$ is supported on $\overline T$,
and
\begin{equation}\label{eq:joint-convexity}
 b_{\overline T}(\overline K)
 \le\int b_{T_\omega}(K_\omega)\,\dd\nu(\omega) .
\end{equation}
\end{lemma}
\begin{proof}
For any $x\in E$, $\langle\overline T x,x\rangle>0$, so
$\overline T$ is positive definite on $E$, and $K$ is supported
on it. Put $A_\omega=\sqrt{T_\omega}$ and $A=\int A_\omega\dd\nu$.
Cauchy--Schwarz in $\omega$ gives $A^2\le\overline T$. By
$\sqrt X$ monotonicity and
$\langle K,\mathsf M^{-1}K\rangle
=\sup_X\{2\langle K,X\rangle-\langle X,\mathsf M X\rangle\}$
(varational formula for a positive selfadjoint operator), the
cost at $(\overline T,\overline K)$ is controlled by the average
of the costs at $(T_\omega,K_\omega)$.
\end{proof}


% The Hessian cost is defined on a fixed support. But our
% matrices CHANGE rank: when entries are switched on or off
% they go from zero to nonzero. The following lemma covers
% this case: it is a first-derivative formula valid across
% rank changes.

\begin{lemma}[Rank changes and entrywise derivatives]\label{lem:rank-change}
For rectangular $R_-,R_+$ of the same size, set
$R(z)=(1-z)R_-+zR_+$ and $D=R_+-R_-$. Then
\begin{equation}\label{eq:rank-change-derivative}
 \Phi(R_+^*R_+)-\Phi(R_-^*R_-)
 =3\int_0^1\operatorname{Re}\langle R(z)\sqrt{R(z)^*R(z)},D\rangle_{\HS}\,\dd z .
\end{equation}
For every $(i,j)$,
\begin{equation}\label{eq:entry-gradient}
 |(R\sqrt{R^*R})_{ij}|
 \le\|\operatorname{row}_iR\|_2\|\operatorname{col}_jR\|_2 .
\end{equation}
\end{lemma}
\begin{proof}
Approximate by $\sqrt{R(z)^*R(z)+\eta I}$ and let $\eta\downarrow0$.
The main claim is the spectral-calculus continuity
$\|\sqrt{R^*R+\eta I}-\sqrt{R^*R}\|_{\op}\le\sqrt\eta$.
Then (4.9) follows from the fundamental theorem of calculus.
(4.10) is Cauchy--Schwarz in the matrix product.
\end{proof}

% The centrepiece of the section: a cyclic product of three
% matrices, each with a diagonal insertion, is dominated by
% the three Hessian costs.

\begin{lemma}[Mixed trace inequality]\label{lem:mixed-trace}
For Hermitian maps $D_v:H_v\to H_v$ put
$D_*=\max_v\|D_v\|_{\op}$. Then
\begin{equation}\label{eq:mixed-trace}
 |\tr(D_0W_0D_1W_1D_2W_2)|
 \le\frac53 D_*\sum_{v=0}^2 b_{T_v}(R_v^*D_vR_v) .
\end{equation}
The constant is independent of dimensions, and no Gram is
assumed invertible.
\end{lemma}
\begin{proof}
The proof proceeds in three stages.
%
First, package the three costs into one quadratic form.
Let $H=H_0\oplus H_1\oplus H_2$ and set
\[
 M=\begin{pmatrix}0&W_0&W_2^*\\W_0^*&0&W_1\\W_2&W_1^*&0\end{pmatrix},
 \quad D=\operatorname{diag}(D_0,D_1,D_2).
\]
If $P_v$ projects onto $H_v$ and $A_v=MP_vM$, $K_v=MP_vDM$,
then after permuting blocks these equal $T_v$ and $R_v^*D_vR_v$,
and $K_v$ is supported on $A_v\le M^2$. If $b$ denotes $b_{M^2}$
on $H$, then
\begin{equation}\label{eq:trace-common-metric}
 \mathcal E:=\sum_v b_{T_v}(R_v^*D_vR_v)
 \ge\sum_v b_{M^2}(K_v)\ge\tfrac13 b_{M^2}(MDM).
\end{equation}
%
Second, bound a cubic trace using the eigenvalue structure
of $M$: in a diagonal eigenbasis $m_i$ of $M$ with $q_i=|m_i|$
nonincreasing, formula (4.6) yields
$\mathcal E\ge\frac12 Q$ with
\[
 Q:=\sum_{i,j:q_i+q_j>0}\frac{q_i^2q_j^2}{q_i+q_j}|D_{ij}|^2 .
\]
Set
$A_i=q_i^3|D_{ii}|^2$, $E_i=q_i\sum_{j>i}q_j^2|D_{ij}|^2$,
$A=\sum A_i$, $E=\sum E_i$. Then $Q\ge A/2+E$ (Hermitian
symmetry with $q_j\le q_i$ for $j>i$).
Now group the cubic trace $\tr((DM)^3)$ by the smallest index
of each triple:
%
(case 1) smallest index once: the contribution is at most
$3D_*E_i$ by Cauchy--Schwarz in the off-diagonal direction.
% (case 2) smallest index twice: the contribution is at most
$3D_*\sqrt{A_iE_i}$ (Cauchy--Schwarz with
$\sum_k|D_{ik}|^2\le D_*^2$).
% (case 3) all three indices equal: modulus at most $D_*A_i$.
%
Summing and using Cauchy--Schwarz,
\[
 |\tr((DM)^3)|
 \le D_*(A+3E+3\sqrt{AE})
 \le D_*(\tfrac52A+\tfrac92E)
 \le 5D_*(\tfrac A2+E)
 \le 5D_*Q .
\]
%
Third, recover the complex phase. Set
$\tau=\tr(D_0W_0D_1W_1D_2W_2)$. The matrix $M$ has zero
diagonal blocks, so each closed three-step block product visits
all three vertices. The three starting points in one
orientation give $\tau$, in the other give $\overline\tau$, so
$\tr((DM)^3)=6\operatorname{Re}\tau$. If $\tau\ne0$ choose
$|z|=1$ with $z\tau=|\tau|$ and replace $W_0\to zW_0$; this
multiplies $R_0,R_1$ on the right by unitary phase diagonals,
preserving every cost. Applying the previous bound gives
$6|\tau|\le5D_*Q\le10D_*\mathcal E$, which is (4.11).
\end{proof}

% For complex $D_v$, split $D_v=D_v^{(0)}+iD_v^{(1)}$ into
% real and imaginary parts, both Hermitian, of operator norm at
% most $\|D_v\|_{\op}$. Expanding gives eight products with
% Hermitian diagonals. Each is bounded by (4.11) with $D_*$
% unchanged; each cost occurs four times across the eight.
% The result is
\begin{equation}\label{eq:complex-mixed-trace}
 |\tr(D_0W_0D_1W_1D_2W_2)|
 \le\frac{20}{3}D_*\sum_{v=0}^2\sum_{\epsilon=0}^1
   b_{T_v}(R_v^*D_v^{(\epsilon)}R_v) .
\end{equation}
% All directions on the right remain Hermitian.

% That concludes the dimension-free matrix engine. What matters
% for the sequel: the cyclic product is controlled by a sum of
% three positive quadratic costs, one for each vertex.

\section{Gaussian heat flow}\label{sec:heat}
% ============================================================

% The matrix costs from Section~\ref{sec:matrix} are second
% derivatives. The heat flow turns those second derivatives
% into derivatives in a scale parameter s, so that integrating
% over s controls the accumulated costs by initial and final
% energies. This is the "energy method": we pay the quadratic
% costs by heat dissipation.

% Setup. Fix a finite collection of vertices u_i=ih with
% -N<=i<=N and h>0. Vertex indices v range over Z/3. Edge
% arrays A_v(i,j) are finite, and we set n_v = (sum h^2
% |A_v(i,j)|^3)^{1/3}. Centers p_v are real; variances sigma_v
% positive. Weights:
\begin{equation}\label{eq:weights}
 g_\sigma(t)=(2\pi\sigma)^{-1/2}e^{-t^2/(2\sigma)},\qquad
 w_v(i)=h g_{\sigma_v}(u_i-p_v),\qquad
 W_v(i,j)=A_v(i,j)\sqrt{w_v(i)w_{v+1}(j)} .
\end{equation}
% The star at vertex v has two blocks
\begin{equation}\label{eq:heat-blocks}
 C_{v,v+1}=W_v,\quad C_{v,v-1}=W_{v-1}^*,\quad
 R_v=[C_{v,v+1}\ C_{v,v-1}] ,
\end{equation}
% and the weights enter two matrix cubes through
$e_v=\Phi(T_v)=\Phi(S_v)$ on the corresponding Gram.
% The Gaussian scores and their insertions are
\begin{equation}\label{eq:heat-scores}
 N_v=\operatorname{diag}_i\frac{u_i-p_v}{\sigma_v},\quad
 U_v=R_v^*N_vR_v,\quad
 P_{v,w}=C_{v,w}N_wC_{v,w}^* .
\end{equation}
% The QUADRATIC COSTS we want to dissipate are
\begin{equation}\label{eq:heat-costs}
 V_v=b_{T_v}(U_v),\quad
 Y_{v,w}=b_{S_v}(P_{v,w}),\quad
 Z_v=\mathcal B_{S_v}(P_{v,v+1},P_{v,v-1}) .
\end{equation}
% All insertions are supported on their own Gram, since a
% sandwiched matrix $B^*DB$ annihilates $\ker B$ both ways.

% The plane over which we integrate is $\Pi_d=\{p_0+p_1+p_2=d\}$,
% with measure inherited from any pair of centers (both changed
% to be the free coordinates; the Jacobian is always one).

% We first record the heat identities. These relate the
% derivative in s to center derivatives and the costs.
\begin{lemma}[Heat identities]\label{lem:heat-identities}
For fixed finite arrays and any positive rates $\alpha_v>0$,
evaluated at $\sigma_i=\alpha_is$, $e_v$ is smooth in the
centers and in $s$. On $\Pi_d$,
\begin{equation}\label{eq:heat-identity}
 2\partial_se_v=\sum_i\alpha_i\partial_{p_i}^2e_v
   -\alpha_vV_v-\sum_{w\ne v}\alpha_wY_{v,w} ,
\end{equation}
and
\begin{equation}\label{eq:heat-mixed}
 \partial_{p_{v+1}}\partial_{p_{v-1}}e_v=Z_v .
\end{equation}
The plane integrals $J_v(d,s)=\int_{\Pi_d}e_v\,\dd\pi_d$ are
finite, smooth, and satisfy
\begin{equation}\label{eq:heat-plane-identities}
 \begin{aligned}
 2\partial_sJ_v
 &=\Sigma_\alpha\partial_d^2J_v
   -\int_{\Pi_d}\Big(\alpha_vV_v+\sum_{w\ne v}\alpha_wY_{v,w}\Big)\dd\pi_d,\\
 \partial_d^2J_v&=\int_{\Pi_d}Z_v\,\dd\pi_d ,
 \end{aligned}
\end{equation}
with $\Sigma_\alpha=\sum\alpha_i$. On any compact
$s$-interval $[s_0,s_1]\subset(0,\infty)$, every such plane
integral grows at most polynomially in $|d|$.
\end{lemma}
\begin{proof}
The support issue: each star factors as
$R_v=D_{\rm row}R_v^0D_{\rm col}$ with positive invertible
diagonal multipliers, so $\ker R_v$, $\supp T_v$,
$\ran R_v$, $\supp S_v$ are locally fixed. Smoothness reduces
to smoothness of the positive eigenvalues of a product of two
positive matrices, which is standard. The rank-zero case is
trivial.

The Gaussian identities
$\partial_pw=Nw$, $\partial_p^2w=(N^2-\sigma^{-1})w$,
$2\partial_\sigma w=\partial_p^2 w$ combined with the chain
rule on $e_v$, restricted to its support, give (5.6) and (5.7).
Dominating by polynomial times $e_v$ and using
$\int_{\Pi_d}e_v\,\dd\pi_d\le$ polynomial in $|d|$, we can
differentiate under the integral. Choosing $p_i$ as dependent
coordinate yields $\partial_d^2J_v=\int\partial_{p_i}^2e_v\dd\pi_d$;
choosing $p_{v+1}$ and then $p_{v-1}$ yields
$\partial_d^2J_v=\int\partial_{p_{v+1}}\partial_{p_{v-1}}e_v\dd\pi_d$.
Then (5.6) and (5.7) give (5.8).
\end{proof}


\subsection{Strict dissipation at comparable variances}
\label{subsec:dissipation}

% The mixed term $Z_v$ does not have a sign. We trade it for
% column costs and then compare them to the row costs at the
% neighboring vertex. The loss is $\sqrt2$, small enough that
% strict dissipation survives when the three variance rates
% are comparable.

\begin{lemma}[Plane dissipation]\label{lem:plane-dissipation}
For finite arrays and any positive rates,
\begin{equation}\label{eq:heat-plane-dissipation}
 2\partial_sJ(d,s)\le\int_{\Pi_d}\sum_w
 \Big[-\alpha_w+\sqrt2\max\{0,\Sigma_\alpha/2-\alpha_w\}\Big]
 V_w\,\dd\pi_d .
\end{equation}
In particular, put $\lambda=11/10$. For each permutation of
$(1,\lambda,\lambda)$,
\begin{equation}\label{eq:heat-strict}
 -\partial_sJ(d,s)\ge c_*\int_{\Pi_d}\sum_wV_w\,\dd\pi_d,
 \qquad c_*=\tfrac12\min\{1-\tfrac35\sqrt2,\tfrac{11}{10}-\tfrac12\sqrt2\}>0 .
\end{equation}
\end{lemma}
\begin{proof}
The core inequality is the neighbor comparison
\begin{equation}\label{eq:heat-neighbors}
 Y_{w+1,w}+Y_{w-1,w}\le\sqrt2\,V_w .
\end{equation}
Let $A=W_w$, $B=W_{w-1}^*$, $N=N_w$. The elementary inequality
$\|Az+Bz'\|^2\le 2\|Az\|^2+2\|Bz'\|^2$ implies
$T_w\le2Q$ with $Q=\operatorname{diag}(A^*A,B^*B)$. By
Lemma~\ref{lem:order},
$V_w\ge2^{-1/2}b_Q([A\ B]^*N[A\ B])$. In a block-preserving
eigenbasis of $Q$, discarding off-diagonal blocks gives
$b_Q([A\ B]^*N[A\ B])\ge b_{A^*A}(A^*NA)+b_{B^*B}(B^*NB)$.
Then $A^*A\le S_{w+1}$ with insertion $P_{w+1,w}$, and
$B^*B\le S_{w-1}$ with insertion $P_{w-1,w}$. A second use of
Lemma~\ref{lem:order} with comparison factor one proves
(5.10). Summing (5.8) over $v$ and using
$2Z_v\le Y_{v,v+1}+Y_{v,v-1}$ gives (5.9); the two column
costs carrying the score at vertex $w$ have common coefficient
$\Sigma_\alpha/2-\alpha_w$. For the stated rates $\Sigma=16/5$;
the two coefficients are $-1+(3/5)\sqrt2$ and $-11/10+(1/2)\sqrt2$,
both negative. This gives (5.10).
\end{proof}

\subsection{Energy bounds and single-edge dissipation}
\label{subsec:energy-bounds}

% The initial energy is bounded by the discrete $L^3$ norms
% once the Gaussian variance is at least the mesh square.
\begin{lemma}[Initial energy]\label{lem:initial-energy}
For all $d\in\R$ and all variances $\sigma_v\ge h^2$,
\begin{equation}\label{eq:initial-energy}
 \begin{gathered}
 \int_{\Pi_d}\|W_v\|_{\HS}^3\,\dd\pi_d\le M n_v^3 ,
 \qquad M=1+\sqrt{2/\pi},\\
 \int_{\Pi_d}\sum_ve_v\,\dd\pi_d\le2\sqrt2\,M\sum_v n_v^3 .
 \end{gathered}
\end{equation}
The bounds also hold after replacing entries by smaller ones
in absolute value.
\end{lemma}
\begin{proof}
The Gaussian grid mass obeys
\begin{equation}\label{eq:heat-mesh-mass}
 \sum_i h g_\sigma(u_i-p)\le 1+\frac{2h}{\sqrt{2\pi\sigma}} .
\end{equation}
For $\sigma\ge h^2$, the right side is at most $M$. Weighted
Holder gives
$\|W_v\|_{\HS}^3\le M\sum|A_v|^3w_vw_{v+1}$.
Using $p_v,p_{v+1}$ as free coordinates on $\Pi_d$ and
integrating gives the first estimate; summing over $v$ and
double-counting each edge once in each star gives the second.
\end{proof}

% When variance rates become very unequal, summed star
% dissipation no longer holds. But a SINGLE EDGE still
% dissipates exactly, because its star has no mixed term.
\begin{lemma}[Single-edge energy]\label{lem:single-edge-energy}
Fix $v\ne w$ and retain only the weighted block $C=C_{v,w}$
at vertex $v$. For arbitrary positive rates, set
\[
 e^e_{v,w}=\Phi(CC^*), \quad
 V^e_{v,w}=b_{C^*C}(C^*N_vC), \quad
 Y^e_{v,w}=b_{CC^*}(CN_wC^*),
\]
\[
 J^e_{v,w}(s)=\iint_{\R^2}e^e_{v,w}\,\dd p_v\dd p_w .
\]
The plane integrals of $e^e,V^e,Y^e$ are independent of $d$,
and
\begin{equation}\label{eq:heat-single-edge}
 -2\partial_sJ^e_{v,w}=\iint_{\R^2}
   (\alpha_vV^e_{v,w}+\alpha_wY^e_{v,w})\,\dd p_v\dd p_w .
\end{equation}
With $n_e$ the edge array's $L^3$ norm, whenever
$\alpha_vs_0,\alpha_ws_0\ge h^2$ and $s_1>s_0$,
\begin{equation}\label{eq:heat-single-edge-integral}
 \int_{s_0}^{s_1}\iint_{\R^2}
  (\alpha_vV^e_{v,w}+\alpha_wY^e_{v,w})\,\dd p_v\dd p_w\dd s
 \le2M n_e^3 .
\end{equation}
For the full two-block star, $Y_{v,w}\le Y^e_{v,w}$.
\end{lemma}
\begin{proof}
The integrands depend only on the two endpoint centers; use
those as free coordinates. In the one-edge star the other
column block is zero, so $Z_v=0$. After permuting blocks the
matrices are
$R=[C\ 0]$, $T=\operatorname{diag}(C^*C,0)$,
$U=\operatorname{diag}(C^*N_vC,0)$, $S=CC^*$. The Hessian
costs are exactly $V^e_{v,w}$ and $Y^e_{v,w}$; (5.8) gives
(5.12). Since $e^e_{v,w}\le\|C\|_{\HS}^3$, the initial-energy
lemma bounds $J^e_{v,w}(s_0)\le Mn_e^3$; integrate (5.12) and
use that the terminal energy is nonnegative. Finally $CC^*\le S_v$
and $CN_wC^*$ is supported on $CC^*$, so Lemma~\ref{lem:order}
with factor one bounds $Y_{v,w}\le Y^e_{v,w}$.
\end{proof}

% is complete. We can now turn to the main smooth-annulus
% argument.


% ============================================================
\section{Smooth annuli and repeated masks}\label{sec:smooth}
% ============================================================

% We prove the smooth-annulus version of the count estimate.
% Same structure: at most n disjoint scale intervals at each
% output point, complex coefficients $|\beta|\le1$, smooth
% derivative windows. The result costs $\sqrt n$ but no more.

% Define the smooth kernels and their scale-integrals by
\begin{equation}\label{eq:smooth-kernels}
 k_s(t)=\frac{t}{\sqrt s}g_s(t),\qquad
 I_{\varepsilon,R}(t)
   =\int_{\varepsilon^2}^{R^2}(k_s*k_s*k_s)(t)\,\frac{\dd s}{s} .
\end{equation}

\begin{proposition}[Smooth annuli with repeated choices]
\label{prop:smooth-switches}
Let $G_v\in C_c^\infty(\R^2)$ be complex. At each output point
$(a,b)$, make a step choice of at most $n\ge1$ disjoint radius
intervals $(\varepsilon_j,R_j)$ and coefficients $|\beta_j|\le1$.
Then for $\mathcal K_{a,b}(t)=\sum_j\beta_jI_{\varepsilon_j,R_j}(t)$,
\[
 |\Lambda_{\mathcal K}(G_0,G_1,G_2)|
 \le C\sqrt n\prod_{v=0}^2\|G_v\|_3 .
\]
The constant is absolute and independent of the step partition,
the menu, and all endpoints.
\end{proposition}

\subsection{The cost of the switches}
\label{subsec:switch-cost}

% Work on the finite grid of the heat section. The masked dual
% array (choosing which annuli are active at each (i,j))
% is
\begin{equation}\label{eq:repeated-mask}
 A_0^s(i,j)=A_0(i,j)\sum_\ell\beta_\ell
      \mathbf 1_{\{\varepsilon_\ell^2<s<R_\ell^2\}} .
\end{equation}
% The sum runs through the list chosen at THIS index pair.
% Keep $A_1,A_2$ fixed.

% The key inputs are that the mask stays bounded by $|A_0|$ and
% that at every switch, the total variation is linear in n.
\begin{equation}\label{eq:mask-switch-count}
 |A_0^s(i,j)|\le|A_0(i,j)|,\qquad
 \sum_\tau|A_0^{\tau+}(i,j)-A_0^{\tau-}(i,j)|\le2n|A_0(i,j)|.
\end{equation}

\begin{lemma}[Repeated-mask dissipation]\label{lem:repeated-dissipation}
With masked arrays above and variances $\sigma_v=\alpha_vs$ where
$\boldsymbol\alpha$ is a permutation of $(1,11/10,11/10)$,
\begin{equation}\label{eq:repeated-dissipation}
 \int_{s_-}^{s_+}\int_{\Pi_0}\sum_vV_v\dd\pi_0\dd s
 \le C\Big(\sum_vn_v^3+n(n_0^3+n_0^2(n_1+n_2))\Big).
\end{equation}
\end{lemma}
\begin{proof}
Between consecutive switch scales the arrays are fixed, so
\eqref{eq:heat-strict} applies. The only new work is bounding
the total absolute energy jump.

At a switch $\tau$, freeze the weights and set
$\delta A(i,j)=A_0^{\tau+}(i,j)-A_0^{\tau-}(i,j)$.
Interpolate all changed entries linearly; convexity keeps
moduli bounded by $|A_0|$. For fixed centers and index, define
the abbreviations
$X^2=\sum_m|A_0(i,m)|^2w_1(m)$,
$Y^2=\sum_m|A_0(m,j)|^2w_0(m)$,
$H^2=\sum_m|A_2(m,i)|^2w_2(m)$,
$Q^2=\sum_m|A_1(j,m)|^2w_2(m)$.
In $R_0=[W_0\ W_2^*]$, the changing entry has row norm at most
$\sqrt{w_0(i)}(X+H)$, column norm $\sqrt{w_1(j)}Y$,
derivative $|\delta A(i,j)|\sqrt{w_0(i)w_1(j)}$; the adjoint
in $R_1=[W_1\ W_0^*]$ has row and column bounds
$\sqrt{w_1(j)}(Y+Q)$ and $\sqrt{w_0(i)}X$. The star at vertex 2
is untouched. Lemma~\ref{lem:rank-change} gives
\[
 |\Delta e_0|+|\Delta e_1|\le C\sum_{i,j}w_0(i)w_1(j)|\delta A(i,j)|
  (2XY+HY+QX).
\]
The measure $h^{-2}w_0(i)w_1(j)\dd\pi_0$ is a probability measure,
under which $p_0,p_1$ are independent normals with means
$u_i,u_j$ and variances $\sigma_0,\sigma_1$, and $p_2=-p_0-p_1$.
Adding Gaussian variances gives
$\mathbb E w_0(m)=hg_{2\sigma_0}(u_m-u_i)$,
$\mathbb E w_1(m)=hg_{2\sigma_1}(u_m-u_j)$,
$\mathbb E w_2(m)=hg_{\sigma_0+\sigma_1+\sigma_2}(u_m+u_i+u_j)$, and
Lemma~\ref{lem:line-maximal} bounds the $\mathbb E$ of
$X^2,Y^2,H^2,Q^2$ by constant multiples of
$X_*^2=M_{\mathbb Z}(|A_0(i,\cdot)|^2)(j)$,
$Y_*^2=M_{\mathbb Z}(|A_0(\cdot,j)|^2)(i)$,
$H_*^2=M_{\mathbb Z}(|A_2(\cdot,i)|^2)(-i-j)$,
$Q_*^2=M_{\mathbb Z}(|A_1(j,\cdot)|^2)(-i-j)$,
all independent of the switch scale. Cauchy--Schwarz under the
probability measure controls the three products. Applying the
$L^{3/2}$ line maximal inequality on slices gives (via
bijection $j\mapsto-i-j$ on relevant slices) that the three
$L^3$ norms of the majorants are at most $Cn_0,Cn_2,Cn_1$.
Integrating, summing the switches using
\eqref{eq:mask-switch-count}, and H\"older on the grid gives
the total jump cost
$\sum_\tau|\Delta\sum_vJ_v(0,\tau)|\le Cn(n_0^3+n_0^2(n_1+n_2))$.
Discarding the terminal energy and adding the initial-energy
bound from Lemma~\ref{lem:initial-energy} finishes the proof.
\end{proof}


\subsection{From the energy to the smooth kernel}
\label{subsec:energy-to-kernel}

\begin{proof}[Proof of Proposition~\ref{prop:smooth-switches}]
Put $\lambda=11/10$ and use base variances $\lambda s$ at every
vertex. In Lemma~\ref{lem:mixed-trace}, insert the diagonal matrix
with entries $k_s(u_i-p_v)/g_{\lambda s}(u_i-p_v)$ at vertex $v$.
It is Hermitian and uniformly bounded: the polynomial factor in
$k_s$ is absorbed by the extra Gaussian decay of $g_{\lambda s}$
relative to $k_s$.

To handle the cost at vertex $v$, narrow just its row variance to
$s$. Write $T_v^{\rm b}$ for the base Gram and $T_v^{\rm n}$ for
the narrowed-row Gram. Then
$(R_v^{\rm b})^*D_vR_v^{\rm b}=\sqrt s\,U_v^{\rm n}$ and
$T_v^{\rm n}\le\sqrt\lambda\,T_v^{\rm b}$. Row scaling preserves
support, so Lemma~\ref{lem:order} bounds the base cost by
$CsV_v^{\rm n}$. The narrowed variances are one of the
permutations allowed in Lemma~\ref{lem:repeated-dissipation}.

Integrate the mixed trace inequality over base centers in $\Pi_0$
and over $s$ with measure $\dd s/s$. Its right side is controlled
by \eqref{eq:repeated-dissipation}. On the left the density at
each vertex is $hk_s(u_i-p_v)$; integration over the center plane
convolves the three windows, giving
\[
 h^3\sum_{i,j,l}A_0(i,j)A_1(j,l)A_2(l,i)
    \sum_{\text{chosen at }(i,j)}\beta\,
    I_{\varepsilon,R}(u_i+u_j+u_l).
\]
If the original norms are nonzero, apply this to arrays normalized
to have norms $n^{-1/2},1,1$. The right side is bounded by an
absolute constant because $n(n^{-3/2}+2n^{-1})$ is bounded.
Rescaling gives $C\sqrt n\prod_vn_v$. The Riemann passage of the
preliminaries gives the proposition for smooth compactly supported
inputs.
\end{proof}

\subsection{The hard endpoint identity}
\label{subsec:hard-endpoint}

% Smooth annuli differ from hard Hilbert annuli by a
% difference of two odd endpoints. We record the exact
% identity now so that the next two sections can estimate
% the endpoint errors together.
%
% Put
%   psi = k_1*k_1*k_1 = -g_3''',
%   P(v) = 2int_0^v psi(z)dz   (v>=0),
%   c_0 = P(inf) = 2g_3''(0) != 0.
% Then P has a smooth even extension vanishing to second order
% at zero. For rho>0 define
\begin{equation}\label{eq:endpoint-kernel}
 E_\rho(t)=\frac{P(|t|/\rho)-c_0\mathbf 1_{\{|t|>\rho\}}}{t},
\end{equation}
% with the continuous value zero at $t=0$. Gaussian scaling and
% the substitution $z=|t|/\sqrt s$ in (6.1) give, away from
% the endpoint values,
\begin{equation}\label{eq:hard-smooth}
 I_{\varepsilon,R}(t)
 =\frac{c_0}{t}\mathbf 1_{\{\varepsilon<|t|<R\}}
   +E_\varepsilon(t)-E_R(t) .
\end{equation}
% This is the BRIDGE between smooth and hard annuli. The hard
% kernel is $c_0/t$ on the annulus $(\varepsilon,R)$; the
% two endpoint errors $E_\varepsilon$ and $E_R$ account for the
% Gaussian tail. Each $E_\rho$ is an odd dilate of one fixed
% kernel with two jumps at $\pm\rho$.

% So the hard count estimate reduces to two problems.
% (i) The smooth-annuli estimate of Proposition~\ref{prop:smooth-switches}
%     bounds the correct $c_0/t$ piece after adding the
%     endpoint errors.
% (ii) A frequency-block estimate, developed next, controls
%     the endpoint errors with square-root loss.


% ============================================================
\section{Uniform frequency-block estimate}\label{sec:frequency}
% ============================================================

% The second main new ingredient: a Fourier estimate for
% oscillatory Gaussian windows, with coefficients that can
% depend jointly on scale, frequency, and output point. Its
% decisive hypothesis is a POINTWISE square-integral bound over
% scales and frequencies. No sum of frequency suprema is
% required.

% Setup. Fix q>0, D>=1, and let Xi_D subset [-2D,2D] be
% measurable with measure dxi/D, so mu(Xi_D)<=4. The frequency
% windows and their convolutions are
\begin{equation}\label{eq:frequency-windows}
 \ell_\xi(z)=D^{-1}\frac{\dd}{\dd z}
                 \bigl(g_q(z)e^{i\xi z}\bigr),\qquad
 H_\xi=\ell_\xi*\ell_\xi*\ell_\xi .
\end{equation}
% The parameters q,D are suppressed in this notation.

\begin{proposition}[Frequency-block estimate]\label{prop:frequency-block}
Let $\mathcal I\subset\mathbb Z$ be finite and nonempty, and
$L_k=2^k$. Let $A_0,A_1,A_2$ be finite arrays on the lattice
$u_i=ih$ with the $L^3$ norms $n_v$ defined as in
Section~\ref{sec:heat}. Suppose measurable coefficients
$\mu_{ij,k}:\Xi_D\to\C$ satisfy
\begin{equation}\label{eq:frequency-coefficient-condition}
 |\mu_{ij,k}(\xi)|\le C_1,\qquad
 \sum_{k\in\mathcal I}\int_{\Xi_D}
  |\mu_{ij,k}(\xi)|^2\,\dd\gamma(\xi)\le C_1^2,
 \quad \dd\gamma=\dd\xi/D,
\end{equation}
for every index pair $(i,j)$, the first inequality holding
almost everywhere. If
\begin{equation}\label{eq:frequency-mesh}
 h^2\le\frac{q\min_{k\in\mathcal I}L_k^2}{16D^2},
\end{equation}
then
\begin{equation}\label{eq:frequency-finite-estimate}
 \Big|h^3\int_{\Xi_D}\sum_{k\in\mathcal I}\sum_{i,j,l}
 A_0(i,j)\mu_{ij,k}(\xi)A_1(j,l)A_2(l,i)
 \mathrm{Dil}_{L_k}H_\xi(u_i+u_j+u_l)\,\dd\gamma(\xi)\Big|
 \le C(q,C_1)\prod_{v=0}^2n_v .
\end{equation}
\end{proposition}

% There are two reasons for refining one Gaussian variance in the
% proof.
% (1) A window oscillating at frequency D becomes an average
%     of score insertions at variance comparable to D^{-2}s.
% (2) The boundary energy from changing the output array can
%     still be estimated uniformly at this small variance.
% The refined row scores are controlled by the heat identity
% through single-edge energies and changes of star energy at
% slab boundaries. Such a change contains the square of the
% changing edge norm times the norm of the unchanged edge; we
% isolate a uniform estimate for that mixed term first.

\begin{lemma}[Uniform mixed boundary estimate]
\label{lem:frequency-boundary}
Fix distinct $v,w\in\{0,1\}$, so $\{v,w\}=\{0,1\}$. The edge
$v-w$ has array $B$, indexed in original order $(0,1)$, and the
edge $v-2$ has array $E$, indexed in the order $(v,2)$. Put
variances $(\theta s,s,s)$ with $0<\theta\le1/16$ and
$h^2\le\theta s$. Let $\dd\nu_s(d)=g_{(1-\theta)s}(d)\dd d$.
There is a nonnegative array $\mathcal H_E(i,j)$, independent of
$(s,\theta)$, such that
\begin{align}
 &\int_{\R}\int_{\Pi_d}
   \|C_{v,w}\|_{\HS}^2\|C_{v,2}\|_{\HS}\,\dd\pi_d\dd\nu_s(d)
 \le C\sum h^2|B(i,j)|^2\mathcal H_E(i,j),\label{eq:frequency-boundary-mixed}\\
 &\Big(\sum h^2\mathcal H_E(i,j)^3\Big)^{1/3}
 \le C\Big(\sum h^2|E(m,l)|^3\Big)^{1/3}.\label{eq:frequency-boundary-norm}
\end{align}
Here $(i,j)$ are in the original endpoint order of edge $0$;
$i_v,i_w$ denote the same two indices in the order $(v,w)$. The
arrays $B,E$ are extended by zero on $\mathbb Z^2$.
\end{lemma}
\begin{proof}
Expand the first Hilbert--Schmidt squared. For each $(i,j)$,
$h^{-2}w_v(i_v)w_w(i_w)\dd\pi_d\dd\nu_s(d)$ is a probability
measure: with $p_v,p_w$ the free coordinates, it makes them
independent normals with means $u_{i_v},u_{i_w}$ and variances
$\theta s,s$, while $d\sim g_{(1-\theta)s}$ and $p_2=d-p_v-p_w$.

Under this measure, $\mathbb E h^{-2}w_v(m)w_2(l)$ is a bivariate
Gaussian density at $(u_m,u_l)$ with mean $(u_{i_v},-u_{i_v}-u_{i_w})$
and covariance
$\Sigma=s\big(\!\begin{smallmatrix}2\theta&-\theta\\-\theta&3\end{smallmatrix}\!\big)$.
The bounds $\Sigma\le4s\operatorname{diag}(\theta,1)$,
$\det\Sigma=\theta(6-\theta)s^2$ and the Gaussian density
formula give
\begin{equation}\label{eq:frequency-covariance-domination}
 \mathbb E[w_v(m)w_2(l)]
 \le Ch^2g_{C'\theta s}(u_m-u_{i_v})
                g_{C's}(u_l+u_{i_v}+u_{i_w})
\end{equation}
with $C'=4$ absolute.

Let $M_{\mathbb Z}^{(1)},M_{\mathbb Z}^{(2)}$ be line maximal
operators in the two coordinates. By Lemma~\ref{lem:line-maximal},
take
\begin{equation}\label{eq:frequency-boundary-majorant}
 \mathcal H_E(i,j)^2
 =C\bigl(M_{\mathbb Z}^{(1)}M_{\mathbb Z}^{(2)}|E|^2\bigr)
                 (i_v,-i_v-i_w).
\end{equation}
Then $\mathbb E\|C_{v,2}\|_{\HS}^2\le\mathcal H_E(i,j)^2$ by
(7.4) and $h^2\le\theta s$. Cauchy--Schwarz under the
probability measure gives (7.2). The map
$(i,j)\mapsto(i_v,-i_v-i_w)$ is a bijection $\mathbb Z^2$;
the $\ell^{3/2}$ bounds for the two line maximal operators
applied to $|E|^2$ give (7.3).
\end{proof}


\subsection{Proof of the frequency-block estimate}
\label{subsec:freq-proof}

\begin{proof}[Proof of Proposition~\ref{prop:frequency-block}]

\paragraph{Reduction to pointwise bounds.}
Discard a common null set of frequencies so that both bounds in
\eqref{eq:frequency-coefficient-condition} hold pointwise.
Set
\[
 B_{0,k}^\xi(i,j)=A_0(i,j)\mu_{ij,k}(\xi),\quad
 a_k=qL_k^2,\quad \theta=\frac1{16D^2}.
\]
For each $\xi$, use $B_{0,k}^\xi$ on the open slab
$2a_k<s<3a_k$, and zero outside; keep $A_1,A_2$ throughout.
Slabs are disjoint because $a_{k+1}=4a_k$. The mesh hypothesis
$h^2\le\theta s$ holds on the whole scale interval
$[s_-,s_+]=[\min a_k,4\max a_k]$.

\paragraph{Refining a row variance.}
Fix a slab, put $L=L_k$, $a=a_k$, $\omega=\xi/L$. The dilated
window is
\[
 f_{\xi,k}(u)=\Dil_L\ell_\xi(u)
   =\frac LD\frac{\dd}{\dd u}\bigl(g_a(u)e^{i\omega u}\bigr).
\]
Initially all three vertices have variance $s$. The diagonal
insertion with entries $f_{\xi,k}(u_i-p_v)/g_s(u_i-p_v)$ has
operator norm at most $C(q)$: $|L\omega/D|\le2$, and Gaussian
decay in $g_a/g_s$ kills the rest. Splitting into real and
imaginary parts gives eight terms, each a Hermitian insertion of
uniformly bounded norm (Lemma~\ref{lem:mixed-trace}).

Consider the row cost at $v$. Replace row variance $s$ by
$b'=\theta s$, keep column variances at $s$, and shift the row
center $p_v\to p_v+d$. Denote the refined Gram and score insertion
by $T_v^d,U_v^d$. Gaussian variance addition gives
\[
 T_v=\int_\R T_v^d\dd\nu_s(d),\qquad
 \dd\nu_s(d)=g_{s-b'}(d)\dd d .
\]
All these Grams share a common support (positive row scaling on
the kernel preserves the kernel of the row matrix).

To represent the inserted Gram in the same way, put
$\omega'=a\omega/(a-b')$. Since $b'<a/2$, completing the square
gives the exact identity
\[
 g_a(u)e^{i\omega u}
 =\exp\!\Big(\frac{\omega^2ab'}{2(a-b')}\Big)
   \int_\R g_{b'}(u-d)e^{i\omega'd}g_{a-b'}(d)\dd d .
\]
Differentiating in $u$, the derivative of $g_{b'}(u-d)$
supplies the negative refined row score $-(u-d)/b'$. The
prefactor is at most $C(q)$; also
$g_{a-b'}(d)\le Cg_{s-b'}(d)$ and $L/D\le C(q)\sqrt{\theta s}$.
Applying Lemma~\ref{lem:joint-convexity} and quadratic
homogeneity to the resulting average gives
\begin{equation}\label{eq:frequency-score-refinement}
 b_{T_v}(K_v)\le C(q)\theta s\int_\R
   V_v(p_0,\ldots,p_v+d,\ldots,p_2;s)\dd\nu_s(d).
\end{equation}
The refined variances are now $(\sigma_v,\sigma_w,\sigma_{v'})=(b',s,s)$
with $b'<a/2$.

\paragraph{Integrated refined scores.}
Fix $v$; all stars, energies, costs refer to the slab edge $0$.
For a center-dependent $Q$, abbreviate
$\widetilde Q(s)=\int_\R\int_{\Pi_d}Q(p,s)\dd\pi_d\dd\nu_s(d)$.
On any open scale interval with fixed arrays, the heat identities
give
\begin{equation}\label{eq:frequency-averaged-heat}
 2\partial_s\widetilde J_v=3\widetilde Z_v-\theta\widetilde V_v
  -\sum_{w\ne v}\widetilde Y_{v,w}.
\end{equation}
The coefficient $3$ comes from the original second derivative
$2+\theta$ in the plane, plus $1-\theta$ from differentiating
$\nu_s$ and integrating by parts. Dominating by polynomials
times $e_v$ and using Gaussian decay of $\nu_s$ justifies the
$s$-differentiation and the two integrations by parts, and gives
one-sided limits at slab boundaries.

The metric Cauchy--Schwarz inequality $2Z_v\le\sum_{w\ne v}Y_{v,w}$
gives
\begin{equation}\label{eq:frequency-score-differential}
 \theta\widetilde V_v
 \le-2\partial_s\widetilde J_v+\tfrac12\sum_{w\ne v}\widetilde Y_{v,w}.
\end{equation}
Pay column costs by single edges: $S_v\ge C_{v,w}C_{v,w}^*$, so
Lemma~\ref{lem:order} yields $Y_{v,w}\le Y^e_{v,w}$. The integrated
single-edge bound (5.13) controls the total of $Y^e_{v,w}$. If the
edge is a fixed one (say at vertices $1,2$), it costs at most
$C(n_1^3+n_2^3)$; if it is the slab edge $0$, it costs at most
$C\sum_k\sum_{i,j}h^2|B_{0,k}^\xi(i,j)|^3$, which after frequency
integration by \eqref{eq:frequency-coefficient-condition} is at
most $C(C_1)n_0^3$. All column terms are thus controlled, and the
coefficient $1$ of the column cost makes this independent of the
small variance ratio $\theta$.

\paragraph{Summing slab boundaries.}
Before the first slab, edge $0$ is zero, so the initial energy is
at most $C(n_1^3+n_2^3)$ by Lemma~\ref{lem:initial-energy}. At a
boundary of each slab, the change in the pointwise star energy at
$v\in\{0,1\}$ is at most
\[
 C\bigl(\|C_{\rm sw}\|_{\HS}^3+\|C_{\rm sw}\|_{\HS}^2\|C_{\rm fix}\|_{\HS}\bigr).
\]
The cubic term integrates by Lemma~\ref{lem:initial-energy} to at
most $C\sum_{i,j}k\sum h^2|B_{0,k}^\xi|^3$, i.e.\ at most
$C(C_1)n_0^3$ after frequency integration.

For the mixed term, let $A^{(v,2)}$ be the fixed unweighted array
on edge $v-2$, indexed in the order $(v,2)$. Apply
Lemma~\ref{lem:frequency-boundary} with this array; its majorant
$\mathcal H_v$ is independent of $(k,\xi,s)$, and its cube norm
is at most $C(n_1+n_2)$. Both boundaries together contribute at
most a constant times
\[
 \sum_{i,j}h^2|A_0(i,j)|^2\mathcal H_v(i,j)
       \sum_k\int_{\Xi_D}|\mu_{ij,k}|^2\dd\gamma
 \le C(C_1)n_0^2(n_1+n_2)
\]
by \eqref{eq:frequency-coefficient-condition} and H\"older. This
step is exactly where averaging coefficient squares over
frequency matters; taking a separate frequency supremum per scale
would not give the bound.

\paragraph{Recovering the frequency-block form.}
Integrate (7.7) on intervals between consecutive boundaries and
telescope. Discard the terminal energy (nonnegative). The
initial energy, the absolute jumps just estimated, and the
single-edge costs give
\begin{equation}\label{eq:frequency-integrated-score}
 \int_{\Xi_D}\int_{s_-}^{s_+}
   \theta\widetilde V_v\,\dd s\,\dd\gamma
 \le C(C_1)\sum_{j=0}^2n_j^3 .
\end{equation}
Values at the finitely many slab boundaries are immaterial.
Measurability in $\xi$ follows from measurability of the arrays
and Borel dependence of finite-matrix energies; Tonelli's theorem
justifies every nonnegative frequency integration above.

Return to equal-variance stars and apply Lemma~\ref{lem:mixed-trace}
to the real and imaginary parts of the three window insertions.
Integrate over base centers on $\Pi_0$. Shifting the row center in
(7.6) maps this plane to $\Pi_d$; the two column centers are free
coordinates, so its Jacobian is one. The cost at vertex $v$ is at
most $C(q)\theta s\widetilde V_v$; integrate over every slab with
$\dd s/s$ and over $\Xi_D$ with $\dd\gamma$; (7.8) bounds the sum.

On the other side, multiplying the three weighted windows in the
trace and integrating the centers gives exactly
\[
 h^3\sum_{i,j,l}B_{0,k}^\xi(i,j)A_1(j,l)A_2(l,i)
    \Dil_{L_k}H_\xi(u_i+u_j+u_l),
\]
the convolution identity on $\Pi_0$: the three window arguments
sum to $u_i+u_j+u_l$. The expression is independent of $s$
within its slab, whose logarithmic length is $\log(3/2)$.

This proves \eqref{eq:frequency-finite-estimate} with the right
side replaced by $C(q,C_1)\sum_vn_v^3$. If all $n_v>0$, apply
this to $A_v/n_v$; coefficients and mesh hypothesis are unchanged,
and trilinearity gives the product bound. If any norm is zero,
the form is zero.
\end{proof}


\subsection{Continuous version}

\begin{corollary}[Continuous frequency-block estimate]
\label{cor:frequency-continuum}
Let $q,D,\Xi_D,\mathcal I$ be as in Proposition~\ref{prop:frequency-block}.
Suppose $\mu_{a,b,k}(\xi)$ is constant in $(a,b)$ on every cell
of one finite rectangular partition, independent of $\xi$ and
common to all $k$. Suppose its values on cells are bounded
measurable functions of $\xi$ and that
\begin{equation}\label{eq:frequency-continuous-condition}
 |\mu_{a,b,k}(\xi)|\le C_1,\qquad
 \sum_k\int_{\Xi_D}|\mu_{a,b,k}(\xi)|^2\dd\gamma\le C_1^2 .
\end{equation}
Define
\begin{equation}\label{eq:frequency-continuous-kernel}
 \mathcal K_{a,b}(t)=\int_{\Xi_D}\sum_{k\in\mathcal I}
   \mu_{a,b,k}(\xi)\Dil_{L_k}H_\xi(t)\dd\gamma(\xi).
\end{equation}
Then for complex $G_v\in L^3(\R^2)$,
\begin{equation}\label{eq:frequency-continuous-estimate}
 |\Lambda_{\mathcal K}(G_0,G_1,G_2)|\le C(q,C_1)\prod_{v=0}^2\|G_v\|_3 .
\end{equation}
\end{corollary}
\begin{proof}
Take $G_v$ smooth compactly supported. On each step choice the
kernel (7.10) is continuous in $t$ (bounded block, bounded
measurable coefficients, Gaussian common majorant). After the
frequency integration, the trilinear integral is the limit of
its lattice Riemann sums. The finitely many rectangular step
boundaries have measure zero. The mesh condition (7.2) holds
for all sufficiently fine lattices, and the discrete norms
converge to $\|G_v\|_3$. Proposition~\ref{prop:frequency-block}
proves (7.11) for these inputs.

For general $L^3$ inputs, the fixed finite block and scale set
admit an integrable majorant $k_*(t)\ge|\mathcal K_{a,b}(t)|$,
uniform in $(a,b)$, obtained as a finite sum of
$\sup_{\xi\in\Xi_D}|H_\xi|$ dilates. For any integrable $k\ge0$,
\[
 \iiint|G_0(a,b)G_1(b,c)G_2(c,a)|k(a+b+c)\dd a\dd b\dd c
 \le\|k\|_1\prod_v\|G_v\|_3 .
\]
Set $t=a+b+c$, apply Holder in $(a,b)$ for fixed $t$, then use
the determinant-one changes of variables for the other two edge
functions. Thus the defining integral is absolutely convergent,
is continuous in the three $L^3$ inputs, and smooth
approximation finishes the proof. The final constant is
independent of $D$, of the number of scales, and of the step
choice.
\end{proof}

\section{Rough kernels and endpoint errors}\label{sec:rough}
% ============================================================

% The comparison (6.3) leaves one error kernel at each annular
% endpoint. Naively estimating them separately would cost the
% number of endpoints. We instead collect endpoints at
% comparable scales, so disjointness of the annuli gives a
% Gaussian bound independent of the number of discontinuities.
% The frequency-block estimate from the previous section
% controls the groups with few discontinuities; the remaining
% groups are few enough to estimate pointwise.

\subsection{A pointwise Gaussian estimate}
\label{subsec:gaussian-envelope}

% For a locally integrable function on R^2, let M_x and M_y
% denote the centered HL maximal operators in the two
% coordinates. For F,G in L^3, put
\begin{equation}\label{eq:rough-maximal-envelope}
 \mathcal M_{F,G}(x,y)
 =\bigl(M_x(|F|^{5/2})(x,y)\bigr)^{2/5}
  \bigl(M_y(|G|^{5/2})(x,y)\bigr)^{2/5}.
\end{equation}
% The one-dimensional maximal inequality on L^{6/5}, then
% Hölder, gives
\begin{equation}\label{eq:rough-maximal-norm}
 \|\mathcal M_{F,G}\|_{3/2}\le C\|F\|_3\|G\|_3 .
\end{equation}

\begin{lemma}[Gaussian envelope]\label{lem:gaussian-envelope}
Suppose $c>0$ and a measurable kernel $f$ satisfies
$|f(z)|\le e^{-cz^2}w(z)$ with $w\ge0$ and $\|w\|_5\le\delta$.
Then for every $L>0$,
\begin{equation}\label{eq:gaussian-envelope}
 \int_\R|F(x+t,y)G(x,y+t)|\,|\Dil_Lf(t)|\dd t
 \le C_c\delta\,\mathcal M_{F,G}(x,y)
\end{equation}
at every point where the right side is finite.
\end{lemma}
\begin{proof}
After setting $t=Lz$, apply Holder with exponents
$(5/2,5/2,5)$ to
$|F(x+Lz,y)|e^{-cz^2/2}$,
$|G(x,y+Lz)|e^{-cz^2/2}$, $w(z)$.
The first two resulting integrals are controlled by
$M_x(|F|^{5/2})$ and $M_y(|G|^{5/2})$, uniformly in $L$, by
splitting $\R$ into $|z|\le1$ and shells $2^j<|z|\le2^{j+1}$.
The last is $\|w\|_5\le\delta$.
\end{proof}

% In the triangular form, $G_1(b,c)=F(b+c,-b)$ and
% $G_2(c,a)=G(-a,c+a)$, both norm-preserving. With
% $t=a+b+c$, the c-integral is bounded by
% $C_c\delta\mathcal M_{F,G}(-a,-b)$, so integrating
% against |G_0(a,b)| gives a contribution at most
% $C_c\delta\prod_v\|G_v\|_3$. Everything can depend on (a,b)
% uniformly, so this statement holds for any kernel family
% with a uniform closed-form Gaussian envelope.


\subsection{Rough kernels with a controlled number of jumps}
\label{subsec:rough-family}

% The following proposition turns the frequency estimate into a
% bound for piecewise smooth kernels. The POINTWISE bound on
% the number of jumps at one scale and the bound on their TOTAL
% number play different roles: they supply the uniform bound and
% the square-integral bound for the frequency coefficients.

\begin{proposition}[Rough kernel families]\label{prop:rough-family}
Fix $A\ge1$, $c>0$. Let $n\ge1$, $\mathcal I\subset\mathbb Z$
finite. For $k\in\mathcal I$, let $K_{a,b,k}:\R\to\C$ and
$m_{a,b,k}\in\{0,1,2,\ldots\}$ be finite-valued step choices in
$(a,b)$ with $K_{a,b,k}=0$ whenever $m_{a,b,k}=0$. When
$m_{a,b,k}\ge1$, suppose $K_{a,b,k}$ is $C^1$ off a set of at
most $Am_{a,b,k}$ points, and that outside this set
\begin{equation}\label{eq:rough-family-decay}
 |K_{a,b,k}(z)|+|K_{a,b,k}'(z)|\le Ae^{-cz^2} .
\end{equation}
Suppose also that
\begin{equation}\label{eq:rough-family-hypotheses}
 \int_\R z^\ell K_{a,b,k}(z)\dd z=0\quad(\ell=0,1,2),
 \quad\sum_k m_{a,b,k}\le An,
 \quad m_{a,b,k}\le A\sqrt n .
\end{equation}
For $\mathcal K_{a,b}(t)=\sum_k\Dil_{2^k}K_{a,b,k}(t)$ and
$G_v\in C_c^\infty$, one has
\begin{equation}\label{eq:rough-family-bound}
 |\Lambda_{\mathcal K}(G_0,G_1,G_2)|
 \le C_{A,c}\sqrt n\log(2+n)\prod_v\|G_v\|_3 .
\end{equation}
\end{proposition}
\begin{proof}

\paragraph{A Gaussian-weighted primitive.}
Seek a representation $K=(g_{3q}f)'''$. Define
\[
 U(z)=\frac12\int_{-\infty}^z(z-u)^2K(u)\dd u .
\]
The three moment conditions imply, for $j=0,1,2$,
\[
 U^{(j)}(z)=-\frac1{(2-j)!}\int_z^\infty(z-u)^{2-j}K(u)\dd u .
\]
Use the defining integral when $z\le0$, the tail integral when
$z\ge0$; the Gaussian tail estimates give
$|U^{(j)}(z)|\le Ce^{-c_1 z^2}$ for $j=0,1,2$, with $c_1>0$ fixed.
Moreover $U'''=K$ as a weak derivative. Choose $q>0$ large enough,
depending on $(A,c)$, and put $f=U/g_{3q}$. Then
\begin{equation}\label{eq:rough-f-decay}
 |f^{(j)}(z)|\le Ce^{-c_2z^2}\quad(0\le j\le3).
\end{equation}
The functions $f,f',f''$ are continuous. Using $K'$ between
exceptional points bounds the classical derivative of $f'''$ by
$Ce^{-c_2z^2}$; the jump at each exceptional point is bounded by
$C$, and there are at most $Am$ of them. Thus
\begin{equation}\label{eq:rough-f-bv}
 \|f^{(j)}\|_\infty\le C,\quad
 \operatorname{Var}(f^{(j)})\le Cm\quad(0\le j\le3).
\end{equation}

\paragraph{Fourier bounds.}
Normalize $\widehat f(\xi)=(2\pi)^{-1}\int f(z)e^{-i\xi z}\dd z$.
Let $\Xi_1=\{|\xi|<2\}$ and $\Xi_D=\{D\le|\xi|<2D\}$ for dyadic
$D\ge2$. For $D\ge2$, the distributional derivative of $f'''$ is
a finite measure of total variation at most $Cm$, so
$|\xi|^4|\widehat f(\xi)|\le Cm$ gives the first bound below. For
the second, the translation inequality for bounded variation
gives
$\|f'''(\cdot+D^{-1})-f'''\|_1\le Cm/D$; applying the supremum
bound \eqref{eq:rough-f-bv} to bound its $L^2$ norm squared by
$Cm/D$, and using Plancherel
$\widehat{f'''}(\xi)=(i\xi)^3\widehat f(\xi)$ together with
$|e^{i\xi/D}-1|\ge2\sin(1/2)$ on $\Xi_D$, yields
\begin{equation}\label{eq:rough-fourier-bounds}
 \sup_{\xi\in\Xi_D}|D^3\widehat f(\xi)|\le\frac{Cm}{D},\quad
 \int_{\Xi_D}|D^3\widehat f(\xi)|^2\dd\xi\le\frac{Cm}{D}.
\end{equation}
For $D=1$, both bounds follow from \eqref{eq:rough-f-decay} and
$m\ge1$.

\paragraph{A finite frequency cutoff.}
Choose $\chi\in C_c^\infty$ supported on $[-2,2]$ equal to one on
$[-1,1]$, set $P_*=(2+n)^8$, and let
$\widehat{f_*}(\xi)=\widehat f(\xi)\chi(\xi/P_*)$,
$K_*=(g_{3q}f_*)'''$. Let $\eta$ be the Schwartz kernel of the
multiplier $\chi$, $\eta_{P_*}(z)=P_*\eta(P_*z)$; then
$f_*^{(j)}=f^{(j)}*\eta_{P_*}$ for $0\le j\le3$. The
translation inequality gives
$\|f^{(j)}-f_*^{(j)}\|_1\le Cm/P_*$. The supremum norms are
uniformly bounded. Interpolation between $L^1$ and $L^\infty$
yields $\|f^{(j)}-f_*^{(j)}\|_5\le C(m/P_*)^{1/5}$.
Since every derivative of $g_{3q}$ is bounded by a constant
multiple of a slightly wider Gaussian, the product rule shows
\begin{equation}\label{eq:rough-cutoff-error}
 |K(z)-K_*(z)|\le e^{-c_3z^2}w(z),\quad
 \|w\|_5\le C(m/P_*)^{1/5}
\end{equation}
for a nonnegative $w$. Apply this separately to each
$K_{a,b,k}$. By \eqref{eq:rough-family-hypotheses},
$(m_{a,b,k}/P_*)^{1/5}\le C/n$ whenever $m_{a,b,k}\ne0$, and at
each $(a,b)$ at most $An$ such indices are nonzero.
Lemma~\ref{lem:gaussian-envelope} bounds the sum of all
replacement errors by $C\mathcal M_{F,G}(-a,-b)$ after
integration in $c$, hence their contribution to
$\Lambda_{\mathcal K}$ is at most $C\prod_v\|G_v\|_3$.

\paragraph{Applying the frequency estimate.}
For each dyadic $D$ use the kernels of
Corollary~\ref{cor:frequency-continuum}:
$\ell_\xi(z)=D^{-1}d/dz(g_q(z)e^{i\xi z})$ and
$H_\xi=\ell_\xi*\ell_\xi*\ell_\xi$. Gaussian convolution and
modulation give $(g_{3q}(z)e^{i\xi z})'''=D^3H_\xi(z)$. For block
$\Xi_D$, the contribution to $K_{a,b,k,*}$ is
\[
 \int_{\Xi_D}c_{a,b,k}(\xi)H_\xi(z)\dd\xi,\quad
 c_{a,b,k}(\xi)=D^3\widehat f_{a,b,k}(\xi)\chi(\xi/P_*).
\]
Set $\mu_{a,b,k}(\xi)=Dc_{a,b,k}(\xi)/\sqrt n$. The two estimates
in \eqref{eq:rough-fourier-bounds} give
\[
 |\mu_{a,b,k}(\xi)|\le\frac{Cm_{a,b,k}}{\sqrt n}\le C,\quad
 \sum_k\int_{\Xi_D}|\mu_{a,b,k}|^2\frac{\dd\xi}{D}
 \le C\sum_k\frac{m_{a,b,k}}{n}\le C .
\]
Every step choice in $(a,b)$ has been preserved; different $k$
share a common finite rectangular partition. The block kernel is
$\sqrt n\int_{\Xi_D}\mu_{a,b,k}(\xi)H_\xi\,\dd\xi/D$. By
Corollary~\ref{cor:frequency-continuum} (with measure $\dd\xi/D$),
this block contributes at most $C\sqrt n\prod_v\|G_v\|_3$. Only
$O(\log(2+n))$ blocks meet the cutoff's support; summing their
bounds and adding the replacement error proves
\eqref{eq:rough-family-bound}.
\end{proof}


\subsection{Removing the first moment of an odd kernel}
\label{subsec:odd-family}

% Endpoint kernels are odd but need not have vanishing first
% moment. A dilation identity reduces them to the previous
% proposition.

\begin{corollary}[Odd kernel families]\label{cor:odd-family}
Proposition~\ref{prop:rough-family} remains valid if the
three moment conditions in \eqref{eq:rough-family-hypotheses}
are replaced by the assumption that every $K_{a,b,k}$ is odd
almost everywhere.
\end{corollary}
\begin{proof}
Choose a real, smooth, compactly supported odd $\varphi$ with
$\int z\varphi(z)\dd z=1$. Put $u_{a,b,k}=\int zK_{a,b,k}\dd z$
(uniformly bounded, vanishing when $m_{a,b,k}=0$). The kernels
$K_{a,b,k}-u_{a,b,k}\varphi$ satisfy all three moment conditions
and the other hypotheses of Proposition~\ref{prop:rough-family},
with enlarged fixed constants and unchanged counts. It remains
to bound the subtracted terms.

Define $\Psi=\varphi-2^{-1}\Dil_2\varphi$; oddness gives its
zeroth and second moments, and its first moment is
$1-2^{-1}\cdot2=0$. For any integer $J\ge1$,
\[
 \Dil_{2^k}\varphi
 =\sum_{j=0}^{J-1}2^{-j}\Dil_{2^{k+j}}\Psi
   +2^{-J}\Dil_{2^{k+J}}\varphi.
\]
For each fixed $j$, apply Proposition~\ref{prop:rough-family} to
shifted scales $k+j$ with kernels $u_{a,b,k}\Psi$ and shifted
counts $m_{a,b,k}$; the constants are independent of $j$ and the
outer coefficients $2^{-j}$ are summable. The remainder
contributes at most $C2^{-J}n\prod_v\|G_v\|_3$ by
Lemma~\ref{lem:gaussian-envelope}, since there are at most $An$
nonzero counts at any output point. Letting $J\to\infty$ proves
the claim.
\end{proof}

\subsection{Grouping the endpoint errors}
\label{subsec:endpoint-errors}

\begin{proposition}[Endpoint errors]\label{prop:endpoint-errors}
At each $(a,b)$, choose at most $n$ pairwise disjoint open radius
intervals $(\varepsilon_j,R_j)$ with $0<\varepsilon_j<R_j<\infty$
and complex coefficients $\beta_j$, $|\beta_j|\le1$. Suppose
these are a finite-valued step function of $(a,b)$. Then the
kernel $\mathcal E_{a,b}(t)=\sum_j\beta_j(E_{\varepsilon_j}-E_{R_j})(t)$
satisfies
\begin{equation}\label{eq:endpoint-error-bound}
 |\Lambda_{\mathcal E}(G_0,G_1,G_2)|
 \le C\sqrt n\log(2+n)\prod_{v=0}^2\|G_v\|_3
\end{equation}
for $G_v\in C_c^\infty$.
\end{proposition}
\begin{proof}
Place each endpoint $\rho$ in the unique half-open dyadic
interval $[L_k,2L_k)$, $L_k=2^k$. Let $m_{a,b,k}$ count
endpoints with multiplicity, collect their contributions into
$\Dil_{L_k}K_{a,b,k}$. Thus $K_{a,b,k}$ is a linear combination
of the dimensionless kernels
\begin{equation}\label{eq:dimensionless-endpoint}
 e_\alpha(z)=\frac{P(|z|/\alpha)-c_0\mathbf 1_{\{|z|>\alpha\}}}{z},
 \qquad 1\le\alpha<2,
\end{equation}
with coefficient $\beta_j$ at a lower endpoint and $-\beta_j$ at
its upper endpoint. Every kernel is odd, its possible jump
points are $\pm\alpha$, and
\begin{equation}\label{eq:endpoint-count}
 \sum_k m_{a,b,k}\le2n .
\end{equation}

The important assertion is the uniform bound, with constants
independent of $m_{a,b,k}$,
\begin{equation}\label{eq:paired-endpoint-bound}
 |K_{a,b,k}(z)|+|K_{a,b,k}'(z)|\le Ce^{-cz^2}
\end{equation}
off at most $2m_{a,b,k}$ points. Fix $(a,b)$ and $k$. Pair the
two endpoints of each annulus whose endpoints both lie in
$[L_k,2L_k)$. After scaling, its contribution is
$\beta_j(e_{\alpha_j}-e_{\gamma_j})$ with
$1\le\alpha_j<\gamma_j<2$. The intervals $(\alpha_j,\gamma_j)$
are disjoint, so $\sum_j(\gamma_j-\alpha_j)\le1$. At most two
endpoints are unpaired, corresponding to intervals that straddle
$L_k$ and $2L_k$ respectively (an interval that strictly
contains $[L_k,2L_k)$ has no endpoint in this group).

For $|z|\le3$, the smooth term $s_\alpha(z)=P(|z|/\alpha)/z$ has
a smooth even extension, and
$\sup_{|z|\le3,1\le\alpha\le2}(|\partial_\alpha s_\alpha|+
|\partial_z\partial_\alpha s_\alpha|)<\infty$. The fundamental
theorem of calculus in $\alpha$, followed by the length bound,
bounds the sum of the paired smooth terms independently of their
number. For the paired hard terms, at most one difference is
nonzero at any $|z|$, and on its support $|z|\ge1$, so the size
and classical derivative are uniformly bounded. The two unpaired
terms obey the same bounds on this compact range.

For $|z|>3$, all indicators equal one and
$P(|z|/\alpha)-c_0=-2g_3''(|z|/\alpha)$. Consequently
$e_\alpha$, $\partial_ze_\alpha$, $\partial_\alpha e_\alpha$,
and $\partial_z\partial_\alpha e_\alpha$ are bounded by
$Ce^{-cz^2}$, uniformly for $1\le\alpha\le2$. The length bound
and the two unpaired terms give the same bound for their sum.
Combining the compact and tail estimates proves
\eqref{eq:paired-endpoint-bound}.

Finish by separating groups by their count. At each output
point, \eqref{eq:endpoint-count} implies at most $2\sqrt n$
indices have $m_{a,b,k}>\sqrt n$. The uniform bound
\eqref{eq:paired-endpoint-bound} and
Lemma~\ref{lem:gaussian-envelope} bound all these groups
together, after $c$-integration, by $C\sqrt n\mathcal M_{F,G}$;
their trilinear contribution is at most
$C\sqrt n\prod_v\|G_v\|_3$.

Set the kernels and counts of those groups to zero. The
remaining family satisfies \eqref{eq:rough-family-decay}, has
odd kernels, and has counts satisfying
$\sum_km_{a,b,k}\le2n$ and $m_{a,b,k}\le\sqrt n$.
Corollary~\ref{cor:odd-family} applies and proves
\eqref{eq:endpoint-error-bound}. Every grouping and deletion is
made at the output point; no common choice of endpoints or
dyadic groups is imposed at different points.
\end{proof}
