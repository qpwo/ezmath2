\title{The complete Crouzeix theorem:\\
optimal similarity and a common positive boundary representation}
\author{}
\date{}
\begin{document}
\maketitle

\begin{abstract}
We prove the sharp constant-two numerical-range inequality for every bounded
operator on a complex Hilbert space and every matrix-valued polynomial, with
no separability assumption. The closure of the numerical range is a complete
$2$-spectral set, and the same estimate holds for finite matrix-valued
functions holomorphic near that closure. On an admissible (analytic, convex,
Jordan) neighborhood $\Omega$ of the numerical range of a matrix $A$, the
optimal similarity making the conformal disk image contractive is
attained with condition number $\kappa\le 2$, and one continuous positive
boundary density of mass the identity represents every matrix-valued
analytic evaluation of the similar matrix at once, at every coefficient
size.
\end{abstract}

\section{A guided tour}\label{sec:intro}
Fix $A\in M_n(\C)$ and a bounded open convex $\Omega\supset W(A)$ with a
regular real-analytic Jordan boundary. Inside $\Omega$ we have a conformal
parametrization $f:\Omega\to\D$; outside we have its exterior analogue
$G:\C\setminus\overline\D\to\C\setminus\overline\Omega$. The map
$\psi=f\circ G:\T\to\T$ measures how much the two coordinate circles
differ. The two main theorems say:

\begin{enumerate}
\item\emph{(optimal metric).} There is a positive definite $H$ with
$I\preceq H\preceq\kappa^2 I$ and $T^*HT\preceq H$, where $T=f(A)$; the
smallest such $\kappa$ is attained and satisfies $\kappa\le 2$. Setting
$S=H^{1/2}$ gives an explicit similarity under which the disk image
$f(A')$ has norm at most one and spectral radius strictly below one.
\item\emph{(common positive density).} With the same $S$, one continuous
$M_n(\C)$-valued positive $\Lambda$ on $\T$, of total mass $I_n$, evaluates
every $M_m(\C)$-valued holomorphic $v$ near $\overline\Omega$ in one
formula $v[A']=\int_\T\Lambda\otimes v(G)\,d\sigma$. The inequality
$\|v[A]\|\le\kappa\max_{\overline\Omega}\|v\|$ follows from a single
isometry rather than coefficient-dependent bookkeeping.
\end{enumerate}

We then extend to all matrices (Gaussian smoothing for arbitrary compact
convex $K$), then to all bounded operators on arbitrary complex Hilbert
spaces (finite compression plus Runge approximation), obtaining the sharp
Crouzeix inequality in the operator form:
\begin{equation}\label{eq:crouzeix-main}
 \|P[A]\|\le 2\sup_{z\in W(A)}\|P(z)\|.
\end{equation}

The guided plan of the actual proof is:
\begin{itemize}
\item[\S\ref{sec:setup}] notation: numerical range, conformal coordinates,
matrix-functional evaluation, tensor convention.
\item[\S\ref{sec:metric}] the metric problem is strictly feasible and
attained; equivalently, $\kappa$ is the best similarity condition number
making $T$ a contraction.
\item[\S\ref{sec:density}] one positive disk density $\Lambda$ captures
all evaluations at all coefficient sizes.
\item[\S\ref{sec:extremal}] if $\kappa>1$ a finite separating hyperplane
in a real affine space yields endpoint vectors $x,y$ and a contractive
matrix function $F$ interchanging them exactly.
\item[\S\ref{sec:boundary}] exterior-coordinate kernel and Cauchy
projection; a positive resolvent field $E_A$ for the original $A$.
\item[\S\ref{sec:bridge}] equality in the common compression forces an
ordered relation between two positive density systems.
\item[\S\ref{sec:compare}] positivity of the resulting block forces
$\kappa\le 2$.
\item[\S\ref{sec:matrices}, \S\ref{sec:operators}] passage to arbitrary
matrices and to bounded operators.
\item[\S\ref{sec:conformal}] a self-contained construction of the
conformal coordinates used throughout.
\end{itemize}

\section{Setup}\label{sec:setup}

\subsection*{Coordinates, numerics, evaluation}
Let $\D=\{z\in\C:|z|<1\}$, $\T=\partial\D$, with normalized arc measure
$d\sigma(e^{i\theta})=d\theta/(2\pi)$. For $A\in M_n(\C)$, the
\emph{numerical range} is
\[
 W(A)=\{x^*Ax:x\in\C^n,\ x^*x=1\}.
\]
Call $\Omega\subset\C$ \emph{admissible} if it is bounded, open, convex,
and its boundary is a regular real-analytic Jordan curve.

\begin{lemma}[Conformal coordinates]\label{lem:conformal}
For every admissible $\Omega$ there are a conformal bijection
$f:\Omega\to\D$ with $f$, $f^{-1}$ univalent through the closures, and a
conformal bijection $G:\{|t|>1\}\to\C\setminus\overline\Omega$
extending univalently to $\{|t|>r\}\cup\{\infty\}$ for some $0<r<1$, as a
sphere map, with a simple pole at infinity and $G(\T)=\partial\Omega$.
Writing $q_G(t)=tG'(t)$ and $\psi=f\circ G$, we have $q_G$ an outward
conormal on $\T$ and $\psi$ a smooth orientation-preserving circle
diffeomorphism.
\end{lemma}

The construction (Riemann map by extremal derivative, boundary harmonic
reflection, collar patching, exterior inversion) is in
\S\ref{sec:conformal}.

\subsection*{Evaluation of matrix-valued functions}
All tensor products are Hilbert tensor products, base space first. For
$B_0,\ldots,B_d\in M_m(\C)$ and $v(z)=\sum_{j=0}^dB_jz^j$, define
$v[A]=\sum_{j=0}^dA^j\otimes B_j$. For general $v$ holomorphic near
$\spec(A)$, evaluate on a Jordan block $J=\beta I+N$ of size $s$ by the
finite Taylor sum $\sum_{j=0}^{s-1}N^j\otimes v^{(j)}(\beta)/j!$,
followed by change of basis on the first factor. Scalar composition
gives $(v\circ f)[J]=v[f(J)]$ and $\spec(f(A))=f(\spec(A))$. Inner
products are conjugate-linear in the first slot; $U\preceq V$ means
$V-U$ is positive semidefinite.

Use the vectorization $\vecop(X)=\sum_{j=1}^nXe_j\otimes e_j$ with
$\|\vecop(X)\|^2=\tr(XX^*)$, and the elementary identity
\begin{equation}\label{eq:vec-mult}
 (D^k\otimes B)\vecop(X)=\vecop(D^kXB^{\mathsf T}).
\end{equation}

\section{The metric problem}\label{sec:metric}

Fix an admissible $\Omega\supset W(A)$ and coordinates $f,G$ from
Lemma~\ref{lem:conformal}. Put $T=f(A)$. Consider
\begin{equation}\label{eq:kappa}
 \kappa^2=\min\{\tau\in\R:I\preceq H\preceq\tau I,\ T^*HT\preceq H
 \text{ for some }H=H^*\}.
\end{equation}
A pair $(H,\tau)$ is \emph{strictly feasible} when
$H-I\succ 0$, $\tau I-H\succ 0$, $H-T^*HT\succ 0$.

\subsection*{Strict feasibility and attainment}
Because $\spec(A)\subset W(A)\subset\Omega$ and $f:\Omega\to\D$,
spectral mapping gives $\rho(T)<1$. Pick $\rho(T)<\eta<1$. Jordan form
supplies $C_\eta$ with $\|T^j\|\le C_\eta\eta^j$. The sum
\[
 H_0=\sum_{j\ge 0}T^{*j}T^j
\]
converges in norm, is $\succeq I$, and satisfies
$H_0-T^*H_0T=I$. For $c>1$ and $\tau>\|cH_0\|$, the pair $(cH_0,\tau)$
has all three slacks positive definite, so the feasible set is nonempty
with nonempty interior. Feasible pairs below any upper bound form a
closed bounded set in $\R\times\Herm_n$, so the minimum is attained.
Every feasible $\tau\ge 1$.

\subsection*{Similarity form of $\kappa$}
If $R$ is invertible with $\|RTR^{-1}\|\le 1$, then $H_R=R^*R$
satisfies $T^*H_RT\preceq H_R$; dividing by the least eigenvalue of $R^*R$
gives a feasible pair with $\tau=\|R\|^2\|R^{-1}\|^2$. Hence
$\kappa\le\|R\|\|R^{-1}\|$. Conversely, from a minimizing $H$ and
$S=H^{1/2}$, congruence gives $D^*D\preceq I$ for $D=STS^{-1}=Sf(A)S^{-1}$
and $I\preceq H\preceq\kappa^2I$ gives $\|S\|\|S^{-1}\|\le\kappa$; so the
inequality is attained:
\begin{equation}\label{eq:opt-sim}
 \kappa=\min\{\|R\|\|R^{-1}\|:\ R\text{ invertible},\
 \|RTR^{-1}\|\le 1\}.
\end{equation}

\noindent\emph{Attained.} Fix a minimizing $H$, put $S=H^{1/2}$, $D=Sf(A)S^{-1}$.
Then $D^*D\preceq I$ (congruence), $\rho(D)=\rho(f(A))<1$ (similarity and
spectral mapping), and $\|S\|\|S^{-1}\|\le\kappa$. The optimizer may have
$\|D\|=1$; that is allowed throughout.

\subsection*{Strict spectral radius} Because $\rho(D)<1$, we have
$\sum_{j\ge 0}\|D^j\|<\infty$; only the spectral-radius bound is needed
below, not the strict norm bound.

\begin{theorem}[Optimal metric]\label{thm:metric-with-bound}
For every admissible $\Omega$, every $A\in M_n(\C)$ with $W(A)\subset\Omega$,
and the conformal coordinate $f$ above, the minimum in \eqref{eq:kappa} is
strictly feasible, attained, and satisfies $1\le\kappa^2\le 4$. If $H$ is
a minimizer and
\[
 S=H^{1/2},\quad A'=SAS^{-1},\quad D=Sf(A)S^{-1}=f(A'),
\]
then
\[
 D^*D\preceq I,\qquad \rho(D)<1,\qquad
 \|S\|\|S^{-1}\|=\kappa\le 2.
\]
\end{theorem}

\begin{theorem}[A common positive representation]\label{thm:representation}
With the same $\Omega,A,f,G$ and any minimizing metric from
Theorem~\ref{thm:metric-with-bound}, there is one continuous
$\Lambda:\T\to M_n(\C)$ with $\Lambda(t)\succeq 0$ and
$\int_\T\Lambda\,d\sigma=I_n$, such that for every $m$ and every
$M_m(\C)$-valued $v$ holomorphic in a neighborhood of $\overline\Omega$,
\[
 v[A']=\int_\T\Lambda(t)\otimes v(G(t))\,d\sigma(t),\qquad
 \|v[A]\|\le\kappa\max_{\overline\Omega}\|v\|.
\]
The same $S$, $\Lambda$, $\kappa$ serve every $m$ and $v$.
\end{theorem}

The remainder of \S\ref{sec:density}--\S\ref{sec:compare} proves these
two theorems; the extension sections use them.

\section{One positive density for every coefficient size}\label{sec:density}

Fix a minimizing $H$, put $D=Sf(A)S^{-1}$ as above. On the unit circle
define the \emph{Delyon double-layer}
\begin{equation}\label{eq:phi}
 \Phi_D(\zeta)=(I-D/\zeta)^{-1}+[(I-D/\zeta)^{-1}]^*-I.
\end{equation}

\begin{proposition}[One density for every coefficient size]\label{prop:density}
Let $\Lambda:\T\to M_n(\C)$ by
\begin{equation}\label{eq:lambda}
 j(t)=\frac{tG'(t)f'(G(t))}{\psi(t)}>0,
 \qquad \Lambda(t)=j(t)\,\Phi_D(\psi(t)).
\end{equation}
Then $\Lambda\succeq 0$ is continuous, $\int_\T\Lambda\,d\sigma=I_n$, and
for every $m$ and every $v$ holomorphic in a neighborhood of
$\overline\Omega$ with values in $M_m(\C)$,
\begin{equation}\label{eq:rep-target}
 v[A']=\int_\T\Lambda(t)\otimes v(G(t))\,d\sigma(t),
 \qquad \|v[A']\|\le\max_{\overline\Omega}\|v\|,
 \qquad \|v[A]\|\le\kappa\max_{\overline\Omega}\|v\|.
\end{equation}
The same $S$, $\Lambda$, $\kappa$ serve every $m$ and $v$.
\end{proposition}

\begin{proof}
For $|\zeta|=1$ let $E_\zeta=I-D/\zeta$; it is invertible, and
$E_\zeta^*\Phi_D(\zeta)E_\zeta=I-D^*D\succeq 0$. Hence $\Phi_D\succeq 0$
and continuous. Since $\sum_{j\ge0}\|D^j\|<\infty$, the Laurent series
\[
 \Phi_D(\zeta)=I+\sum_{j\ge1}(\zeta^{-j}D^j+\zeta^{j}D^{*j})
\]
converges absolutely and uniformly on $\T$, and each Fourier moment
satisfies
\begin{equation}\label{eq:moment}
 \int_\T\zeta^j\Phi_D(\zeta)\,d\sigma(\zeta)=D^j\qquad(j\ge 0).
\end{equation}
For any $M_m$-valued $h$ holomorphic near $\overline\D$, integration of
its uniformly convergent Taylor series against \eqref{eq:moment} gives
\[
 h[D]=\int_\T\Phi_D(\zeta)\otimes h(\zeta)\,d\sigma(\zeta).
\]
Apply with $h=v\circ f^{-1}$. The conformal collar makes $h$ holomorphic
near $\overline\D$, and scalar composition gives $h[D]=v[A']$.

Local expansion of $\psi(e^{i\theta})=e^{i\eta(\theta)}$ shows
$j(e^{i\theta})=\eta'(\theta)>0$. The substitution $\zeta=\psi(t)$ in
the last identity gives the first claim of \eqref{eq:rep-target}. The
zeroth moment $\int\Phi_D=I$ combines with $\int j=1$ to give
$\int\Lambda=I_n$.

For each $m$, set
\[
 V_m:\C^n\otimes\C^m\longrightarrow L^2(\T,d\sigma;\C^n\otimes\C^m),
 \qquad(V_m b)(t)=(\Lambda(t)^{1/2}\otimes I_m)b.
\]
Normalization gives $\|V_mb\|_2=\|b\|$: the squared $L^2$ norm of the right
side is $\int\tr((\Lambda\otimes I_m)bb^*)\,d\sigma=\tr(bb^*)$.
If $M_v$ multiplies by $I_n\otimes v(G(t))$, then $v[A']=V_m^\dagger M_vV_m$
and
\[
 \|v[A']\|\le\|M_v\|_{L^2\to L^2}\le\max_{\overline\Omega}\|v\|.
\]
Similarity $A'=SAS^{-1}$ gives the last bound with $\kappa$.
Neither $S$ nor $\Lambda$ depends on $m$ or $v$.
\end{proof}

\section{Endpoint extremal: a contractive $F$ with $F[D]x=y$}\label{sec:extremal}

Assume $\kappa>1$; otherwise Proposition~\ref{prop:density} already proves
Theorem~\ref{thm:metric-with-bound} below. The proof splits into four
paragraphs: separation, dual balance, endpoint supports, unitary
realization.

\subsection*{Separation}
In the finite-dimensional real space $\R\oplus\Herm_n^3$, the affine set
\[
 \mathcal A_T=\{(s,J-I,sI-J,J-T^*JT):s\in\R,\ J=J^*\}
\]
is disjoint from the open convex set
\[
 \mathcal O=\{(s,U,V,W):\ s<\kappa^2,\ U,V,W\succ 0\}.
\]
Otherwise a point of $\mathcal A_T$ would be a strictly feasible pair
below the minimizing $\tau$. Write $\mathcal A_T=a_0+L$ and let
$\pi:\R\oplus\Herm_n^3\to L^\perp$ be orthogonal projection. Then
$\mathcal K=\pi(\mathcal O-a_0)$ is a nonempty open convex subset of
$L^\perp$ not containing $0$. The classical separation step, proved
for these specific sets below, provides a nonzero linear functional
$\ell$ with
\[
 \ell(a)\ge \ell(o)\qquad(a\in\mathcal A_T,\ o\in\mathcal O).
\]
First case, $0\notin\overline{\mathcal K}$: a nearest point $z_0$ to $0$ in
the closed convex $\overline{\mathcal K}$ has $\langle z_0,c\rangle\ge
\|z_0\|^2>0$ for every $c\in\mathcal K$; take $\nu=z_0$.

Second case, $0\in\overline{\mathcal K}$: fix $c_*\in\mathcal K$ and set
$z_j=-c_*/j$. Each $z_j\notin\overline{\mathcal K}$; otherwise $0$ would
lie on an open segment from an interior point $c_*$ to a point of the
closure, hence in $\mathcal K$, which it does not. Let $y_j$ be nearest
to $z_j$ in the closed convex $\overline{\mathcal K}$; since $0$ is in
$\overline{\mathcal K}$, $\|y_j-z_j\|\le\|z_j\|\to 0$, so $y_j\to 0$;
a subsequence of $\nu_j=(y_j-z_j)/\|y_j-z_j\|$ tends to a unit $\nu$, and
the nearest-point inequality $\langle\nu_j,c-y_j\rangle\ge 0$ gives
$\langle\nu,c\rangle\ge 0$ for all $c\in\mathcal K$.

In either case $\ell(v)=-\langle\nu,\pi v\rangle$ is nonzero and
$\ell(a)\ge\ell(o)$ for all $a\in\mathcal A_T$, $o\in\mathcal O$.
Since $\mathcal A_T$ is affine, $\ell$ is constant on it. Bounded above on
$\mathcal O$, it must have the sign pattern
\[
 \ell(s,U,V,W)=\lambda s-\tr(Y_0U)-\tr(X_0V)-\tr(Z_0W),
 \qquad\lambda\ge 0,\ X_0,Y_0,Z_0\succeq 0.
\]
If $\lambda=0$, strict feasibility produces a negative value on
$\mathcal A_T$, while separation requires $\ge\sup_{\mathcal O}\ell=0$.
So $\lambda>0$; scale to $\lambda=1$. Constancy on $\mathcal A_T$ gives
\begin{equation}\label{eq:dual}
 X_0-Y_0=Z_0-TZ_0T^*,\qquad\tr X_0=1.
\end{equation}
The constant value is $\tr Y_0\ge \kappa^2$. Evaluating at $(\kappa^2,H)$:
\[
 \kappa^2-\tr(Y_0(H-I))-\tr(X_0(\kappa^2I-H))
       -\tr(Z_0(H-T^*HT))\le\kappa^2.
\]
Each subtracted trace is nonnegative; each vanishes; hence
$\tr Y_0=\kappa^2$.

\subsection*{Endpoint support and balance}
For positive $B,C$, $\tr(BC)=\|C^{1/2}B^{1/2}\|_{\HS}^2$, and
$\tr(BC)=0$ forces $\mathrm{range}(B)\subset\ker(C)$. Every subtraction
above vanishes, so $X_0$ is supported on the $\kappa^2$-eigenspace of
$H$, and $Y_0$ on the $1$-eigenspace. Choose
\[
 XX^*=\kappa^{-2}SX_0S,\qquad YY^*=\kappa^{-2}SY_0S,\qquad Q=\kappa^{-2}SZ_0S,
\]
with $X,Y,Q$ of the indicated positive shapes. The trace and endpoint
relations give
\begin{equation}\label{eq:endpoints1}
 \tr(XX^*)=\tr(YY^*)=1,\quad SX=\kappa X,\quad SY=Y,\quad X^*Y=0.
\end{equation}
(Congruence of \eqref{eq:dual} multiplied by $\kappa^{-2}S$ gives the last
identity $XX^*-YY^*=Q-DQD^*$, whose column ranges are orthogonal by the
disjoint support of $X_0$ and $Y_0$.)

\subsection*{Unitary realization}
Put $L=Q^{1/2}$. The rows $R_1=[L\ Y]$ and $R_2=[DL\ X]$ satisfy
$R_1R_1^*=R_2R_2^*$. Hence $R_2^*v\mapsto R_1^*v$ is a well-defined
isometry of their ranges; extending by a unitary on the
(equal-dimensional) orthogonal complements and taking the adjoint gives a
unitary
\[
 U=\begin{pmatrix}a&b\\ c&d\end{pmatrix},\qquad [L\ Y]=[DL\ X]\,U.
\]
Thus $L=DLa+Xc$ and $Y=DLb+Xd$. Since $\|a\|\le 1$ and $\rho(D)<1$, the
identity
\begin{equation}\label{eq:series}
 Y=Xd+\sum_{j\ge 0}D^{j+1}Xca^j b
\end{equation}
is norm-convergent.

Define, for $|\zeta|<1$,
\[
 R(\zeta)=d+\zeta c(I-\zeta a)^{-1}b,\qquad F_\D(\zeta)=R(\zeta)^{\mathsf T}.
\]
For $e\in\C^n$ and $v=(I-\zeta a)^{-1}be$, unitarity gives
$U\binom{\zeta v}{e}=\binom{v}{R(\zeta)e}$, so
$\|e\|^2-\|R(\zeta)e\|^2=(1-|\zeta|^2)\|v\|^2\ge 0$. Hence $R$ is
contractive. Ordinary transpose preserves norm and holomorphy. Since $R$
is rational and bounded in $\D$, any apparent pole on $\T$ is removable;
after cancellation, finitely many poles remain strictly outside
$\overline\D$. Continuity extends contractivity to $\T$.

Finally, using \eqref{eq:vec-mult} with $x=\vecop(X)$, $y=\vecop(Y)$ and
the coefficientwise transpose in $F_\D$, identity \eqref{eq:series}
gives $F_\D[D]x=y$. Set $F_\Omega=F_\D\circ f$, $F=F_\Omega\circ G$;
scalar composition gives $F_\Omega[A']x=y$ and $\|F(t)\|\le 1$ on $\T$.

\begin{proposition}[Endpoint extremal]\label{prop:extremal}
If $\kappa>1$, there exist $M_n(\C)$ matrices $X,Y$ with
\eqref{eq:endpoints1}, plus a contractive rational $M_n$-valued $F_\D$
holomorphic near $\overline\D$ with $F_\D[D]x=y$ for $x=\vecop(X)$,
$y=\vecop(Y)$. The functions $F_\Omega=F_\D\circ f$ and $F=F_\Omega\circ G$
are holomorphic near $\overline\Omega$ and contractive, respectively.
\end{proposition}

\section{Local boundary tools}\label{sec:boundary}

Fix the admissible $\Omega$, exterior coordinate $G$, admissibly
normalized at infinity by $G(t)=at+b+O(t^{-1})$, $a\ne 0$. Put
$q_G(t)=tG'(t)$.

\subsection*{Faber polynomials and exterior kernel}
For fixed $z\in\Omega$,
\begin{equation}\label{eq:faber-repr}
 \frac{q_G(t)}{G(t)-z}=\sum_{k\ge 0}b_k(z)\,t^{-k}
 \quad\text{near }t=\infty,
\end{equation}
where $b_k$ is a polynomial of degree $k$ with leading coefficient
$a^{-k}$, and $b_0=1$.

The mixed correction kernel
\begin{equation}\label{eq:B-kernel}
 B(w,t)=\frac{q_G(t)}{G(t)-G(w)}-\frac{t}{t-w}
      =\sum_{i,j\ge 1}\beta_{ij}w^{-i}t^{-j},
 \qquad b_j(G(w))=w^j+\sum_{i\ge 1}\beta_{ij}w^{-i}
\end{equation}
is holomorphic through the diagonal and both infinities of the exterior
product collar. For each $r<\rho<1$ there is $C_\rho$ with
$|\beta_{ij}|\le C_\rho\rho^{i+j}$, so the double series converges
normally throughout the collar and absolutely uniformly on $\T^2$.

Put $m(w,t)=1+B(w,t)+\overline{B(w,t)}$ on $\T^2$. Then $m\ge 0$,
continuous, with each normalized-circle marginal one:
$\int m(w,t)\,d\sigma(t)=1=\int m(w,t)\,d\sigma(w)$.

\begin{proof}[Sketch of the exterior calculations]
Expand \eqref{eq:faber-repr} in powers of $1/t$ after factoring $at$. For
$B$, put $Q(w,t)=(G(t)-G(w))/(t-w)$ and write $G(t)=at+g(u)$, $u=1/t$;
then $Q=a-uv\,(g(u)-g(v))/(u-v)$, with brackets jointly holomorphic. Off
the diagonal $Q\ne 0$ by univalence; on the diagonal $Q=G'\ne 0$; along
either coordinate axis $Q=a\ne 0$. Hence
\[
 B=t\partial_tQ/Q=-u\partial_uQ/Q
\]
is holomorphic on the reciprocal bidisk and vanishes on both axes; only
strictly mixed powers $u^iv^j$ occur. Cauchy estimates on the bidisk of
radius $\rho^{-1}$ give $|\beta_{ij}|\le C_\rho\rho^{i+j}$. The marginals
follow by term-by-term integration. Positivity: for $w,t\in\T$, $w\ne t$,
\[
 m(w,t)=2\operatorname{Re}\frac{q_G(t)}{G(t)-G(w)}\ge 0,
\]
since $q_G(t)$ is the outward conormal and $\Omega$ is convex; continuity
extends the inequality to the diagonal.
\end{proof}

\subsection*{Resolvent field $E_A$}
For $A\in M_n(\C)$ with $W(A)\subset\Omega$, define for $t\in\T$
\begin{equation}\label{eq:EA}
 K_A(t)=q_G(t)(G(t)I-A)^{-1}=\sum_{k\ge 0}b_k(A)\,t^{-k},
 \qquad E_A(t)=K_A(t)+K_A(t)^*\succeq 0.
\end{equation}

\begin{proof}[Positivity]
Every eigenvalue of $A$ lies in $W(A)$, so $G(t)I-A$ is invertible on and
outside $\T$, and the Laurent expansion \eqref{eq:EA} converges
absolutely uniformly by Cauchy estimates. Put $R_t=G(t)I-A$. Then
\[
 R_t^*(K_A+K_A^*)R_t=q_G(t)R_t^*+\overline{q_G(t)}R_t.
\]
At a unit vector $x$, the quadratic form is
$2\operatorname{Re}(\overline{q_G(t)}(G(t)-x^*Ax))\ge 0$, by the supporting
half-plane property of $\Omega$ and $W(A)\subset\Omega$. Since $R_t$ is
invertible, $E_A\succeq 0$.
\end{proof}

The mass identity $\int_\T E_A\,d\sigma=2I_n$ is a case of the resolvent
contract \eqref{eq:resolvent-contract} below.

\subsection*{Cauchy projection and adjoints}
Let $\cH_k=L^2(\T,d\sigma;M_k(\C))$ with inner product
$\langle u,v\rangle=\int_\T\tr(u^*v)\,d\sigma$, Hilbert--Schmidt pointwise
norm. Let $P_0,P_+,P_-$ be Fourier projections onto constant, strictly
positive, strictly negative frequencies; write $u=u_0+u_++u_-$. Define
$\M:\cH_k\to\cH_k$ by
\[
 (\M u)(w)=\int_\T m(w,t)u(t)\,d\sigma(t).
\]
The mixed expansion gives
\begin{equation}\label{eq:M-fourier}
 \M(t^j)(w)=\sum_{i\ge 1}\beta_{ij}w^{-i},\qquad
 \M(t^{-j})(w)=\sum_{i\ge 1}\overline{\beta_{ij}}w^{i}\quad(j\ge 1),
 \qquad \M(1)=1,
\end{equation}
so $\M$ maps constants to constants, positive to strictly negative,
negative to strictly positive, and preserves the mean. Its actual Hilbert
adjoint $\M^\dagger$ has kernel $m(t,w)$ and
\begin{equation}\label{eq:Mdag-fourier}
 \M^\dagger(t^{-i})(w)=\sum_{j\ge 1}\overline{\beta_{ij}}w^{j},\qquad
 \M^\dagger(t^{i})(w)=\sum_{j\ge 1}\beta_{ij}w^{-j}\quad(i\ge 1),
 \qquad\M^\dagger(1)=1.
\end{equation}

Define
\[
 \cC=P_0+P_++\M P_+,\qquad \widetilde\cC u=(\cC(u^*))^*.
\]
For finite Laurent data $u$, the physical Cauchy integral
$c_u(z)=\int_\T u(t)q_G(t)/(G(t)-z)\,d\sigma(t)$ is a polynomial in $z$;
substituting $z=G(w)$ traces out $\cC u(w)$. Hence $\cC$ extends to a
bounded projection on every $\cH_k$, and
\begin{equation}\label{eq:C-sq}
 \cC^2=\cC,\qquad \cC(F\cC u)=F\cC u,\qquad
 \cC((\cC u)F)=(\cC u)F
\end{equation}
for every $F=F_\Omega\circ G$ with $F_\Omega$ holomorphic near
$\overline\Omega$.

\begin{proposition}[Cauchy adjoints and modules]\label{prop:proj}
For every $k\ge 1$ and every $u\in\cH_k$,
\begin{equation}\label{eq:adj}
 \cC^\dagger u=u_0+u_++\M^\dagger u_-, \qquad
 \widetilde\cC^\dagger u=u_0+u_-+\M^\dagger u_+,
\end{equation}
and the weighted inequality
\begin{equation}\label{eq:weighted}
 \|\cC^\dagger u\|_2^2+\|(\cC^\dagger u)_0\|_2^2\le 2\|u\|_2^2
\end{equation}
holds. For finite Laurent data $u$ and every $n$,
\begin{equation}\label{eq:resolvent-contract}
 \int_\T E_A\otimes u\,d\sigma=c_u[A]+c_{u^*}[A]^*,
 \qquad\int_\T E_A\,d\sigma=2I_n.
\end{equation}
\end{proposition}

\begin{proof}
Boundedness of $\M$ and the marginal identity $\int m(w,t)\,d\sigma(t)=1$
show that $\M$ is a contraction on $\cH_k$ (Jensen at $w$, integrate). The
Fourier action in \eqref{eq:M-fourier} extends to the closed positive
subspace by boundedness and density of Fourier polynomials; the same for
$\M^\dagger$. Equations \eqref{eq:adj} below verify they are
adjoints.

Since $\cC=P_0+P_++\M P_+$ and the three terms live in orthogonal
frequency sectors (constant, positive, negative correction), $\cC$ is
bounded; the same orthogonality gives $\cC^2=\cC$. Taking adjoints:
\[
 \cC^\dagger=P_0+P_++P_+\M^\dagger=P_0+P_++\M^\dagger P_-
\]
since $P_+\M^\dagger=\M^\dagger P_-$ by the Fourier exchange in
\eqref{eq:Mdag-fourier}. Conjugating by the antiunitary $u\mapsto u^*$
gives the second identity in \eqref{eq:adj}.

For \eqref{eq:weighted}, expand $u=u_0+u_++u_-$. Then
$\cC^\dagger u=u_0+u_++\M^\dagger u_-$ and
$\widetilde\cC^\dagger u=u_0+u_-+\M^\dagger u_+$; using
$\|\M^\dagger\|\le 1$,
\[
 \|\cC^\dagger u\|_2^2+\|(\cC^\dagger u)_0\|_2^2
 =2\|u_0\|_2^2+\|u_++\M^\dagger u_-\|_2^2\le 2\|u\|_2^2.
\]

For \eqref{eq:resolvent-contract}, finite Laurent $u$ gives by
term-by-term integration and \eqref{eq:faber-repr}
$\int_\T K_A\otimes u\,d\sigma=\sum_{j\ge 0}b_j(A)\otimes U_j=c_u[A]$.
Take the full matrix adjoint and add the conjugate with $u^*$.
\end{proof}

\begin{remark}
In the second adjoint identity in \eqref{eq:adj} (which we use
below) the second factor is $(\cC(u^*))^*$, not an independent analytic
function evaluated at $A$; this is why the comparison in \S\ref{sec:compare}
keeps left and right multiplication orders.
\end{remark}

\section{From extremal vectors to two positive density systems}\label{sec:bridge}

Return to Proposition~\ref{prop:extremal}, assume $\kappa>1$, and let
$X,Y,F_\D,F_\Omega,F$ be as there. Put $x=\vecop(X)$, $y=\vecop(Y)$,
$R_S=S\otimes I_n$. The endpoint relations are
\begin{equation}\label{eq:endpoints2}
 \|x\|=\|y\|=1,\quad R_Sx=\kappa x,\quad R_Sy=y,\quad F_\Omega[A']x=y.
\end{equation}
Throughout this section the auxiliary coefficient size is $k=n$, matching
the base size; the density and its bound then extend to every target size
via Proposition~\ref{prop:density}.

\subsection*{Equality in the common compression}
Use the isometry $V=V_n$ of Proposition~\ref{prop:density} and set
\[
 x_\Lambda=Vx=(\Lambda^{1/2}\otimes I_n)x,\qquad
 y_\Lambda=Vy=(\Lambda^{1/2}\otimes I_n)y.
\]
Let $M_F$ multiply by $I_n\otimes F(t)$. The interpolation identity
$v[A']=V^\dagger M_vV$ at $v=F$, together with \eqref{eq:endpoints2},
gives
\[
 1=\|V^\dagger M_FVx\|\le\|M_FVx\|\le\|Vx\|=1.
\]
Equality in the first inequality makes $M_FVx$ lie in the range of $V$:
the orthogonal projection $VV^\dagger$ cannot preserve the norm of a
vector with a nonzero perpendicular component. Thus $M_FVx=Vy$.
Equality in the second inequality means the positive operator
$I-M_F^\dagger M_F$ has zero quadratic form at $Vx$, hence annihilates
$Vx$; so, almost everywhere on $\T$,
\begin{equation}\label{eq:fiber-eq}
 y_\Lambda=(I_n\otimes F(t))x_\Lambda,\qquad
 x_\Lambda=(I_n\otimes F(t)^*)y_\Lambda.
\end{equation}

\subsection*{Partial traces and ordered densities}
Take the partial trace over the base factor (sum of diagonal base blocks):
\[
 p=\mathrm{Tr}_{\rm base}(x_\Lambda x_\Lambda^*),\quad
 q=\mathrm{Tr}_{\rm base}(y_\Lambda y_\Lambda^*),\quad
 r=\mathrm{Tr}_{\rm base}(y_\Lambda x_\Lambda^*).
\]
They are in $\cH_n$ and $p,q\succeq 0$. Writing
$x_\Lambda=\sum_a e_a\otimes\xi_a$ and $y_\Lambda=\sum_a e_a\otimes\eta_a$
gives $r=\sum_a\eta_a\xi_a^*$; the fiber identity \eqref{eq:fiber-eq}
gives $\eta_a=F\xi_a$ and $\xi_a=F^*\eta_a$. Hence, with both
multiplication orders preserved,
\begin{equation}\label{eq:ordered-densities}
 p=F^*r,\qquad r=Fp,\qquad q=rF^*.
\end{equation}
Writing $p,q,r$ directly through $X,Y,\Lambda$:
\begin{equation}\label{eq:transpose-densities}
 p=(X^*\Lambda X)^{\mathsf T},\quad
 q=(Y^*\Lambda Y)^{\mathsf T},\quad
 r=(X^*\Lambda Y)^{\mathsf T}.
\end{equation}
Indeed, the $(j,k)$ entry of the cross density is
$\sum_a(\Lambda^{1/2}Y)_{aj}\overline{(\Lambda^{1/2}X)_{ak}}
=(X^*\Lambda Y)_{kj}$. Integrating \eqref{eq:transpose-densities}
against $\int\Lambda=I_n$ and using $X^*Y=0$ from \eqref{eq:endpoints1}:
\begin{equation}\label{eq:matrix-zero}
 r_0=(X^*Y)^{\mathsf T}=0,\qquad \tr(p_0)=\tr(X^*X)=1.
\end{equation}
The first identity is a full \emph{matrix} mean-zero; the scalar
$\tr(r_0)=0$ alone would not supply the Fourier orthogonality used in
\S\ref{sec:compare}.

The cross density has the testing orientation
\begin{equation}\label{eq:testing}
 \langle r,u\rangle
 =\int_\T y_\Lambda^*(I_n\otimes u)x_\Lambda\,d\sigma
 =y^*\left(\int_\T\Lambda\otimes u\,d\sigma\right)x\qquad(u\in\cH_n),
\end{equation}
which is $\tr(r^*u)=\sum_a\eta_a^*u\xi_a$; the diagonal testings use
$(x,x)$ for $p$ and $(y,y)$ for $q$.

\subsection*{The original-matrix positive block}
Using $E_A\succeq 0$ from \S\ref{sec:boundary}, put
\[
 x_e=(E_A^{1/2}\otimes I_n)x,\qquad y_e=(E_A^{1/2}\otimes I_n)y,
\]
and define $p_e,q_e,r_e$ by the same three partial traces, now using
$(x_e,y_e)$. Writing $x_e=\sum_a e_a\otimes\alpha_a$ and
$y_e=\sum_a e_a\otimes\beta_a$:
\begin{equation}\label{eq:positive-block}
 \Delta=\begin{pmatrix}p_e&r_e^*\\ r_e&q_e\end{pmatrix}
 =\sum_a\binom{\alpha_a}{\beta_a}\binom{\alpha_a}{\beta_a}^{\!*}\succeq 0.
\end{equation}
This is the only place where $W(A)\subset\Omega$ enters for the original
$A$: $E_A\succeq 0$ is built from $A$, not from $A'=SAS^{-1}$.

\subsection*{Identification with projected densities}
For an $M_n$-valued $v$ holomorphic near $\overline\Omega$, similarity and
\eqref{eq:endpoints2} give
\begin{equation}\label{eq:kappa-factors}
 v[A]=R_S^{-1}v[A']R_S,\quad v[A]^*=R_Sv[A']^*R_S^{-1},\quad
 y^*v[A]x=\kappa y^*v[A']x,\quad y^*v[A]^*x=\kappa^{-1}y^*v[A']^*x.
\end{equation}
For diagonal pairings the similar matrix $R_S$, $R_S^{-1}$ cost factors
that cancel in both the operator and the adjoint.

Take a matrix Laurent polynomial $u\in\cH_n$ with physical representative
$c_u$ (holomorphic near $\overline\Omega$). By
\eqref{eq:resolvent-contract} and \eqref{eq:kappa-factors},
\[
 \langle r_e,u\rangle
 =y^*\bigl(c_u[A]+c_{u^*}[A]^*\bigr)x
 =\kappa\langle r,\cC u\rangle
  +\kappa^{-1}\langle r,\widetilde\cC u\rangle
 =\langle\kappa\cC^\dagger r+\kappa^{-1}\widetilde\cC^\dagger r,u\rangle.
\]
Here the first identity uses \eqref{eq:resolvent-contract} and
\eqref{eq:testing}; the second uses \eqref{eq:kappa-factors} together with
$c_w[A']=\int_\T\Lambda\otimes w(G(t))\,d\sigma$, so the boundary
integrand of $c_{u^*}[A']^*$ is the analytic adjoint $\widetilde\cC u$ and
not an independent analytic function. The diagonal computations have no
similarity factor. Laurent polynomials are dense in $\cH_n$, so
\begin{equation}\label{eq:projected-densities}
 p_e=\cC^\dagger p+\widetilde\cC^\dagger p,\qquad
 q_e=\cC^\dagger q+\widetilde\cC^\dagger q,\qquad
 r_e=\kappa\cC^\dagger r+\kappa^{-1}\widetilde\cC^\dagger r.
\end{equation}
All five fields lie in the same $L^2$ space $\cH_n$.

\section{The ordered comparison and the constant two}\label{sec:compare}

The previous section produces two positive density systems from the same
endpoint vectors. Let $g=\cC^\dagger r$. Then the diagonal norms
$\|p_e\|_2,\|q_e\|_2$ are controlled by $\|g\|_2$, whereas the cross
norm $\|r_e\|_2$ is controlled from below by $\kappa\|g\|_2$. The
positivity of the resulting $2k\times 2k$ block forces $\kappa\le 2$.

\begin{lemma}[Ordered comparison]\label{lem:comparison}
Let $k\ge 1$, and let $F=F_\Omega\circ G$ be $M_k$-valued with $F_\Omega$
holomorphic near $\overline\Omega$ and $\|F(t)\|\le 1$ on $\T$. Suppose
$p,q,r\in\cH_k$ satisfy
\[
 p=p^*,\quad q=q^*,\quad p=F^*r,\quad q=rF^*,\quad r_0=0,\quad \tr(p_0)>0.
\]
For $\kappa>0$ set
\[
 p_e=\cC^\dagger p+\widetilde\cC^\dagger p,\quad
 q_e=\cC^\dagger q+\widetilde\cC^\dagger q,\quad
 r_e=\kappa\cC^\dagger r+\kappa^{-1}\widetilde\cC^\dagger r.
\]
If
\[
 \Delta(t)=\begin{pmatrix}p_e(t)&r_e(t)^*\\ r_e(t)&q_e(t)\end{pmatrix}
 \succeq 0\quad\text{for a.e. }t\in\T,
\]
then $\kappa\le 2$.
\end{lemma}

\begin{proof}
Put $g=\cC^\dagger r$, $P_1=\cC^\dagger p$, $P_2=\cC^\dagger q$. Each
$P_j$ has only constant and strictly positive Fourier frequencies; its
mean $(P_j)_0$ is Hermitian.

The adjoint of left multiplication by $F$ is left multiplication by
$F^*$; the adjoint of right multiplication is right multiplication by
$F^*$ (because $\tr((uF)^*v)=\tr(F^*u^*v)=\tr(u^*vF^*)$). Taking
Hilbert-space adjoints of the two module identities \eqref{eq:C-sq}
gives
\[
 \cC^\dagger L_{F^*}\cC^\dagger=\cC^\dagger L_{F^*},\qquad
 \cC^\dagger R_{F^*}\cC^\dagger=\cC^\dagger R_{F^*},
\]
where $L_{F^*}u=I_k\otimes F^*u$ and $R_{F^*}u=uF^*$ are left/right
multiplication by $F^*$. Therefore
\[
 P_1=\cC^\dagger(F^*r)=\cC^\dagger L_{F^*}\cC^\dagger r=\cC^\dagger L_{F^*}g
     =\cC^\dagger(F^*g),
 \qquad
 P_2=\cC^\dagger(rF^*)=\cC^\dagger(gF^*).
\]
If $g=0$, the first identity gives $P_1=0$, contradicting mean preservation
$(P_1)_0=p_0$ and $\tr(p_0)>0$. Thus $g\ne 0$.

Since $\|F\|\le 1$ on $\T$, $\|F^*g\|_2\le\|g\|_2$ and
$\|gF^*\|_2\le\|g\|_2$. The weighted inequality \eqref{eq:weighted}
applied to $F^*g$ and to $gF^*$ gives
\[
 \|P_j\|_2^2+\|(P_j)_0\|_2^2\le 2\|g\|_2^2\qquad(j=1,2).
\]
Fourier orthogonality for a matrix field with only constant and positive
modes gives
\[
 \|P_j+P_j^*\|_2^2
 =4\|(P_j)_0\|_2^2+2\|(P_j)_+\|_2^2
 =2\bigl(\|P_j\|_2^2+\|(P_j)_0\|_2^2\bigr).
\]
Since $p=p^*$ and $q=q^*$, the antiunitary $u\mapsto u^*$
gives $\widetilde\cC^\dagger p=(\cC^\dagger p)^*=P_1^*$ and
$\widetilde\cC^\dagger q=P_2^*$. Therefore
\begin{equation}\label{eq:diagbound}
 p_e=P_1+P_1^*,\quad q_e=P_2+P_2^*,\quad
 \|p_e\|_2^2\le 4\|g\|_2^2,\quad \|q_e\|_2^2\le 4\|g\|_2^2.
\end{equation}

The matrix identity $r_0=0$ says that every entry has zero mean; hence
$g$ has constant and positive frequencies only (no constant part), and
$h=\widetilde\cC^\dagger r$ has constant and negative frequencies only
(no constant part). Their inner product vanishes. So
\begin{equation}\label{eq:crossbound}
 \|r_e\|_2^2=\kappa^2\|g\|_2^2+\kappa^{-2}\|h\|_2^2\ge \kappa^2\|g\|_2^2.
\end{equation}

Put $J=\diag(I_k,-I_k)$. Positivity of both $\Delta$ and $J\Delta J$ shows
that the Hermitian square $(\Delta^{1/2}J\Delta^{1/2})^2$ is positive
semidefinite, so its trace is nonnegative. That trace equals
$0\le\tr(\Delta J\Delta J)$. Direct $2\times 2$ block multiplication gives
\[
 \tr(\Delta J\Delta J)
 =\|p_e\|_{\HS}^2+\|q_e\|_{\HS}^2-2\|r_e\|_{\HS}^2.
\]
The pointwise entries lie in $L^2$, so the inequality is integrable.
Combining \eqref{eq:diagbound} and \eqref{eq:crossbound},
\[
 2\kappa^2\|g\|_2^2
 \le 2\|r_e\|_2^2
 \le \|p_e\|_2^2+\|q_e\|_2^2
 \le 8\|g\|_2^2.
\]
Since $g\ne 0$, dividing proves $\kappa\le 2$.
\end{proof}

\begin{proof}[Proof of Theorems~\ref{thm:metric-with-bound} and~\ref{thm:representation}]
Fix a minimizing $H$ and set $D=Sf(A)S^{-1}$ as in \S\ref{sec:metric}.
The conclusions $D^*D\preceq I$, $\rho(D)<1$, $\|S\|\|S^{-1}\|=\kappa$ were
established there. Proposition~\ref{prop:density} already gives the
representation and the operator inequality at every target size $m$; the
only remaining bound is $\kappa\le 2$.

If $\kappa=1$, then $H=I$ is admissible, so $T^*T\preceq I$ and
$\kappa=1\le 2$ trivially.

Suppose $\kappa>1$. Proposition~\ref{prop:extremal} provides $X,Y$,
$F_\Omega$, $F$; \S\ref{sec:bridge} provides $p,q,r\in\cH_n$ with
\eqref{eq:ordered-densities}, \eqref{eq:matrix-zero}, and $p_e,q_e,r_e$
in \eqref{eq:projected-densities}. The hypotheses of
Lemma~\ref{lem:comparison} are:
\begin{itemize}
\item $p=p^*$ and $q=q^*$: from \eqref{eq:transpose-densities}, since
$\Lambda(t)=\Lambda(t)^*$.
\item $p=F^*r$, $q=rF^*$: \eqref{eq:ordered-densities}.
\item $r_0=0$ (matrix identity) and $\tr(p_0)=1>0$:
\eqref{eq:matrix-zero}.
\item $\Delta(t)\succeq 0$ a.e.: \eqref{eq:positive-block} with the
identifications \eqref{eq:projected-densities}.
\end{itemize}
Lemma~\ref{lem:comparison} then gives $\kappa\le 2$. With that,
Theorem~\ref{thm:metric-with-bound} and Theorem~\ref{thm:representation}
are both complete.
\end{proof}

\section{All finite matrices}\label{sec:matrices}

Prior sections presumed an admissible $\Omega\supset W(A)$. We now build
admissible $\Omega_\epsilon$ arbitrarily close to $W(A)$ and pass to the
limit.

\subsection*{Convexity and compactness of $W(A)$}
$W(A)$ is the continuous image of the compact unit sphere of $\C^n$,
hence compact; its closure is itself.

\begin{lemma}[Toeplitz--Hausdorff]\label{lem:th}
For every $A\in M_n(\C)$, $W(A)$ is compact and convex.
\end{lemma}
\begin{proof}
Compactness is immediate. For convexity, take $z_j=x_j^*Ax_j$ with
$\|x_j\|=1$, $j=1,2$. Let $V=\mathrm{span}(x_1,x_2)$, $P_V$ the
orthogonal projection, $A_V=P_VA|_V$. Since $x_j\in V$,
$x_j^*A_Vx_j=x_j^*Ax_j=z_j$, so $z_j\in W(A_V)\subset W(A)$. It remains
to prove convexity in dimension at most two.

Dimension one is trivial. In dimension two, rank-one projections are
\[
 \pi(u,v,w)=\frac12\begin{pmatrix}1+u&v-iw\\ v+iw&1-u\end{pmatrix},
 \qquad u^2+v^2+w^2=1,
\]
and every unit $x\in\C^2$ gives $xx^*=\pi(u,v,w)$ for exactly one triple.
The map $(u,v,w)\mapsto\tr(A_V\pi(u,v,w))$ is real-affine into
$\C\simeq\R^2$. For a rank-2 real-affine map $g:\R^3\to\R^2$, the image
$g(S^2)$ equals $g(B^3)$: every line fiber of $g$ through a point of the
solid ball $B^3$ also meets $S^2$, at least in a tangency point. So
$g(S^2)$ is a filled ellipse (or, for rank-1 maps, a segment; for rank-0,
a point); each image is convex. Thus $W(A_V)$ is convex.
\end{proof}

\subsection*{Gaussian analytic outer approximation}
Identify $\C\simeq\R^2$. For a nonempty compact convex $K\subset\C$, let
$d_K(x)=\dist(x,K)$ and $R_K=\max_K|x|$. Both are finite.

\begin{lemma}[Gaussian outer]\label{lem:gaussian}
Fix $\epsilon>0$ and let $Z$ be standard 2D Gaussian with $c=E|Z|$. For
small enough $\delta>0$, the map $h_\delta(x)=E\,d_K(x-\delta Z)+\delta|x|^2$
is real analytic, proper, strictly convex with Hessian $\ge 2\delta I$,
and its sublevel set $\Omega_\epsilon=\{h_\delta<\epsilon/2\}$ is admissible
with
\[
 K\subset\Omega_\epsilon,\qquad \overline\Omega_\epsilon\subset\{d_K<\epsilon\}.
\]
\end{lemma}
\begin{proof}
$d_K$ is convex (as the infimum of the jointly convex $f(x,y)=\|x-y\|$
over $y\in K$) and 1-Lipschitz. Hence $|E d_K(x-\delta Z)-d_K(x)|\le c\delta$.
The convolution $E\,d_K(\cdot-\delta Z)$ is real analytic: complexify $x$
and bound by Gaussian decay. It is also convex (average of convex
functions). Adding $\delta|x|^2$ (Hessian $2\delta I\succ 0$) gives a
strictly convex, proper, real-analytic $h_\delta$ with
$\nabla^2 h_\delta\succeq 2\delta I$.

Choose $\delta>0$ with $c\delta+\delta R_K^2<\epsilon/2$ and
$c\delta<\epsilon/2$. For $x\in K$,
\[
 h_\delta(x)\le d_K(x)+c\delta+\delta R_K^2=c\delta+\delta R_K^2<\epsilon/2.
\]
For $x$ with $d_K(x)\ge\epsilon$,
\[
 h_\delta(x)\ge d_K(x)-c\delta\ge\epsilon-c\delta>\epsilon/2.
\]
Thus $K\subset\Omega_\epsilon\subset\{d_K<\epsilon\}$. On the boundary
$h_\delta=\epsilon/2$, so $d_K\le h_\delta+c\delta=\epsilon/2+c\delta<\epsilon$;
hence $\overline\Omega_\epsilon\subset\{d_K<\epsilon\}$.

Boundedness follows from properness of $h_\delta$. The level set
$\{h_\delta=\epsilon/2\}$ is nonempty (contained near $K$) and lies
strictly above the global minimum of $h_\delta$ (which is smaller than
$\epsilon/2$ because $K$ is in the sublevel set). At a global minimum of
a differentiable function the gradient vanishes; here it does not, because
the Hessian is positive definite everywhere. The implicit function
theorem gives a regular real-analytic boundary on $\partial\Omega_\epsilon$.
Radial projection from the minimum is a continuous bijection to the circle,
hence a homeomorphism. So $\partial\Omega_\epsilon$ is Jordan.
\end{proof}

\subsection*{The matrix case}

\begin{corollary}[Sharp matrix polynomial inequality]\label{cor:polynomial}
For all $n,m\ge 1$, $A\in M_n(\C)$, $d\ge 0$, and $B_0,\ldots,B_d\in M_m(\C)$,
with $P(z)=\sum_{j=0}^dB_jz^j$,
\[
 \left\|\sum_{j=0}^dA^j\otimes B_j\right\|
 \le 2\max_{z\in W(A)}\|P(z)\|.
\]
The constant is optimal.
\end{corollary}
\begin{proof}
By Lemma~\ref{lem:th}, $K=W(A)$ is compact convex. For each $\epsilon>0$,
Lemma~\ref{lem:gaussian} supplies admissible $\Omega_\epsilon$ with
$W(A)\subset\Omega_\epsilon$. Theorem~\ref{thm:representation} together
with $\kappa\le 2$ at every target size gives
\[
 \left\|\sum_{j=0}^dA^j\otimes B_j\right\|
 =\|P[A]\|\le 2\max_{\overline\Omega_\epsilon}\|P\|.
\]
As $\epsilon\downarrow 0$, the maxima of $\|P\|$ on
$\overline\Omega_\epsilon$ tend to the maximum on $W(A)$; this holds by
uniform continuity on a fixed compact neighborhood.

Sharpness: $A_0=\binom{0\ 2}{0\ 0}$, $P_0(z)=z$, $m=1$. Then
$\|A_0\|=2$. For $x=(x_1,x_2)$ with $|x|^2=1$,
$x^*A_0x=2\bar x_1 x_2$, so $|x^*A_0x|\le |x_1|^2+|x_2|^2=1$ with
equality at $x_1=x_2=1/\sqrt2$. Thus $W(A_0)=\{|z|\le 1\}$ and
$\max_{W(A_0)}|P_0|=1$, giving quotient two.
\end{proof}

\section{Bounded operators on Hilbert space}\label{sec:operators}

We now pass from matrices to all bounded operators on complex Hilbert
spaces, without a separability assumption. The compression trick is
that a polynomial $\sum_{j=0}^dA^j\otimes B_j$ applied to any single
vector uses only finitely many vectors, so the calculation is a matrix
calculation on a finite-dimensional subspace. Then Runge approximation
extends from polynomials to holomorphic and rational functions.

\subsection*{Spectral containment and convexity}
Let $H$ be a complex Hilbert space and $A\in \mathcal B(H)$ bounded. Put
$K=\overline{W(A)}$. As $W(A)\subset\overline{B(0,\|A\|)}$, $K$ is
compact.

\begin{lemma}\label{lem:spectral}
$K$ is compact and convex, and $\spec(A)\subset K$.
\end{lemma}
\begin{proof}
Compactness is clear. For convexity, given $z_j=(x_j,Ax_j)$ with
$\|x_j\|=1$, let $V=\mathrm{span}(x_1,x_2)$ and $A_V=P_VA|_V$. As in
Lemma~\ref{lem:th}, $z_j\in W(A_V)$; and $W(A_V)$ is convex for a
$2\times 2$ matrix.

For spectral containment: take $\lambda\notin K$, $\delta=\dist(\lambda,K)>0$.
For every $x$ with $\|x\|=1$, Cauchy--Schwarz on the real part gives
$|(x,(\lambda I-A)x)|\ge\delta$. Hence $\|\lambda I-A\|\ge\delta$ on the
unit sphere; $\lambda I-A$ has closed range and is injective. Applying
the same to $A^*$ (whose numerical range is $\overline{W(A)}$) shows
$\lambda I - A$ is also injective on the orthogonal complement of its
range, hence surjective. Thus $\lambda I-A$ is invertible on $H$.
\end{proof}

\subsection*{Polynomial bound by finite compression}

\begin{corollary}[Bounded-operator polynomial bound]\label{cor:hilbert-poly}
Let $A\in\mathcal B(H)$, $m,d\ge 0$, $B_0,\ldots,B_d\in M_m(\C)$,
$P(z)=\sum_jB_jz^j$. Then
\[
 \|P[A]\|=\left\|\sum_{j=0}^dA^j\otimes B_j\right\|_{\mathcal B(H\otimes\C^m)}
 \le 2\sup_{W(A)}\|P\|=2\max_K\|P\|.
\]
\end{corollary}
\begin{proof}
Fix a unit $\xi\in H\otimes\C^m$, $\xi=\sum_{j=1}^m\xi_j\otimes e_j$. Let
$E\subset H$ be the span of $\{A^k\xi_j:0\le k\le d,1\le j\le m\}$;
$\dim E\le (d+1)m<\infty$. Let $A_E=P_EA|_E$. By induction,
$A_E^k\xi_j=A^k\xi_j$ for $0\le k\le d$: at each step, the next iterate
is already in $E$. Hence $P[A_E]\xi=P[A]\xi$ as vectors in $E\otimes\C^m$.

For every unit $x\in E$, $(x,A_Ex)=(x,P_EAx)=(x,Ax)$; thus
$W(A_E)\subset W(A)$. Corollary~\ref{cor:polynomial} gives
\[
 \|P[A]\xi\|=\|P[A_E]\xi\|\le 2\max_{W(A_E)}\|P\|\cdot\|\xi\|
 \le 2\max_K\|P\|.
\]
Supremum over unit $\xi$ finishes. The final equality
$\sup_{W(A)}=\max_K$ follows from continuity of $P$ and density of $W(A)$
in $K$.
\end{proof}

\subsection*{Holomorphic evaluation via Runge}

\begin{corollary}[Bounded-operator holomorphic bound]\label{cor:hilbert-holo}
Let $U$ be an open neighborhood of $K$ and $F:U\to M_m(\C)$ holomorphic.
Then
\[
 F[A]=\sum_{i,j}f_{ij}(A)\otimes E_{ij}
\]
defines a bounded operator on $H\otimes\C^m$ and
\[
 \|F[A]\|\le 2\max_K\|F\|.
\]
\end{corollary}
\begin{proof}
Choose $\epsilon>0$ with $K_\epsilon=\{z:\dist(z,K)\le\epsilon\}\subset U$;
$K_\epsilon$ is compact convex, and its complement is connected. Scalar
Runge approximation (applied entrywise) gives matrix polynomials $P_l$
with $\max_{K_\epsilon}\|P_l-F\|\to 0$.

By Lemma~\ref{lem:gaussian}, take an admissible $\Omega$ with
$K\subset\Omega$ and $\overline\Omega\subset K_\epsilon$. Let $\Gamma=\partial\Omega$,
positively oriented. The scalar resolvent identity defines the entrywise
holomorphic calculus, and tensoring with $E_{ij}$ gives
\[
 F[A]=\frac{1}{2\pi i}\int_\Gamma(zI-A)^{-1}\otimes F(z)\,dz,
\]
convergent in operator norm (the resolvent is continuous on the compact
$\Gamma$, and $\Gamma$ lies in the resolvent set by spectral containment
$\spec(A)\subset K\subset\overline\Omega$). Similarly by Cauchy's theorem,
\[
 P_l[A]=\frac{1}{2\pi i}\int_\Gamma(zI-A)^{-1}\otimes P_l(z)\,dz,
\]
which also equals $\sum_j A^j\otimes(B_j)$ on polynomials
(expand resolvent as $\sum A^jz^{-j-1}$ and integrate against $z^j$).

Subtract:
\[
 \|F[A]-P_l[A]\|\le\frac{|\Gamma|}{2\pi}
 \cdot\max_{z\in\Gamma}\|(zI-A)^{-1}\|_{op}\cdot\max_{z\in\Gamma}\|F(z)-P_l(z)\|
 \longrightarrow 0.
\]
Therefore $\|F[A]\|=\lim_l\|P_l[A]\|$. By Corollary~\ref{cor:hilbert-poly}
applied to $P_l$ on $\Omega$, plus $\max_{\overline\Omega}\|P_l\|\le\|P_l\|_{K_\epsilon}$
and $\|P_l-F\|_{K_\epsilon}\to 0$,
\[
 \|F[A]\|\le 2\max_K\|F\|.
\]
\end{proof}

\subsection*{Complete 2-spectral set}

\begin{theorem}[$K$ is a complete 2-spectral set]\label{thm:complete}
Let $A\in\mathcal B(H)$ and $K=\overline{W(A)}$. Then for every $m\ge 1$
and every rational $R:U\to M_m(\C)$ with $U$ a neighborhood of $K$ and
the poles of $R$ outside $K$,
\[
 \|R[A]\|\le 2\max_K\|R\|.
\]
The constant $2$ is optimal, attained by $A_0=\binom{0\ 2}{0\ 0}$,
$R(z)=z$, $m=1$, $H=\C^2$.
\end{theorem}
\begin{proof}
$R$ is holomorphic on some such $U$ (the poles, being outside $K$, are
excluded). Apply Corollary~\ref{cor:hilbert-holo}. The evaluation
$R[A]=\sum_{ij}r_{ij}(A)\otimes E_{ij}$ agrees with the rational functional
calculus entry-by-entry.

Sharpness: $A_0$ and $R(z)=z$ give $\|R[A_0]\|=\|A_0\|=2$, and
$\max_K|z|=1$ because $K=W(A_0)=\overline{B(0,1)}$. The quotient is $2$.
\end{proof}

\section{Construction of the conformal coordinates}\label{sec:conformal}

We prove Lemma~\ref{lem:conformal}. The construction has five steps:
Riemann map by extremal derivative; holomorphic reflection across the
analytic boundary; collar patching to a full neighborhood of the closure;
exterior coordinate via inversion; orientation and conormal direction.
Standard references for the boundary behaviour of conformal maps and the
reflection principle are \cite{Ahlfors1979,Pommerenke1992}.

\subsection*{Step 1. Riemann map by extremal derivative}

Let $U\subset\C$ be a bounded simply connected domain and $z_*\in U$. Let
$\mathcal F$ be the set of injective holomorphic $\phi:U\to\D$ with
$\phi(z_*)=0$. This set is nonempty (small multiples of $z-z_*$ belong).
Let $\alpha=\sup_{\phi\in\mathcal F}|\phi'(z_*)|$. A closed disk
$B(z_*,r)\subset U$ gives, by Cauchy's derivative estimate,
$|\phi'(z_*)|\le 1/r$ for all $\phi\in\mathcal F$; so $0<\alpha<\infty$.

Choose $\phi_j\in\mathcal F$ with $|\phi_j'(z_*)|\to\alpha$. By Cauchy's
estimates and compact exhaustion, a subsequence converges locally uniformly
to a holomorphic $\phi:U\to\C$. Since $|\phi_j|\le 1$, the limit satisfies
$|\phi|\le 1$, and $\phi\not\equiv$ constant since $|\phi'(z_*)|=\alpha>0$;
max principle gives $\phi(U)\subset\D$. Moreover $\phi$ is injective: if
$\phi(a)=\phi(b)$ for distinct $a,b$, choose disjoint small closed disks
$D_a,D_b$ about them; by the argument principle (applied to
$\phi_j-\phi_j(a)$, with $\phi_j\to\phi$ uniformly) each of
$D_a,D_b$ contains a point with common image for large $j$, contradicting
injectivity of $\phi_j$.

If $w\in\D\setminus\phi(U)$, then $w\ne 0$ (as $\phi(z_*)=0$), and
\[
 q(z)=\frac{\phi(z)-w}{1-\bar w\phi(z)}
\]
is holomorphic on the simply connected $U$, nonvanishing, and
disk-valued. It therefore has a holomorphic square root $h$ with $|h|<1$,
injective (because $h^2=q$). Set $\beta=h(z_*)$, $|\beta|=\sqrt{|w|}$, and
$\psi(z)=(h(z)-\beta)/(1-\bar\beta h(z))$. Then $\psi$ is injective
holomorphic into $\D$ with $\psi(z_*)=0$. Its derivative at $z_*$:
\[
 |\psi'(z_*)|=\frac{|\phi'(z_*)|(1-|w|^2)}{2\sqrt{|w|}(1-|w|)}
 =\frac{1+|w|}{2\sqrt{|w|}}\,\alpha>\alpha,
\]
contradicting the maximality of $\alpha$. So $\phi:\Omega\to\D$ is a
conformal bijection.

\subsection*{Step 2. Boundary reflection}

For $0<r<1$, $\phi^{-1}(r\overline\D)$ is compact in $U$ (conformal inverse
continuous on $\D$); hence $|\phi(z)|\to 1$ as $z\to\partial U$ from inside.

Fix $p\in\partial U$. A regular real-analytic parametrization of the
boundary near $p$ extends holomorphically with nonzero derivative; its
local holomorphic inverse straightens the boundary. We obtain a holomorphic
chart $\chi$ sending $0$ to $p$, a real diameter onto the boundary arc,
and the upper half of a small disk onto $U\cap$ (image neighborhood).
Choose the neighborhood avoiding $z_*$.

Define $u(\xi)=\log|\phi(\chi(\xi))|$ on the upper half disk. Then $u$ is
harmonic, strictly negative (because $|\phi|<1$), and extends continuously
to the real diameter as $u=0$ (because $|\phi|\to 1$ at the boundary).

Consider the harmonic reflection $\widehat u$: on the boundary of a smaller
full disk, prescribe $u$ on the upper semicircle and $-\!u(\bar\cdot)$ on
the lower; Poisson extension to the full disk is odd in the real axis, so
it vanishes on the diameter. By uniqueness of the Dirichlet problem on
the upper half disk, $\widehat u$ agrees with $u$ there.

Its first nonvanishing homogeneous Taylor term at $0$ has the form
$c\,\operatorname{Im}(\xi^k)$, $c\ne 0$, $k\ge 1$, because $\widehat u$ is
harmonic and vanishes on the real diameter. If $k\ge 2$, $\sin(k\theta)$
takes both signs on $0<\theta<\pi$; along two small rays the leading term
would have both signs, contradicting $u<0$. So $k=1$, and $\nabla\widehat u(0)\ne 0$.

Let $\widehat v$ be a harmonic conjugate of $\widehat u$ on the full disk;
$\Phi(\xi)=\exp(\widehat u + i\widehat v)$ is holomorphic, nonvanishing,
and has nonzero derivative at $0$. On the upper half disk,
$|\Phi|=e^{\widehat u}=e^u=|\phi\circ\chi|$. Thus $\Phi/(\phi\circ\chi)$
is holomorphic, of constant modulus one, hence constant $C$ with $|C|=1$.
Therefore $\phi\circ\chi=\Phi/C$, extending holomorphically across the
real diameter; and $(\phi\circ\chi)'(0)=\Phi'(0)/C\ne 0$.
Returning to $z=\chi(\xi)$ shows that $\phi$ extends holomorphically
across $p$ with nonzero derivative.

\subsection*{Step 3. Collar patching}

For each $p\in\partial U$, the local extension is holomorphic on a disk
$U_p=B(p,R_p)$ containing $p$ and agreeing with $\phi$ on
$U\cap U_p$. Fix finitely many $p_1,\ldots,p_N$ and radii $R_i=R_{p_i}$
such that the smaller disks $V_i=B(p_i,R_i/3)$ cover $\partial U$. Let
$\phi_i:U_i=B(p_i,R_i)\to\C$ be the corresponding local extension.

Overlaps patch. If $V_i\cap V_j\ne\varnothing$ and $R_i\ge R_j$, then
$|p_i-p_j| < (R_i+R_j)/3\le 2R_i/3 < R_i$, so $U_i\cap U_j$ is a
nonempty open lens containing $p_j$ and meeting $U$ in a nonempty open
set. Both $\phi_i,\phi_j$ agree with $\phi$ on $U\cap U_i\cap U_j$; the
identity theorem extends agreement to all of $U_i\cap U_j$.

Therefore a global holomorphic extension
\[
 \Phi:\,U\cup\bigcup_i U_i\longrightarrow\C
\]
exists; $U\cup\bigcup_i U_i$ is an open neighborhood of $\overline U$.

The boundary image is a local diffeomorphism onto $\T$ (nonzero derivative
plus extended continuity from the interior), hence is open in $\T$. It is
also nonempty and compact, hence $=\T$.

Injectivity on $\overline U$: interior injectivity is inherited from $\phi$;
boundary injectivity follows because two distinct boundary preimages of
the same $\eta\in\T$ would give two distinct preimages of nearby
$\eta'\in\D$ in $U$ (rationale: each of two disjoint local inverses
restricted to a small ball around $\eta$ maps into $U$, since the
boundary arc maps to $\T$ and the interior side maps to $\D$),
contradicting interior injectivity. Since interior images lie in $\D$ and
boundary images lie in $\T$, there is no interior/boundary collision either.

Injectivity on a neighborhood of $\overline U$: if not, distinct sequences
$z_j,w_j$ approaching $\overline U$ with $\Phi(z_j)=\Phi(w_j)$ exist.
Subsequence limits $z,w\in\overline U$ satisfy $\Phi(z)=\Phi(w)$, hence
$z=w$; both sequences eventually lie in a single local inverse neighborhood
of $z$, contradicting injectivity of that local inverse.

Therefore $\Phi$ restricted to some open neighborhood $W_1$ of $\overline U$
is injective. Its image is open and contains $\overline\D$; the inverse is
holomorphic on a neighborhood of $\overline\D$.

Applying this to the admissible $\Omega$ produces the interior coordinate
$f:\Omega\to\D$ with the asserted extensions.

\subsection*{Step 4. The exterior coordinate}

Fix $z_0\in\Omega$ and put $\iota(z)=1/(z-z_0)$, $\iota(\infty)=0$. The
image $U_{\mathrm{ext}}=\iota(\C\setminus\overline\Omega)$ is a bounded
domain containing zero. Convexity of $\Omega$ and $z_0\in\Omega$ show each
ray from $z_0$ meets $\partial\Omega$ at one positive distance $R(\theta)$;
so
\[
 U_{\mathrm{ext}}=\{0\}\cup\{s e^{-i\theta}:0<s<R(\theta)^{-1}\}.
\]
This is star-shaped about zero, hence simply connected. Boundedness follows
also from a small ball about $z_0$ contained in $\Omega$ and a small ball
about $\infty$ in the image because $\Omega$ is bounded. Inversion preserves
regular real-analytic Jordan boundaries (its pole $z_0$ is interior to
$\Omega$, so not on $\partial\Omega$).

By Steps 1--3, a conformal $\phi_{\mathrm{ext}}:U_{\mathrm{ext}}\to\D$ with
$\phi_{\mathrm{ext}}(0)=0$ exists, whose inverse $h$ extends univalently
to $|\zeta|<\rho$ for some $\rho>1$; $h(0)=0$ is a simple zero.
Define
\[
 G(t)=z_0+\frac{1}{h(1/t)},\qquad |t|>r_0:=\rho^{-1}.
\]
Univalence of $h$ and $\iota$ shows $G$ is univalent on $|t|>r_0$. It maps
$|t|>1$ onto the exterior of $\overline\Omega$, sends $\infty$ to $\infty$
with a simple pole there, and its Laurent expansion reads
\[
 G(t)=at+b+\sum_{\ell\ge 1}c_\ell t^{-\ell},\qquad a=h'(0)^{-1}\ne 0.
\]
On $\T$, $G$ is a smooth diffeomorphism onto $\partial\Omega$.

\subsection*{Step 5. Orientation and conormal}

For $t=e^{i\theta}$, the radial and angular derivatives of $G$ at $r=1$ are
\[
 \partial_r G(re^{i\theta})|_{r=1}=tG'(t),
 \qquad\frac{d}{d\theta}G(e^{i\theta})=itG'(t).
\]
Since $G$ is conformal at $\T$, these are nonzero and perpendicular. Since
increasing $r$ enters the exterior of $\overline\Omega$, $tG'(t)$ points
outward; consequently $itG'(t)$ is the positive boundary tangent.
The interior conformal $f$ preserves boundary orientation (a biholomorphism
between Jordan domains), so $\psi=f\circ G$ is a smooth orientation-preserving
circle diffeomorphism. All assertions of Lemma~\ref{lem:conformal} are
proved.

\section*{Attribution and further reading}
Crouzeix posed the scalar constant-two question in 2004 and obtained the
universal complete bound $11.08$ in 2007
\cite{Crouzeix2004,Crouzeix2007}. Delyon--Delyon's positive integral
representation \cite{DelyonDelyon1999} and the Crouzeix--Palencia
double-layer argument \cite{CrouzeixPalencia2017} (simplified by
Ransford--Schwenninger \cite{RansfordSchwenninger2018}) give the complete
$(1+\sqrt{2})$ bound. Jin, Lorist--Schwenninger, and Luo give scalar
constant-two proofs \cite{Jin2026,LoristSchwenninger2026,Luo2026};
\AA hag--Czy{\.{z}}--Virtanen give the complete constant-two estimate in
base dimension three at arbitrary coefficient size \cite{AhagCzyzVirtanen2026}.
The optimal-similarity formulation is classical
\cite{Paulsen1984Similarity,Arveson1972}, and the dilation machinery used
for the endpoint lemma is from
\cite{SzNagy1953,Brodskii1978,KatsnelsonKheifetsYuditskii1997}.
The Stein criterion for the metric problem is \cite{Stein1952}; the
affine--convex separation tool is \cite[Theorem 11.2]{Rockafellar1970}.
The exterior kernel $m$ is the differentiated Grunsky kernel; the $b_k$
are the classical Faber polynomials
\cite{Beckermann2005,BeckermannCrouzeix2014,BeckermannStylianopoulos2018}.
The conformal reflection and collar are from \cite{Ahlfors1979,Pommerenke1992};
Runge approximation from \cite{Runge1885}; Toeplitz--Hausdorff convexity
of the numerical range from \cite{Toeplitz1918,Hausdorff1919}.
