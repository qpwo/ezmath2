\author{}

\begin{abstract}
We prove the Mahler conjecture for general convex bodies, in every
dimension and with no symmetry or regularity hypothesis. For every
convex body $K\subset\R^n$ and its Santal\'o point $s(K)$,
\[
 |K|\,|(K-s(K))^\circ|\ \ge\ \frac{(n+1)^{n+1}}{(n!)^2},
\]
with equality if and only if $K$ is a simplex. Two consequences are
given: a sharp functional (log-concave) Mahler inequality, and an
entropy--transport inequality.
\end{abstract}

\section{Introduction and overview}\label{sec:intro}

A \emph{convex body} in $\R^n$ is a compact convex set with nonempty
interior.  Write $|A|$ for $n$-dimensional Lebesgue volume and
$\langle\cdot,\cdot\rangle$ for the Euclidean inner product.  A simplex
is the convex hull of $n+1$ affinely independent points.

\paragraph{Polarity and the Santal\'o point.}
For $z\in\operatorname{int}K$ let
\[
 (K-z)^\circ=\{y\in\R^n:\ \langle y,x-z\rangle\le 1\ \text{for all }x\in K\}
\]
be the polar body of $K$ re-centered at $z$.  The function
$z\mapsto |(K-z)^\circ|$ is strictly convex on $\operatorname{int}K$ and
tends to $+\infty$ at the boundary; hence it has a unique interior
minimizer, the \emph{Santal\'o point} $s(K)$.  (We prove this in
Section~\ref{sec:cones}.)  The \emph{volume product} is
\[
 P(K)=|K|\,|(K-s(K))^\circ|
     =\inf_{z\in\operatorname{int}K}|K|\,|(K-z)^\circ|.
\]
It is invariant under invertible affine maps.

\begin{theorem}[Mahler conjecture]\label{thm:mahler}
For every $n\ge1$ and every convex body $K\subset\R^n$,
\[
 P(K)\ \ge\ \frac{(n+1)^{n+1}}{(n!)^2}.
\]
Equality holds if and only if $K$ is a simplex.
\end{theorem}

Two remarks.  First, because $P(K)$ is the minimum over translation
points, Theorem~\ref{thm:mahler} also gives the displayed lower bound
for $|K|\,|(K-z)^\circ|$ at \emph{every} interior point $z$.
Second, the \emph{symmetric} Mahler conjecture asks for the stronger
constant $4^n/n!$ on the restricted class of centrally symmetric bodies.
That sharper statement is different and is proved separately in the
companion paper~\cite{OpenAISymmetricMahler2026}; the argument here does
not use symmetry.

\paragraph{Context.}
The problem originates with Mahler, who settled the planar case and its
triangle equality among polygons \cite{Mahler1938}, and who formulated
the higher-dimensional symmetric version \cite{Mahler1939}.  Meyer gave
a new proof of the planar general inequality with a complete equality
characterization (triangles are the only minimizers)
\cite{Meyer1991}.  The affine center used here belongs to Santal\'o's
theory \cite{Santalo1949}.  Asymptotic lower bounds (inverse
Santal\'o / Bourgain--Milman type) go back to
\cite{BourgainMilman1987, Kuperberg2008, Nazarov2012,
MastrantonisRubinstein2024}; these give exponential constants but not
the sharp simplex value.  Sharp results for special classes include the
polytope case with at most $n+3$ vertices via shadow systems
\cite{MeyerReisner2006}, strict local minimality at every simplex
\cite{KimReisner2011}, the three-dimensional symmetric case
\cite{IriyehShibata2020}, and the general three-dimensional case
\cite{ChenLiXiXu2026}.  The present argument works for arbitrary bodies
in all dimensions and also pins down the global equality case.

\paragraph{How to read the proof.}
The proof has a single spine:
\begin{enumerate}[label=\arabic*.]
\item \emph{Cone reduction} (Section~\ref{sec:cones}):  translate $K$
  and lift it to a cone $C\subset\R^{n+1}$.  The volume product equals,
  up to an explicit constant, a product of two Laplace integrals on the
  cone and its positive dual $D$.  It suffices to show that product is
  $\ge1$.
\item \emph{Gaussian projections} (Section~\ref{sec:proj}):  build two
  maps of a Gaussian into the cones using orthogonal projections, with
  carefully chosen biases and covariance.  Their Jacobians turn the
  Laplace product into an entropy-type lower bound.
\item \emph{Entropy inequality} (Section~\ref{sec:entropy}):  change of
  variables plus Jensen reduces the cone inequality to $\mathcal E\le0$
  for an explicit error functional $\mathcal E$.
\item \emph{Matrix comparison} (Sections~\ref{sec:quad}--\ref{sec:layers}):
  compare the whole field of projection derivatives to spectral
  thresholds of a linear Gaussian matrix $L=\sum_i G_iM_i$.  Hermite
  calculus separates the linear part; the leftover ``layer'' terms are
  bounded below by zero.
\item \emph{Linear field and equality}
  (Section~\ref{sec:linear}):  express the remaining terms as averages
  over eigenvalue pairs of $L$, absorb them with a strict one-variable
  inequality, and show equality forces the projection to be
  coordinatewise, hence $K$ a simplex.
\end{enumerate}
The dimension-independent scalar inequalities used throughout are
collected once (Proposition~\ref{prop:scalar-input}) and proved in the
two appendices, Sections~\ref{sec:scalar} and~\ref{sec:segments}.

\paragraph{Functional and transport consequences.}
The geometric theorem implies a sharp functional (log-concave) Mahler
inequality for arbitrary convex potentials with no centering
assumption:
\begin{corollary}[Functional Mahler]\label{cor:functional-mahler}
For every $n\ge1$ and every proper lower-semicontinuous convex
$\varphi:\R^n\to\R\cup\{+\infty\}$ with
$0<\int e^{-\varphi}<\infty$,
\[
 \Big(\int_{\R^n}e^{-\varphi(x)}\,dx\Big)
 \Big(\int_{\R^n}e^{-\varphi^*(y)}\,dy\Big)\ \ge\ e^n,
\]
where $\varphi^*(y)=\sup_x\{\langle x,y\rangle-\varphi(x)\}$.
The second integral may be infinite, in which case the inequality is
trivial.
\end{corollary}
This is sharp ($e^n$ is attained).  Via Gozlan's equivalence it yields
an entropy--transport inequality (Corollary~\ref{cor:entropy-transport}).
Both are proved in Section~\ref{sec:consequences} and are \emph{not}
inputs to the geometric argument.

\section{Polarity, the cone reduction, and the scalar input}\label{sec:cones}

\subsection{The minimizing center}
The support function is $h_K(u)=\sup_{x\in K}\langle u,x\rangle$.  For
$z\in\operatorname{int}K$ and $n\ge2$, polar coordinates give
\[
 |(K-z)^\circ|=\frac1n\int_{S^{n-1}}\big(h_K(u)-\langle z,u\rangle\big)^{-n}\,d\sigma(u),
\]
where $\sigma$ is surface measure on the unit sphere.  As a function of
$z$ this is strictly convex: $t\mapsto t^{-n}$ is strictly convex for
$t>0$, and two distinct centers have inner products differing on a set
of positive surface measure.  It blows up at $\partial K$: at a limiting
boundary point with supporting normal $u_0$, the quantity
$\varepsilon=h_K(u_0)-\langle z,u_0\rangle\to0$; the integrands share a
Lipschitz constant in $z\in K$, are $O(\varepsilon)$ on a spherical cap
of radius $\varepsilon$, and that cap has measure $\gtrsim
\varepsilon^{n-1}$, so its contribution is $\gtrsim\varepsilon^{-1}$.
Hence a unique interior minimizer exists.  In dimension one, with
$K=[a,b]$, the same conclusion follows from
$(z-a)^{-1}+(b-z)^{-1}$.

For an invertible affine map $x\mapsto Bx+a$ we have
$(BK+a-(Bz+a))^\circ=B^{-\mathsf T}(K-z)^\circ$, the two volume factors
change by reciprocal determinants, and $s(BK+a)=Bs(K)+a$.  Thus $P$ is
affine-invariant and the Santal\'o point is affine-equivariant.

\subsection{The cone reduction}
Put $m=n+1$.  Fix $z_0\in\operatorname{int}K$ and define the cone
\[
 C=\{(y,t):\ t\ge0,\ y\in t(K-z_0)\},\qquad
 D=C^*=\{w:\ \langle w,x\rangle\ge0\ \text{for all }x\in C\}.
\]
Here the star is \emph{positive} duality.  Both cones are closed, convex,
solid (nonempty interior) and pointed (contain no line).  For interior
vectors $V\in D$ and $U\in C$ write the Laplace integrals
\[
 \chi_C(V)=\int_C e^{-\langle V,x\rangle}\,dx,\qquad
 \chi_D(U)=\int_D e^{-\langle U,x\rangle}\,dx
\]
in $m$-dimensional Lebesgue measure.

\begin{lemma}[Cone reduction]\label{lem:cone-reduction}
For $U=(0,m)$ and $V=(0,1)$,
\[
 \chi_C(V)\,\chi_D(U)=\frac{(n!)^2}{m^m}\,|K|\,|(K-z_0)^\circ|.
\]
For \emph{any} closed convex solid pointed cone $C$ with dual $D=C^*$
and interior $U\in C$, $V\in D$, both Laplace integrals are finite, and
their product and the inner product $\langle U,V\rangle$ are preserved
under $(C,D,U,V)\mapsto(BC,B^{-\mathsf T}D,BU,B^{-\mathsf T}V)$ for
every invertible linear map $B$.
\end{lemma}

\begin{proof}
A point $(w,t)$ lies in $D$ iff $t\ge\sup_{x\in K-z_0}\langle -w,x\rangle$.
Since $0$ is interior to $K-z_0$ this forces $t\ge0$, and the slice at
height $t$ is $-t(K-z_0)^\circ$.  Integrating slices,
\[
 \chi_C((0,1))=n!\,|K|,\qquad
 \chi_D((0,m))=n!\,m^{-m}\,|(K-z_0)^\circ|,
\]
giving the product formula.  Interiority of $V$ gives
$\langle V,x\rangle\ge c|x|$ on $C$, and interiority of $U$ the analogue
on $D$, so both integrals are finite.  Under the linear rescaling the
two change-of-variables factors are $|\det B|$ and $|\det B|^{-1}$,
while inner products in the exponentials are unchanged.
\end{proof}

By Lemma~\ref{lem:cone-reduction} it suffices to prove
\begin{equation}\label{eq:goal}
 \chi_C(V)\,\chi_D(U)\ \ge\ 1
\end{equation}
whenever $C\subset\R^m$ is closed, convex, solid, pointed, $D=C^*$,
$U\in\operatorname{int}C$, $V\in\operatorname{int}D$, and
$\langle U,V\rangle=m$.  These are the \emph{standing cone
assumptions}; the constant $m$ in the inner product is what makes the
Gaussian normalization below come out exactly.  We will additionally
see that equality forces $C$ to be a linear image of an orthant, which
by Section~\ref{sec:linear} is equivalent to $K$ being a simplex.

\subsection{Scalar profiles and the analytic input}
The argument uses a fixed family of one-variable functions, independent
of $C$ and of the dimension.  Let
$\phi(x)=(2\pi)^{-1/2}e^{-x^2/2}$, $p(x)=\int_{-\infty}^x\phi$, and
$\gamma f=\int f\,\phi$.  Put $\mathcal N=x\partial_x-\partial_x^2$.
Temporarily write
\[
 X=\frac{p}{\phi},\qquad Y=\frac{1-p}{\phi},\qquad
 f_0=-(1-p)-\log p,\qquad j_0=-p-\log(1-p),
\]
and define the three even profiles
\begin{equation}\label{eq:profiles}
 d=(1+xX)^2 f_0+(1-xY)^2 j_0,\qquad
 g=\tfrac12\big(1-X^2f_0-Y^2j_0\big),\qquad v=\log(XY).
\end{equation}
The functions $d,g,v$ are smooth, even, and $v'=-\log\frac{p}{1-p}$ up
to an additive constant.

\begin{lemma}[Profile identities]\label{lem:profile-ids}
With $d,g,v$ as above,
\begin{equation}\label{eq:profile-ids}
 (\mathcal N+2)g=d,\qquad d'=2xd-v'-2g',\qquad
 \mathcal N d-v''=-(\mathcal N+2)(d+g'').
\end{equation}
\end{lemma}
\begin{proof}
One checks $X'=1+xX$, $Y'=xY-1$, $f_0'=-\phi(1-p)/p$, $j_0'=\phi p/(1-p)$.
Direct substitution gives
$X^2f_0'+Y^2j_0'=0$, $XX'f_0'+YY'j_0'=-1$,
$(X')^2f_0'+(Y')^2j_0'=-v'$.  Also $X''=xX'+X$ and $Y''=xY'+Y$.
Differentiating $g'=-(XX'f_0+YY'j_0)$ once more gives
$(\mathcal N+2)g=d$.  Differentiating $d=(X')^2f_0+(Y')^2j_0$ gives the
second identity, and differentiating the first identity twice and
combining with the second gives the third.
\end{proof}

\section{Biased Gaussian projections and simultaneous normalization}
\label{sec:proj}

We now attach to the cone data a family of Gaussian projections whose
means move along a fixed curve in $C$.  Their derivative field supplies
both the Jacobians in the entropy estimate and the matrix fields in the
later quadratic analysis.  Throughout, fix cone data satisfying the
standing assumptions and a symmetric $T$ with
\[
 t_-I\le T\le t_+I,\qquad \Sigma=I+T,\qquad Z=\Sigma^{1/2}G,
\]
where $G$ is standard Gaussian in $\R^m$.  Thus every eigenvalue of
$\Sigma$ lies in $[.978,1.064]$.  Expectation is in $G$ unless stated.

\paragraph{Matrix trace conventions.}
For matrix fields, set
\[
 \tau B=\tfrac1m\E\operatorname{tr}B,\quad
 \tau_\Lambda B=\tau(\Lambda B),\quad
 \langle B,E\rangle_\Lambda=\tau_\Lambda(B\circ E),\quad
 B\circ E=\tfrac12(BE+EB),
\]
for symmetric $B,E$ and deterministic symmetric weight $\Lambda$;
subscript $I$ is omitted.  Transposition and cyclicity give
$\operatorname{tr}(\Lambda BE)=\operatorname{tr}(\Lambda EB)$ without
assuming commutation, so $\langle\cdot,\cdot\rangle_\Sigma$ is a
positive inner product (as $\Sigma>0$).  We write
$\|B\|_2^2=\tau(B^{\mathsf T}B)$ and $[B,E]=BE-EB$.
